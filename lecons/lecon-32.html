26 Ash-Shu'ara' — 227 آيَة

(1) تَا، سِينْ، مِيمْ.

(2) طࣹنْ كٝ آيࣹيجࣹ دࣹفْتࣹرࣹ ࢠَنْنگُندࣹ ندࣹنْ.

(3) بࣹلَجٝؤٝ أَن أَ تَمْپِنِرَيْ هٝورࣹ مَاطَ كِنْ فِي ࢠࣹ لَاتَاكِ گٝمْطِنْࢠࣹ.

(4) سِ مࣹنْ مُويِي مࣹنْ جِپِّنَيْ إࣹ مَࢠّࣹ إِمّٝرْدࣹ كَ كَمُّ آيَ دَاطࣹ مَࢠّࣹ طࣹنْ ݧَلَّ كٝ طࣹ يَنْكِنَنِيطࣹ مٝ.

(5) أَرَتَا إࣹ مَࢠّࣹ وَاجُ إِمّٝرْدࣹ كَ هِنّٝرَنْتࣹ أٝن فُطَّانگُ سِ وَنَا ࢠࣹ لَاتِكࣹ طُورِيࢠࣹ نگُ.

(6) گٝمْطِي ࢠࣹ فࣹنِّي أَرٝيَيْ إࣹ مَࢠّࣹ كَبَرُوجِ كٝ ࢠࣹ لَاتِنٝو كٝࢠࣹ جَلْكِتَنٝ.

(7) إࣹنࣹي، ࢠࣹ يِئَالِ فَادࣹ إࣹ لࣹيْدِ ندِنْ، نٝ فٝتِ هࣹوْدࣹ كٝ مࣹنْ فُطِنِ إࣹ مَيْرِ إِمّٝرْدࣹ إࣹ كَلَ نٝونࣹ تࣹدُّطٝ?

(8) پࣹلّࣹتْ، وٝنِي إࣹ طُمْ كٝ جَݝْتَا طٝو آيَ، ࢠُرْࢠࣹ هࣹيوُࢠࣹ إࣹ مَࢠّࣹ لَاتَاكِ گٝمْطِنْࢠࣹ.

(9) پࣹلّࣹتْ، جٝومِ مَاطَ كٝ كَنْكٝ وٝنِ تࣹدُّطٝ يُرْمࣹتࣹيطٝ أٝنْ.

(10) جَݝْتٝ تُمَ ندࣹ جٝومِ مَاطَ نٝدُّنٝو موسى، طُمْ وٝنِ أَرُ إࣹ يِمْࢠࣹ تٝوݧٝوࢠࣹ ࢠࣹنْ.

(11) يِمْࢠࣹ فرعون، كَا أٝن هُلَتَا

(12) أٝ مَاكِ: "جٝومْ أَم ، مِينْ مِيطُ هُلِ وَتَ ࢠࣹ فࣹنَّمْ.

(13) ࢠࣹرْندࣹ أَم ندࣹنْ ࢠِتّٝو هَرَ طࣹمْنگَلْ أَم نگَلْ وࣹلِّتَاكِ، أَوَ نُلُ هَارُونَ.

(14) نٝ وٝودَنِ ࢠࣹ إࣹ دٝوْ أَم بَكَّاتُ مِيطُ هُلِ فِي وَتَ ࢠࣹ وَرَمْ.

(15) أٝ دَالِ: "وَنَا نٝنْ، يَهࣹي أٝنٝنْ طِطٝ، مࣹنْطࣹنْ وٝنْندِ إࣹ مٝوطٝنْ كٝ مࣹنْ هࣹطِيࢠࣹ".

(16) أَرࣹي كَ فرعون وِئٝنْ: "مࣹنࣹنْ دࣹيْ كٝ مࣹنْ نُلَاࢠࣹ جٝومَ وِنْندࣹرࣹ ندࣹنْ."

(17) طُمْ وٝنِ أَݖِّتُ ࢠِطْࢠࣹ إِسْرَئِيلَ ࢠࣹنْ وٝنْندُدࣹ إࣹ أَمࣹنْ.

(18) أٝ وِئِ: "ࣹنࣹي، مࣹنْ نࣹئَانٝ مَ كَ أَمࣹنْ طٝو هَرِ كٝ أَ بٝوبٝ، ݧِيࢠُطَا كَ أَمࣹنْ طٝو دُوࢠِ إࣹ نگُرْندَمْ مَاطَمْ?"

(19) گٝلُّطَا گٝلَّلْ مَاطَ نگَلْ گٝلُّنٝطَا نگَلْ، أَن كٝ إࣹ يࣹدُّࢠࣹ ࢠࣹنْ جࣹيَطَا.

(20) أٝ مَاكِ: مِي" وَطِي طُمْ، أٝنْتُمَ هَرَ مِينْ كٝ مِي جࣹيَاطٝ إࣹ مَجُّࢠࣹ ࢠࣹنْ".

(21) مِي دٝگِ أٝنْ، تُمَ ندࣹ مِي هُلُنٝو أٝنْ، جٝومِ أَم أٝكِّنْ مِي گَنْندَلْ أٝ وَطِنْ مِي جࣹيَاطٝ إࣹ نُلَاࢠࣹ ࢠࣹنْ.

(22) طُمْ كٝ نࣹيمَ مٝ وَجِتِرْتَا مِي طُمْ، فِي أَ مَݖِّنْطِنِي ࢠِطْࢠࣹ إسرائيل ࢠࣹنْ.

(23) فرعون وِئِ: " كٝ هٝمْنبٝ جࣹيِ وِنْندࣹرࣹ ندࣹنْ?"

(24) أٝ مَاكِ: "كٝ جٝومَ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ إࣹ كٝ وٝنِ كٝنْ هَكُّندࣹ مَجِّ، سِ أٝن لَاتِكࣹ يَنَنَاࢠࣹ."

(25) أٝ وِئَنِ ࢠࣹنْ وٝنُࢠࣹ بَنْݝࣹ مَكّٝ: "ࣹنࣹي; أٝن هࣹطَتَاكٝ?"

(26) أٝ مَاكِ: جٝومِ" مٝوطٝنْ إࣹ جٝومَ بَابِرَاࢠࣹ أَرْدِيࢠࣹ ࢠࣹنْ."

(27) فرعون وِئِ: "پࣹلّࣹتْ، نُلَاطٝ مٝوطٝنْ أٝن نُلَاطٝ فَادࣹ إࣹ مٝوطٝنْ كٝ أٝ فࣹيتُطٝ."

(28) أٝ مَاكِ: "كٝ جٝومَ فُنَّانگࣹ إࣹ هِرْنَانگࣹ إࣹ كٝ وٝنِ كٝنْ هَكُّندࣹ مَجِّ، سِ أٝن لَاتِكࣹ هَكِّلٝوࢠࣹ."

(29) فرعون وِئِ: مِي" وٝوندِي سِ جٝگِتِكࣹ رࣹوࣹتࣹيطٝ تَنَا أَمْ، مِي وَطࣹتࣹ جࣹيَاطٝ دُمْنبَاࢠࣹ ࢠࣹنْ".

(30) أٝ مَاكِ: "ࣹنࣹي، هَيْ سِ مِي أَدَّنِي مَ هُوندࣹ ࢠَنْنگُندࣹ?"

(31) فرعون وِئِ: "َوَ أَدُّ طُمْ، سِ تَوِي كٝ إࣹ گٝونگُ ࢠࣹنْ جࣹيَاطَا."

(32) أٝ وࣹرْلِييي تُگّٝرْدُ مَكّٝ ندُنْ، جَكَ كٝ ندُ بِيدَاندࣹ(نگٝولْوَ) ࢠَنْنگُبَ.

(33) أٝ يَلْتِنِ جُنْنگٝ مَكّٝ نگٝنْ، جَكَ يٝو كٝ نگٝ رَوْنُنگٝ وٝنَنْندࣹ ندَارٝوࢠࣹ ࢠࣹنْ.

(34) فرعون وِئَنِ نبَتُ نگُنْ فࣹي مَكّٝ: "پࣹلّࣹتْ، أٝوطٝو كٝ نبِلࣹيجٝ هࣹيوُطٝ گَنْندَلْ".

(35) هِمٝ فَاندِي يَلْتِنِرْگٝلْ أٝن كَ لࣹيْدِ مٝوطٝنْ نبِلࣹوُ مَكّٝ نگُنْ. كٝ هُنْطُنْ يَمِرْطٝنْ?

(36) ࢠࣹ وِئِ : "نࣹنُّ مٝ إࣹ مُسِطّٝ مَكّٝ أٝنْ، نُلَا إࣹ ݖَئࣹ طࣹنْ مٝوࢠٝوࢠࣹ (نبِلࣹ)".

(37) ࢠࣹ أَدَّنࣹ كَلَ نبِلࣹيجٝ أَنْندُطٝ.

(38) نبِلࣹيࢠࣹ ࢠࣹنْ مٝوࢠَا فِي فٝدٝورࣹ ݧَلَاندࣹ أَنْندَاندࣹ.

(39) يِمْࢠࣹ ࢠࣹنْ وِئَنَا: "تَوْ سِ أٝنٝنْ كٝ أٝن مٝوࢠِكࣹ(ندࣹنْندِ)"

(40) بࣹلَجٝؤٝ مࣹنࣹنْ مࣹنْ جٝكَّيْ نبِلࣹيࢠࣹ ࢠࣹنْ، سِ ࢠࣹ لَاتِكࣹ كٝ كَمْࢠࣹ وٝنِ فٝولُࢠࣹ ࢠࣹنْ.

(41) تُمَ ندࣹ نبِلࣹيࢠࣹ ࢠࣹنْ أَرُنٝو; ࢠࣹ وِئَنِ فرعون: "ࣹنࣹي، هَرَ نٝ وٝودَنِ مࣹنْ نجٝࢠْدِ سِ تَوِي كٝ مࣹنْ وٝنِ فٝولُࢠࣹ ࢠࣹنْ?"

(42) أٝ وِئِ : "ࣹيّٝ، هَرَيْ أٝنْتُمَ أٝن جࣹيَامَ إࣹ ࢠَطْتِنَاࢠࣹ ࢠࣹنْ".

(43) موسى مَاكَنِ ࢠࣹ: "نبࣹدّࣹي، كٝ وࣹرْلٝتٝنْ كٝنْ".

(44) ࢠࣹ لِࢠِ ࢠٝكّٝيْ مَࢠّࣹ كٝيْ إࣹ ݖَوٝيْ مَࢠّࣹ كٝنْ، ࢠࣹ وِئِ: "سَبُ تࣹدُّنگَلْ فرعون، كٝ مࣹنࣹنْ وٝنِ فٝولٝوࢠࣹ ࢠࣹنْ".

(45) موسى وࣹرْلِي تُگّٝرْدُ مَكّٝ ندُنْ، تَوِ كَيْرُ هِندُ مٝطَ كٝ ࢠࣹ فࣹفِندِي كٝنْ.

(46) نبِلࣹيࢠࣹ ࢠࣹنْ سُويِ ࢠࣹ سُجِّ.

(47) ࢠࣹ وِئِ : مࣹنْ" گٝمْطِنِي جࣹيْطٝ وِنْندࣹرࣹ ندࣹنْ،

(48) جࣹيْطٝ موسى إࣹ هَارُونَ".

(49) فرعون وِئِ: "ࣹنࣹي، أٝن گٝمْطِنَيْ مٝ، كٝ أَدِي مِي دُݝَنَادࣹ أٝنْ? پࣹلّࣹتْ، كٝ كَنْكٝ وٝنِ مَوْطٝ مٝوطٝنْ أَنْندِنْطٝ أٝن أٝن نبِلࣹوُ نگُنْ، كٝنٝ أَرَيْ أَنْندٝنْ، مَمِ تَڃُ جُوطࣹ مٝوطٝنْ إࣹ كٝيْطࣹ مٝوطٝنْ تَڃَ لُوتِندِرَ. مَمِ فࣹتُ أٝنْ، أٝن فٝوْ مٝوطٝنْ".

(50) ࢠࣹ وِئِ: "تَنَا أَلَا، مࣹنࣹنْ كٝ فَادࣹ كَ جٝومِ أَمࣹنْ مࣹنْ رُتّٝتٝو".

(51) مࣹنࣹنْ مࣹطࣹنْ تَمْڃِنِي ندࣹ جٝومِ أَمࣹنْ يَوْتَنِ مࣹنْ گٝوپِ أَمࣹنْ طِنْ،تَوْدࣹ مࣹنْ لَاتِكࣹ أَدِيࢠࣹ گٝمْطِنْندࣹ ࢠࣹنْ.

(52) مࣹنْ وَهَيِنِ فَادࣹ إࣹ موسى: "طُمْ وٝنِ رٝڃِدُ إࣹ جࣹيَاࢠࣹ أَم ࢠࣹنْ. پࣹلّࣹتْ، أٝنٝنْ كٝ أٝن جٝكِّتࣹتࣹيࢠࣹ".

(53) فرعون نُلِ كَ ݖَئࣹ مٝوࢠٝوࢠࣹ.

(54) أٝ وِئِ: پࣹلّࣹتْ، ࢠࣹي كٝ ࢠࣹ مٝجٝبُنْ پَمَرُنْ.

(55) پࣹلّࣹتْ، كَمْࢠࣹ هِࢠࣹ تِكِّنْندࣹ مࣹنْ.

(56) إࣹنࣹنْ نٝنْ إࣹن فٝوْ كٝ إࣹن رࣹنْتِيࢠࣹ.

(57) مࣹنْ يَلْتِنِ ࢠࣹ إِمّٝرْدࣹ إࣹ گࣹسࣹ إࣹ ݖَنْطِ.

(58) إࣹ نگَلُوجِ إࣹ ݧِيࢠِرْدࣹ تࣹدُّندࣹ.

(59) كٝ وَنٝ نِي، مࣹنْ رٝنِنِ طُمْ ࢠِطْࢠࣹ إسرائيل ࢠࣹنْ.

(60) ࢠࣹ جٝكِّࢠࣹ كَ پُطَلْ.

(61) تُمَ ندࣹ جَمَاجِ طِنْ طِنْ يِأْندِرْنٝو، يِمْࢠࣹ موسى ࢠࣹنْ وِئِ: "ࣹنࣹنْ إࣹن هࣹوْتٝيْتࣹ".

(62) أٝ مَاكِ: "طُمْ وَطَتَا، پࣹلّࣹتْ، جٝومِ أَم نٝ وٝنْندِ إࣹ أَم أٝ فࣹوْنَيْ لَمْ".

(63) مࣹنْ وَهَيِنِ إࣹ موسى، طُمْ وٝنِ پِيِرْ تُگّٝرْدُ مَاطَ ندُنْ مَايٝ نگٝنْ، نگٝ سࣹيكِي سࣹيكٝلْ كَلَ سࣹيكٝلْ نَ فٝتِ إࣹ تُلْدࣹ مَوْندࣹ.

(64) مࣹنْ ࢠَنِّ تٝنْ هࣹدِّيࢠࣹ ࢠࣹنْ.

(65) مࣹنْ دَنْندِ موسى إࣹ وٝنْندُࢠࣹ إࣹ مَكّٝ ࢠࣹنْ فٝوْ مَࢠّࣹ.

(66) رࣹفْتِ مࣹنْ يٝولِ هࣹدِّي ࢠࣹ ࢠࣹنْ.

(67) پࣹلّࣹتْ، وٝنِي إࣹ طُمْ كٝ جَݝْتَا طٝو آيَ، ࢠُرْࢠࣹ هࣹيوُدࣹ إࣹ مَࢠّࣹ لَاتَكِ گٝمْطِنْࢠࣹ.

(68) پࣹلّࣹتْ، جٝومِ مَاطَ، كٝ كَنْكٝ وٝنِ تࣹدُّطٝ يُرْمࣹتࣹيطٝ.

(69) جَنْنگَنْ ࢠࣹ كُمْپِتَلْ فِي إبراهيم.

(70) تُمَ ندࣹ أٝ وِئَنّٝو بࣹنْ مَكّٝ إࣹ يِمْࢠࣹ مَكّٝ ࢠࣹنْ; كٝ هٝنْطُمْ رࣹوٝتٝنْ?

(71) ࢠࣹ وِئِ: "مࣹطࣹنْ رࣹوُدࣹ سَنَمُوجِ، مࣹنْ ݧَلَّيْ أُفِنَنَادࣹ طِ".

(72) أٝ مَاكِ: "يَلِّ هِطِ نَنَنَ أٝنْ، ندࣹ نٝدّٝتٝنْ".

(73) كَا هِطِ نَفَ أٝن مَاطُمْ طِ لٝرَّ أٝنْ?

(74) ࢠࣹ وِئِ: "كٝ وٝنِ، مࣹنْ تَوِي بَابِرَاࢠࣹ أَمࣹنْ ࢠࣹنْ كٝنِي ࢠࣹ گٝلِّرَيْنٝو".

(75) أٝ مَاكِ: "ࣹنࣹي، أٝن يِئِي كٝنْ كٝ وٝنُطٝنْ إࣹ رࣹوُدࣹ?"

(76) أٝنٝنْ إࣹ بَابِرَاࢠࣹ مٝوطٝنْ أَدِنٝوࢠࣹ ࢠࣹنْ.

(77) پࣹلّࣹتْ، كَمْࢠࣹ كٝ ࢠࣹ أَيْࢠࣹ أَمْ، سِ وَنَا جٝومَ وِنْندِرࣹ ندࣹنْ.

(78) أٝن تَگُطٝ لَمْ، كٝ كَنْكٝ فࣹوْنَتَ لَمْ.

(79) أٝن كٝ كَنْكٝ ݧَمِّنْتَ لَمْ أٝ يَرْنَنْ-مِ.

(80) سِ مِي نَوْنِي، كٝ كَنْكٝ سࣹلِّنْتَ لَمْ.

(81) إࣹ أٝن وَرَيْطٝ لَمْ رࣹفْتِ أٝ وُرْنِتَنْ مِي.

(82) أٝن مٝ مِي تَمْڃِنِي كَدِ ندࣹ أٝ يَافَتٝومِ گٝوپِ أَم طِنْ ݧَلْندࣹ نجٝࢠْدِ دِينَ كَنْ.

(83) جٝومَمْ، أٝكَّمْ گَنْندَلْ وَتِّدَا مِي إࣹ مٝڃُّࢠࣹ ࢠࣹنْ.

(84) وَطَنَا لَمْ طࣹمْنگَلْ گٝونگُنگَلْ كَ سَكِّتِيࢠࣹ ࢠࣹنْ.

(85) وَطَا مِي جࣹيَاطٝ إࣹ رٝنٝوࢠࣹ الجنة نࣹيمَاجِ طِنْ.

(191) پࣹلّࣹتْ، جٝومِ مَاطَ، كٝ كَنْكٝ وٝنِ تࣹدُّطٝ يُرْمࣹتࣹيطٝ.

(192) إࣹ پࣹلّࣹتْ، كَيْرࣹ كٝ ندࣹ جِپِّنَاندࣹ جٝومَ وِنْندࣹرࣹ ندࣹنْ.

(193) جِپِّنِي ندࣹ رُوهُ هٝولَاطٝ أٝنْ.

(194) كَ ࢠࣹرْندࣹ مَاطَ فِي يٝ أَ لَاتٝ جࣹيَاطٝ إࣹ جࣹرْتِنٝوࢠࣹ ࢠࣹنْ.

(195) إࣹ طࣹمْنگَلْ آرَبُ ࢠَنْنگُنگَلْ.

(196) پࣹلّࣹتْ، كَيْرࣹ هِندࣹ كَ طࣹرِ أَرَنْࢠࣹ ࢠࣹنْ.

(197) إࣹنࣹي، هَرَ لَاتَنَاكِ ࢠࣹ مَاندࣹ، ندࣹ تَوَتَ أَنْندِينٝ ندࣹ ࢠِطْࢠࣹ إسرائيل أَنْندُࢠࣹ ࢠࣹنْ?

(198) سِنْندٝ مࣹنْ جِپِّنّٝ ندࣹ إࣹ دٝوْ يٝگَ إࣹ أَجَمِيَنْكٝوࢠࣹ ࢠࣹنْ.

(199) أٝ جَنْنگِ ندࣹ إࣹ دٝوْ مَࢠّࣹ، ࢠࣹ لَاتٝتَاكٝ گٝمْطِنْࢠࣹ ندࣹ.

(200) كٝ وَنٝ نِي مࣹنْ نَادِرِ طُمْ إࣹ ࢠࣹرْطࣹ بٝنْࢠࣹ ࢠࣹنْ.

(201) ࢠࣹ گٝمْطِنْتَا ندࣹ، هَا ࢠࣹ يِئَ لࣹپْتࣹ مُوسُطࣹ طࣹنْ.

(202) طࣹ أَرَ إࣹ مَࢠّࣹ إࣹ جُهَيࣹي، هَرَ ࢠࣹ سٝؤَا.

(203) سَكّٝ ࢠࣹ وِئَ: "تَوْ سِ كٝ مࣹنْ نࣹنَّنْتࣹيࢠࣹ?"

(204) إࣹنࣹي، كٝ لࣹپْتࣹ أَمࣹنْ طࣹنْ ࢠࣹ هࣹݧٝرْتَ?

(205) أَ تِنِي سِ مࣹنْ دَكْمِتِي ࢠࣹ دُوࢠِ.

(206) رࣹفْتِ أَرِ إࣹ مَࢠّࣹ كٝنْ كٝ ࢠࣹ لَاتِنٝو هِࢠࣹ فٝدࣹي.

(207) نَفَالِ ࢠࣹ كٝ ࢠࣹ لَاتِنٝو كٝنْ هِࢠࣹ دَكْمِتٝو.

(208) مࣹنْ هَلْكَالِ سَارࣹ وٝو، سِ وَنَا هَرَ نٝ وٝودَنِ ندࣹ جࣹرْتِنٝوࢠࣹ.

(209) طُمْ كٝ وَاجُ، مࣹنْ لَاتَاكِ تٝوݧٝوࢠࣹ.

(210) وَنَا سࣹيْتَنُوجِ طِنْ جِپِّنِ ندࣹ.

(211) طُمْ هَانَنَاطِ، طِ هَتَّنْتَاكࣹ كَدِ.

(212) پࣹلّࣹتْ، كَنْنجِ كٝ طِ فٝلّتَاطِ گَايِ هࣹطَگٝلْ نگٝلْ.

(213) وَتَ أَ نٝدُّ رࣹوࣹتࣹيطٝ گٝو وٝنْندُدࣹ إࣹ الله، سَكّٝ لَاتٝطَا جࣹيَاطٝ إࣹ لࣹپْتࣹتࣹيࢠࣹ ࢠࣹنْ.

(214) جࣹرْتِنْ گٝرٝلْ مَاطَ ࢠُرْࢠࣹ ࢠَدَادࣹ ࢠࣹنْ.

(215) لࣹسْطِنَنْ بَلْبࣹ مَاطࣹنْ، وٝنَنْندࣹ ࢠࣹنْ جٝكُّࢠࣹ مَ إِمّٝرْدࣹ إࣹ گٝمْطِنْࢠࣹ ࢠࣹنْ.

(216) سِ ࢠࣹ يࣹدِّي مَ، هَرَيْ مَاكُ: مِينْ" دࣹيْ، كٝ مِي دَنْندِيطٝ كٝنْ كٝ گٝلّٝتٝنْ".

(217) هٝولٝرْ أٝن تࣹدُّطٝ يُرْمࣹتࣹيطٝ.

(218) أٝن سُتِيطٝ مَ فࣹوْندٝ دَرٝتٝطَا.

(219) إࣹ وَيْلِتٝگٝلْ مَاطَ إࣹ هَكُّندࣹ سُجّٝوࢠࣹ ࢠࣹنْ.

(220) پࣹلّࣹتْ، كٝ كَنْكٝ وٝنِ نَنٝووٝ أَنْندُطٝ أٝنْ.

(221) تَوْ سِ مِي هُمْپِتَيْ أٝن كٝ إࣹ هٝمْنبٝ سࣹيْتَنُوجِ طِنْ جِپّٝتٝو?

(222) هِطِ جِپّٝو إࣹ دٝوْ كَلَ فࣹفِندٝتٝوطٝ بَكّٝدِنْطٝ.

(223) هِطِ وࣹرْلٝو نَنْطࣹ طࣹنْ، كٝ ࢠُرْࢠࣹ هࣹيوُدࣹ إࣹ مَجِّ كٝ فࣹنٝوجِ.

(224) يِمٝوࢠࣹ ࢠࣹنْ كٝ جٝكَّتَ ࢠࣹ، كٝ بࣹوْࢠࣹ ࢠࣹنْ.

(225) إࣹنࣹي، أَ تِنَالِ وٝنْندࣹمَ كَمْࢠࣹ هِࢠࣹ يِنٝو إࣹ كَلَ أَيْندࣹ?

(226) إࣹ تَوْدࣹ كَمْࢠࣹ هِࢠࣹ وٝوْلَ كٝ ࢠࣹ هُووَتَا.

(227) سِ وَنَا ࢠࣹنْ گٝمْطِنْࢠࣹ ࢠࣹ گٝلِّ گٝلّࣹ مٝڃّࣹ ࢠࣹ جَݝْتِي الله كٝ طُوطِ ࢠࣹ يٝࢠْتِي ࢠَاوٝ ࢠࣹ تٝوݧَامَ. أَرَيْ كَ ࢠࣹنْ تٝوݧُࢠࣹ أَنْندَيْ كٝ هٝندࣹ إࣹ وَيْلِتٝرْدࣹ ࢠࣹ وَيْلِتٝتٝو.

27 An-Naml — 93 آيَة

(1) تَا سِينْ، طࣹنْ كٝ آيࣹيجࣹ القرآن ندࣹنْ إࣹ دࣹفْتࣹرࣹ ࢠَنْنگُندࣹ.

(2) كٝ ندࣹ پࣹيوَلْ إࣹ وࣹلٝ-وࣹلٝ وٝنَنْندࣹ گٝمْطِنْࢠࣹ ࢠࣹنْ.

(3) ࢠࣹنْ ࢠࣹ نٝ ݧِيࢠَ إࣹ جُلْدࣹ ندࣹنْ ࢠࣹ تٝتَّ جَكَّ أٝنْ، هَرَ كَمْࢠࣹ هِࢠࣹ يَنَنَا فِي لَاكَرَ أٝنْ.

(4) پࣹلّࣹتْ، ࢠࣹنْ ࢠࣹ گٝمْطِنَالِ لَاكَرَ أٝنْ، مࣹنْ ݖُطَنِي ࢠࣹ گٝلّࣹ مَࢠّࣹ طࣹنْ، كَمْࢠࣹ هِࢠࣹ يِنٝو.

(5) كٝ ࢠࣹنْ وٝنِ ࢠࣹنْ، ࢠࣹ وٝودَنِ طُمْ لࣹپْتࣹ بٝنْطࣹ طࣹنْ، كَمْࢠࣹ كَ لَاكَرَ كٝ كَمْࢠࣹ وٝنِ هَيْرُࢠࣹ ࢠࣹنْ.

(6) إࣹ پࣹلّࣹتْ، أَن هِطَ لٝنْنگِنࣹي القرآن ندࣹنْ إِمّٝرْدࣹ كَ ݧࣹيݧٝ أَنْندُطٝ.

(7) تُمَ ندࣹ موسى مَاكَنّٝو ࢠࣹيْنگُورࣹ مَكّٝ ندࣹنْ: مِينْ" دࣹيْ، مِي هَيْنِكࣹ يِيتࣹ، أَرَيْ مِي أَدَّنَ أٝن كَ نگࣹ وٝنِ طٝنْ كُمْپِتَلْ مَاطُمْ مِي أَدَّنَ أٝن دٝݖٝتَلْ هُࢠُّنگَلْ، بࣹلَ جٝؤٝ أٝنٝنْ أٝن إِوْلٝتٝ.

(8) تُمَ ندࣹ أٝ أَرُنٝو إࣹ مَگّࣹ، أٝ نٝدَّا: طُمْ وٝنِ بَرْكِنَامَ أٝن وٝنْطٝ كَ يِيتࣹ إࣹ ࢠࣹنْ وٝنُࢠࣹ بَنْݝࣹ مَگّࣹ، إࣹ سࣹنَيࣹي وٝنَنِي الله جٝومَ وِنْندࣹرࣹ ندࣹنْ.

(9) كٝ أَن يٝ موسى، كٝ وٝنِ، كٝ مِينْ وٝنِ الله تࣹدُّطٝ ݧࣹيݧُطٝ أٝنْ.

(10) لِࢠُ تُگّٝرْدُ مَاطَ ندُنْ، تُمَ ندࣹ أٝ يِئُنٝو ندُ هِندُ دِلَّ، أَ وِئَيْمَ كَيْرُ كٝ ندُ نبٝدِّ، أٝ هُݖِّتِ كٝ أٝ دٝگُطٝ أٝ يِلْتِتَاكࣹ. كٝ أَن يٝ موسى: "وَتَ أَ هُلُ، مِينْ دࣹيْ نُلَاࢠࣹ ࢠࣹنْ هُلَتَا كَ أَمْ".

(11) سِ وَنَا أٝن تٝوݧُطٝ، رࣹفْتِ أٝ لٝنْتِنِ كٝ مٝڃِّ ࢠَاوٝ بٝنࣹ طࣹنْ. پࣹلّࣹتْ، مِينْ كٝ مِي سُرّٝووٝ يُرْمࣹتࣹيطٝ.

(12) نَادُ جُنْنگٝ مَاطَ نگٝنْ كَ جِيبَ مَاطَ، نگٝ يَلْتَيْ كٝ نگٝ رَنࣹوٝ كٝ أَلْدَا إࣹ بٝنࣹ، طُمْ نٝ جࣹيَا إࣹ ندࣹرْ آيࣹيجࣹ جࣹينَيْ نُلَاطࣹ إࣹ فرعون إࣹ يِمْࢠࣹ مَكّٝ ࢠࣹنْ، پࣹلّࣹتْ، كَمْࢠࣹ ࢠࣹ لَاتِكࣹ يِمْࢠࣹ فَاسِقِيࢠࣹ.

(13) تُمَ ندࣹ أَرُنٝو إࣹ مَࢠّࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ كٝ يِئࣹتࣹيطࣹ، ࢠࣹ وِئِ: "طُؤُمْ، كٝ نبِلࣹوُ ࢠَنْنگُنگُ".

(14) ࢠࣹ يࣹدِّ آيࣹيجࣹ طࣹنْ ، تَوِ وٝنْكِيجِ مَࢠّࣹ طِنْ نٝ يَنَنَا فِي مَجّࣹ، تٝوݧࣹ إࣹ مَوْنِنْتِنَارࣹ. ندَارُ كٝ هٝنّٝ لَاتٝرْنٝو بَتَّنࣹ فࣹنُّࢠࣹ ࢠࣹنْ.

(15) گٝمْطِي مࣹنْ أٝكِّي دَاوُودَ إࣹ سُلَيْمَانَ گَنْندَلْ ࢠࣹ مَاكِ: "يࣹتّٝورࣹ ندࣹنْ وٝودَنِي الله أٝن ࢠُرْنُطٝ مࣹنْ إࣹ دٝوْ طُوطُࢠࣹ إِمّٝرْدࣹ إࣹ جࣹيَاࢠࣹ مَكّٝ گٝمْطِنْࢠࣹ ࢠࣹنْ".

(16) سُلَيْمَانَ رٝنِ دَاوُودَ، أٝ مَاكِ: "كٝ أٝنٝنْ يٝ يِمْࢠࣹ، مࣹنْ أَنْندِنَامَ هَالَ ݖٝلِّ طِنْ، مࣹنْ أٝكَّامَ إِمّٝرْدࣹ إࣹ كَلَ هُوندࣹ، پࣹلّࣹتْ، طُمْطٝو كٝ طُمْ وٝنِ ࢠُرَلْ ࢠَنْنگُنگَلْ".

(17) سُلَيْمَانَ مٝوࢠَنَا كٝنࣹيلِ مَكّٝ طِنْ إِمّٝرْدࣹ كَ جِنَّاجِ إࣹ كَ يِمْࢠࣹ إࣹ كَ ݖٝلِّ، كَمْࢠࣹ كٝ ࢠࣹ يُࢠِّنَاࢠࣹ.

(18) هَا ندࣹ ࢠࣹ أَرُنٝو إࣹ أَيْندࣹ كٝرْندٝلِّ طِنْ، كٝرْندٝلِلْ وِئِ: "كٝ أٝنٝنْ يٝ كٝرْندٝلِّ، نَاتࣹي كَ كٝطَنْطِ مٝوطٝنْ، فِي وَتَ سُلَيْمَانَ إࣹ كٝنࣹيلِ مُؤُنْ طِنْ مُنْݖُ أٝنْ، هَرَ كَمْࢠࣹ ࢠࣹ سٝؤَا".

(19) سُلَيْمَانَ مٝوسِ كٝ أٝ جَلُطٝ إِمّٝرْدࣹ إࣹ كٝنْنگٝلْ مَگِّ نگٝلْ، أٝ مَاكِ: "جٝومَ، لٝنْنگِنَمْ نٝ مِي يࣹتِّرَ نࣹيمَ مَاطَ مٝ نࣹيمِنْطَا إࣹ دٝوْ أَم أٝنْ، إࣹ جِبِبِنْࢠࣹ لَمْ ࢠࣹنْ، إࣹ ندࣹ مِي گٝلَّتَ كٝ مٝڃِّ وࣹلْتٝرَا طُمْ، نَادَامِ كَ يُرْمࣹيندࣹ مَاطَ وٝنْندُدࣹ إࣹ جࣹيَاࢠࣹ مَاطَ مٝڃُّࢠࣹ ࢠࣹنْ".

(20) أٝ ڃࣹوْتِي ݖٝلِّ طِنْ، أٝ مَاكِ: "كٝ هࣹࢠِ لَمْ، مِي يِئَالِ هُدْ-هُدَ. كَا ندُ لَاتِ جࣹيَاندُ إࣹ وِرْنِيطِ طِنْ?"

(21) مَا مِي لࣹپْتَ ندُ سَنّࣹ لࣹپْتࣹ سَطْتُطࣹ مَاطُمْ مَا مِي هِرْسَندُ سَنّࣹ، مَاطُمْ ندُ أَدَّنَ لَمْ هُجَّ ࢠَنْنگُطٝ.

(22) ندُ ݧِيࢠࣹ كٝ وٝطَّا، ندُ وِئِ: مِي" هُࢠِندِكࣹ كٝ أَ هُࢠِندَاكِ طُمْ، مِي أَدَّنِي مَ إِمّٝرْدࣹ تٝ سَبَئِ كُمْپِتَلْ يَنَنَانگَلْ".

(23) مِينْ دࣹيْ، مِي تَوِي سٝنَّاجٝ نٝ لَامِي ࢠࣹ هِمٝ أٝكَّا إِمّٝرْدࣹ إࣹ كَلَ هُوندࣹ، نٝ وٝودَمِ مٝ كَدِ جُلّࣹرࣹ لَامُ مَوْندࣹ.

(24) مِي تَوِي مٝ كَنْكٝ إࣹ يِمْࢠࣹ مَكّٝ ࢠࣹنْ هِࢠࣹ سُجَّنَ نَانگࣹ نگࣹنْ گَانِنْ الله; سࣹيْتَانࣹ نٝ ݖُطَنِࢠࣹ گٝلّࣹ مَࢠّࣹ طࣹنْ أٝ فَدِّي ࢠࣹ گَايِ لَاوٝلْ نگٝلْ، كَمْࢠࣹ ࢠࣹ فࣹيوَتَا.

(25) كٝ هَطَتَ ࢠࣹ سُجَّنَ الله أٝن يَلْتِنٝووٝ كٝ سُوطِي كٝنْ كَ كَمُّولِ إࣹ كَ لࣹيْدِ هِمٝ أَنْندِ كٝ سُوطٝتٝنْ إࣹ كٝ فࣹݧِنْتٝنْ.

(26) الله، رࣹوࣹتࣹيطٝ أَلَا سِ وَنَا كَنْكٝ جٝومَ أَرْسِ تࣹدُّكِ كِنْ.

(27) أٝ مَاكِ: "َرَيْ مࣹنْ ندَارَ سِ أَ گٝونگُ، مَاطُمْ سِ لَاتِ جࣹيَاطٝ إࣹ فࣹنُࢠࣹ ࢠࣹنْ".

(28) يَهِدُ إࣹ ندُو تَلْكُرُ أَم وࣹرْلٝطَا ندُ كَ مَࢠّࣹ، رࣹفْتِ هُݖَّا ࢠࣹ ندَارَا كٝ هٝنْطُمْ ࢠࣹ رُتَّتَ.

(29) سٝنَّاجٝ أٝن وِئِ: "كٝ أٝنٝنْ يٝ نبَتُ، مِينْ دࣹيْ وࣹرْلَامَ كَ أَم تَلْكُرُ تࣹدُّندُ".

(30) پࣹلّࣹتْ، ندُو كٝ إِمّٝرْدࣹ إࣹ سُلَيْمَانَ، پࣹلّࣹتْ، كٝ وٝنِ إࣹ مَيْرُ كٝ: "ِنْندࣹ الله هُوࢠُطٝ يُرْمࣹيندࣹ هࣹيرِيطٝ يُرْمࣹيندࣹ أٝنْ".

(31) وَتَ أٝن تٝوْنِتَنٝ لَمْ، أَرࣹي إࣹ أَم كٝ جࣹبِّلِيࢠࣹ.

(32) أٝ وِئِ: "كٝ أٝنٝنْ يٝ نبَتُ، وَطْتࣹي لَمْ إࣹ فࣹيرࣹ إࣹ طࣹي فِيَاكࣹ أَمْ، مِي لَاتَاكِ تَڃٝووٝ فِيَاكࣹ هَا تَوْدࣹطٝنْ إࣹ أَمْ".

(33) ࢠࣹ وِئِ: "ࣹنࣹنْ كٝ إࣹن جٝومَ دٝولࣹ إࣹ جٝومَ هَرࣹ سَطْتُطࣹ، فِيَاكࣹ طࣹنْ كٝ كَ مَاطَ وٝنِ، ندَارُ كٝ هٝنْطُمْ يَمِرْتَا".

(34) دࣹبّٝ أٝن وِئِ: "پࣹلّࣹتْ، لَمْࢠࣹ ࢠࣹنْ سِ ࢠࣹ نَاتِي إࣹ سَارࣹ ࢠࣹ بٝنَّيْ ندࣹ، ࢠࣹ وَطَ تࣹدُّࢠࣹ مَيْرࣹ ࢠࣹنْ هٝيْࢠࣹ، كٝ وَنٝنِي ࢠࣹ گٝلِّرْتَ.

(35) أَوَ مِينْ، مِي نُلَيْ إࣹ مَࢠّࣹ نࣹلْدَارِ، مِي ندَارَ كٝ هٝنْطُمْ نُلَاࢠࣹ ࢠࣹنْ رُتّٝدَتَ.

(36) تُمَ ندࣹ ࢠࣹ أَرُنٝو كَ سُلَيْمَانَ، أٝ مَاكِ: "ࣹنࣹي، أٝن ࢠࣹيْدِرَيْ لَمْ جَوْدِ، كٝنْ كٝ الله أٝكِّنْ مِي، نٝ ࢠُرِ مٝڃُّدࣹ دِينِ كٝنْ كٝ أٝ أٝكِّ أٝنْ، كٝ وٝنِ، أٝنٝنْ هِطٝنْ فَرْنٝرَ نࣹلْدَارِ مٝوطٝنْ ندِنْ".

(37) رُتّٝ كَ مَࢠّࣹ، مَمࣹنْ أَدَّنَ ࢠࣹ كٝنࣹيلِ طِ ࢠࣹ مَرَا فࣹيرࣹ مُوطُمْ، مَمࣹنْ يَلْتِنْ ࢠࣹ تٝنْ كٝ ࢠࣹ جَاسُࢠࣹ هَرَ كَمْࢠࣹ كٝ هٝيْࢠࣹ.

(38) أٝ مَاكِ: "كٝ أٝنٝنْ يٝ نبَتُ، كٝ هٝمْنبٝ إࣹ مٝوطٝنْ أَدَّنْتَ لَمْ جُلّࣹرࣹ لَامُ مَكّٝ ندࣹنْ، كٝ أَدِي ࢠࣹ أَرُدࣹ كَ أَم كٝ ࢠࣹ جࣹبِّلِيࢠࣹ?"

(39) نگَارِ جِنَّ ندِنْ وِئِ: مِينْ"، مِي أَدَّنْتࣹ ندࣹ كٝ أَدِي أَ إِمَّادࣹ إࣹ دَرْندࣹ مَاطَ ندࣹنْ، إࣹ پࣹلّࣹتْ، مِينْ إࣹ دٝوْ طُمْ كٝ مِي دٝلْنُطٝ هٝولَاطٝ".

(40) أٝن مٝ وٝنِ كَ مُوطُمْ گَنْندَلْ إِمّٝرْدࣹ كَ دࣹفْتࣹرࣹ وِئِ: مِينْ" مِي أَنْندَنْتࣹ ندࣹ كٝ أَدِي أَ ندࣹ ندَرْطࣹ مَاطَ طࣹنْ رُتّٝتٝو إࣹ مَاطَ". تُمَ ندࣹ أٝ يِئُنٝو ندࣹ كٝ ندࣹ ݧِيࢠُندࣹ كَ مَكّٝ، أٝ مَاكِ: "طُمْ طٝو، كٝ إِمّٝرْدࣹ إࣹ ࢠُرَلْ جٝومِ أَمْ، فِي يٝ أٝ ندَرْندٝ لَمْ سِ مِي جَرْنَيْ مٝ مَاطُمْ سِ مِي يࣹدَّيْ مٝ ، كَلَ جَرْنُطٝ أَنْندُ كٝ أٝ جَرْنِ كٝنْ وٝنَنَيْ وٝنْكِي مَكّٝ كِنْ، كَلَ يࣹدُّطٝ كَدِ پࣹلّࣹتْ، جٝومِ أَم كٝ أٝ گَلٝ تࣹدُّطٝ".

(41) أٝ مَاكِ: "فَيْتِنَنࣹي مٝ جُلّࣹرࣹ لَامُ مَكّٝ ندࣹنْ، ندَارࣹنْ سِ أٝن أَنْندِتَيْ، كَا أٝ لَاتٝتٝ جࣹيَاطٝ إࣹ ࢠࣹنْ ࢠࣹ أَنْندِتَتَا".

(42) تُمَ ندࣹ أٝ أَرُنٝو، أٝ وِئَا: "ࣹنࣹي، كٝ نِي جُلّࣹرࣹ مَاطَ ندࣹنْ وَئِ?" أٝ وِئِ: "َوِئَيْمَ كٝ كَيْرࣹ تِگِ نِي". مࣹنْ أٝكَّامَ گَنْندَلْ كٝ أَدِي مٝ، مࣹنْ لَاتِي جࣹبِّلِيࢠࣹ.

(43) فَدِّي مٝ، طُمْ كٝ أٝ لَاتِنٝو هِمٝ رࣹوَ گَانِنْ الله، پࣹلّࣹتْ، كَنْكٝ أٝ لَاتِنٝكࣹ جࣹيَاطٝ إࣹ يِمْࢠࣹ يࣹدُّࢠࣹ.

(44) وِئَنَا مٝ: "نَاتُ كَ هُࢠࣹيرࣹ". تُمَ ندࣹ أٝ يِئُنٝو ندࣹ أٝ سِكِّ كٝ ندࣹ دُنࣹيرࣹ(تࣹوْرُ)، أٝ ݧٝرِ دࣹنْسࣹ (كٝرْلࣹ)مَكّٝ طࣹنْ. سُلَيْمَانَ مَاكِ: "پࣹلّࣹتْ، ندࣹي كٝ هُࢠࣹيرࣹ نࣹوْطِنِرَاندࣹ ندَارٝرْطࣹ". دࣹبّٝ أٝن وِئِ: "جٝومَ، مِينْ دࣹيْ مِي تٝوݧِي وٝنْكِي أَمْ، مِي جࣹبِّلَنِكࣹ الله جٝومَ وِنْندِرࣹ ندࣹنْ، وٝنْندُدࣹ إࣹ سُلَيْمَانَ".

(45) گٝمْطِي مࣹنْ نُلِي فَادࣹ إࣹ سَمُودِنْكٝوࢠࣹ ࢠࣹنْ مُسِطّٝ مَࢠّࣹ أٝن سَالِهُ، طُمْ وٝنِ رࣹوࣹي الله. تَوِ كَمْࢠࣹ كٝ ࢠࣹ فࣹدّࣹيجِ طِطِ هِࢠࣹ وࣹنْنجٝندِرَ.

(46) أٝ مَاكِ: "كٝ أٝنٝنْ يٝ يِمْࢠࣹ أَمْ، كٝ فِي هٝنْطُنْ هࣹݧٝرْنْتٝنْ بٝنࣹ طࣹنْ كٝ أَدِي مٝڃّࣹرࣹ ندࣹنْ?" "كٝ هَطِ إِنْسِنَنِنٝ الله بࣹلَ جٝؤٝ أٝنٝنْ أٝن يُرْمِنٝيْتࣹ".

(47) ࢠࣹ وِئِ: مࣹنْ" كِيتِنٝرِي مَ إࣹ ࢠࣹنْ وٝنْندُࢠࣹ إࣹ مَاطَ". أٝ مَاكِ: "كِيتࣹ مٝوطٝنْ أٝن كٝ كَ الله وٝنِ، كٝ وٝنِ، أٝنٝنْ كٝ أٝن يِمْࢠࣹ فِتْنࣹتࣹيࢠࣹ"

(48) لَاتِنٝكࣹ كَ سَارࣹ تٝنْ وٝرْࢠࣹ نجࣹينَيٝ هِࢠࣹ بٝنَّ كَ لࣹيْدِ ࢠࣹ مٝڃِّنْتَا.

(49) ࢠࣹ وِئِ: ࢠࣹ وٝوندِرِي الله، مَ مࣹنْ وَالَنَنْ مٝ كَ سُودُ كَنْكٝ إࣹ ࢠࣹيْنگُورࣹ مَكّٝ ندࣹنْ رࣹفْتِ مَ مࣹنْ وِئَنْ جࣹيْࢠࣹ مٝ ࢠࣹنْ; مࣹنْ تَوَاكَ وَرُگٝلْ يِمْࢠࣹ مَكّٝ ࢠࣹنْ، إࣹ پࣹلّࣹتْ، مࣹنࣹنْ كٝ مࣹنْ گٝونگُࢠࣹ".

(50) ࢠࣹ هٝدِ هٝدٝورࣹ مࣹنࣹنْ كَدِ مࣹنْ فࣹوْجِ فࣹوْجٝورࣹ، تَوِ كَمْࢠࣹ ࢠࣹ سٝؤَتَا.

(51) ندَارُ، كٝ هٝنّٝ لَاتٝرْنٝ بَتَّنࣹ هٝدٝ مَࢠّࣹ نگٝنْ، مࣹنࣹنْ مࣹنْ يِرْبِنّٝ ࢠࣹ كَمْࢠࣹ إࣹ يِمْࢠࣹ مَࢠّࣹ ࢠࣹنْ ࢠࣹ فٝوْ مَࢠّࣹ.

(52) طِنْ ݖُوطِ مَࢠّࣹ كٝ يࣹيوُطِ سَبُ كٝ ࢠࣹ تٝوݧُنٝو كٝنْ. پࣹلّࣹتْ، وٝنِي إࣹ طُمْ مَاندࣹ وٝنَنْندࣹ يِمْࢠࣹ هَقِّلٝوࢠࣹ ࢠࣹنْ.

(53) مࣹنْ دَنْندِ ࢠࣹنْ گٝمْطِمْࢠࣹ، ࢠࣹ وٝنِ هُلٝوࢠࣹ الله .

(54) إࣹ لوط، تُمَ ندࣹ أٝ وِئَنّٝ يِمْࢠࣹ مَكّٝ ࢠࣹنْ: إࣹنࣹي، أٝن أَرَيْ پَنْكَرࣹ طࣹنْ هَرَ هِطٝنْ سُتِي?"

(55) إࣹيٝو، أٝنٝنْ أٝن أَرَيْ إࣹ وٝرْࢠࣹ ࢠࣹنْ كَ تُويٝ أَݖّٝنْ رࣹوْࢠࣹ ࢠࣹنْ?! سِ كٝ وٝنِ، أٝنٝنْ كٝ أٝن يِمْࢠࣹ مَجَّاࢠࣹ.

(56) جَابَوٝلْ يِمْࢠࣹ مَكّٝ ࢠࣹنْ وٝنَانٝ سِ وَنَا كٝ ࢠࣹ وِئِ : "يَلْتِنࣹي يِمْࢠࣹ لوط ࢠࣹنْ كَ هٝطٝ (وُرَ) مٝوطٝنْ ; كَمْࢠࣹ كٝ ࢠࣹ يِمْࢠࣹ لَࢠِّنْتِنْتٝوࢠࣹ".

(57) مࣹنْ دَطْندِ مٝ كَنْكٝ إࣹ يِمْࢠࣹ مَكّٝ ࢠࣹنْ، سِ وَنَا دࣹبّٝ مَكّٝ أٝنْ; أٝن طٝنْ مࣹنْ هٝدِّرَنِي مٝ كٝ أٝ جࣹيَاطٝ إࣹ هَلْكࣹتَاࢠࣹ ࢠࣹنْ.

(58) مࣹنْ تٝࢠِنِ إࣹ دٝوْ مَࢠّࣹ تٝࢠٝ، بٝنِي تٝࢠٝ جࣹرْتِنَاࢠࣹ ࢠࣹنْ.

(59) مَاكُ: يࣹتّٝورࣹ ندࣹنْ كٝ الله وٝودَنِ، كِسِيࣹي وٝنِي إࣹ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ ࢠࣹ أٝ سُࢠِي. إࣹنࣹي، كٝ الله ࢠُرِ مٝڃُّدࣹ، كَا كٝ طِنْ طِࢠࣹ سِرْكَنْتَ?

(60) كَا كٝ هٝمْنبٝ تَگِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ أٝ جِپِّنَنِ أٝن إِمّٝرْدࣹ كَ كَمُّ ندِيَمْ? مࣹنْ فُطِنِرِ طَمْ گࣹسࣹ جٝومِرَاجࣹ ݝَرِ. وٝنَنْتَا أٝن ندࣹ فُطِنٝنْ لࣹطّࣹ مَجّࣹ. إࣹنࣹي، هَرَ رࣹوࣹتࣹيطٝ نٝ وٝودِ وٝنْندُدࣹ إࣹ الله? كٝ وٝنِ، كَمْࢠࣹ كٝ ࢠࣹ يِمْࢠࣹ أٝوݧٝتٝوࢠࣹ.

(61) كَا كٝ هٝمْنبٝ وَطِ لࣹيْدِ ندِنْ ݧِيࢠُندِ، أٝ وَطِ هَكُّندࣹ مَيْرِ ݖَلُّطِ أٝ وَطَنِ ندِ پࣹلّࣹ أٝ وَطِ هَكُّندࣹ مَاجࣹ طࣹنْ طِطِ هࣹيدٝ? إࣹنࣹي، هَرَ نَ وٝودِ رࣹوࣹتࣹيطٝ گٝو وٝنْندُدࣹ إࣹ الله? كٝ وٝنِ، ࢠُرْࢠࣹ طُوطُدࣹ ࢠࣹنْ إࣹ مَࢠّࣹ ࢠࣹ أَنْندَا.

(62) كَا كٝ هٝمْنبٝ جَابَتٝ ࢠِتِّيطٝ أٝن سِ أٝ نٝدِّ مٝ أٝ هُنْݖَ بٝنࣹ طࣹنْ أٝ وَطَ أٝن لٝنْتِيࢠࣹ كَ لࣹيْدِ? نَ وٝودِ كٝرࣹوࣹتࣹيطٝ گٝو وٝنْندُدࣹ إࣹ الله? فَنْطِي كٝ وَاجِتٝطٝنْ.

(63) كَا كٝ هٝمْنبٝ فࣹوْنَتَ أٝن كَ نِࢠࣹ نجٝورْندِ إࣹ كَ مَايٝ، إࣹ كٝ هٝمْنبٝ أَݖِّتَتَ كࣹنِ طِنْ وࣹوْلِنِرَيْطِ فِي يُرْمࣹيندࣹ مَكّٝ ندࣹنْ? إࣹنࣹي، كٝ رࣹوࣹتࣹيطٝ گٝو وٝنْندُدࣹ إࣹ الله? الله تٝووِي گَايِ كٝ ࢠࣹ سِرْكَتَ كٝنْ.

(64) كَا كٝ هٝمْنبٝ وٝنِ فُطُّطٝ تَگُ نگُنْ رࣹفْتِ أٝ رُتَّ نگُ، إࣹ كٝ هٝمْنبٝ أَرْسِكَتَ أٝن إِمّٝرْدࣹ كَ كَمُّ إࣹ كَ لࣹيْدِ? إࣹنࣹي، كٝ رࣹوࣹتࣹيطٝ گٝو وٝنْندُدࣹ إࣹ الله? مَاكُ: "َدّࣹي دَلِيلِ مٝوطٝنْ أٝن سِ تَوِي كٝ أٝن گٝونگُࢠࣹ".

(65) مَاكُ: "َنْندَا كٝ وِرْنِي كٝنْ ࢠࣹنْ وٝنُࢠࣹ كَ كَمُّولِ إࣹ كَ لࣹيْدِ سِ وَنَا الله. ࢠࣹ سٝؤَتَا كَدِ كٝ هٝنْندࣹ تُمَ ࢠࣹ إِمِّنْتِنٝيْتࣹي".

(66) كٝ وٝنِ، گَنْندَلْ مَࢠّࣹ نگَلْ هَتِّي فِي لَاكَرَ أٝنْ، كٝ وٝنِ، كَمْࢠࣹ هِࢠࣹ إࣹ سِكّࣹ فِي لَاكَرَ ، كٝ وٝنِ، كَمْࢠࣹ كٝ ࢠࣹ وُمُࢠࣹ فِي طُمْ.

(67) ࢠࣹنْ يࣹدُّࢠࣹ وِئِ: " إࣹنࣹي، هَيْ سِ مࣹنْ لَاتِكࣹ نبُلَّارِ إࣹ بَابِرَاࢠࣹ أَمࣹنْ ࢠࣹنْ كَدِ، إࣹ هَرَ مࣹنࣹنْ كٝ مࣹنْ يَلْتِنٝيْتࣹيࢠࣹ?"

(68) گٝمْطِي مࣹنْ فٝدَنٝومَ طُمْ مࣹنࣹنْ إࣹ بَابِرَاࢠࣹ أَمࣹنْ كٝ أَدِي، طُمْ وٝنَالِ سِ وَنَا تِنْندِ أَدِنٝوࢠࣹ.

(69) مَاكُ : "يࣹهࣹي كَ لࣹيْدِ ندَارٝنْ كٝ هٝنّٝ بَتَّنࣹ بٝنْࢠࣹ ࢠࣹنْ لَاتٝرْنٝو".

(70) وَتَ أَ سُنٝ فِي مَࢠّࣹ، وَتَ أَ لَاتٝ إࣹ ࢠِتّࣹيندࣹ إِمّٝرْدࣹ إࣹ كٝ ࢠࣹ فࣹوْجَتَ كٝنْ.

(71) هِࢠࣹ وِئَ: "كٝ هٝنْندࣹ تُمَ وٝنِ ندࣹي فٝدٝورࣹ، سِ أٝن لَاتِكࣹ گٝونگُࢠࣹ?"

(72) مَاكُ: "نٝ تِجَّا ندࣹ تَوَتَ ࢠَدَنِكࣹ أٝن يٝگَ إࣹ كٝنْ كٝ طَࢠُّطٝنْ يَݖِّنَنࣹيدࣹ".

(73) پࣹلّࣹتْ، جٝومِ مَاطَ كٝ أٝ جٝمْ ࢠُرَلْ إࣹ دٝوْ يِمْࢠࣹ ࢠࣹنْ، كٝنٝ ࢠُرْࢠࣹ طُوطُدࣹ ࢠࣹنْ إࣹ مَࢠّࣹ ࢠࣹ جَرْنَتَا الله .

(74) پࣹلّࣹتْ، جٝومِ مَاطَ هِمٝ أَنْندِ كٝ ࢠࣹرْطࣹ مَࢠّࣹ طࣹنْ سٝومِ إࣹ كٝ طࣹ فࣹݧِّنْتَ كٝنْ.

(75) أَلَا إࣹ وِرْنِيكُنْ كَ كَمُّ إࣹ كَ لࣹيْدِ سِ وَنَا فِي مَكُّنْ نٝ إࣹ ندࣹرْ دࣹفْتࣹرࣹ ࢠَنْنگُندࣹ.

(76) پࣹلّࣹتْ ندࣹي القرآن، هِندࣹ فِلّٝو إࣹ دٝوْ ࢠِطْࢠࣹ إسرائيل ࢠࣹنْ فِي هُوندࣹ ندࣹ ࢠࣹ ࢠُرِ لُوتٝندِرْدࣹ ندࣹنْ إࣹ فِي مُوطُمْ.

(77) پࣹلّࣹتْ، كَيْرࣹ كٝ ندࣹ پࣹيوَلْ إࣹ يُرْمࣹيندࣹ وٝنَنْندࣹ گٝمْطِنْࢠࣹ ࢠࣹنْ.

(78) پࣹلّࣹتْ، جٝومِ مَاطَ أٝ ݧَاوِرَيْ هَكُّندࣹ مَࢠّࣹ ݧَاوٝورࣹ مَكّٝ نگَلْ، كٝ كَنْكٝ وٝنِ تࣹدُّطٝ أَنْندُطٝ.

(79) وَكِّلٝ إࣹ الله، پࣹلّࣹتْ، أَن هِطَ إࣹ هٝورࣹ گٝونگَ ࢠَنْنگُكَ كَنْ.

(80) پࣹلّࣹتْ، أَن أَ نَنِنْتَا مَيْࢠࣹ ࢠࣹنْ، أَ نَنِنْتَا فَاطُࢠࣹ ࢠࣹنْ نٝدَّاندُ ندُنْ سِ ࢠࣹ طُورْنِكࣹ ࢠࣹ هُݖِّتِي.

(81) أَن أَ وٝنَالِ فࣹوْنٝووٝ بُمْࢠࣹ ࢠࣹنْ إࣹ گَايِ مَجّࣹرࣹ مَࢠّࣹ ندࣹنْ. أَ نَنِنْتَا سِ وَنَا ࢠࣹنْ گٝمْطِنْࢠࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ هَرَ كَمْࢠࣹ كٝ ࢠࣹ جࣹبِّلِيࢠࣹ.

(82) سِ كٝنْنگٝلْ نگٝلْ يَنِي إࣹ دٝوْ مَࢠّࣹ، مࣹنْ يَلْتِنَنَ ࢠࣹ دَابࣹوَ إِمّٝرْدࣹ كَ لࣹيْدِ هِنبَ هَالَنَ ࢠࣹ وٝنْندࣹ يِمْࢠࣹ ࢠࣹنْ لَاتِكࣹ ࢠࣹ يَنَنَاكَ آيࣹيجࣹ أَمࣹنْ طࣹنْ.

(83) إࣹ ݧَلْندࣹ مࣹنْ مٝوࢠٝيْتَ إِمّٝرْدࣹ إࣹ كَلَ مٝفْتࣹ دࣹنْتَلْ جࣹيَاࢠࣹ إࣹ ࢠࣹنْ فࣹنّٝوࢠࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ هَرَ كَمْࢠࣹ هِࢠࣹ سٝگّࣹي.

(84) هَا سِ ࢠࣹ أَرِي، أٝ دَالَ: "ࣹيٝو، أٝن فࣹنِّينٝ آيࣹيجࣹ أَم طࣹنْ، هَرَ أٝن هُࢠِندٝرَالِ طِ گَنْندَلْ. كَا كٝ هٝنْطُمْ وٝنْطٝنْ إࣹ گٝلُّدࣹ?"

(85) كٝنْنگٝلْ لࣹپْتࣹ نگٝلْ يَنِ إࣹ مَࢠّࣹ سَبُ كٝ ࢠࣹ تٝوݧِ كٝنْ، كَمْࢠࣹ ࢠࣹ وٝوْلَتَا.

(86) إࣹنࣹي، هَرَ ࢠࣹ يِئَالِ، مࣹنࣹنْ مࣹنْ وَطِي جࣹمَّ أٝن فِي يٝ ࢠࣹ دࣹيڃُ إࣹ مَاكٝ، إࣹ ݧَلٝرْمَ أٝن يِئِرْطٝ? پࣹلّࣹتْ، وٝنِي إࣹ طُمْ آيࣹيجࣹ وٝنَنْندࣹ يِمْࢠࣹ گٝمْطِنٝوࢠࣹ ࢠࣹنْ.

(87) إࣹ ݧَلْندࣹ وُتّٝيْتࣹي كَ أَلَادُ، طࣹنْڃٝو ࢠࣹنْ وٝنُࢠࣹ كَ كَمُّولِ إࣹ ࢠࣹنْ وٝنُࢠࣹ كَ لࣹيْدِ، سِ وَنَا ࢠࣹنْ ࢠࣹ الله مُويِ، گٝوتٝ كَلَ إࣹ مَࢠّࣹ أَرَيْ إࣹ مَكّٝ هَرَ كٝ ࢠࣹ هٝيْكِنِيࢠࣹ.

(88) هِطَ تِنَ پࣹلّࣹ طࣹنْ، سِكَّ كٝ طࣹ فࣹنْندِيطࣹ، هَرَ لࣹ كَنْنجࣹ هِطࣹ وُلٝو وُلَگٝلْ دُولࣹ طࣹنْ، طُمْ كٝ كُوگَلْ الله، أٝن ݝَرْطِنْطٝ كَلَ هُوندࣹ، پࣹلّࣹتْ، كَنْكٝ كٝ أٝ هُمْپِتِيطٝ كٝ گٝلّٝتٝنْ كٝنْ.

(89) مٝ أَرْدِي إࣹ مٝڃّࣹرࣹ ندࣹنْ هَرَيْ نٝ وٝودَنِ مٝ كٝ ࢠُرِ ندࣹ مٝڃُّدࣹ، هَرَ كَمْࢠࣹ كٝ ࢠࣹ هٝولِيࢠࣹ إِمّٝرْدࣹ إࣹ طࣹنْڃࣹرࣹ ندࣹنْ ݧَلْندࣹ.

(90) مٝ أَرْدِي كَدِ إࣹ بٝنࣹ طࣹنْ، گࣹيݖࣹ مَࢠّࣹ طࣹنْ هِپّࣹتࣹ كَ يِيتࣹ. يَلِّ أٝن يٝࢠࣹتࣹ سِ وَنَا كٝنْ كٝ لَاتِنٝطٝنْ هِطٝنْ گٝلَّ?

(91) أَنْندُ كٝ مِي يَمِرَا كٝ يٝ مِي رࣹوُ جٝومَ ندِي لࣹيْدِ ندِ الله هَرْمِنِ طُمْ، نٝ وٝودَنِ مٝ كَلَ هُوندࣹ. مِي يَمِرَامَ كَدِ يٝ مِي لَاتٝ جࣹيَاطٝ إࣹ جࣹبِّلِيࢠࣹ ࢠࣹنْ.

(92) إࣹ يٝ مِي جَنْنگُ القرآن ندࣹنْ، مٝ فࣹيوِي هَرَيْ كٝ هٝورࣹ-مَكّٝ أٝ فࣹيوَنِ، مٝ مَجِّي كَدِ هَرَيْ مَاكُ: مِينْ" دࣹيْ، كٝ مِي جࣹيَاطٝ إࣹ جࣹرْتِنٝوࢠࣹ ࢠࣹنْ.

(93) مَاكُ كَدِ: "يࣹتّٝورࣹ ندࣹنْ وٝودَنِي الله، أَرَيْ أٝ هٝلَّ أٝن آيࣹيجࣹ مَكّٝ طࣹنْ أَنْندٝنْ طࣹ". جٝومِ مَاطَ وٝنَالِ وࣹلْسِندِيطٝ گَايْ إࣹ كٝنْ كٝ گٝلّٝتٝنْ.

28 Al-Qasas — 88 آيَة

(1) تَا، سِينْ، مِيمْ.

(2) طࣹي كٝ آيࣹيجࣹ دࣹفْتࣹرࣹ ࢠَنْنگُندࣹ ندࣹنْ.

(3) مࣹطࣹنْ جَنْنگِرَ إࣹ مَاطَ هٝورࣹ گٝونگَ، كُمْپِتَلْ موسى إࣹ فرعون، وٝنَنْندࣹ يِمْࢠࣹ گٝمْطِنٝوࢠࣹ ࢠࣹنْ.

(4) پࣹلّࣹتْ، فرعون تٝوْنِتِنٝ كَ لࣹيْدِ; أٝ وَطِ يِمْࢠࣹ مَيْرِ ندِنْ پࣹݖّࣹ-پࣹݖّࣹ: هِمٝ لٝئِنَ فࣹدّࣹ گٝو إࣹ مَࢠّࣹ، أٝ هِرْسَ ࢠِطْࢠࣹ مَࢠّࣹ وٝرْࢠࣹ ࢠࣹنْ أٝ أَݖَّ رࣹوْࢠࣹ مَࢠّࣹ ࢠࣹنْ. پࣹلّࣹتْ، كٝ إࣹ بٝنّٝوࢠࣹ ࢠࣹنْ أٝ جࣹيَنٝو.

(5) مࣹطࣹنْ فَاندِنٝو مٝڃَّگٝلْ إࣹ لٝئِنَنٝوࢠࣹ ࢠࣹنْ كَ لࣹيْدِ; مࣹنْ وَطَ ࢠࣹ كٝوهٝوࢠࣹ، مࣹنْ وَطَ ࢠࣹ كَدِ رٝنٝوࢠࣹ ࢠࣹنْ.

(6) مࣹنْ هُࢠِندِنَ ࢠࣹ إࣹ لࣹيْدِ ندِنْ، مࣹنْ هٝلَّ فرعون إࣹ هَامَانَ إࣹ كٝنࣹيلِ مَࢠّࣹ طِنْ، كٝنْ كٝ ࢠࣹ هُلَيْنٝو.

(7) مࣹنْ وَهَيِنِ إࣹ يُمَّ موسى، وٝنْندࣹ: "مُيْنِنْ مٝ". سِ أَ هُلَنٝيِي مٝ، كٝ يٝ أَ بُگٝيٝ مٝ كَ مَايٝ. وَتَ أَ هُلُ، وَتَ أَ سُنٝ; پࣹلّࣹتْ، مࣹنْ رُتَّيْ مٝ كَ مَاطَ، مࣹنْ وَطٝيَ مٝ گٝوتٝ إࣹ نُلَاࢠࣹ ࢠࣹنْ".

(8) يِمْࢠࣹ فرعون ࢠࣹنْ هٝݖِّ مٝ، فِي يٝ لَاتَنٝ ࢠࣹ گَݧٝ إࣹ سُنٝ. پࣹلّࣹتْ، فرعون إࣹ هَامَانَ إࣹ كٝنࣹيلِ مَࢠّࣹ طِنْ، ࢠࣹ لَاتِنٝ وٝوپُࢠࣹ.

(9) دࣹبّٝ فرعون أٝن وِئِ: "(كُنْ إࣹنْتُمْ كٝ) ࢠُوࢠࣹيندِ گِتࣹ أَم إࣹ أَن كَدِ! وَتَ أٝن وَرُ مٝ، هِنٝ گَسَ كَ أٝ نَفَتَ إࣹنْ، مَا جٝگِتٝرࣹنْ مٝ ࢠِطّٝ". هَرِ ࢠࣹ سٝؤَانِ.

(10) ࢠࣹرْندࣹ يُمَّ موسى ندࣹنْ وٝنْتِ يࣹيوُندࣹ. هَا أٝ إࣹࢠِّ فࣹݧِّندࣹ فِي مَكّٝ، سِ وَنَانٝ مࣹنْ تَبِنْتِنْندࣹ ࢠࣹرْندࣹ مَكّٝ ندࣹنْ، فِي يٝ أٝ لَاتٝ إࣹ گٝمْطِنْࢠࣹ ࢠࣹنْ.

(11) يُمَّ موسى وِئَنِ بَنْندِرَاوٝ مَكّٝ أٝنْ: "جٝكِّتُ مٝ"; أٝ هَيْنِي مٝ كَ دࣹيسْتِي، هَرِ ࢠࣹ سٝؤَا-َنْندُدࣹ.

(12) مࣹنْ هَرْمِنَنِ مٝ كٝ أَدِي، (ࣹنْطِ) مُيْنِرْطِ طِنْ، (بَنْندِرَاوٝ مَكّٝ أٝنْ) وِئِ: "ٝن أَݖَّيْ مِي تِنْندِنَيْ أٝن فِي يِمْࢠࣹ سُودُ، ࢠࣹ نࣹئَنَ أٝن مٝ، هَرَ كَدِ كٝ لَاࢠَنَيْ ࢠࣹ مٝ?"

(13) مࣹنْ رُتِّ مٝ كَ يُمَّ مَكّٝ، فِي يٝ ࢠࣹرْندࣹ مَكّٝ ࢠُوࢠُ، هَرَ أٝ سُنَاكِ، إࣹ فِي يٝ أٝ أَنْندُ پࣹلّࣹتْ، فٝدٝورࣹ الله ندࣹنْ، كٝ گٝونگَ. كٝنٝ ࢠُرْࢠࣹ طُوطُدࣹ إࣹ مَࢠّࣹ أَنْندَا.

(14) ندࣹ أٝ يٝتِّنٝو سَگَتَلَاكُ مَكّٝ نگُنْ، أٝ فٝتٝندِ- رِ، مࣹنْ أٝكِّ مٝ فَامُ إࣹ گَنْندَلْ. كٝ وَنٝ نٝنْ مࣹنْ يٝࢠِرْتَ مٝڃِّنٝوࢠࣹ ࢠࣹنْ.

(15) أٝ نَاتِرِ إࣹ سَارࣹ ندࣹنْ، فࣹوْندٝ وࣹلْسِندَارࣹ يِمْࢠࣹ مَيْرࣹ; أٝ تَوِ وٝرْࢠࣹ طِطٝ نٝ هَࢠَ تٝنْ: گٝطّٝ أٝن كٝ جࣹيَاطٝ إࣹ لࣹݧٝلْ مَكّٝ، أٝيَ أٝن كٝ جࣹيَاطٝ إࣹ أَيْࢠࣹ مَكّٝ ࢠࣹنْ. جࣹيَاطٝ إࣹ لࣹݧٝلْ مَكّٝ نگٝلْ طَࢠِّرِ فَابٝ إࣹ هٝورࣹ جࣹيَاطٝ إࣹ گَݧٝ مَكّٝ أٝنْ. موسى أُتِّي مٝ، ندࣹ گَيْنِ فِي مَكّٝ. موسى مَاكِ: "طُمْ طٝو، كٝ جࣹيَاطُمْ إࣹ كُوطࣹ سࣹيْتَانࣹ. پࣹلّࣹتْ، كٝ أٝ گَݧٝ، مَجِّنٝووٝ ࢠَنْنگُطٝ".

(16) أٝ مَاكِ: جٝومِ" أَمْ، مِي تٝوݧِي هٝورࣹ-َمْ; هَفٝرَنَمْ." أٝ هَفٝرَنِ مٝ. پࣹلّࣹتْ، كٝ كَنْكٝ وٝنِ هَفٝرٝووٝ، هِنّٝتٝوطٝ أٝنْ.

(17) أٝ مَاكِ: جٝومِ" أَمْ، سَبُ كٝ نࣹيمِنْطَا كٝنْ إࣹ أَمْ، مِي وٝنْتَتَا بَلّٝ بٝمْࢠࣹ ࢠࣹنْ".

(18) أٝ وٝنْتِ هُلْطٝ كَ سَارࣹ، هِمٝ هَبِّي نَنْنگࣹيدࣹ . تَوِ فَابِنٝرْنٝوطٝ مٝ أٝن هَنْكِ، هِنٝ فَابِنٝرَ مٝ هَنْندࣹ كَدِ . موسى مَاكَنِ مٝ: "پࣹلّࣹتْ، كٝ أَ بࣹوْطٝ ࢠَنْنگُطٝ".

(19) ندࣹ موسى فَاندِنٝو نَنْنگُگٝلْ تَوْطٝ أٝن كٝ گَݧٝ مَࢠّࣹ كَمْࢠࣹ طِطٝ، أٝ وِئِ: "كٝ أَن يٝ موسى، أَ فَالَ وَرُگٝلْ لَمْ، وَنٝ إِتِّرْنٝطَا وٝنْكِي نٝنْ هَنْكِ? أَ فَاندَاكِ، سِ وَنَا وٝنُگٝلْ دُنْندَرَنْكࣹيجٝ كَ لࣹيْدِ، أَ فَاندَاكِ جࣹيࣹيدࣹ إࣹ مٝڃِّنٝوࢠࣹ ࢠࣹنْ".

(20) گٝرْكٝ إِوْرِ كَ ࢠُرِ وٝطُّدࣹ إࣹ سَارࣹ ندࣹنْ، هِمٝ يَݖّٝو، أٝ وِئِ: "كٝ أَن يٝ موسى، نبَتُ نگُنْ نٝ فࣹوْجَنَ فِي وَرُگٝلْ مَا. يَلْتُ (ࣹ لࣹيْدِ ندِنْ). كٝ مِي جࣹيَاطٝ إࣹ لَاࢠَنٝوࢠࣹ مَ".

(21) أٝ يَلْتِ تٝنْ، كٝ أٝ هُلُطٝ، هِمٝ تِجِّي نَنْنگࣹيدࣹ . أٝ مَاكِ: جٝومِ" أَمْ، دَنْطْندَمْ إࣹ يِمْࢠࣹ تٝوݧٝوࢠࣹ ࢠࣹنْ".

(22) ندࣹ أٝ فࣹوْتُنٝو هࣹلَّ مَدْيَنَ، أٝ مَاكِ: "هِنٝ هَسِي كَ جٝومِ أَم فࣹوْنِرْتَ لَمْ، إࣹ لَاوٝلْ فٝݖِّينگٝلْ نگٝلْ".

(23) ندࣹ أٝ هࣹوْتُنٝو كَ ࢠُنْندُ مَدْيَنَ، أٝ تَوِ طٝنْ دࣹنْتَلْ يِمْࢠࣹ، هِࢠࣹ يَرْنَ دَابࣹيجِ مَࢠّࣹ ، أٝ تَوِ گَانِنْ مَࢠّࣹ، رࣹوْࢠࣹ طِطٝ، نٝ سٝگَّ دَمِّ مَࢠّࣹ ، أٝ مَاكِ: "كٝ هٝنْطُمْ هَبِّ-طٝنْ?" ࢠࣹ وِئِ: مࣹنْ" يَرْنَتَا هَا أَيْنٝوࢠࣹ ࢠࣹنْ دࣹيسْتٝو، بَابَ أَمࣹنْ، كٝ مَوْطٝ نَيࣹيجٝ".

(24) أٝ يَرْنَنِ ࢠࣹ دَمِّ طِنْ ، رࣹفْتِ أٝ هُݖِّتِرِ كَ طٝوْدِ، أٝ مَاكِ: جٝومِ" أَمْ، مِينْ كٝ جِپِّنْطَا إࣹ أَم وٝو إࣹ مٝڃّࣹرࣹ، كٝ مِي هَتٝنجِطٝ".

(25) گٝوتٝ إࣹ جِوْࢠࣹ ࢠࣹنْ أَرِ إࣹ مَكّٝ، هِمٝ يَهْرَ هࣹرْسَ، أٝ وِئِ: "بَابَ أَم نٝ نٝدّࣹ، فِي يٝ أٝ يٝࢠࣹ نجٝࢠْدِ كٝنْ كٝ يَرْنَنْ-طَا مࣹنْ". تُمَ أٝ هࣹوْتُنٝو كَ مَوْطٝ أٝنْ، أٝ سِفَنِي مٝ فِي مَكّٝ ، أٝ مَاكِ: "وَتَ أَ هُلُ، أَ دَطِي يِمْࢠࣹ تٝوݧٝوࢠࣹ ࢠࣹنْ".

(26) گٝوتٝ إࣹ جِوْࢠࣹ ࢠࣹنْ وِئِ: "ڃࣹتُّ مٝ گٝلّࣹ، يٝ بَابَ. پࣹلّࣹتْ، كٝ ࢠُرِ مٝڃُّدࣹ إࣹ مٝ ڃࣹتَّتَا گٝلّࣹ، كٝ دٝولْنُطٝ هٝولْنِيطٝ".

(27) مَوْطٝ أٝن مَاكِ: مِيطُ" يِطِ رࣹسِنگٝلْ مَ گٝوتٝ إࣹ جِوْࢠࣹ أَم ࢠࣹنْ طٝو طِطٝ، إࣹ هٝورࣹ يٝ أَ گٝلَّ-نَمْ دُوࢠِ جࣹيتَتِ. سِ أَ تِمِّنِي سَپّٝ، هَرَيْ كٝ إࣹ سَگٝ مَا; مِي فَالَاكَ سَطْتِنَنْندࣹ مَ. أَ تَوَيْ لَمْ، سِ الله جَࢠِي، كٝ إࣹ مٝڃِّنٝوࢠࣹ ࢠࣹنْ مِي جࣹيَا".

(28) موسى مَاكِ: "طُمْ وٝنِي هَكُّندࣹ أَم إࣹ مٝنْ. هَپُّ وٝو إࣹ مَجِّ طِطِ، طِ مِي هُنِّ، هَرَيْ ࢠِطْتࣹيدࣹ أَلَنَا إࣹ لَمْ. إࣹن وَطِي الله فَوْتِيطٝ إࣹ كٝ وٝوْلُطࣹنْ كٝنْ".

(29) ندࣹ موسى تِمِّنّٝو هَپُّ نگُنْ، أٝ يَهْدِ إࣹ ࢠࣹيْدْنُورࣹ مَكّٝ، أٝ هَيْنِي يِيتࣹ كَ سࣹنْنگٝ (فࣹلّٝ) تُورِ، أٝ مَاكِ ࢠࣹــُٔنگُورࣹ مَكّٝ ندࣹنْ: "وٝنࣹي طٝو، مِينْ مِي هَيْنِكࣹ يِيتࣹ. بࣹلَجٝؤٝ مِي أَدَّنَ أٝن إࣹ مَگّࣹ كُمْپِتَلْ مَا تࣹئَنْندࣹ(دٝݖٝتَلْ) يِيتࣹ نگࣹنْ، بࣹلَجٝؤٝ إِوْلٝرٝنْ".

(30) ندࣹ أٝ هࣹوْتُنٝو نگࣹ، أٝ نٝدِّرَا كَ فٝنْنگٝ ݧَامٝ أَيْندࣹ، كَ تُمْنبࣹرࣹ بَرْكِنَاندࣹ، كَ (تَكّٝ) لࣹكِّ، وٝنْندࣹ: "كٝ أَن يٝ موسى! كٝ مِينْ وٝنِ الله، جࣹيْطٝ وِنْندࣹرࣹ ندࣹنْ".

(31) كَدِ: "بُگٝ تُگّٝرْدُ مَا ندُنْ"; ندࣹ أٝ تِنُنٝو هِندُ دِلِّرَ وَ نبٝدِّ،ٝ هُݖِّتِرِ دٝگُدُ، أٝ يࣹڃِّتَاكِ. أٝ نٝدَّا : "كٝ أَن يٝ موسى! أَرْتُ، وَتَ أَ هُلُ; كٝ أَ جࣹيَاطٝ إࣹ هٝولِيࢠࣹ ࢠࣹنْ.

(32) نَادُ جُونگٝ مَا نگٝنْ كَ جَيْبَ مَا، نگٝ يَلْتَيْ رَنࣹوٝ، كٝ أَلْدَا إࣹ ݧَوُ. بَكَا إࣹ مَاطَ، جُوطࣹ مَا طࣹنْ، فِي كُلٝلْ نگٝلْ. طِنْ كٝ هُجَّاجِ طِطِ، إِمّٝرْدࣹ كَ جٝومِ مَا، هَا إࣹ فرعون إࣹ دࣹنْتَلْ مُنْ، پࣹلّࣹتْ، ࢠࣹنْ لَاتِكࣹ يِمْࢠࣹ فَاسِقِيࢠࣹ".

(33) أٝ مَاكِ: جٝومِ" أَمْ، مِي إِتِّينٝ وٝنْكِي إࣹ مَࢠّࣹ، مِيطُ هُلِ ندࣹ ࢠࣹ وَرَتَ لَمْ.

(34) كٝ مُسِطّٝ أَم أٝن هَارُونَ، ࢠُرِ لَمْ لَاࢠُدࣹ طࣹمْنگَلْ. إِمِّندِنْ-مٝ إࣹ أَمْ، كٝ وَلِّتٝو لَمْ، گٝونگِنَنْ-مِ; مِيطُ هُلِ ندࣹ ࢠࣹ فࣹنَّتَ لَمْ".

(35) الله دَالِ: مࣹنْ" سࣹمْنبِنِرَيْ تࣹرࣹ مَا طࣹنْ، مُسِطّٝ مَا أٝنْ، مࣹنْ وَطَنَ أٝن هُجَّ (ࣹ ݖࣹمْنبࣹ); ࢠࣹ هࣹوْتِرْتَا أٝن طِطٝ، سَبُ آيࣹيجࣹ أَمࣹنْ طࣹ مࣹنْ نُلِ إࣹ مٝؤٝنْ، إࣹ ࢠࣹنْ جٝكُّࢠࣹ أٝن فٝولُࢠࣹ".

(36) ندࣹ موسى أَرْدُنٝو إࣹ مَࢠّࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ ࢠَنْنگُطࣹ، ࢠࣹ وِئِ: "طُمْ وٝنَالِ، سِ وَنَا نبِلࣹوُ فࣹفِندَانگُ. مࣹنْ نَنِرَالِ طُمْ بَابِرَاࢠࣹ أَمࣹنْ أَدِنٝوࢠࣹ ࢠࣹنْ".

(37) موسى مَاكِ: "كٝ جٝومِ أَم ࢠُرِ أَنْندُدࣹ أٝن أَرْدُطٝ إࣹ پࣹيوَلْ، إِمّٝرْدࣹ كَ مَكّٝ، إࣹ أٝن مٝ بَتَّنࣹ مٝڃّࣹ وٝودَنِ كَ لَاكَرَ. پࣹلّࣹتْ، أٝ مَلْنَتَا تٝوݧٝوࢠࣹ ࢠࣹنْ".

(38) فرعون وِئِ: "كٝ أٝنٝنْ دࣹنْتَلْ، مِي أَنْندَنَا أٝن رࣹوࣹتࣹيطٝ، سِ وَنَا مِينْ. كٝ أَن يٝ هَامَانَ، هُࢠِّنَنَمْ يِيتࣹ إࣹ لٝوپَلْ نگَلْ، مَهِرَنَا لَمْ هُࢠࣹيرࣹ تٝووُندࣹ، نٝ گَسَ كَ مِي ڃࣹلِّتٝو إࣹ رࣹوࣹتࣹيطٝ موسى أٝنْ; تَوْدࣹ مِيطُ سِكِّ مٝ إࣹ فࣹنٝوࢠࣹ ࢠࣹنْ".

(39) أٝ تٝوْنِتِي كَ لࣹيْدِ، كَنْكٝ إࣹ كٝنࣹيلِ مَكّٝ طِنْ، كٝ أَلْدَا إࣹ گٝونگَ. ࢠࣹ سِكِّ وٝندࣹ ࢠࣹ رُتِّتࣹتَاكٝ كَ أَمࣹنْ.

(40) مࣹنْ نَنْنگِ مٝ، كَنْكٝ إࣹ كٝنࣹيلِ مَكّٝ طِنْ، مࣹنْ بُگِي ࢠࣹ كَ مَايٝ. تَسْكٝ، كٝ هٝنّٝ بَتَّنࣹ تٝوݧُࢠࣹ ࢠࣹنْ وَؤُنٝو.

(41) مࣹنْ وَطِ ࢠࣹ أَرْدِيࢠࣹ، نٝدّٝو ࢠࣹ إࣹ يِيتࣹ. ݧَلْندࣹ دَرْنگَلْ، ࢠࣹ وَلّࣹتَاكࣹ.

(42) مࣹنْ جٝكِّنْتِنِرِ ࢠࣹ كَ أَدُنَ طٝو، كُدِّ. ݧَلْندࣹ دَرْنگَلْ، كٝ ࢠࣹ جࣹيَاࢠࣹ إࣹ وٝطِّنْتِنٝيْتࣹيࢠࣹ إࣹ يُرْمࣹيندࣹ الله .

(43) گٝمْطِي مࣹنْ أٝكِّي موسى دࣹفْتࣹرࣹ ندࣹنْ، ࢠَاوٝ ندࣹ مࣹنْ هَلْكُنٝو گِرࣹ أَرَنࣹ طࣹنْ، كٝ ندࣹ ندَيْگِنٝرْدࣹ يِمْࢠࣹ ࢠࣹنْ، إࣹ پࣹيوَلْ إࣹ يُرْمࣹيندࣹ. بࣹلَجٝؤٝ، ࢠࣹ وَاجِتٝرَيْ.

(44) أَ تَوَنٝوكَ كَ سࣹنْنگٝ هِيرْنَانگࣹ، فࣹوْندٝ مࣹنْ وَهْيِنّٝو إࣹ موسى يَمِرٝورࣹ ندࣹنْ، أَ جࣹيَنٝوكَ كَدِ إࣹ سࣹيدِيࢠࣹ ࢠࣹنْ.

(45) كٝنٝ مࣹنࣹنْ وٝنِ تَگُࢠࣹ گِرࣹ، دُوࢠِ مَࢠّࣹ جُوتِ. أَ هٝطَانٝ كَدِ إࣹ يِمْࢠࣹ مَدْيَنَ ࢠࣹنْ، هِطَ جَنْنگَنَ ࢠࣹ آيࣹيجࣹ أَمࣹنْ; كٝ مࣹنࣹنْ وٝنِ نُلْࢠࣹ.

(46) أَ تَوَنٝوكَ كَ سࣹنْنگٝ فࣹلّٝ تُورِ، فࣹوْندٝ مࣹنْ نٝدُّنٝو; كٝ يُرْمࣹيندࣹ إِمّٝرْدࣹ كَ جٝومِ مَا وٝنِ كٝ مࣹنْ نُلِرْ-مَا ، فِي نٝ جࣹرْتِنِرَا يِمْࢠࣹ، ࢠࣹ رࣹينْتِنٝووٝ أَرَالِ إࣹ مَࢠّࣹ، أَدٝ مَاطَ، بࣹلَجٝؤٝ، ࢠࣹ وَاجِتٝتٝ.

(47) سِ وَنَانٝ فِي يٝ لࣹپْتࣹ يَنِرْ إࣹ مَࢠّࣹ، سَبُ كٝ جُوطࣹ مَࢠّࣹ أَرْدِنِ، ࢠࣹ وِئَيْنٝ: جٝومِ" أَمࣹنْ، كٝ هَطْنٝ إِمِّنَا نُلَاطٝ إࣹ أَمࣹنْ? مࣹنْ جٝكَّ نُنْ آيࣹيجࣹ مَا طࣹنْ، مࣹنْ جࣹيࣹي إࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ".

(48) ندࣹ أٝ أَدِّرَنّٝوࢠࣹ گٝونگَ كَنْ، إِمّٝرْدࣹ كَ أَمࣹنْ، ࢠࣹ وِئِ: "كٝ هَطْنٝ أٝ أٝكّࣹي يࣹرُ كٝ موسى أٝكَّنٝو كٝنْ?" كࣹرࣹ ࢠࣹ يࣹدَّانٝ كٝ موسى أٝكَّنٝو كٝنْ كٝ أَدِي? ࢠࣹ وِئِ: "كٝ نبِلࣹيجِ طِطِ، طِ نٝ وَلِّندِرَ!" ࢠࣹ وِئِ كَدِ: "كٝ مࣹنْ يࣹدُّࢠࣹ فٝوْ إࣹ مُنْ".

(49) مَاكُ: "َدّࣹي دࣹفْتࣹرࣹ إِمّٝرْدࣹ كَ الله، هَرَ هِندࣹ ࢠُرِ طࣹي طِطِ فࣹيوُدࣹ، مِي جٝكَّيْندࣹ، سِ أٝن لَاتِكࣹ گٝونگُنْࢠࣹ".

(50) سِ ࢠࣹ جَابَاكِ مَ، أَنْندُ كٝ نبࣹلࣹيطࣹ مَࢠّࣹ ࢠࣹ جٝكَّتَ. كٝ هٝمْنبٝ ࢠُرِ مَجُّدࣹ ࢠُرِ أٝن جٝكِّرْطٝ نبࣹلࣹيطࣹ مُنْ، كٝ أَلْدَا إࣹ پࣹيوَلْ إِمّٝرْدࣹ كَ الله? پࣹلّࣹتْ، الله فࣹوْنَتَا يِمْࢠࣹ تٝوݧٝوࢠࣹ ࢠࣹنْ.

(51) گٝمْطِي مࣹنْ هࣹوْتِنِي ࢠࣹ كٝنْنگٝلْ نگٝلْ، بࣹلَجٝؤٝ، ࢠࣹ وَاجِتٝرَ.

(52) ࢠࣹنْ ࢠࣹ مࣹنْ رٝكِّ دࣹفْتࣹرࣹ تَوْرࣹيتَ ندࣹنْ، كٝ أَدِي القرآن ندࣹي، كٝ ࢠࣹ گٝمْطِمْࢠࣹ ندࣹ.

(53) سِ ندࣹ جَنْنگَامَ إࣹ دٝوْ مَࢠّࣹ، ࢠࣹ وِئَ: مࣹنْ" گٝمْطِنِي ندࣹ. كٝ ندࣹ گٝونگَ إِمّٝرْدࣹ كَ جٝومِ أَمࣹنْ. پࣹلّࣹتْ مࣹنْ وٝنِينٝ جࣹبِّلِيࢠࣹ، أَدٝ مَيْرࣹ".

(54) ࢠࣹنْ طٝنْ، ࢠࣹ جٝنِّرٝيْتࣹ نجٝࢠْدِ مَࢠّࣹ ندِنْ، لَابِ طِطِ، سَبُ كٝ ࢠࣹ مُݧِنٝو كٝنْ، هِࢠࣹ دُݧِرَيْنٝو مٝڃّࣹرࣹ ندࣹنْ، بٝنࣹ أٝنْ، إࣹ هُوندࣹ ندࣹ مࣹنْ أَرْسِكِ ࢠࣹ ندࣹنْ، هِࢠࣹ نَفْقَ.

(55) سِ ࢠࣹ نَنِي مࣹيرࣹنْتࣹ، ࢠࣹ طُورْنٝو إࣹ طُمْ، ࢠࣹ وِئَ: "كٝ مࣹنࣹنْ گٝلّࣹ أَمࣹنْ وٝودَنِ، كٝ أٝنٝنْ كَدِ گٝلّࣹ مٝنْ وٝودَنِ. كِسَلْ وٝنِي إࣹ مٝؤٝنْ. مࣹنْ طَࢠَّتَا وٝنْندُگٝلْ إࣹ مَجُّࢠࣹ ࢠࣹنْ".

(56) پࣹلّࣹتْ، أَن (نُلَاطٝ)، أَ فࣹوْنَتَا مٝ يِطُطَا; كٝنٝ الله نٝ فࣹوْنَ أٝن مٝ أٝ مُويِ. كٝ كَنْكٝ ࢠُرِ أَنْندُدࣹ فࣹيوُࢠࣹ ࢠࣹنْ.

(57) ࢠࣹ وِئِ: "سِ مࣹنْ جٝكِّدِي پࣹيوَلْ نگَلْ إࣹ مَا، مࣹنْ بِفْتࣹتࣹ كَ لࣹيْدِ أَمࣹنْ". إࣹ مࣹنْ هٝطِنِرَالِ ࢠࣹ إࣹ نٝكُّ هٝرْمِنَاطٝ، هٝولْنِيطٝ، مٝ دِمْطࣹ نٝونࣹ كَكَ أَدّࣹتࣹي إࣹ مُنْ، أَرْسِكࣹ إِمّٝرْدࣹ كَ أَمࣹنْ? سِ كٝ وٝنِ، كٝ ࢠُرِ طُوطُدࣹ إࣹ مَࢠّࣹ، ࢠࣹ أَنْندَا.

(58) طُوطِي كٝ مࣹنْ هَكْلِ، إࣹ يِمْࢠࣹ ݖَئࣹ، بࣹوْرُنٝوࢠࣹ نگُورࣹي مُنْ، إࣹ هِنٝ نٝنْ گَلّࣹيجِ مَࢠّࣹ طِنْ، هٝطِرَاكَ ࢠَاوٝ مَࢠّࣹ، سِ وَنَا سࣹيطَ. مࣹنْ لَاتِي وٝنْندࣹ رٝنٝوࢠࣹ ࢠࣹنْ.

(59) جٝومِ مَا سِفٝرَالِ وٝنْندࣹ هَلْكٝووٝ يِمْࢠࣹ ݖَئࣹ، هَا أٝ إِمِّنَ إࣹ لࣹنْيٝلْ مَࢠّࣹطࣹنْ، نُلَاطٝ، كٝ جَنْنگَنَ ࢠࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ. مࣹنْ سِفٝرَالِ كَدِ هَلْكُگٝلْ يِمْࢠࣹ ݖَئࣹ، سِ وَنَا هَرَ يِمْࢠࣹ مَجّࣹ كٝ تٝوݧُࢠࣹ.

(60) كَلَ كٝ أٝكَّطٝنْ إࣹ هُوندࣹ، كٝ دَكَمّࣹ نگُورْندَمْ أَدُنَ إࣹ ݖُطَارِ مُنْ; كٝ كٝ وٝنِ كَ الله كٝنْ، ࢠُرِ مٝڃُّدࣹ، وٝنِ كَدِ كٝ لُتَّتَ. إࣹ أٝن هَقِّلْتَا?

(61) إࣹ كٝ أٝن مٝ مࣹنْ فٝدِ فٝدَارِ مٝڃُّرِ، ندِ أٝ هَوْرَتَ إࣹ مُؤُنْ، وَأَتَ وَ أٝن مٝ مࣹنْ دَكْمِتِرِ دَكَمّࣹ نگُورْندَمْ أَدُنَ، رࣹفْتِ هَرَ كَنْكٝ أٝنْ، ݧَلْندࣹ دَرْنگَلْ، كٝ إࣹ تَوْنࣹتࣹيࢠࣹ (كَ يِيتࣹ) ࢠࣹنْ أٝ جࣹيَا?

(62) ݧَلْندࣹ أٝ نٝدّٝيْتَ ࢠࣹ، أٝ وِئَ: "كٝ هٝنْتٝ كَفِدَاطِ أَمْ، طِنْ طِ آڃٝرَيْنٝطٝنْ وٝنِ?"

(63) ࢠࣹنْ ࢠࣹ كٝنْنگٝلْ جٝجِّ إࣹ مُؤُنْ وِئَ: جٝومِ" أَمࣹنْ، كٝ ࢠࣹي وٝنِ كٝ مࣹنْ بࣹوِنّٝو. مࣹنْ بࣹوِنِرْنٝ ࢠࣹ، وَنٝ مࣹنْ بࣹوِرْنٝو. مࣹنْ دَطْندِنِكࣹ إࣹ مَاطَ. وَنَا مࣹنࣹنْ ࢠࣹ رࣹوَيْنٝو".

(64) ࢠࣹ وِئَنࣹي: "نٝدّࣹي نٝنْ كَفِدَاطِ مٝنْ طِنْ". ࢠࣹ نٝدَّ طِ، كٝنٝ طِ جَابٝتَاكٝ ࢠࣹ. ࢠࣹ يِئَ لࣹپْتࣹ طࣹنْ، (ࢠࣹ يࣹلٝو) سِنّٝ هَرَ ࢠࣹ فࣹيوُنٝ.

(65) ݧَلْندࣹ أٝ نٝدّٝيْتَ ࢠࣹ، أٝ وِئَ: "كٝ هٝنْطُمْ جَابِنٝطٝنْ نُلَاࢠࣹ ࢠࣹنْ?"

(66) لَمِتٝرْطِ طِنْ وُمِطَنَ ࢠࣹ ندࣹنْ ݧَلْندࣹ، ࢠࣹ لَمْندٝندِرْتَا.

(67) سِ كٝ تُوبُطٝ، أٝ گٝمْطِنِ، أٝ گٝلِّ مٝڃُّطِ، هِنٝ هَسِي كَ أٝ جࣹيࣹي إࣹ مَلَاࢠࣹ ࢠࣹنْ.

(68) جٝومِ مَا نٝ تَگَ كٝ أٝ مُويِ، أٝ سُࢠٝو; سُࢠَگٝلْ أَلَنَا ࢠࣹ. سࣹنَادࣹ وٝودَنِي الله! أٝ تٝووِي إࣹ كٝنْ كٝ ࢠࣹ سِرْكَتَ!

(69) جٝومِ مَا نٝ أَنْندِ كٝ ࢠࣹرْطࣹ مَࢠّࣹ طࣹنْ سُوطَتَ، إࣹ كٝنْ كٝ ࢠࣹ فࣹݧِّنْتَ.

(70) كٝ كَنْكٝ الله وٝنِ، مٝ رࣹوࣹتࣹيطٝ أَلَا، سِ وَنَا كَنْكٝ. كٝ كَنْكٝ يࣹتّٝورࣹ وٝودَنِ أَدُنَ إࣹ لَاكَرَ. هِنٝ وٝودَنِ مٝ ݧَاوٝورࣹ ندࣹنْ. كٝ كَ مَكّٝ رُتّࣹتࣹطٝنْ.

(71) مَاكُ: "يࣹيتࣹي لَمْ، سِ الله وَطِرَنِي أٝن جࣹمَّ أٝنْ، دُومِيطٝ هَا ݧَلْندࣹ دَرْنگَلْ، كٝ رࣹوࣹتࣹيطٝ هٝمْنبٝ، تَنَا الله، أَدَّنْتَ أٝن ندَيْنگُ? إࣹ أٝن نَنَتَا?"

(72) مَاكُ: "يࣹيتࣹي لَمْ، سِ الله وَطِرَنِي أٝن ݧَلٝرْمَ أٝنْ، دُومِيطٝ هَا ݧَلْندࣹ دَرْنگَلْ، كٝ رࣹوࣹتࣹيطٝ هٝمْنبٝ، تَنَا الله، أَدَّنْتَ أٝن جࣹمَّ، مٝ دࣹيڃٝتٝنْ إࣹ مُنْ? إࣹ أٝن يِئَتَا?"

(73) هِنٝ جࣹيَا إࣹ يُرْمࣹيندࣹ مَكّٝ، أٝ وَطِرَنِي أٝن جࣹمَّ إࣹ ݧَلٝرْمَ; فِي نٝ دࣹيڃِرٝنْ إࣹ (جࣹمَّ) أٝنْ، إࣹ نٝ طَࢠِّرٝنْ ( إࣹ ݧَلٝرْمَ أٝنْ)، ࢠُرَلْ مَكّٝ، إࣹ نٝ بࣹلَجٝؤٝ يࣹتِّرٝنْ مٝ .

(74) ݧَلْندࣹ أٝ نٝدّٝيْتَ ࢠࣹ، أٝ وِئَ: "كٝ هٝنْتٝ كَدِدَاطِ(رࣹنْندِدِيطِ) أَمْ، طِنْ طِنْ آڃٝرَيْنٝطٝنْ وٝنِ?"

(75) مࣹنْ طٝوفِ إࣹ كَلَ مٝفْتࣹ، سࣹيدٝتٝوطٝ، مࣹنْ وِئَ: "َدّࣹي هُجَّاجِّ مٝنْ طِنْ". ࢠࣹ أَنْندِ وٝنْندࣹ گٝونگَ كَنْ، كٝ الله وٝودَنِ; كٝ ࢠࣹ فࣹفِندٝتٝنٝو كٝنْ مَجِّتٝرَ ࢠࣹ.

(76) پࣹلّࣹتْ،قَارُونَ كٝ إࣹ يِمْࢠࣹ موسى ࢠࣹنْ جࣹيَنٝو،كٝنٝ أٝ بࣹوِ إࣹ مَࢠّࣹ.مࣹنْ أٝكِّ مٝ نگَلُوجِ تَوَيْ-طِ سٝكْتِرْطࣹ مُنْ،نٝ تࣹدَّنَ جَمَا دٝولْنُطٝ.ندࣹ يِمْࢠࣹ مَكّٝ ࢠࣹنْ،وِئَنّٝ مٝ:وَتَ أَ وࣹلْتٝرْ تٝوْنِتَارࣹ.پࣹلّࣹتْ الله يِطَا وࣹلْتٝرْتٝوࢠࣹ تٝووْنِتَارࣹ.

(77) طَࢠِّرْ كٝ الله أٝكُّ-مَا كٝنْ، گَلّࣹ لَاكَرࣹ طࣹنْ. وَتَ أَ يࣹجِّتُ گࣹࢠَلْ مَا نگَلْ أَدُنَ. مٝڃِّرْ نٝ الله مٝڃّٝرِي نٝنْ إࣹ مَاطَ. وَتَ أَ طَࢠُّ بٝنُّگٝلْ كَ لࣹيْدِ. أَنْندُ پࣹلّࣹتْ، الله يِطَا بٝنّٝوࢠࣹ ࢠࣹنْ".

(78) أٝ وِئِ: "كٝ مِي أٝكِّرَا طُمْ، كٝ إࣹ هٝورࣹ گَنْندَلْ نگَلْ مِي جٝگِي". إࣹ أٝ أَنْندَا وٝنْندࣹ گٝمْطِي الله هَلْكِينٝ گِرࣹ، أَدِيࢠࣹ مٝ، ࢠُرْࢠࣹ دٝولࣹ، ࢠُرِ مٝ طُوطُدࣹ جَوْلࣹ? ࢠࣹ لَمْندٝيْتَاكࣹ فِي جُنُوبِ مَࢠّࣹ، كَمْࢠࣹ بٝمْࢠࣹ ࢠࣹنْ!

(79) أٝ يَلْتِتِرِ إࣹ يِمْࢠࣹ مَكّٝ ࢠࣹنْ، ݖُطَارِ مَكّٝ ندِنْ. يِطُࢠࣹ نگُورْندَمْ أَدُنَ ࢠࣹنْ وِئِ: "هَرَ نُنْ مࣹنْ هࣹࢠَيْنٝ سُگُ كٝو كٝ قَارُونَ أٝكَّا. كَنْكٝ أٝنْ، پࣹلّࣹتْ، كٝ أٝ جٝمْ-گࣹࢠَلْ مَوْنُنگَلْ".

(80) أٝكَّاࢠࣹ گَنْندَلْ ࢠࣹنْ كَنْ وِئِ: "ࣹي بٝنࣹ مٝنْ! كٝ نبَرْجَارِ الله ندِنْ ࢠُرِ مٝڃَّنْندࣹ أٝن گٝمْطِنْطٝ أٝ گٝلِّ كٝ مٝڃِّ". كٝنٝ هَوْرَتَا إࣹ طُمْ، سِ وَنَا مُݧِّيࢠࣹ ࢠࣹنْ.

(81) مࣹنْ مُتِندِنِ مٝ إࣹ گَلّࣹ مَكّٝ طࣹنْ، كَ لࣹيْدِ. فࣹدّࣹ وٝو أَلَنَا مٝ، كٝ فَابٝو مٝ گَانِنْ الله، أٝ وٝنَانٝ فَابٝتٝوطٝ هٝورࣹ-مُنْ.

(82) ࢠࣹنْ يࣹلِنٝوࢠࣹ نٝكُّ مَكّٝ أٝن هَنْكِ مُنْ، ࢠࣹ وَطْتِ وِيدࣹ: "ࣹي بٝنࣹ! الله نٝ وࣹرْتَنَ أَرْسِكࣹ أٝنْ، مٝ أٝ مُويَنِ إࣹ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ، أٝ ࢠِطْتِنَنَ مٝ أٝ مُويَنِ . سِ وَنَا الله مٝڃِّنٝو إࣹ أَمࣹنْ، أٝ مُتِنَيْنٝ مࣹنْ. إࣹي بٝنࣹ! جَكَ أٝ مَلْنَتَا يࣹدُّࢠࣹ ࢠࣹنْ".

(83) طࣹنْ گَلّࣹ لَاكَرَ، مࣹنْ وَطَنِي طࣹ ࢠࣹنْ ࢠࣹ يِطَا تٝوْنِتَارࣹ كَ لࣹيْدِ، وَنَا كَدِ بٝنُّگٝلْ. بَتَّنࣹ مٝڃّࣹ طࣹنْ، كٝ فِي گٝمْطُࢠࣹ ࢠࣹنْ.

(84) كَلَ أَرْدُطٝ إࣹ مٝڃّࣹرࣹ، هِنٝ وٝودَنِ مٝ كٝ ࢠُرِ ندࣹ مٝڃُّدࣹ; كَلَ كَدِ أَرْدُطٝ إࣹ بٝنَنْندࣹ، (يٝ أٝ أَنْندُ) هُووُࢠࣹ بٝنْطِ طِنْ يٝࢠࣹتَاكࣹ، سِ وَنَا كٝ ࢠࣹ هُووَيْنٝو كٝنْ.

(85) پࣹلّࣹتْ، جِپِّنْطٝ القرآن أٝن إࣹ مَاطَ، كٝ رُتّٝو-وٝ مَ كَ وٝووَاندِ مَا. مَاكُ: "كٝ جٝومِ أَم ࢠُرِ أَنْندُدࣹ أَرْدُطٝ إࣹ پࣹيوَلْ، إࣹ وٝنْطٝ إࣹ مَجّࣹرࣹ ࢠَنْنگُندࣹ".

(86) أَ وٝنَانٝ تَمْڃِنِنٝوطٝ جِپِّنࣹيدࣹ دࣹفْتࣹرࣹ ندࣹنْ إࣹ مَاطَ. سِ وَنَا تُنْ، كٝ يُرْمࣹيندࣹ إِمّٝرْدࣹ كَ جٝومِ مَا. وَتَ وٝنُ مُكْ، وَلّٝووٝ هࣹيفࣹرْࢠࣹ ࢠࣹنْ.

(87) وَتَ ࢠࣹ فَلٝ مَ كَدِ إࣹ آيࣹيجࣹ الله طࣹنْ، ࢠَاوٝ طࣹ جِپِّنࣹيدࣹ إࣹ مَاطَ. نٝدُّ (يِمْࢠࣹ ࢠࣹنْ) فَادࣹ إࣹ جٝومِ مَا، وَتَ أَ وٝنُ مُكْ، جࣹيَاطٝ إࣹ سِرْكٝوࢠࣹ ࢠࣹنْ.

(88) وَتَ أَ رࣹوَدُ إࣹ الله، رࣹوࣹتࣹيطٝ گٝو. رࣹوࣹتࣹيطٝ گٝو أَلَا، سِ وَنَا كَنْكٝ. هُوندࣹ فٝوْ رࣹئَيْ، سِ وَنَا جَاتِ مَاكٝ كِنْ. كٝ كَنْكٝ وٝودَنِ ݧَاوُگٝلْ; كٝ كَ مَكّٝ كَدِ رُتّࣹتࣹطٝنْ.

29 Al-'Ankabut — 69 آيَة

(1) أَلِفْ، لَامْ، مِيمْ.

(2) إࣹ يِمْࢠࣹ ࢠࣹنْ سِكُّ ࢠࣹ أَݖِّرْتࣹ نِي، فِي ࢠࣹ وِئِي: مࣹنْ" گٝمْطِنِي!" هَرَ ࢠࣹ فِتْنِنَاكَ?

(3) گٝمْطِي، مࣹنْ فِتْنِنِي ࢠࣹنْ أَدِنٝوࢠࣹ ࢠࣹ. مَا الله فࣹݧِّنْ، فِي گٝونگُࢠࣹ ࢠࣹنْ، مَا أٝ فࣹݧِّنْ كَدِ، فِي فࣹنٝوࢠࣹ ࢠࣹنْ.

(4) كَا هُووُࢠࣹ بٝنْطِ ࢠࣹنْ سِكُّ، وٝنْندࣹ ࢠࣹ لَاوٝتٝ مࣹنْ? كٝ ࢠࣹ ݧَاوَتَ كٝنْ بٝنِي!

(5) كَلَ تَوْطٝ هِمٝ جٝرْتِي هَوْرٝيْگٝلْ إࣹ الله، كٝ پࣹلّࣹتْ، لَجَلْ الله نگَلْ، كٝ هࣹوْتَيْگَلْ. كٝ كَنْكٝ وٝنِ نَنٝووٝ، أَنْندُطٝ أٝنْ.

(6) كَلَ كَدِ تِيطْنِيطٝ، كٝ هٝورࣹ-مَكّٝ أٝ تِيطْنَنِي، تَوْدࣹ پࣹلّࣹتْ، الله كٝ يٝنْندِنِيطٝ إࣹ تَگࣹيفٝ نگٝنْ.

(7) ࢠࣹنْ نٝنْ گٝمْطِمْࢠࣹ، ࢠࣹ گٝلِّ گٝلّࣹ مٝڃّࣹ، مَ مࣹنْ مٝمْتَنْ ࢠࣹ بٝنْطِ مَࢠّࣹ طِنْ، مࣹنْ يٝࢠِتْرَ ࢠࣹ، كٝ ࢠُرِ مٝڃُّدࣹ إࣹ كٝ ࢠࣹ گٝلَّيْنٝو كٝنْ.

(8) مࣹنْ يَمِرِي نࣹطَّنْكࣹ أٝنْ، مٝڃُّگٝلْ إࣹ مَوْࢠࣹ مَكّٝ ࢠࣹنْ. سِ ࢠࣹ دٝولِي مَا فِي سِرْكَنگٝلْ لَمْ، كٝ أَ أَلَنَا طُمْ گَنْندَلْ، وَتَ أَ طٝفْتٝ ࢠࣹ ! كٝ كَ أَمْ، وٝنِ رُتّٝرْدࣹ مٝنْ، مِي هَبْرَ أٝنْ، كٝ گٝلَّيْنٝطٝنْ.

(9) ࢠࣹنْ گٝمْطِمْࢠࣹ الله، ࢠࣹ گٝلِّ گٝلّࣹ مٝڃّࣹ، مَ مࣹنْ وَطْتِدُࢠࣹ إࣹ مٝڃُّࢠࣹ ࢠࣹنْ.

(10) هِنٝ إࣹ جِمْࢠࣹ ࢠࣹنْ، وِئٝووٝ: مࣹنْ" گٝمْطِنِي الله"; هَرَ نٝنْ سِ أٝ تَمْپِنِرَامَ فِي الله، أٝ وَطَيْ لٝرَّ يِمْࢠࣹ ࢠࣹنْ، وَنٝ لࣹپْتࣹ الله طࣹنْ. سِ بَلَّلْ كَدِ إِمّٝرِي كَ جٝومِ مَا، پࣹلّࣹتْ، ࢠࣹ وِئَيْ: "مࣹطࣹنْ وٝندُنٝو إࣹ مٝؤٝنْ!" كࣹرࣹ وَنَا الله ࢠُرِ أَنْندُدࣹ كٝ وٝنِ إࣹ ࢠࣹرْطࣹ تَگࣹيفٝ نگٝنْ?

(11) مَ الله أَنْندِنْ فِي گٝمْطِمْࢠࣹ ࢠࣹنْ، مَ أٝ فࣹݧِّنَ كَدِ فِي نَافِقِيࢠࣹ ࢠࣹنْ.

(12) هࣹيفࣹرْࢠࣹ ࢠࣹنْ وِئَنِ گٝمْطِمْࢠࣹ ࢠࣹنْ: "جٝكّࣹي لَاوٝلْ أَمࣹنْ نگٝلْ، مࣹنْ فَوْتٝتٝ فِي گٝوپِ مٝنْ طِنْ". كٝنٝ ࢠࣹ وٝنَالِ فَوْتٝتٝوࢠࣹ هَيْ هُوندࣹ إࣹ گٝوپِ مَࢠّࣹ. پࣹلّࣹتْ، كٝ ࢠࣹ فࣹنٝوࢠࣹ.

(13) كٝنٝ مَا ࢠࣹ فَوْتٝو فِي تࣹدّࣹيندِ جُنُوبِ مَࢠّࣹ طِنْ، إࣹ تࣹدّࣹيندِ گٝو، وٝنْندُدࣹ إࣹ تࣹدّࣹيندِ مَࢠّࣹ ندِنْ. كٝ مَا ࢠࣹ لَمْندٝيࣹي، ݧَلْندࣹ دَرْنگَلْ، گَايْ إࣹ كٝنْ كٝ ࢠࣹ فࣹفِندٝتٝنٝو.

(14) گٝمْطِي مࣹنْ نُلِي نوح إࣹ يِمْࢠࣹ مَكّٝ ࢠࣹنْ، أٝ ݧِيࢠِ هَكُّندࣹ مَࢠّࣹ، دُوࢠِ أُجُنࣹرࣹ، ݝَكُّطُمْ دُوࢠِ ݖَپَّنْطࣹ جٝوِ. وَامࣹ أٝن نَنْنگِ ࢠࣹ، هَرِ كٝ ࢠࣹ تٝوݧُࢠࣹ.

(15) مࣹنْ دَطْندِ مٝ، كَنْكٝ إࣹ يِمْࢠࣹ لَانَ كَنْ; مࣹنْ وَطِ كَ كَاوِسَ تَگࣹيفٝ نگٝنْ.

(16) (جَنْتٝ) كَدِ إبراهيم، ندࣹ أٝ وِئَنّٝو يِمْࢠࣹ مَكّٝ ࢠࣹنْ: "رࣹوࣹي الله، هُلٝنْ مٝ; كٝ طُمْ ࢠُرِ مٝڃَّنْندࣹ أٝنْ، سِ وٝنِ هِطٝنْ أَنْندِ".

(17) أَنْندࣹي كٝ رࣹوࣹتٝنْ كٝنْ گَانِنْ الله، كٝ سَنَمُوجِ، هِطٝنْ تَڃَ فࣹنَاندࣹ. پࣹلّࣹتْ، طِنْ طِ رࣹوٝتٝنْ گَانِنْ الله، طِ وَاوَا أٝن أَرْسِكُدࣹ; طَࢠِّرࣹي أَرْسِكࣹ أٝن كَ الله، رࣹوٝنْ مٝ، يࣹتّٝنْ مٝ. كٝ كَ مَكّٝ وٝنِ رُتّࣹتࣹطٝنْ.

(18) سِ أٝن فࣹنِّي، هَرَيْ مٝفْتࣹ أَدِنٝوࢠࣹ كَدِ فࣹنِّينٝ (نُلَاࢠࣹ مُنْ). أَلَا كٝ فَوِي إࣹ نُلَاطٝ أٝنْ، سِ وَنَا يٝتِّنگٝلْ ࢠَنْنگِنْنگٝلْ.

(19) إࣹ ࢠࣹي يِئَالِ نٝ الله فُطّٝرْتَ تَگُ نگُنْ، رࣹفْتِ أٝ رُتَّيْ نگُ? پࣹلّࣹتْ، طُمْ وٝنِي كٝ نࣹوَنِي الله.

(20) مَاكُ: "يِيلࣹي كَ لࣹيْدِ، ندَارٝنْ نٝ أٝ فُطّٝرِ تَگُ نگُنْ. رࣹفْتِ الله تَگَ تَگُ سَكِّتٝرُ نگُنْ. پࣹلّࣹتْ، الله كٝ هَتَّنْطٝ كَلَ هُوندࣹ".

(21) هِمٝ لࣹپْتࣹ مٝ أٝ مُويَنِ، أٝ يُرْمٝ مٝ أٝ مُويَنِ; كٝ كَ مَكّٝ رُتّࣹتࣹطٝنْ.

(22) أٝن وٝنَالِ لَاوٝتٝوࢠࣹ كَ لࣹيْدِ، وَنَا كَ كَمُّ; أَلَنَا أٝن گَانِنْ الله، وٝنِ گِطٝ، وَنَا بَلّٝ.

(23) ࢠࣹنْ يࣹدُّࢠࣹ آيࣹيجࣹ الله طࣹنْ، إࣹ هَوْرٝيْگٝلْ إࣹ مَكّٝ نگٝلْ، ࢠࣹنْ تَڃِكࣹ إࣹ يُرْمࣹيندࣹ أَمْ، ࢠࣹنْ طٝنْ، هِنٝ وٝودَنِ ࢠࣹ لࣹپْتࣹ مُوسُطࣹ.

(24) جَابَوٝلْ يِمْࢠࣹ مَكّٝ ࢠࣹنْ وٝنَانٝ، سِ وَنَا وِيدࣹ ࢠࣹ: "وَرࣹي مٝ، مَا سُنّٝنْ مٝ". كٝنٝ الله دَطْندِ مٝ إࣹ يِيتࣹ نگࣹنْ. پࣹلّࣹتْ، وٝنِي إࣹ طُمْ، مَاندࣹيجِ، وٝنَنْندࣹ يِمْࢠࣹ گٝمْطِمْࢠࣹ.

(25) أٝ مَاكِ: "َنْندࣹي كٝ رࣹوٝتٝنْ كٝنْ گَانِنْ الله، كٝ سَنَمُوجِ، فِي يِطِندِرٝ هَكُّندࣹ مٝنْ، كَ نگُورْندَمْ أَدُنَ، هٝوتِ نٝنْ، ݧَلْندࣹ دَرْندَلْ، يٝگَ مٝنْ يࣹدّٝيَيْ يٝگَ، يٝگَ مٝنْ كَدِ هُطٝ يٝگَ، هَرَ نٝنْ، جَاڃٝرْدࣹ مٝنْ ندࣹنْ، كٝ كَ يِيتࣹ، هَرَ وَلّٝوࢠࣹ أَلَنَا أٝنْ".

(26) لوط گٝمْطِنِ مٝ. أٝ مَاكِ: مِينْ" أٝنْ، كٝ مِي فࣹرٝووٝ كَ جٝومِ أَمْ، كٝ كَنْكٝ وٝنِ فٝولُطٝ، ݧࣹيݧٝ أٝنْ".

(27) مࣹنْ يࣹطِ مٝ إِسْمَائِيلَ إࣹ يعقوب، مࣹنْ وَطِ إࣹ گࣹيْنگٝلْ مَكّٝ نگٝلْ، أَنَّبَاكُ إࣹ دَفْتࣹرࣹ. مࣹنْ أٝكِّ مٝ نبَرْجَارِ مَكّٝ ندِنْ أَدُنَ، پࣹلّࣹتْ، كَنْكٝ أٝن كَ لَاكَرَ، كٝ أٝ جࣹيَاطٝ إࣹ مٝڃُّࢠࣹ ࢠࣹنْ.

(28) (جَنْتٝ) كَدِ لوط، ندࣹ أٝ مَاكَنّٝو يِمْࢠࣹ مَكّٝ ࢠࣹنْ : " إࣹ أٝن أَرَيْ إࣹ پَنْكَرࣹ، طࣹ گٝوتٝ إࣹ وِنْندࣹرࣹ ندࣹنْ أَدَاكِ أٝن إࣹ مُنْ?

(29) إࣹ أٝن أَرَيْ يٝتَّگٝلْ إࣹ وٝرْࢠࣹ ࢠࣹنْ? پَلٝطٝنْ كَدِ إࣹ لَاوٝلْ? هُووٝنْ كٝ نبَتِرْدُ مٝنْ كٝ أَݧِنِي?" كٝنٝ جَابَوٝلْ يِمْࢠࣹ مَكّٝ ࢠࣹنْ وٝنَانٝ، سِ وَنَا وِيدࣹ ࢠࣹ: "َدَّنْ مࣹنْ لࣹپْتࣹ الله طࣹنْ، سِ تَوِي كٝ أَ جࣹيَاطٝ إࣹ گٝونگُࢠࣹ ࢠࣹنْ".

(30) أٝ مَاكِ: جٝومِ" أَمْ، وَلَّمْ إࣹ هٝورࣹ يِمْࢠࣹ بٝنّٝوࢠࣹ ࢠࣹنْ".

(31) تُمَ ندࣹ إِمِّنَاࢠࣹ أَمࣹنْ ࢠࣹنْ أَدَّنّٝو إبراهيم وࣹلٝ-وࣹلٝ نگٝنْ، ࢠࣹ وِ"ِ: "مࣹنࣹنْ، كٝ مࣹنْ هَلْكٝوࢠࣹي يِمْࢠࣹ ندࣹي سَارࣹ. تَوْدࣹ پࣹلّࣹتْ، يِمْࢠࣹ مَيْرࣹ لَاتِكࣹ تٝوݧٝوࢠࣹ".

(32) أٝ مَاكِ: "كٝنٝ لوط نٝ تٝنْ!" ࢠࣹ وِئِ: "كٝ مࣹنࣹنْ ࢠُرِ أَنْندُدࣹ فِي وٝنْࢠࣹ تٝنْ. كٝ مَ مࣹنْ دَطْندُ مٝ، كَنْكٝ إࣹ يِمْࢠࣹ مَكّٝ ࢠࣹنْ، سِ وَنَا دࣹبّٝ مَكّٝ أٝنْ، أٝ جࣹيَامَ إࣹ هࣹلْكࣹتࣹيࢠࣹ ࢠࣹنْ".

(33) تُمْ ندࣹ إِمِّنَاࢠࣹ أَمࣹنْ ࢠࣹنْ هࣹوْتُنٝو كَ لوط، أٝ سَطْتِرَا فِي مَࢠّࣹ، ࢠࣹرْندࣹ مَكّٝ ࢠِطْتٝرِي فِي مَࢠّࣹ. ࢠࣹ وِئِ: "وَتَ أَ هُلُ، وَتَ أَ سُنٝ. كٝ مࣹنْ دَطْندٝوࢠࣹ مَ أَن إࣹ ࢠࣹيْنگُورࣹ مَا ندࣹنْ، سِ وَنَا دࣹبّٝ مَا أٝنْ، أٝ جࣹيَامَ إࣹ هَلْكَاࢠࣹ ࢠࣹنْ.

(34) كٝ مࣹنْ جِپِّنٝوࢠࣹ لࣹپْتࣹ إࣹ يِمْࢠࣹ ندࣹي سَارࣹ، إِمّٝرْدࣹ كَ كَمُّ، سَبُ كٝ ࢠࣹ يَلْتُنٝو إࣹ طٝفْتَارࣹ كٝنْ".

(35) گٝمْطِي مࣹنْ أَݖِّي إࣹ ندࣹنْ (سَارࣹ)، مَاندࣹ ࢠَنْنگُندࣹ، وٝنَنْندࣹ يِمْࢠࣹ هَقِّلٝوࢠࣹ ࢠࣹنْ.

(36) مࣹنْ( نُلِ) كَدِ إࣹ مَدْيَنَ، مُسِطّٝ مَࢠّࣹ أٝنْ، شُؤَيْبُ. أٝ مَاكِ: "كٝ أٝنٝنْ يٝ يِمْࢠࣹ أَمْ، رࣹوࣹي الله، جٝرْتٝرٝنْ ݧَلَنْندࣹ سَكِّتِيندࣹ ندࣹنْ، وَتَ وٝنِرِ إࣹ لࣹيْدِ ندِنْ، بٝنّٝوࢠࣹ".

(37) كٝنٝ ࢠࣹ فࣹنِّ مٝ. دِلَّنْندࣹ ندࣹنْ نَنْنگِ ࢠࣹ، ࢠࣹ وَطْتِنِ كَ كٝطٝولِ مَࢠّࣹ، كٝ ࢠࣹ جِݖِّيࢠࣹ-مَايِ.

(38) مࣹنْ( هَلْكِ) كَدِ آدِئࣹنْ، إࣹ سَامُدَئࣹنْ. گٝمْطِي ݖَانبْبࣹجࣹ مَࢠّࣹ طِنْ ࢠَنْنگَنِي أٝنْ. سࣹيْتَانࣹ ݖُطِنَنِ ࢠࣹ، گٝلّࣹ مَࢠّࣹ طࣹنْ، ندُ سَكِّي ࢠࣹ إࣹ لَاوٝلْ نگٝلْ; تَوِ لࣹ، كٝ ࢠࣹ رࣹنْتُنٝوࢠࣹ.

(39) مࣹنْ( هَلْكِ) كَدِ قَارُونَ، إࣹ فرعون إࣹ هَامَانَ. گٝمْطِي لࣹ، موسى أَدَّنِي ࢠࣹ ࢠَنْنگَنْندُيࣹيجِ، كٝنٝ ࢠࣹ تٝوْنِتِي كَ لࣹيْدِ، ࢠࣹ وٝنَانٝ لَاوٝتٝوࢠࣹ.

(40) كَلَ مَࢠّࣹ، مࣹنْ لࣹپْتِرْنٝ جُنُوبِ مُنْ طِنْ: نٝ إࣹ مَࢠّࣹ، ࢠࣹ مࣹنْ وُورْتِ هࣹنْندُ إࣹ مُنْ; نٝ إࣹ مَࢠّࣹ، ࢠࣹ هَاݖَانگٝ هَلْكِ; نٝ إࣹ مَࢠّࣹ، مٝ مࣹنْ مُتِنِ طُمْ كَ لࣹيْدِ، نٝ إࣹ مَࢠّࣹ كَدِ، ࢠࣹ مࣹنْ يٝولُنٝو. كٝنٝ نٝنْ، الله سِفٝرَالِ تٝوݧٝووٝ ࢠࣹ; سِ كٝ وٝنِ، كٝ كَمْࢠࣹ تٝوݧُنٝو كٝئࣹ مَࢠّࣹ.

(41) مِسَلْ ࢠࣹنْ جٝگِتِيࢠࣹ گَانِنْ الله رࣹوࣹتࣹيطِ، كٝ وَنٝ مَرِجَاوٝوَلْ(نجٝنبْلٝ) سَݖُّنگَلْ سُودُ. كٝ ࢠُرِ لٝؤُدࣹ نٝنْ إࣹ ݖُوطِ طِنْ، كٝ سُودُ مَرِجَاوٝ. سِنّٝ ࢠࣹ أَنْندُنٝ!

(42) پࣹلّࣹتْ، الله نٝ أَنْندِ كَلَ هُوندࣹ ندࣹ ࢠࣹ نٝدَّتَ گَانِنْ مَكّٝ. كٝ كَنْكٝ وٝنِ فٝولُطٝ، ݧࣹيݧٝ أٝنْ.

(43) طࣹي كٝ مِسࣹ طࣹ مࣹنْ پِيَنْتَ يِمْࢠࣹ ࢠࣹنْ; كٝ وٝنِ تُنْ، فَامَتَا طࣹنْ، سِ وَنَا أَنْندُࢠࣹ ࢠࣹنْ.

(44) الله تَگِرِي كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، إࣹ هٝورࣹ گٝونگَ. پࣹلّࣹتْ، وٝنِي طُمْ، مَاندࣹ، وٝنَنْندࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ.

(45) جَنْنگُ كٝنْ كٝ وَايِنَا إࣹ مَاطَ كَ دࣹفْتࣹرࣹ، ݧِيࢠَا إࣹ جُلْدࣹ ندࣹنْ. پࣹلّࣹتْ، جُولْدࣹ ندࣹنْ نٝ هَطَ پَنْكَرࣹ إࣹ كٝ أَݧِنِي. كٝ جَنْتَگٝلْ الله نگٝلْ ࢠُرِ (هُوندࣹ كَلَ) مَوْنُدࣹ. الله نٝ أَنْندِ كٝ گٝلّٝتٝنْ كٝنْ.

(46) وَتَ أٝن وࣹنْنجِرْ يِمْࢠࣹ دࣹفْتࣹ ࢠࣹنْ سِ وَنَا إࣹ نٝ ࢠُرِ إࣹ مٝڃِّرْدࣹ، سِ وَنَا هَرَ كٝ ࢠࣹنْ تٝوݧُࢠࣹ إࣹ مَࢠّࣹ. وِئٝنْ: مࣹنْ" گٝمْطِنِي ندࣹنْ جِپِّنَاندࣹ إࣹ أَمࣹنْ إࣹ طࣹنْ جِپِّنَاطࣹ إࣹ مٝنْ. رࣹوࣹتࣹيطٝ أَمࣹنْ أٝن إࣹ رࣹوࣹتࣹيطٝ مٝنْ أٝن كٝ گٝوتٝ، مࣹنࣹنْ نٝنْ، كٝ مࣹنْ جࣹبِّلَنِي ࢠࣹ مٝ".

(47) كٝ وَنٝ نٝنْ مࣹنْ جِپِّنِرِ إࣹ مَاطَ دࣹفْتࣹرࣹ ندࣹنْ. ࢠࣹنْ ࢠࣹ مࣹنْ أٝكِّ دࣹفْتࣹرࣹ ندࣹنْ، هِࢠࣹ گٝمْطِنِ ندࣹ. هِنٝ إࣹ مَࢠّࣹ أٝن گٝمْطِنْطٝ ندࣹ. يࣹدَّتَا آيࣹيجࣹ أَمࣹنْ طࣹنْ، سِ وَنَا هࣹيفࣹرْࢠࣹ ࢠࣹنْ.

(48) أَ وٝنَانٝ هِطَ جَنْنگَيْنٝو دࣹفْتࣹرࣹ أَدٝ مَيْرࣹ، أَ وِنْندِرْتَانٝ ݧَامٝ مَا نگٝنْ. هَرَيْنٝ، بٝنّٝوࢠࣹ ࢠࣹنْ سِكِّتّٝنٝ.

(49) سِ كٝ وٝنِ، كٝ ندࣹ آيࣹيجࣹ ࢠَنْنگُطࣹ إࣹ ࢠࣹرْطࣹ ࢠࣹنْ أٝكَّاࢠࣹ گَنْندَلْ. يࣹدَّتَا آيࣹيجࣹ أَمࣹنْ طࣹنْ، سِ وَنَا تٝوݧُࢠࣹ ࢠࣹنْ.

(50) ࢠࣹ وِئِ: "هَنّٝ جِپِّنࣹي إࣹ مَكّٝ مَاندࣹيجِ إِمّٝرْدࣹ كَ جٝومِ مَكّٝ?" مَاكُ: "َايَاجِ كٝ جٝومَمْ أَدَّتَ، أَنْندࣹي مِينْ، كٝ مِي جࣹرْتِنٝووٝ ࢠَنْنگُطٝ".

(51) كࣹرࣹ يٝنَالِ ࢠࣹ وٝنْندࣹ مࣹنْ جِپِّنِي إࣹ مَاطَ دࣹفْتࣹرࣹ ندࣹنْ هِندࣹ جَنْنگࣹي إࣹ مَࢠّࣹ? وٝنِي إࣹ طُمْ، يُرْمࣹيندࣹ إࣹ وَاجُ وٝنَنْندࣹ يِمْࢠࣹ گٝمْطِنٝوࢠࣹ.

(52) مَاكُ: الله" يٝنِي هَكُّندࣹ أَم إࣹ مٝوطٝنْ سࣹيدࣹي هِنٝ أَنْندِ كٝ وٝنِ كٝنْ كَ كَمُّولِ إࣹ كَ لࣹيْدِ. ࢠࣹنْ نٝنْ گٝمْطِمْࢠࣹ مࣹيرࣹ ندࣹنْ ࢠࣹ يࣹدِّ الله، كٝ ࢠࣹنْ وٝنِ سٝويُࢠࣹ ࢠࣹنْ.

(53) هِࢠࣹ هَوْجِرَنࣹ لࣹپْتࣹ طࣹنْ. سِ وَنَانٝ فِي لَجَلْ إِنَّانگَلْ، لࣹپْتࣹ طࣹنْ أَرْدِرَيْنٝ إࣹ مَࢠّࣹ جُهَلْ، هَرَ كَمْࢠࣹ ࢠࣹ سٝؤَا-َنْندُدࣹ.

(54) هِࢠࣹ هَوْجِرَنࣹ لࣹپْتࣹ طࣹنْ، هَرَ نٝنْ پࣹلّࣹتْ، جهنم كٝ فِيلْتِينگࣹ هࣹيفࣹرࣹࢠࣹ ࢠࣹنْ.

(55) ݧَلْندࣹ لࣹپْتࣹ طࣹنْ سُدِّرٝيْتَ ࢠࣹ سࣹنْنگٝ دٝوْ مَࢠّࣹ إࣹ كَ لࣹيْ كٝيْطࣹ مَࢠّࣹ، أٝ دَالَ نَ ࢠࣹ : "مࣹيطࣹي كٝ گٝلَّيْنٝطٝنْ كٝنْ!"

(56) كٝ أٝنٝنْ يٝ جࣹيَاࢠࣹ أَم گٝمْطِمْࢠࣹ! أَنْندࣹي لࣹيْدِ أَم ندِنْ كٝ يَاجُندِ. كٝ مِينْ وٝنِ كٝ رࣹوٝتٝنْ!

(57) كَلَ وٝنْكِي مࣹيطَيْ مَايْدࣹ. رࣹفْتِ كٝ كَ أَمࣹنْ رُتّࣹتࣹطٝنْ.

(58) ࢠࣹنْ گٝمْطِمْࢠࣹ ࢠࣹ گٝلِّ گٝلّࣹ مٝڃّࣹ، مَ مࣹنْ هٝطْنُ ࢠࣹ كَ الجنة إࣹ ݖُوطِ طِ ݖَنْطِ إِلَتَ سࣹنْنگٝ لࣹيْ مَجِّ، كٝࢠࣹ دُومٝتٝوࢠࣹ تٝنْ. إِسْكِنْ(هࣹسْكࣹيْ)° مٝڃّࣹرࣹ نجٝࢠْدِ گٝلُّنٝوࢠࣹ،

(59) ࢠࣹنْ مُݧِّيࢠࣹ، ࢠࣹ تُوگِي إࣹ جٝومِ مَࢠّࣹ.

(60) طُوطِي إࣹ رُوگَيْطِ طِ يٝوࢠٝتَاكٝ أَرْسِكࣹ مَجِّ! كٝ الله أَرْسِكَتَ طِ وٝنْندُدࣹ إࣹ أٝنٝنْ. كٝ كَنْكٝ وٝنِ نَنٝووٝ، أَنْندُطٝ.

(61) پࣹلّࣹتْ، سِ أَ لَنْندِكࣹ ࢠࣹ: "كٝ هٝمْنبٝ تَگِ كَمُّو- لِ طِنْ إࣹ لࣹيْدِ ندِنْ، أٝ إࣹلْتِ نَانگࣹ نگࣹنْ إࣹ لࣹوْرُ ندُ?" مَ ࢠࣹ وِئُ: "كٝ الله". كٝ هٝنّٝ ࢠࣹ فࣹرْلِتٝرَا (ࣹ گٝمْطِنگٝلْ نگٝلْ)?

(62) الله نٝ يَاجِنَنَ أَرْسِكࣹ أٝن مٝ أٝ مُويِ إࣹ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ، هِمٝ ࢠِتَّنَ مٝ. پࣹلّࣹتْ، الله كٝ أَنْندُطٝ كَلَ هُوندࣹ.

(63) پࣹلّࣹتْ، سِ أَ لَنْندِكࣹ ࢠࣹ: "كٝ هٝمْنبٝ جِپِّنِرِ ندِيَمْ كَ كَمُّ طَمْ أٝ وُورْنِتِرِ لࣹيْدِ ندِنْ ࢠَاوٝ ندِ يٝورُمࣹ?"، مَ ࢠࣹ وِئُ: "كٝ الله". مَاكُ: "يࣹتّٝورࣹ وٝودَنِي الله!". سِ كٝ وٝنِ، كٝ ࢠُرِ طُوطُدࣹ إࣹ مَࢠّࣹ هَقِّلْتَا.

(64) طَمْ نگُورْندَمْ أَدُنَ وٝنَالِ، سِ وَنَا نبࣹلࣹيطࣹ إࣹ فِجِندَارُ. پࣹلّࣹتْ، كٝ هٝطٝ لَاكَرَ نگٝنْ وٝنِ نگُورَندَمْ. سِنّٝ ࢠࣹ أَنْندُنٝ.

(65) سِ ࢠࣹ وَطِّكࣹ كَ لَانَ، ࢠࣹ نٝدَّ الله ࢠࣹ لَࢠِّنَنٝ مٝ نٝدَّاندُ ندُنْ. تُمَ ندࣹ أٝ دَطْندِ ࢠࣹ كَ نجٝرْندِ، أٝنْتُمَ ࢠࣹ سِرْكَ.

(66) نٝ ࢠࣹ يࣹدِّرَ كٝ مࣹنْ أٝكِّ ࢠࣹ كٝنْ، أَوَ يٝ ࢠࣹ دَكْمِتٝ! أَرَيْ كَࢠࣹ أَنْندَ.

(67) إࣹ ࢠࣹي يِئَالِ وٝنْندࣹ مࣹنْ وَطِي هٝرْمٝرْدࣹ هٝولْنِيندࣹ، يِمْࢠࣹ ࢠࣹنْ نٝ ݖِفࣹي(بِفْتࣹ) بَنْنگࣹ مَࢠّࣹ? إࣹ كٝ مࣹيرࣹ ندࣹنْ ࢠࣹ گٝمْطِنْتَ ࢠࣹ يࣹدَّ نࣹيمَاجِ الله طِنْ?

(68) إࣹ كٝ هٝمْنبٝ ࢠُرِ تٝوݧِرْدࣹ وَ أٝن فࣹفِندِيطٝ فࣹنَاندࣹ إࣹ هٝورࣹ الله، مَا أٝ فࣹنِّ گٝونگَ كَنْ ندࣹ كَ أَرُنٝو إࣹ مَكّٝ? إࣹ وٝنَا كَ جهنم وٝنِ جَاڃٝرْدࣹ يࣹدُّࢠࣹ ࢠࣹنْ?

(69) ࢠࣹنْ تِيطْنٝرْࢠࣹ إࣹ (وࣹلْتِنگٝلْ) مࣹنْ، مَ مࣹنْ فࣹوْنِرْࢠࣹ إࣹ لَاوِ أَمࣹنْ، كٝ پࣹلّࣹتْ، الله نٝ وٝنْندِ إࣹ مٝڃِّنٝوࢠࣹ ࢠࣹنْ.

30 Ar-Rum — 60 آيَة

(1) أَلِفْ، لَامْ، مِيمْ

(2) رُومِيَنْكٝوࢠࣹ ࢠࣹنْ فٝولَامَ،

(3) كَ ࢠُرِ ࢠَدَادࣹ إࣹ لࣹيْدِ ندِنْ، كَمْࢠࣹ كَدِ ࢠَاوٝ ࢠࣹ فٝولࣹيدࣹ، ࢠࣹ فٝولٝيَ،

(4) إࣹ ندࣹرْ دُوࢠِ سࣹيطَاجِ. كٝ الله وٝودَنِ فِيَاكُ (پٝولَلْ نگَلْ) أَدٝ إࣹ ࢠَاوٝ (مَگَّلْ). ندࣹنْ ݧَلْندࣹ، گٝمْطِمْࢠࣹ ࢠࣹنْ وࣹلْتٝتَ.

(5) بَلَّلْ الله نگَلْ. هِمٝ وَلَّ مٝ أٝ مُويِ. كٝ كَنْكٝ وٝنِ فٝولٝووٝ، هِنّٝتٝوطٝ.

(6) طُمْ كٝ فٝدٝورࣹ الله. الله لُنْندٝتَاكٝ فٝدٝورࣹ مَكّٝ ندࣹنْ. كٝنٝ كٝ ࢠُرِ طُوطُدࣹ إࣹ يِمْࢠࣹ ࢠࣹنْ ࢠࣹ أَنْندَا.

(7) هِࢠࣹ أَنْندِ كٝ فࣹيݧِ إࣹ نگُورْندَمْ أَدُنَ، كٝنٝ كَمْࢠࣹ إࣹ فِي لَاكَرَ أٝنْ، كٝ ࢠࣹ وࣹلْسِندِيࢠࣹ.

(8) إࣹ ࢠࣹي مِيجِتَاكِ فِي كٝئࣹ-مَࢠّࣹ? الله تَگِرَالِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ إࣹ كٝ وٝنِ هَكُّندࣹ طِنْ طِطِ، سِ وَنَا گٝونگَ إࣹ لَجَلْ إِنَّانگَلْ. أَنْندࣹي كٝ طُوطِ إࣹ يِمْࢠࣹ ࢠࣹنْ، كٝ يࣹدُّࢠࣹ هَوْرٝيْگٝلْ إࣹ جٝومِ مَࢠّࣹ.

(9) إࣹ ࢠࣹي يِلَّاكِ كَ لࣹيْدِ ࢠࣹ تَسْكٝو كٝ هٝنّٝ بَتَّنࣹ ࢠࣹنْ أَدِنٝوࢠࣹ ࢠࣹ وَؤُنٝو? ࢠࣹنْ وٝنُنٝ ࢠُرْࢠࣹ ࢠࣹ دٝولࣹ ࢠࣹ رࣹمِ إࣹ لࣹيْدِ ندِنْ ࢠࣹ وُورِ إࣹ مَيْرِ ࢠُرِ كٝ ࢠࣹي وُورِ إࣹ مَيْرِ كٝنْ. نُلَاࢠࣹ مَࢠّࣹ ࢠࣹنْ أَدَّنِ ࢠࣹ ࢠَنْنگَنْندُيࣹيجِ. الله وٝنَالِ تٝوݧٝووٝ ࢠࣹ، كٝنٝ نٝنْ كٝ كٝئࣹ مَࢠّࣹ ࢠࣹ وٝنّٝ إࣹ تٝوݧُدࣹ.

(10) رࣹفْتِ كٝ وٝنُنٝو بَتَّنࣹ بٝنُّࢠࣹ ࢠࣹنْ، كٝ بٝنَندࣹ فِي ࢠࣹ فࣹنِّي آيࣹيجࣹ الله طࣹنْ، ࢠࣹ لَاتِي إࣹ جَلْكِتُگٝلْ طࣹ.

(11) كٝ الله فُطِّ تَگُ نگُنْ; رࣹفْتِ أٝ رُتَّ نگُ; رࣹفْتِ كٝ كَ مَكّٝ رُتّࣹتࣹطٝنْ.

(12) ݧَلْندࣹ دَرْنگَلْ نگَلْ دَرِي، بٝمْࢠࣹ ࢠࣹنْ تَڃٝتٝ.

(13) وٝودَنٝيْتَاࢠࣹ إࣹ كَفِدِيطِ مَࢠّࣹ طِنْ تࣹفٝوجِ، ࢠࣹ وٝنِينٝ يࣹدُّࢠࣹ كَفِدُࢠࣹ ࢠࣹنْ.

(14) ݧَلْندࣹ دَرْنگَلْ نگَلْ دَرِي،ندࣹنْ ݧَلْندࣹ ࢠࣹ سࣹيرْتٝيَيْ.

(15) سِ كٝ ࢠࣹنْ گٝمْطِمْࢠࣹ ࢠࣹ گٝلِّ گٝلّࣹ مٝڃّࣹ، ࢠࣹنْ طٝنْ وٝنٝيَيْ إࣹ بࣹيلِ الجنة() هِࢠࣹ سَطْنࣹي(وࣹلْتِنࣹ).

(16) سِ تَوِي نٝنْ كٝ ࢠࣹنْ يࣹدُّࢠࣹ ࢠࣹ فࣹنِّ آيࣹيجࣹ أَمࣹنْ طࣹنْ إࣹ هَوْرُگٝلْ نگٝلْ إࣹ لَاكَرَ، ࢠࣹنْ طٝنْ كٝ تَوْنٝيْتࣹيࢠࣹ إࣹ لࣹپْتࣹ طࣹنْ.

(17) سُبُّنْهِنَنࣹي الله، تُمَ هِيرٝتٝنْ إࣹ تُمَ وࣹيتٝتٝنْ.

(18) كٝ كَنْكٝ يࣹتّٝورࣹ وٝودَنِ كَ كَمُّولِ إࣹ كَ لࣹيْدِ، إࣹ كِيكِيطࣹ إࣹ سَاگَ ݧَلّٝتٝنْ.

(19) هِمٝ يَلْتِنَ كٝ وُورِ إࣹ ندْرْ كٝ مَايِ، أٝ يَلْتِنَ كَدِ كٝ مَايِ إࣹ ندࣹرْ كٝ وُورِ. هِمٝ وُورْنِتَ لࣹيْدِ ندِنْ ࢠَاوٝ ندِ يٝورُدࣹ. كٝ وَنٝ نٝنْ يَتَــِٔنِرْتࣹ- طٝنْ (ࣹ گࣹنَالࣹ مٝنْ).

(20) هِنٝ جࣹيَا إࣹ مَاندࣹيجِ مَكّٝ طِنْ: وٝنْندࣹ أٝ تَگِي أٝن إِمّٝرْدࣹ إࣹ نبُمْ-نبُلْدِ، رࣹفْتِ جَكَ كٝ أٝن ࢠَنْندِكࣹ سَاكٝتٝوࢠࣹ إࣹ لࣹيْدِ ندِنْ .

(21) هِنٝ جࣹيَا إࣹ مَاندࣹيجِ مَكّٝ طِنْ : وٝنْندࣹ أٝ تَگِرَنِي أٝن إࣹ وٝنْكِيجِ مٝنْ طِنْ، رࣹسٝندِرْࢠࣹ، فِي يٝ أٝن دࣹيڃُ إࣹ مَࢠّࣹ، أٝ وَطِ هَكُّندࣹ مٝنْ گِگّٝلْ إࣹ يُمࣹيندࣹ. وٝنِي إࣹ طُمْ، مَاندࣹيجِ وٝنَنْندࣹ يِمْࢠࣹ مِيجِتٝتٝوࢠࣹ.

(22) هِنٝ جࣹيَا إࣹ مَاندࣹيجِ مَكّٝ طِنْ: تَگُگٝلْ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، إࣹ لُوتٝندِرْگٝلْ طࣹمْطࣹ مٝنْ إࣹ نبَادِيجِ مٝنْ. وٝنِي إࣹ طُمْ، مَاندࣹيجِ أَندُࢠࣹ ࢠࣹنْ.

(23) هِنٝ جࣹيَا إࣹ مَاندࣹيجِ مَكّٝ طِنْ: طٝيْنگٝلْ مٝنْ نگٝلْ جࣹمَّ إࣹ ݧَلٝرْمَ، إࣹ طَࢠُّگٝلْ مٝنْ ࢠُرَلْ مَكّٝ. وٝنِي إࣹ طُمْ، مَاندࣹيجِ وٝنَنْندࣹ يِمْࢠࣹ نَنٝوࢠࣹ.

(24) هِنٝ جࣹيَا إࣹ مَاندࣹيجِ مَكّٝ طِنْ: هِمٝ هٝلِّرَ أٝن مَڃࣹ طࣹنْ كُلٝلْ إࣹ تَمَؤُ (تٝࢠٝ)، هِمٝ جِپِّنِرَ كَ كَمُّ ندِيَمْ، أٝ وُورْنِتِرَ طَمْ لࣹيْدِ ندِنْ ࢠَاوٝ ندِ يٝورُدࣹ. وٝنِي إࣹ طُمْ، مَاندࣹيجِ، وٝنَنْندࣹ يِمْࢠࣹ هَقِّلٝوࢠࣹ.

(25) هِنٝ جࣹيَا إࣹ مَاندࣹيجِ مَكّٝ طِنْ: ندࣹ كَمُّ نگُنْ إࣹ لࣹيْدِ ندِنْ دَرٝتٝ إࣹ يَمِرٝورࣹ مَكّٝ. رࣹفْتِ سِ أٝ نٝدِّي أٝن نٝدَّاندُ كَ لࣹيْدِ، إࣹ جَكَ أٝنٝنْ هِطٝنْ يَلْتَ (ࣹ گࣹنَالࣹ مٝنْ).

(26) هِنٝ وٝودَنِ (لَامَگٝلْ، إࣹ تَگُگٝلْ إࣹ تٝپِّتَگٝلْ) ࢠࣹنْ وٝنْࢠࣹ كَ كَمُّولِ إࣹ كَ لࣹيْدِ: كَلَ مَࢠّࣹ كٝ طٝفْتِيࢠࣹ مٝ.

(27) كٝ كَنْكٝ وٝنِ فُطُّطٝ تَگُ نگُنْ، رࣹفْتِ أٝ رُتِّتَ نگُ; كٝ طُمْ ࢠُرِ نࣹوَنَاندࣹ مٝ. هِنٝ وٝودَنِ سِفَ ࢠُرْطٝ تٝووُدࣹ كَ كَمُّولِ إࣹ كَ لࣹيْدِ. كٝ كَنْكٝ وٝنِ أٝن فٝولُطٝ، ݧࣹيݧُطٝ أٝنْ.

(28) أٝ پِيَنِي أٝن مِسَلْ إِمّٝرْدࣹ إࣹ كٝئࣹ-مٝنْ: هَرَ نٝ وٝودَنِ أٝن إࣹ ࢠࣹنْ ࢠࣹ جࣹيْرُطٝنْ ݧَامࣹ مٝنْ، إِمّٝرْدࣹ إࣹ كَفِدَلْ إࣹ كٝنْ كٝ مࣹنْ أَرْسِكِ أٝنْ، هَرَ هِطٝنْ فٝتَ إࣹ طُمْ، هِطٝنْ هُلِرَ ࢠࣹ وَنٝ هُلِرْتٝنْ كٝئࣹ-مٝنْ? كٝ وٝنِ نِي مࣹنْ فࣹنْسِتِرْتَ آيࣹيجࣹ طࣹنْ، وٝنَنْندࣹ يِمْࢠࣹ هَقِّلٝوࢠࣹ.

(29) سِ كٝ وٝنِ، تٝوݧُࢠࣹ ࢠࣹنْ جٝكِّرِي بࣹلࣹيطࣹ مَࢠّࣹ طࣹنْ كٝ أَلْدَا إࣹ گَنْندَلْ. كٝ هٝمْنبٝ فࣹوْنَتَ مٝ الله مَجِّنِ? أَلَنَا ࢠࣹ وَلّٝوࢠࣹ.

(30) فࣹوْتِنِرْ يࣹيسٝ مَا نگٝنْ إࣹ دِينَ كَنْ ݧِيࢠُگٝلْ إࣹ مُنْ . طُمْ كٝ تَگُ الله نگُ أٝ تَگِرِ يِمْࢠࣹ ࢠࣹنْ إࣹ مُنْ - وَطْتِتُگٝلْ أَلَا إࣹ تَگُ الله نگُنْ -. كٝ طُمْ وٝنِ دِينَ فٝݖِّيكَ كَنْ; كٝنٝ ࢠُرْࢠࣹ طُوطُدࣹ إࣹ يِمْࢠࣹ ࢠࣹنْ أَنْندَا.

(31) هَرَ كٝ أٝن رُتّٝتٝوࢠࣹ إࣹ مَكّٝ; هُلٝنْ مٝ، ݧِنّٝنْ جُلْدࣹ ندࣹنْ، وَتَ أٝن جࣹيࣹ إࣹ سِرْكٝوࢠࣹ ࢠࣹنْ.

(32) (وَنٝ) ࢠࣹيَ سࣹنْندُنٝوࢠࣹ دِينَ مُنْ كَنْ، ࢠࣹ وٝنِ پࣹݖّࣹ-پࣹݖّࣹ، كَلَ فࣹدّࣹ نٝ وࣹلْتٝرَ كٝ وٝنِ كَ مُنْ كٝنْ.

(33) سِ سَطْتࣹيندࣹ مࣹمِي يِمْࢠࣹ ࢠࣹنْ، ࢠࣹ نٝدِّرَ جٝومِ مَࢠّࣹ كٝ ࢠࣹ رُتِّتِيࢠࣹ إࣹ مَكّٝ; رࣹفْتِ سِ أٝ مࣹيطِنِي ࢠࣹ يُمࣹيندࣹ إِمّٝرْدࣹ إࣹ مَكّٝ، إࣹ جَكَ فࣹدّࣹ گٝو إࣹ مَࢠّࣹ نٝ كَفِدَ إࣹ جٝومِ مُنْ،

(34) يٝࢠࣹ يࣹدُّ كٝ مࣹنْ أٝكِّ ࢠࣹ كٝنْ. "َوَ دَكْمِتࣹي، أَرْمَ أَنْندٝنْ".

(35) كَا مࣹنْ جِپِّنْ إࣹ مَࢠّࣹ هُجَّ (دࣹفْتࣹرࣹ)، هَرَ هٝندࣹ يࣹوْتَ فِي كٝ ࢠࣹ سِرْكَيْنٝو كٝنْ?

(36) سِ مࣹنْ مࣹيطِنِرِي يِمْࢠࣹ ࢠࣹنْ نࣹيمَ، ࢠࣹ وࣹلْتٝرٝو مٝ. سِ بٝنࣹ مࣹمِرِي ࢠࣹ سَبُ كٝ جُوطࣹ مَࢠّࣹ طࣹنْ أَرْدِنِرِ كٝنْ، إࣹ جَكَ هِࢠࣹ تَڃٝو.

(37) إࣹ ࢠࣹ يِئَالِ وٝنْندࣹ الله نٝ يَݧَّنَ أَرْسِكࣹ أٝن مٝ أٝ مُويَنِ، أٝ ࢠِلِّنَ? وٝنِي إࣹ طُمْ، مَاندࣹيجِ وٝنَنْندࣹ يِمْࢠࣹ گٝمْطِنٝوࢠࣹ.

(38) جٝنُّ جٝمْ-ࢠَدِيدٝ أٝ هَكّࣹ مَكّٝ أٝنْ، إࣹ ࢠِلِّيࢠࣹ إࣹ ࢠِي-لَاوٝلْ. طُمْ ࢠُرِ مٝڃُّدࣹ وٝنَندࣹ ࢠࣹينْ فَالَاࢠࣹ يَرْلُيࣹي الله أٝنْ; كٝ ࢠࣹنْ وٝنِ مَلٝيْتࣹيࢠࣹ ࢠࣹنْ.

(39) كَلَ كٝ جٝنُّطٝنْ إࣹ رِبَ، فِي أٝن ࢠࣹيْدٝرْ إࣹ جَوْلࣹ يِمْࢠࣹ ࢠࣹنْ، طُمْ ࢠࣹيْدٝتَاكٝ كَ الله، كَلَ نٝنْ كٝ أٝكُّطٝنْ إࣹ جَكَّ، مٝ فَاندٝرِطٝنْ وࣹلَيࣹي الله، ࢠࣹنْ طٝنْ، كٝ سٝوَنٝيْتࣹيࢠࣹ نجٝࢠْدِ ࢠࣹنْ.

(40) كٝ الله وٝنِ تَگُطٝ أٝنْ، رࣹفْتِ أٝ أَرْسِكِ أٝنْ، رࣹفْتِ أٝ وَرَ أٝنْ، رࣹفْتِ أٝ وُورْنِتَ أٝنْ. هَرَ نٝ إࣹ كَفِدَاطِ مٝنْ طِنْ وَطَيْطِ گٝطُّمْ إࣹ طُمْ? سࣹنَيࣹيوٝو-دَنِ مٝ! أٝ تٝووِي إࣹ كٝنْ كٝ ࢠࣹ سِرْكَتَ.

(41) بٝنࣹ أٝن فࣹيݧِرِ كَ نجٝورْندِ إࣹ كَ مَاجࣹ، سَبُ كٝ جُوطࣹ يِمْࢠࣹ ࢠࣹنْ فَگِّتِي كٝنْ; فِي يٝ أٝ مࣹيطِنْ ࢠࣹ يٝگَ إࣹ كٝ ࢠࣹ هُووِ; بࣹلَجٝؤٝ، ࢠࣹ رُتّٝتٝ.

(42) مَاكُ: "يِيلࣹي كَ لࣹيْدِ، ندَارٝنْ كٝ هٝنّٝ بَتَّنࣹ أَدِنٝوࢠࣹ ࢠࣹنْ وَؤُنٝو. تَوْنٝ كٝ ࢠُرِ طُوطُدࣹ إࣹ مَࢠّࣹ كٝ سِرْكٝوࢠࣹ".

(43) فࣹوْتِنِرْ يࣹيسٝ مَا نگٝنْ إࣹ دِينَ فٝݖِّيكَ كَنْ، أَدٝ ݧَلَاندࣹ أَرْدࣹ إِمّٝرْدࣹ كَ الله ندࣹ رُتّࣹتَاكࣹ. ندࣹ ݧَلْندࣹ، ࢠࣹ سࣹيدَيْ:

(44) كَلَ يࣹدُّطٝ، هَرَيْ گࣹدِّ مَكّٝ طِنْ نٝ فَوِي مٝ. كَلَ نٝنْ گٝلُّطٝ مٝڃّࣹرࣹ، هَرَيْ كٝ كٝئࣹ-مَࢠّࣹ ࢠࣹ هࣹبِلَنِ الجنة()،

(45) فِي يٝ أٝ يٝࢠُ ࢠࣹنْ گٝمْطِمْࢠࣹ ࢠࣹ گٝلِّ مٝڃُّطِ، إࣹ ࢠُرَلْ مَكّٝ نگَلْ. كَنْكٝ أٝنْ، أٝ يِطَا هࣹيفࣹرْࢠࣹ ࢠࣹنْ.

(46) هِنٝ جࣹيَا إࣹ مَاندࣹيجِ مَكّٝ طِنْ: وُرْتُگٝلْ مٝ كࣹنࣹيلِ طِنْ وࣹوْلِنٝوجِ، إࣹ فِي يٝ أٝ مࣹيطِنْ أٝن إࣹ يُرْمࣹيندࣹ مَكّٝ ندࣹنْ، إࣹ نٝ لَاطࣹ طࣹنْ دٝگِرَ إࣹ يَمِرٝورࣹ مَكّٝ، إࣹ فِي يٝ أٝ طَࢠُّ إࣹ ࢠُرَلْ مَكّٝ. بࣹلَجٝؤٝ، أٝن يࣹتَّيْ مٝ .

(47) گٝمْطِي مࣹنْ نُلِي أَدٝ مَاطَ نُلَاࢠࣹ هَا إࣹ يِمْࢠࣹ مَࢠّࣹ ࢠࣹنْ، ࢠࣹ أَدَّنِ ࢠࣹ ࢠَنْنگَنْندُيࣹيجِ. مࣹنْ يٝتِّي إࣹ ࢠࣹنْ بٝنُّࢠࣹ; تَوِ نٝ جٝجِّ إࣹ أَمࣹنْ فَابَگٝلْ گٝمْطِمْࢠࣹ ࢠࣹنْ.

(48) كٝ الله وٝنِ نُلٝووٝ كࣹنࣹيطِ طِنْ، طِ سٝگَّ دُولࣹ طࣹنْ; أٝ سَاكِتِرَ طُمْ كَ وࣹييٝ نٝ أٝ يِطِرِ; أٝ وَطْتَ طُمْ تَڃࣹ-تَڃࣹ. أَ يِئَيْ تٝࢠٝ نگٝنْ نٝ يَلْتَ هَكُّدࣹ مُنْ. سِ أٝ هࣹوْتِنِي نگٝ إࣹ أٝن مٝ أٝ مُويِ إࣹ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ، إࣹ جَكَ كَمْࢠࣹ هِࢠࣹ وࣹلْتٝو.

(49) گٝمْطِي ࢠࣹ لَاتِنٝكࣹ أَدٝ (تٝࢠٝ) نگٝنْ جِپِّنࣹيدࣹ إࣹ مَࢠّࣹ، كٝ تَڃِنٝوࢠࣹ.

(50) ندَارِي إࣹ بَتَّنࣹ يُرْمࣹيندࣹ الله ندࣹنْ، نٝ أٝ وُورْنِتٝرْتَ لࣹيْدِ ࢠَاوٝ ندِ يٝورُدࣹ. پࣹلّࣹتْ، وَطٝووٝ طُمْ أٝنْ، كٝ وُورْنِتٝووٝ مَايْࢠࣹ ࢠࣹنْ، كَنْكٝ كٝ أٝ هَتَّنْطٝ كَلَ هُوندࣹ.

(51) هَيْ سِ مࣹنْ نُلِي هࣹنْندُ، ࢠࣹ يِئِ گࣹسࣹ مَࢠّࣹ طࣹنْ نٝ سٝيْطِ، ࢠࣹ جٝكَّيْ ࢠَاوٝ طُمْ، هِࢠࣹ يࣹدَّ.

(52) پࣹلّࣹتْ، أَن أَ نَنِنْتَا مَايْࢠࣹ ࢠࣹنْ، أَ نَنِنْتَا فَاطُࢠࣹ ࢠࣹنْ كَدِ نٝدَّاندُ ندُنْ، سِ ࢠࣹ رُنْتِيمَ ࢠࣹ هُݖِّنِي ࢠَبّࣹ (مَࢠّࣹ).

(53) أَ وٝنَالِ فࣹوْنٝووٝ وُمْࢠࣹ ࢠࣹنْ إࣹ مَجّࣹرࣹ مَࢠّࣹ. أَ نَنِنْتَا سِ وَنَا ࢠࣹنْ گٝمْطِمْࢠࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ، هَرَ كٝ ࢠࣹ جࣹبِّلِيࢠࣹ.

(54) كٝ الله وٝنِ تَگُطٝ أٝن إࣹ لٝئࣹرࣹ، رࣹفْتِ أٝ وَطِ ࢠَاوٝ لٝئࣹرࣹ ندࣹنْ ݖࣹمْنبࣹ، رࣹفْتِ أٝ وَطِ ࢠَاوٝ ݖࣹمْنبࣹ طࣹنْ لٝئࣹرࣹ إࣹ پُرِ(دَندَنْندِ). هِمٝ تَگَ كٝ أٝ مُويِ، كٝ كَنْكٝ وٝنِ أَنْندُطٝ، هَتَّنْطٝ.

(55) ݧَلْندࣹ دَرْنگَلْ نَلْ دَرٝتٝو، بٝمْࢠࣹ ࢠࣹنْ وٝوندَيْ ࢠࣹ ݧِيࢠَالِ سِ وَنَا سَانگَ. كٝ وَنٝ نِي ࢠࣹ وٝنَنٝو فࣹرْلِسࣹيدࣹ (ࣹ گٝونگَ).

(56) أٝكَّاࢠࣹ گَنْندَلْ إࣹ گٝمْطِنَلْ ࢠࣹنْ وِئَ: "گٝمْطِي أٝن ݧِيࢠِي إࣹ كٝ الله وِنْندِ كٝنْ، هَا ݧَلْندࣹ أُمِّتَلْ. كٝ ندࣹي وٝنِ ݧَلَاندࣹ أُمِّتَلْ، كٝنٝ أٝنٝنْ أٝن أَنْندَانٝ".

(57) ندࣹنْ ݧَلْندࣹ، نگَنْتٝوجِ طِنْ نَفٝيْتَا تٝوݧُࢠࣹ ࢠࣹنْ، وَنَا كَمْࢠࣹ وٝنِ كٝ طَࢠِّرࣹتࣹي وࣹلَيࣹي الله .

(58) گٝمْطِي مࣹنْ پِيَنِي يِمْࢠࣹ ࢠࣹنْ إࣹ ندࣹ أَلْقُرَانَا- رࣹ، كَلَ مِسَلْ. پࣹلّࣹتْ، سِ أَ أَدَّنِي ࢠࣹ آيَ، ࢠࣹنْ يࣹدُّࢠࣹ وِئَيْ: "ٝن وٝنَالِ، سِ وَنَا مࣹيرࣹنْتࣹيࢠࣹ".

(59) كٝ وَنٝ نِي الله نٝتِرَ إࣹ ࢠࣹرْطࣹ ࢠࣹنْ ࢠࣹ أَنْندَا.

(60) مُݧّٝ، پࣹلّࣹتْ، فٝدٝورࣹ الله ندࣹنْ، كٝ گٝونَ. وَتَ ࢠࣹ هٝيْنُمَ ࢠࣹنْ هٝيْفِنࣹ.
