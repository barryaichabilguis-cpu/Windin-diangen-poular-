39 Az-Zumar — 75 آيَة

(1) جِپَّگٝلْ دࣹفْتࣹرࣹ ندࣹنْ، كٝ إِمّٝرْدࣹ كَ الله، فٝولُطٝ، ݧࣹيݧٝ أٝنْ.

(2) مࣹنْ جِپِّنِيرِي دࣹفْتࣹرࣹ ندࣹنْ إࣹ مَاطَ، إࣹ هٝورࣹ گٝونگَ. رࣹوُ الله، لَࢠِّنَنَا مٝ دِينَ كَنْ.

(3) أَنْندࣹي، كٝ الله وٝودَنِ دِينَ لَاࢠُكَ كَنْ. ࢠࣹنْ نٝنْ ࢠࣹ ࢠࣹيَ جٝگِتٝرِ يِطْࢠࣹ گَانِنْ مَكّٝ (هَرَ هِࢠࣹ وِئَ): "مࣹنْ أَلَا رࣹوِرْدࣹ طِ، سِ وَنَا تُنْ نٝ طِ ࢠَنِّرَ مࣹنْ كَ الله، ࢠَدَلْ". پࣹلّࣹتْ، الله ݧَاوٝيَيْ هَكُّندࣹ مَࢠّࣹ، إࣹ كٝنْ كٝ ࢠࣹ وٝنِ لُرُّدࣹ إࣹ مُنْ. پࣹلّࣹتْ، الله فࣹوْنَتَا أٝنْ وٝنْطٝ فࣹوْنٝووٝ، طُوطُطٝ يࣹدُّگٝلْ.

(4) سِنّٝ الله يِطُنٝ، ندࣹ أٝ سِفٝرْتَ ࢠِطّٝ، أٝ سُࢠَيْنٝ كٝ أٝ يِطِ وٝو، إࣹ كٝ أٝ تَگِ كٝنْ. سࣹنَيࣹي وٝودَنِي مٝ ! كٝ كَنْكٝ وٝنِ الله، بَجّٝ، فٝولُطٝ أٝنْ.

(5) أٝ تَگِرِي كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، گٝونگَ.هِمٝ سُدِّرَ جࣹمَّ أٝنْ ݧَلٝرْمَ أٝنْ، أٝ سُدِّرَ ݧَلٝرْمَ أٝنْ جࣹمَّ أٝنْ. أٝ إࣹلْتِ نَانگࣹ نگࣹنْ إࣹ لࣹوْرُ ندُنْ، كَلَ مَجِّ نٝ دٝگَ هَا إࣹ لَجَلْ تَڃَانگَلْ. أَنْندࣹي، كٝ كَنْكٝ وٝنِ فٝولُطٝ، هَفٝرٝووٝ أٝنْ.

(6) أٝ تَگِي أٝنْ إࣹ فِتَّاندُ وٝوتُرُ، رࣹفِ أٝ وَطِ إࣹ ندُنْ، گࣹنْندٝ مَيْرُ. أٝ جِپِّنَنِ أٝنْ إࣹ نࣹيمٝرَاطِ طِنْ، نٝونࣹيجِ جࣹيتَتِ. هِمٝ تَگَ أٝنْ كَ تࣹيگࣹ يُمَّ مٝنْ إࣹنْ، تَگُ ࢠَاوٝ تَگُ، إࣹ ندࣹرْ نِࢠࣹ تَتِ. كٝ أٝنْ مٝوطٝنْ الله، وٝنِ جٝومِ مٝنْ ! كٝ كَنْكٝ لَامُ وٝودَنِ. رࣹوࣹتࣹيطٝ أَلَا، سِ وَنَا الله. كٝ هٝنّٝ فࣹرْلِتٝرَطٝنْ إࣹ رࣹوُگٝلْ مٝ ?

(7) سِ أٝنْ يࣹدِّي، أَنْندٝنْ الله، كٝ يٝندِنِيطٝ إࣹ مٝنْ. أٝ جَࢠَنْتَا جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ، كࣹيفࣹرَاكُ. سِ أٝنْ يࣹتِّي مٝ نٝنْ، أٝ وࣹلْتَنٝو أٝنْ طُمْ.هَيْ گٝوتٝ رٝوندَنْتَاكٝ گٝطّٝ وٝو دٝنْنگَلْ مُمْ. رࣹفْتِ كٝ كَ جٝومِ مٝنْ، وٝنِ رُتّٝرْدࣹ مٝنْ; أٝ يࣹيتِتٝو أٝنْ، فِي كٝنْ كٝ گٝلَّيْنٝطٝنْ. پࣹلّࣹتْ، كٝ أٝ أَنْندُطٝ، كٝ وٝنِ إࣹ ࢠࣹرْطࣹ.

(8) سِ لٝرَا مࣹمِي نࣹطّٝ أٝنْ، أٝ نٝدِّرَ جٝومِ مَكّٝ، رُتَّگٝلْ إࣹ مَكّٝ. رࣹفْتِ سِ أٝ أٝكِّي مٝ مٝڃّࣹرࣹ، إِمّٝرْدࣹ كَ مَكّٝ، أٝ يࣹجِّتَ كٝ أٝ نٝدَّيْنٝو كٝنْ أَدٝ طُمْ، أٝ وَطَنَ الله كَفِدِيطِ، فِي نٝ أٝ مَجِّنِرَ (يِمْࢠࣹ ࢠࣹنْ) إࣹ لَاوٝلْ مَكّٝ نگٝلْ. مَاكُ: "دَكْمِتٝرْ كࣹيفࣹرَاكُ مَا نگُنْ، سࣹيطَ. پࣹلّࣹتْ، كٝ أَ جࣹيَاطٝ إࣹ يِمْࢠࣹ يِيتࣹ ࢠࣹنْ".

(9) كَا كٝ أٝنْ يَنْكِنٝتٝوطٝ تُمْنبٝ جࣹمَّ، كٝ أٝ سُجّٝووٝ، دَرٝتٝوطٝ (نَافࣹ)، هَرَ هِمٝ لَاكَرَ، هِمٝ جٝرْتِي يُرْمࣹيندࣹ جٝومِ مَكّٝ ندࣹنْ... مَاكُ: "هَرَ أَنْندُࢠࣹ ࢠࣹنْ فٝتَيْ إࣹ ࢠࣹ أَنْندَا ࢠࣹنْ?" أَنْندُ كٝ وَاجِتٝتٝو، كٝ جٝومِرَاࢠࣹ هَقِّلّࣹ ࢠࣹنْ.

(10) مَاكُ: "كٝ أٝنٝنْ يٝ جࣹيَاࢠࣹ أَمْ گٝمْطِمْࢠࣹ، هُلࣹي جٝومِ مٝنْ". هِنٝ وٝودَنِ مٝڃِّمْࢠࣹ (گٝلّࣹ) ࢠࣹنْ إࣹ أٝو أَدُنَ، نبَرْجَارِ. لࣹيْدِ الله ندِنْ كٝ يَاجُندِ. أَنْندࣹي مُݧِّيࢠࣹ ࢠࣹنْ تِمِّنِرَنْتࣹ نجٝࢠْدِ مَࢠّࣹ ندِنْ، كٝ أَلْدَا إࣹ لِمٝورࣹ.

(11) مَاكُ: "مِينْ مِي يَمِرَامَ رࣹوُگٝلْ الله، لَࢠِّنَنَ مٝ دِينَ كَنْ،

(12) مِي يَمِرَا كَدِ، وٝنُگٝلْ أَرَنٝ إࣹ جࣹبِّلِيࢠࣹ ࢠࣹنْ".

(13) مَاكُ : "مِينْ مِطُ هُلِ، سِ مِي يࣹدِّي جٝومِ أَمْ، لࣹپْتࣹ ݧَلَاندࣹ مَوْندࣹ".

(14) مَاكُ: "كٝ الله مِي رࣹوَتَ، مِي لَࢠِّنَنْ مٝ دِينَ أَمْ.

(15) رࣹوࣹي كٝ يِطُطٝنْ أٝنٝنْ، هَرَ وَنَا كَنْكٝ. مَاكُ: "كٝ وٝنِ هَيْرُࢠࣹ ࢠࣹنْ، كٝ ࢠࣹنْ ࢠࣹ تَوَتَ ݧَلْندࣹ دَرْنگَلْ، ࢠࣹ هَيْرِنِي كٝؤࣹ مَࢠّࣹ، إࣹ ࢠࣹيْنگُورࣹيجِ مَࢠّࣹ". أَنْندࣹي كٝ ندࣹنْ وٝنِ هَيْرࣹرࣹ ࢠَنْنگُندࣹ ندࣹنْ.

(16) هِنٝ وٝودِرَنِ ࢠࣹ سࣹنْنگٝ دٝوْ مَࢠّࣹ، كِپّٝودِ يِيتࣹ، إࣹ كِپّٝودِ سࣹنْنگٝ لࣹيْ مَࢠّࣹ. كٝ طُمْ وٝنِ كٝ الله هُلْࢠِنِرْتَ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ. كٝ أٝنٝنْ يٝ جࣹيَاࢠࣹ أَمْ، كٝرِ هُلࣹي لَمْ!

(17) ࢠࣹنْ وٝطِّتٝرْࢠࣹ رࣹوُگٝلْ بࣹوْنٝوجِ طِ، ࢠࣹ رُتِّي إࣹ الله، وࣹلٝ-وࣹلٝ نٝ وٝودَنِ ࢠࣹ. وࣹوْلِنِرْ جࣹيَاࢠࣹ أَمْ ࢠࣹنْ،

(18) نَنٝوࢠࣹ كٝنْنگٝلْ ࢠࣹنْ، ࢠࣹ جٝكَّ كٝ ࢠُرِ مٝڃُّدࣹ إࣹ مَگّٝلْ. كٝ ࢠࣹنْ وٝنِ ࢠࣹ الله فࣹوْنِ ࢠࣹنْ، كٝ ࢠࣹنْ وٝنِ جٝومِرَاࢠࣹ هَقِّلّࣹ ࢠࣹنْ.

(19) إࣹ كٝ أٝنْ مٝ كٝنْنگٝلْ لࣹپْتࣹ جٝجِّ إࣹ مُنْ... إࣹ كٝ أَنْ دَطْندَتَ وٝنْطٝ كَ يِيتࣹ?

(20) سِ كٝ ࢠࣹنْ هُلُࢠࣹ جٝومِ مُنْ، هِنٝ وٝودَنِ ࢠࣹ كُࢠࣹيجࣹ كَ الجنة إࣹ كُࢠࣹيجࣹ مَهَاطࣹ دٝوْ مَجّࣹ، هَرَ ݖَنْطِ نٝ إِلَ لࣹيْ مَجّࣹ. كٝ فٝدٝورࣹ الله ! الله نٝنْ فِرْتَتَا فٝدٝورࣹ.

(21) إࣹ أَ يِئَالِ، وٝنْندࣹ الله سَاڃِنِرِي ندِيَمْ كَ كَمُّ، أٝ نَادِ طَمْ إࣹ ندࣹرْ ࢠُلِّ كَ لࣹيْدِ; رࣹفْتِ أٝ يَلْتِنِرَ طَمْ رࣹمُرُ سࣹيدُندُ نٝونࣹ، رࣹفْتِ ندُ طَيْطَ، يِئَا هِندُ سٝيْطِ; رࣹفْتِ أٝ وَطَ طُمْ كٝ مُنْݖَا. پࣹلّࣹتْ، وٝنِي إࣹ طُمْ، وَاجُ، وٝنَنْندࣹ جٝومِرَاࢠࣹ هَقِّلّࣹ ࢠࣹنْ.

(22) إࣹ كٝ أٝنْ مٝ الله يَݧَّنِ ࢠࣹندْرࣹ مُنْ ندࣹنْ، فِي لِسْلَامُ، هَرَ هِمٝ إࣹ ندَيْگُ إِمّٝرْدࣹ كَ جٝومِ مَكّٝ... بٝنࣹ وٝودَنِي ࢠࣹنْ يٝورُࢠࣹ ࢠࣹرْطࣹ مُنْ، إࣹ جَنْتَگٝلْ الله. ࢠࣹنْ وٝنِي إࣹ مَجّࣹرࣹ ࢠَنْنگُندࣹ.

(23) الله جِپِّنِي كٝ ࢠُرِ مٝڃُّدࣹ إࣹ يࣹوْتࣹرࣹ، دࣹفْتࣹرࣹ نَنْندُندࣹ ݧَاوٝوجࣹ ، طِمِّتُندࣹ. ندࣹ گُرِ ࢠࣹنْ هُلٝوࢠࣹ جٝومِ مُنْ رِࢠِرْتَ (نَنُگٝلْ ندࣹ)، رࣹفْتِ گُرِ إࣹ ࢠࣹرْطࣹ مَࢠّࣹ دࣹيڃِرَ جَنْتٝورࣹ الله ندࣹنْ. كٝ طُمْ وٝنِ پࣹيوَلْ الله نگَلْ أٝ فࣹوْنِرْتَ أٝنْ مٝ مُويَنِ. كَلَ نٝنْ مٝ الله مَجِّنِ، هَرَيْ فࣹوْنٝووٝ مٝ أَلَا.

(24) إࣹ كٝ أٝنْ رࣹينْتَرِيطٝ يࣹيسٝ مُنْ نگٝنْ، إࣹ لࣹپْتࣹ بٝنْطࣹ طࣹنْ ݧَلْندࣹ دَرْنگَلْ... تٝوݧُࢠࣹ ࢠࣹنْ وِئَنࣹي أٝنْتُمَ: "مࣹيطࣹي كٝنْ كٝ وٝنُنٝطٝنْ فَگِّتَنَادࣹ".

(25) أَدِنٝوࢠࣹ ࢠࣹنْ فࣹنِّينٝ، لࣹپْتࣹ طࣹنْ أَرْدِرَنِ ࢠࣹ، كَ ࢠࣹ سٝؤَا-أَنْندُدࣹ.

(26) الله مࣹيطِنِرِ ࢠࣹ كٝيࣹيرَ، كَ نگُورْندَمْ أَدُنَ. پࣹلّࣹتْ نٝنْ، كٝ لࣹپْتࣹ لَاكَرَ طࣹنْ نٝنْ، ࢠُرِ مَوْنُدࣹ، سِ تَوْنٝ هِࢠࣹ أَنْندُنٝو!

(27) گٝمْطِي مࣹنْ پِيَنِي يِمْࢠࣹ ࢠࣹنْ، إࣹ ندࣹي القرآن، كَلَ مِسَلْ، بࣹلَجٝؤٝ، ࢠࣹ وَاجِتٝرَ.

(28) القرآن (جَنْنگࣹتࣹيندࣹ) إࣹ هَالَ آرَبُ، ندࣹ أَلْدَا إࣹ أٝوݧَارࣹ، فِي نٝ ࢠࣹ گٝمْطِرَ.

(29) الله وَطِي مِسَلْ، گٝرْكٝ مٝ لُرُّࢠࣹ هَوْتِ طُمْ، هِࢠࣹ پٝوطٝندِرَ مٝ، إࣹ گٝرْكٝ لَاࢠَنْطٝ گٝرْكٝ گٝو : هَرَ ࢠࣹنْ طِطٝ فٝتَيْ إࣹ مِسَلْ? يࣹتّٝورࣹ وٝودَنِي الله! كٝنٝ ࢠُرْࢠࣹ طُوطُدࣹ إࣹ مَࢠّࣹ، ࢠࣹ أَنْندَا.

(30) پࣹلّࣹتْ، كٝ أَ مَايٝووٝ، كَمْࢠࣹ كَدِ كٝ ࢠࣹ مَايٝوࢠࣹ;

(31) رࣹفْتِ أٝنٝنْ، ݧَلْندࣹ دَرْنگَلْ، كٝ كَ جٝومِ مٝنْ، يࣹدّٝندِرٝيْتٝنْ.

(32) إࣹ كٝ هٝمْنبٝ ࢠُرِ تٝوݧْدࣹ وَ أٝنْ فࣹنُطٝ إࣹ هٝورࣹ الله، مَا أٝ فࣹنِّ گٝونگَ كَنْ ندࣹ كَ أَرُنٝو إࣹ مَكّٝ? إࣹ وٝنَا جهنم وٝنِ جَاڃٝرْدࣹ يࣹدُّࢠࣹ ࢠࣹنْ?

(33) أٝنْ نٝنْ أَرْدُطٝ إࣹ گٝونگَ، أٝ گٝونگِنِ طُمْ، كٝ ࢠࣹنْ وٝنِ گٝمْطُࢠࣹ ࢠࣹنْ.

(34) هِنٝ وٝودَنِ كَلَ كٝ ࢠࣹ فَالࣹتࣹي، كَ جٝومِ مَࢠّࣹ; كٝ طُمْ نجٝࢠْدِ مٝڃِّنٝوࢠࣹ ࢠࣹنْ.

(35) فِي نٝ الله يَوْتِرَنَ ࢠࣹ، كٝ ࢠُرِ بٝنْندࣹ إࣹ كٝ ࢠࣹ گٝلِّ كٝنْ، أٝ يٝࢠِتِرَ ࢠࣹ نجٝࢠْدِ مَࢠّࣹ ندِنْ، كٝ ࢠُرِ مٝڃُّدࣹ إࣹ كٝ ࢠࣹ گٝلَّيْنٝو كٝنْ.

(36) الله سِفٝرَالِ يٝنُگٝلْ جࣹيَاطٝ مَكّٝ أٝنْ? هِࢠࣹ هُلْࢠِنِرࣹ طِنْ طِ وٝنَا مٝ. كَلَ مٝ الله مَجِّنِ، هَرَيْ فࣹوْنٝووٝ أَلَنَا مٝ.

(37) كَلَ نٝنْ مٝ الله فࣹوْنِ، مَجِّنٝووٝ مٝ أَلَا. الله سِفٝرَالِ وٝنْندࣹ فٝولُطٝ، جٝمْ لࣹپْتࣹ?

(38) پࣹلّࣹتْ، سِ أَ لَمْندِكࣹ ࢠࣹ: "كٝ هٝمْنبٝ تَگِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ?"، مَ ࢠࣹ وِئُ: "كٝ الله". مَاكُ: "يࣹيتࣹي لَمْ فِي طِنْ طِ رࣹوٝتٝنْ گَانِنْ الله، سِ الله فَاندَنِكࣹ لَمْ لٝرَّ، هَرَ طِ هُنْݖَنَيْ لَمْ لٝرَّ مَكّٝ كَنْ? مَا سِ أٝ فَاندَنِكࣹ لَمْ يُرْمࣹيندࣹ، تَوْ سِ طِ هَطَيْ لَمْ يُرْمࣹيندࣹ مَكّٝ ندࣹنْ? مَاكُ: "الله يٝنِي لَمْ، كٝ إࣹ مَكّٝ هٝولْنٝتٝوࢠࣹ ࢠࣹنْ تُوگِنٝتٝو".

(39) مَاكُ: "كٝ أٝنٝنْ يٝ يِمْࢠࣹ أَمْ، گٝلّࣹي إࣹ لَاوٝلْ مٝنْ، مِينْ كٝ إࣹ لَاوٝلْ أَمْ مِي گٝلِّرْتَ. أَرَيْ كَ أَنْندٝنْ.

(40) كٝ هٝمْنبٝ لࣹپْتࣹ هٝيْنَيْطࣹ أَرٝيَتْ إࣹ مُوطُمْ، لࣹپْتࣹ دُومِيطࣹ جِپّٝو إࣹ مَكّٝ.

(41) مࣹنْ جِپِّنِرِي دࣹفْتࣹرࣹ ندࣹنْ إࣹ مَاطَ، فِي يِمْࢠࣹ ࢠࣹنْ، إࣹ هٝورࣹ گٝونگَ. كَلَ فࣹيوُطٝ، كٝ هٝورࣹ مَكّٝ أٝ فࣹيوَنِ ; كَلَ كَدِ مَجُّطٝ، كٝ هٝورࣹ مَكّٝ أٝ مَجَّنِ. أَ وٝنَالِ وَكِيلْ إࣹ دٝوْ مَࢠّࣹ.

(42) الله هِنٝ ڃࣹتِّتَ پِتَّالِ طِنْ، سَانگَ مَايْدࣹ مَجِّ، إࣹ طِنْ طِ مَايَالِ، كَ طٝيْنگٝلْ مَجِّ. أٝ جٝگِتٝو طِنْ طِ ݧَاوَنِ مَايْدࣹ، أٝ أَݖِّتَ طِيَ، هَا إࣹ لَجَلْ تَڃَانگَلْ. پࣹلّࣹتْ، مَاندࣹيجِ وٝنِ إࣹ طُمْ، وٝنَنْندࣹ يِمْࢠࣹ مِيجِتٝتٝوࢠࣹ.

(43) كَا ࢠࣹ جٝگِتِ تࣹفٝوجِ، گَانِنْ الله? مَاكُ: "هَيْ سِ تَوِي طِ جࣹيَا هُوندࣹ، طِ هَقِّلْتَا كَدِ?"

(44) مَاكُ: "تࣹفٝورࣹ ندࣹنْ فٝوْ، كٝ الله هࣹيرَنِي. كٝ كَنْكٝ وٝودَنِ لَامَگٝلْ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ. رࣹفْتِ كٝ كَ مَكّٝ رُتّࣹتࣹطٝنْ".

(45) سِ فِي الله تُنْ جَنْتَامَ، ࢠࣹرْࢠࣹ ࢠࣹنْ ࢠࣹ گٝمْطِنَا لَاكَرَ ࢠࣹنْ، تَمٝو. سِ فِي طِيَ، طِ وٝنَا مٝ كَنْكٝ الله جَنْتَامَ، إࣹ جَكَ يٝو، هِࢠࣹ وࣹوْلَ.

(46) مَاكُ: "يَا الله، فُطُّطٝ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، أَنْندُطٝ وِرْنِيطِ إࣹ فࣹيݧُطِ، كٝ أَنْ ݧَاوَتَ هَكُّندࣹ جࣹيَاࢠࣹ مَا ࢠࣹنْ، كٝنْ كٝ ࢠࣹ لُوتٝندِرَيْنٝو إࣹ مُنْ".

(47) سِ تَوْنٝ نٝ وٝودَنّٝ تٝوݧُࢠࣹ ࢠࣹنْ، كٝ وٝنِ كٝنْ كَ لࣹيْدِ فٝوْ، إࣹ يࣹرُ مُنْ، وٝنْندُدࣹ إࣹ مُمْ، ࢠࣹ سٝتّٝرَيْنٝ طُمْ، بٝنࣹ لࣹپْتࣹ طࣹنْ، ݧَلْندࣹ دَرْنگَلْ; فࣹيݧِرَنَ ࢠࣹ كَ الله، كٝنْ كٝ ࢠࣹ تِجَّنٝوكِ.

(48) بٝنࣹيجِ طِ ࢠࣹ فَگِّتِنٝو طِنْ فࣹݧَنَ ࢠࣹ، كٝ ࢠࣹ جَلْكِتَيْنٝو كٝنْ هُوࢠِتَ ࢠࣹ.

(49) سِ لٝرَّ هࣹࢠِي نࣹطّٝ أٝنْ، أٝ نٝدَّ مࣹنْ. رࣹفْتِ سِ مࣹنْ أٝكِّي مٝ نࣹيمَ، إِمّٝرْدࣹ كَ أَمࣹنْ، أٝ وِئَ: "أَنْندࣹي كٝ إࣹ گَنْندَلْ أَمْ مِي أٝكِّيرَ طُمْ". طُمْ كٝ جَرَّبُيࣹي كَنْ; كٝنٝ كٝ ࢠُرِ طُوطُدࣹ إࣹ مَࢠّࣹ، ࢠࣹ أَنْندَا.

(50) گٝمْطِي ࢠࣹنْ أَدِنٝوࢠࣹ ࢠࣹ وِئِينٝ طُمْ. كٝنٝ كٝنْ كٝ ࢠࣹ فَگِّتٝتٝنٝو، نَفَانِ ࢠࣹ.

(51) بٝنࣹيجِ طِ ࢠࣹ فَگِّتٝتٝنٝو طِنْ مࣹمِ ࢠࣹ. تٝوݧُࢠࣹ ࢠࣹنْ إࣹ مَࢠّࣹ، كٝنْ كٝ ࢠࣹ وٝنُنٝو إࣹ فَگِّتَگٝلْ، مࣹمٝيَيْ ࢠࣹ، ࢠࣹ وٝنَالِ لَاوٝتٝوࢠࣹ.

(52) إࣹ ࢠࣹ أَنْندَا وٝنْندࣹ الله نٝ يَݧَّنَ أَرْسِكࣹ أٝنْ مٝ أٝ مُويَنِ، أٝ ࢠِلِّنَ مٝ أٝ مُويَنِ ? پࣹلّࣹتْ، مَاندࣹيجِ وٝنِي إࣹ طُمْ، وٝنَنْندࣹ يِمْࢠࣹ گٝمْطِنٝوࢠࣹ.

(53) مَاكُ: "كٝ أٝنٝنْ يٝ جࣹيَاࢠࣹ أَمْ فَنْتِمْࢠࣹ إࣹ هٝورࣹ مُنْ، وَتَ أٝنْ تَڃٝ يُرْمࣹيندࣹ الله ندࣹنْ. پࣹلّࣹتْ، الله كٝ هَفٝرٝووٝ جُنُبَاجٝ طِنْ فٝوْ. كٝ كَنْكٝ وٝنِ هَفٝرٝووٝ، هِنّٝتٝوطٝ أٝنْ".

(54) تُوبࣹي رُتّٝطٝنْ كَ جٝومِ مٝنْ، جࣹبِّلَنٝطٝنْ مٝ، أَدٝ لࣹپْتࣹ طࣹنْ أَرْدࣹ إࣹ مٝنْ، هَرَ أٝنْ فَابࣹتَاكࣹ.

(55) جٝكّࣹي كٝ ࢠُرِ مٝڃُّدࣹ إࣹ كٝ جِپِّنَا إࣹ مٝنْ، إِمّٝرْدࣹ كَ جٝومِ، أَدٝ لࣹپْتࣹ طࣹنْ أَرْدُدࣹ أٝنْ، جُهَلْ، هَرَ أٝنْ سٝؤَا أَنْندُدࣹ.

(56) فِي وَتَ وٝنْكِي وِئٝيْ: "إࣹي نِمْسࣹ أَمْ، إࣹ كٝ مِي وࣹلْسِندِي إࣹ فِي الله. پࣹلّࣹتْ، مِي جࣹيَنٝ إࣹ ࢠࣹنْ جَلْكِتٝوࢠࣹ".

(57) مَا كِ وِئَ: "سِ أَلَهْ فࣹوْنُنٝ لَمْ، مِي وٝنَنْيْنٝ جࣹيَاطٝ إࣹ گٝمْطُࢠࣹ ࢠࣹنْ".

(58) مَا كِ وِئَ، تُمَ كِ يِئَتَ لࣹپْتࣹ طࣹنْ: "سِ فࣹيرࣹ رُتَّگٝلْ نٝ وٝودَنّٝو لَمْ، مِي جࣹيࣹي إࣹ مٝڃِّنٝوࢠࣹ ࢠࣹنْ".

(59) "گَسَتَا ! تَوْدࣹ آيࣹيجࣹ أَمْ طࣹنْ أَرُنٝ إࣹ مَاطَ، فࣹنُّطَا طࣹ، مَوْنِنْتِنِطَا، لَاتِطَا جࣹيَاطٝ إࣹ يࣹدُّࢠࣹ ࢠࣹنْ".

(60) ݧَلْندࣹ دَرْنگَلْ، أَ يِئٝيَ فࣹنّٝنٝوࢠࣹ إࣹ هٝورࣹ الله ࢠࣹنْ، هِنٝ ࢠَوْلِ گࣹيݖࣹ. وٝنَا كَ جهنم وٝنِ جَاڃٝرْدࣹ مَوْنِنْتِنِيࢠࣹ ࢠࣹنْ?

(61) الله نَ دَطْندِرَ گٝمْطُࢠࣹ ࢠࣹنْ، سَبُ مَلٝورࣹ مَࢠّࣹ ندࣹنْ. بٝنࣹ مࣹمَتَا ࢠࣹنْ، وَنَا كَمْࢠࣹ وٝنِ سُنٝيْتٝوࢠࣹ.

(62) كٝ الله وٝنِ تَگُطٝ كَلَ هُوندࣹ، كٝ كَنْكٝ كَدِ وٝنِ فَوْتِيطٝ كَلَ هُوندࣹ.

(63) كٝ كَنْكٝ سٝتِّرْطࣹ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ وٝودَنِ. يࣹدُّࢠࣹ آيࣹيجࣹ الله ࢠࣹنْ نٝنْ، كٝ ࢠࣹنْ وٝنِ هَيْرُࢠࣹ(پࣹرْتُࢠࣹ) ࢠࣹنْ.

(64) مَاكُ: "إࣹ كٝ وٝورِ الله، يَمِرْتٝنْ مِي، يٝ مِي رࣹوُ، أٝنٝنْ ࢠࣹي مَجُّࢠࣹ?"

(65) گٝمْطِي وَهَيِنَامَ إࣹ مَاطَ، إࣹ ࢠࣹنْ أَدِنٝوࢠࣹ مَ: "سِ أَ سِرْكِي، پࣹلّࣹتْ، گٝلّࣹ مَا طࣹنْ بٝنَيْ جࣹيࣹطَا إࣹ هَيْرُࢠࣹ ࢠࣹنْ.

(66) أَلَهْ رࣹوْ الله، جࣹيࣹطَا إࣹ يࣹتّٝوࢠࣹ (مٝ) ࢠࣹنْ.

(67) ࢠࣹ مَوْنِنِرَالِ الله، نٝ أٝ فٝتِ مَنِنِرࣹيدࣹ، هَرَ لࣹ، لࣹيْدِ ندِنْ فٝوْ، كٝ تَمَنْندࣹ مَكّٝ، ݧَلْندࣹ دَرْنگَلْ، كَمُّولِ طِنْ كَدِ تَگِّرࣹي ݧَامٝ مَكّٝ. سࣹنَيࣹي وٝودَنِي مٝ ! أٝ تٝووِي إࣹ كٝنْ كٝ ࢠࣹ سِرْكَنْتَ مٝ.

(68) وُتّࣹي كَ الله، كَلَ كٝ وٝنِ كَ كَمُّولِ إࣹ كَ لࣹيْدِ كٝنْ فَطّࣹي، سِ وَنَا ࢠࣹنْ ࢠࣹ الله مُويَنِ دَطُگٝلْ . رࣹفْتِ وُتّࣹي إࣹ مَيْرُ طِطَࢠٝلْ، تَوَ كٝ ࢠࣹ دَرِيࢠࣹ هَبِّتِي.

(69) لࣹيْدِ ندِنْ جَلْبِرَ ندَيْگٝ جࣹيْدٝ ندِ أٝنْ; تَالْكُرُ ندُنْ وَلِّنࣹي، أَنَّبَاࢠࣹ إࣹ سࣹيدࣹيࢠࣹ مَࢠّࣹ ࢠࣹنْ أَدّࣹي; گٝونگَ ݧَاوِرࣹي هَكُّندࣹ مَࢠّࣹ، وَنَا ࢠࣹ تٝوݧࣹتࣹيࢠࣹ;

(70) وٝنْكِ كَلَ يٝࢠࣹي كٝ گٝلِّ كٝنْ. كٝ كَنْكٝ الله ࢠُرِ أَنْندُدࣹ كٝ ࢠࣹ گٝلَّتَ.

(71) يࣹدُّࢠࣹ ࢠࣹنْ سٝگِّرٝيࣹي كَ جهنم دࣹنْتࣹ-دࣹنْتࣹ. هَا سِ ࢠࣹ هࣹوْتِي نگࣹ، دَمࣹ مَگّࣹ أُدِّتࣹي، رࣹينُࢠࣹ نگࣹ ࢠࣹنْ وِئَنَ ࢠࣹ: "إࣹ نُلَاࢠࣹ جࣹيَاࢠࣹ إࣹ مٝؤٝنْ هࣹوْتَانٝ أٝنْ، جَنْنگَيْنٝوࢠࣹ آيࣹيجࣹ جٝومِ مٝنْ، إࣹ مٝوطٝنْ، ࢠࣹ رࣹنْتِنَ أٝنْ هَوْرُگٝلْ إࣹ ندࣹي ݧَلَاندࣹ مٝنْ?" ࢠࣹ وِئَ: "كٝ نٝنْ تِگِرِ، كٝنٝ كٝنْنگٝلْ لࣹپْتࣹ نگٝلْ، جٝجِّينٝ إࣹ يࣹدُّࢠࣹ ࢠࣹنْ".

(72) ࢠࣹ وِئَنࣹي: "نَاتِرࣹي دَمࣹ جهنم طࣹنْ لُتّٝنْ تٝنْ. نگࣹ بٝنِي جَاڃٝرْدࣹ مَوْنِنْتِنِيࢠࣹ ࢠࣹنْ !".

(73) هُلٝوࢠࣹ جٝومِ مُنْ ࢠࣹنْ كَدِ طٝوِتِرࣹي كَ الجنة، دࣹنْتࣹ-دࣹنْتࣹ. هَا سِ ࢠࣹ هࣹوْتِي مٝ، دَمࣹ مَكّٝ أُدِّتࣹي، رࣹينُࢠࣹ مٝ ࢠࣹنْ وِئَنَ ࢠࣹ: "كِسِيࣹي وٝنِي إࣹ مٝنْ، أٝنْ لَاࢠِي، نَاتِرࣹي مٝ، دُومَگٝلْ".

(74) ࢠࣹ وِئَ أٝنْتُمَ: "يࣹتّٝورࣹ وٝودَنِي الله، لَاتِنَنْطٝ مࣹنْ فٝدٝورࣹ مُنْ، أٝ رٝنِنِ مࣹنْ لࣹيْدِ الجنة ندِنْ، مࣹطࣹنْ وࣹيرِرَ إࣹ الجنة أٝنْ، كَلَ كَ مࣹنْ يِطِ". نبَرْجَارِ گٝلُّنٝوࢠࣹ مٝڃّࣹرࣹ ࢠࣹنْ مٝڃِّي.

(75) يِئٝيَا الملائكة، هِࢠࣹ تَارِي أَرْسِ كِنْ، هِࢠࣹ سُبُّنْهِنٝرَ يࣹتُّگٝلْ جٝومِ مَࢠّࣹ. گٝونگَ ݧَاوِرࣹي هَكُّندࣹ مَࢠّࣹ. وِئࣹي أٝنْتُمَ: "يࣹتّٝورࣹ وٝودَنِي الله، جࣹيْطٝ تࣹگࣹيفٝ نگٝنْ".

40 Ghafir — 85 آيَة

(1) هَا، مِيمْ.

(2) جِپَّگٝلْ دࣹفْتࣹرࣹ ندࣹنْ، كٝ إِمّٝرْدࣹ كَ الله، فٝولُطٝ، أَنْندُطٝ أٝنْ.

(3) هَفٝرٝووٝ جُنُبَاجِ طِنْ، جَࢠٝووٝ تُوبُوبُيࣹي، سَطْتُطٝ لࣹپْتࣹ، جٝمْ يَاجࣹيندࣹ. رࣹوࣹتࣹيطٝ أَلَا، سِ وَنَا كَنْكٝ. كٝ كَ مَكّٝ وٝنِ رُتّٝرْدࣹ ندࣹنْ.

(4) أَلَا كٝ وࣹنْنجَتَ إࣹ آيࣹيجࣹ الله طࣹنْ، سِ وَنَا يࣹدُّࢠࣹ ࢠࣹنْ. وَتَ أَ هٝدِتٝرْ كٝ ࢠࣹ دِلِّرْتَ كٝنْ كَ لࣹيْدِ.

(5) أَدٝ مَࢠّࣹ، يِمْࢠࣹ نوح ࢠࣹنْ إࣹ فࣹدّࣹيجِ طِنْ ࢠَاوٝ مَࢠّࣹ، فࣹنُّنٝ (نُلَاࢠࣹ مُنْ ࢠࣹنْ)، مٝفْتࣹ كَلَ هِمِّرِ نَنْنگُگٝلْ نُلَاطٝ مُنْ أٝنْ. ࢠࣹ وࣹنْنجِرِ مࣹيرࣹ، فِي نٝ بٝنِّتِرَ (گٝونگَ) كَنْ. كٝنٝ مِي نَنْنگِ ࢠࣹ. تَسْكِي نٝ نَنْنگَلْ أَمْ نگَلْ وَطْتِنِرْنٝو !

(6) كٝ وَنٝ نِي كٝنْنگٝلْ جٝومِ مَا نگٝلْ، جٝجِّرِ إࣹ يࣹدُّࢠࣹ ࢠࣹنْ : "وٝنْندࣹ پࣹلّࣹتْ، كٝ ࢠࣹ يِمْࢠࣹ يِيتࣹ".

(7) ࢠࣹنْ (الملائكة) رٝوندِيࢠࣹ أَرْسِكِ كِنْ، إࣹ وٝنْࢠࣹ بَنْنگࣹ مَكِّ ࢠࣹنْ، هِࢠࣹ سُبُّنْهِنٝرَ يࣹتُّگٝلْ جٝومِ مَࢠّࣹ، ࢠࣹ گٝمْطِنَ مٝ، ࢠࣹ إِنْسِنٝرَنَ گٝمْطِمْࢠࣹ ࢠࣹنْ (وِيدࣹ): "جٝومِ أَمࣹنْ، أَ يَݧِّرِي كَلَ هُوندࣹ، يُرْمࣹيندࣹ إࣹ گَنْندَلْ. يَوْتَنْ ࢠࣹنْ تُوبُࢠࣹ، ࢠࣹ جٝكِّ لَاوٝلْ مَا نگٝلْ، دَطْندَا ࢠࣹ كَدِ لࣹپْتࣹ دُلْࢠٝووࣹ نگࣹنْ.

(8) جٝومِ أَمࣹنْ، نَادُࢠࣹ كَ الجنة ݧِيࢠَلْ، طِنْ طِ فٝدُطَا ࢠࣹ، إࣹ كَلَ مٝڃُّطٝ إࣹ بَابِرَاࢠࣹ مَࢠّࣹ، إࣹ گࣹنْندِرَاࢠࣹ مَࢠّࣹ، إࣹ گࣹيْنگٝلْ مَࢠّࣹ. پࣹلّࣹتْ، كٝ أَنْ وٝنِ فٝولُطٝ، ݧࣹيݧُطٝ أٝنْ.

(9) دَطْندَاࢠࣹ كُوطࣹ إࣹ بٝنْطࣹ طࣹنْ. كَلَ مٝ دَطْندُطَا كُوطࣹ مُنْ بٝنْطࣹ طࣹنْ ندࣹنْ ݧَلْندࣹ، هَرَيْ پࣹلّࣹتْ، أَ يُرْمِيِ مٝ". كٝ طُمْ وٝنِ مَلٝورࣹ مَوْنُندࣹ ندࣹنْ.

(10) يࣹدُّࢠࣹ ࢠࣹنْ نٝدِّرٝيْتࣹ: "كٝ تِكّࣹرࣹ الله ندࣹنْ ࢠُرِ مَوْنُدࣹ، إࣹ دِي كٝ تِكَّنْطٝنْ كٝئࣹ مٝنْ كٝنْ، تُمَ نٝدّࣹتࣹنٝطٝنْ إࣹ گٝمْطِنَلْ، يࣹدّٝنْ.

(11) ࢠࣹ وِئَ: "جٝومِ أَمࣹنْ، أَ وَرِي مࣹنْ لَاوِ طِطِ، وُورْنِتُطَا مࣹنْ لَاوِ طِطِ: مࣹنْ قِرِّكࣹ جُنُوبِ أَمࣹنْ طِنْ. تَوْ سِ لَاوٝلْ يَلْتِرْنگٝلْ هِنٝ وٝودِ?".

(12) "طُمْ مٝوطٝنْ، كٝ تَوْدࣹ سِ الله تُنْ نٝدَّنٝومَ، أٝنْ يࣹدّٝنْ; سِ أٝ كَفِدَامَ، گٝمْطِنٝنْ. ݧَاوٝورࣹ ندࣹنْ كٝ الله هࣹيرَنِي، تٝووُطٝ، مَوْطٝ أٝنْ".

(13) كٝ كَنْكٝ وٝنِ هٝلّٝووٝ أٝنْ مَاندࣹيجِ مَكّٝ طِنْ، أٝ جِپِّنِرَنَ أٝنْ تٝࢠٝ كَ كَمُّ. كٝنٝ تَسْكٝتَاكٝ، سِ وَنَا رُتّٝتٝوطٝ إࣹ الله .

(14) رࣹوِرࣹي الله، لَࢠِّنِرَنگٝلْ مٝ دِينَ كَنْ، هَيْ سِ هࣹيفࣹرْࢠࣹ ࢠࣹنْ يِطَا.

(15) كٝ أٝ تٝووُطٝ دَرْجَاجِ، جࣹيْطٝ أَرْسِ كِنْ. هِمٝ جِپِّنِرِــَٔ وَهَيُ أٝنْ، يَمِرٝورࣹ مَكّٝ، إࣹ أٝنْ مٝ أٝ مُويَنِ إࣹ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ، نٝ أٝ جࣹرْتِنِرَ ݧَلْندࣹ كَوْرَلْ نگَلْ،

(16) ݧَلْندࣹ ࢠࣹ فࣹيݧٝيْتَ. هُوندࣹ سُوطٝتَاكٝ الله، إࣹ فࣹي مَࢠّࣹ. كٝ هٝمْنبٝ لَامُ وٝودَنِ هَنْندࣹ? كٝ الله بَجّٝ، فٝولُطٝ أٝنْ.

(17) هَنْندࣹ، وٝنْكِي كَلَ يٝࢠࣹتࣹ كٝ فَگِّتِي. تٝوݧࣹ وٝو أَلَا هَنْندࣹ، پࣹلّࣹتْ، الله كٝ يَاوُطٝ ݧَاوٝورࣹ.

(18) جࣹرْتْنِرْتْࢠࣹ، ݧَلَاندࣹ دِلّࣹرࣹ ندࣹنْ، تُمَ ࢠࣹرْطࣹ طࣹنْ ڃَوْتِرْتَ كَ كٝنٝندِ، هَرَ هِࢠࣹ (هࣹيوُࢠࣹ) نگٝڃٝ. گِطٝ وٝو أَلَنَا تٝوݧٝوࢠࣹ ࢠࣹنْ، وَنَا تࣹفٝووٝ جَࢠَنْتࣹيطٝ.

(19) هِمٝ أَنْندِ جَنْفَ گِتࣹ، إࣹ كٝ ࢠࣹرْطࣹ طࣹنْ سُوطِ.

(20) الله كٝ گٝونگَ ݧَاوِرْتَ. طِ ࢠࣹ رࣹوَتَ طِنْ نٝنْ، گَانِنْ مَكّٝ، هَيْ هُوندࣹ طِ ݧَاوِرْتَا. پࣹلّࣹتْ، كٝ كَنْكٝ وٝنِ نَنٝووٝ، يِئٝووٝ أٝنْ.

(21) هَرَ ࢠࣹي يِيلَاكِ كَ لࣹيْدِ، ࢠࣹ ندَارَ نٝ بَتَّنࣹ أَدٝ مَبّࣹ ࢠࣹنْ وَأُنٝو? ࢠࣹنْ لࣹ نٝ ࢠُرُنٝو تِيطُدࣹ دٝولࣹ إࣹ بَتّࣹيجِ، كَ لࣹيْدِ. الله نَنْنگِتِرِ ࢠࣹ جُنُوبِ مَࢠّࣹ طِنْ، ࢠࣹ هࣹࢠَانٝ دَطْندٝووٝ ࢠࣹ إࣹ (لࣹپْتࣹ) الله طࣹنْ.

(22) كٝ وَطِ طُمْ، كٝ تَوْدࣹ نُلَاࢠࣹ مَࢠّࣹ ࢠࣹنْ أَدَّيْنٝ ࢠࣹ ࢠَنْنگَنْندُيࣹيجِ، ࢠࣹ يࣹدَّ. الله هَلْكِ ࢠࣹ. پࣹلّࣹتْ، كٝ أٝ دٝولْنُطٝ، سَطْتُطٝ لࣹپْتࣹ.

(23) گٝمْطِي مࣹنْ نُلِرِي موسى آيࣹيجࣹ أَمࣹنْ طࣹنْ، إࣹ هُجَّ ࢠَنْنگُطٝ.

(24) هَا إࣹ فرعون، إࣹ هَامَانَ، إࣹ قَارُونَ. كٝنٝ ࢠࣹ وِئِ: "كٝ نبِلࣹيجٝ، طُوطُطٝ فࣹنَاندࣹ!".

(25) ندࣹ أٝ أَدَّنّٝو ࢠࣹ گٝونگَ كَنْ، إِمّٝرْدࣹ إࣹ أَمࣹنْ، ࢠࣹ وِئِتِ: "وَرࣹي ࢠِطْࢠࣹ وٝرْࢠࣹ گٝمْطِمْدِمْࢠࣹ ࢠࣹنْ إࣹ مَكّٝ، أَݖّٝنْ رࣹوْࢠࣹ مَࢠّࣹ ࢠࣹنْ". پࣹوْجࣹ هࣹيفࣹرْࢠࣹ ࢠࣹنْ وٝنَالِ، سِ وَنَا إࣹ مَجّࣹرࣹ.

(26) فرعون وِئِ: "أَݖّࣹي مِي وَرَ موسى. يٝ أٝ نٝدُّ جٝومِ مَكّٝ ! مِطُ هُلِ ندࣹ أٝ وَيْلَتَ دِينَ مٝنْ كَنْ، مَا أٝ فࣹݧِّنَ بٝنّࣹرࣹ كَ لࣹيْدِ".

(27) مُوسَ مَاكِ: "مِي مٝولٝرِي جࣹيْطٝ لَمْ أٝنْ، جࣹيِ أٝنْ، إࣹ بٝنࣹ كَلَ مَوْنِنْتِنِيطٝ، مٝ گٝمْطِنَا ݧَلَاندࣹ هَسْبٝورࣹ".

(28) گٝرْكٝ گٝمْطِنٝطٝ گٝو، جࣹيَاطٝ إࣹ يِمْࢠࣹ فرعون ࢠࣹنْ - سُوطُنٝوطٝ گٝمْطِنَلْ مُنْ - :"إࣹ أٝنْ وَرَيْ گٝرْكٝ، فِي أٝ وِئِ: "جٝومِ أَمْ كٝ الله?" هَرَ لࣹ أٝ أَدَّنِي أٝنْ ࢠَنْنگَنْندُيࣹيجِ، إِمّٝرْدࣹ كَ جٝومِ مٝنْ. سِ تَوِي أٝ فࣹنُ، هَرَيْ فࣹنَاندࣹ مَكّٝ كٝ كَنْكٝ فَوِي; سِ تَوِي كَدِ أٝ گٝونگُ، هَرَيْ يٝگَ إࣹ كٝنْ كٝ كَمْنبِرَنِ أٝنْ، هࣹࢠَيْ أٝنْ". پࣹلّࣹتْ، الله فࣹوْنَتَا أٝنْ فَنْتِنٝووٝ فࣹنّٝووٝ!

(29) كٝ أٝنٝنْ يٝ يِمْࢠࣹ أَمْ، لَامُ نٝ وٝودَنِ أٝنْ هَنْندِ، كٝ أٝنْ فٝولُࢠࣹ كَ لࣹيْدِ. كٝنٝ كٝ هٝمْنبٝ وَلَّتَ إࣹنْ، إࣹ لࣹپْتࣹ الله طࣹنْ، سِ طࣹ أَرِي إࣹ مࣹنْ?" فرعون وِئِ: "مِي هٝلَّتَا أٝنْ، سِ وَنَا كٝ مِي يِئِ كٝنْ. مِي تِندِنْتَا أٝنْ كَدِ، سِ وَنَا إࣹ لَاوٝلْ پࣹيوَلْ نگٝلْ".

(30) گٝمْطِنْطٝ أٝنْ وِئِتِ: "كٝ أٝنٝنْ يٝ يِمْࢠࣹ أَمْ، مِينْ أٝنْ مِطُ هُلَنِ أٝنْ، سُگُ ݧَلَاندࣹ هࣹࢠُنٝوندࣹ فࣹدّࣹيجِ طِنْ.

(31) سُگُ نبٝوْكَ يِمْࢠࣹ نوح ࢠࣹنْ، إࣹ آدِئࣹنْ، إࣹ سَمُودَئࣹنْ، إࣹ أَرْنٝوࢠࣹ ࢠَاوٝ مَࢠّࣹ ࢠࣹنْ". الله فَاندَنَاكِ تٝوݧُدࣹ جࣹيَاࢠࣹ ࢠࣹنْ.

(32) "كٝ أٝنٝنْ يٝ يِمْࢠࣹ أَمْ، مِينْ مِطُ هُلَنِ أٝنْ، ݧَلَاندࣹ نٝدِّندِرْتٝنْ

(33) ݧَلْندࣹ هُݖِّتِرٝيْتٝنْ، هٝطٝنْ ندٝگَ، هَرَ دَطْندٝووٝ أٝنْ أَلَنَا أٝنْ إࣹ الله". كَلَ مٝ الله مَجِّنِ، هَرَيْ فࣹوْنٝووٝ أَلَنَا مٝ.

(34) گٝمْطِي يوسف أَرْدَنِي أٝنْ إࣹ ࢠَنْنگَندُيࣹيجِ، كٝ أَدِي، أٝنْ سࣹيرَالِ إࣹ سِكِّتَگٝلْ كٝنْ كٝ أٝ أَدَّنِ أٝنْ. هَا ندࣹ أٝ فَاتِنٝو، وِئُطٝنْ: "الله نُلْتَتَا نُلَاطٝ گٝو، ࢠَاوٝ مَكّٝ". كٝ نٝنْ مَجِّنِرْتَ كَلَ وٝنٝطٝ فَنْتِنٝووٝ سِكِّتِيطٝ.

(35) ࢠࣹنْ وࣹنجِرٝوࢠࣹ إࣹ آيࣹيجࣹ الله طࣹنْ كٝ أَلْدَا إࣹ هُجَّ أَرْطٝ ࢠࣹ، كُوطࣹ مَࢠّࣹ طࣹنْ مَوْنِي تِكّࣹرࣹ كَ الله، إࣹ كَ گٝمْطِمْࢠࣹ ࢠࣹنْ. كٝ وَنٝ نٝنْ الله نٝتِرْتَ إࣹ ࢠࣹرْندࣹ كَلَ مَوْنِنْتِنِيطٝ دُندَرَنْكࣹيجٝ.

(36) فرعون وِئِ: " كٝ أَنْ يٝ هَامَانَ، مَهَنَمْ سٝورٝ، نٝ گَسَ مِي يٝتّٝو طَتࣹ طࣹنْ.

(37) طَتࣹ كَمُّولِ طِنْ، مِي ڃࣹيوٝيَ رࣹوࣹتࣹيطٝ موسى أٝنْ; سَبُ مِي سِكِّ مٝ فࣹنٝووٝ". كٝ وَنٝ نٝنْ فرعون ݖُطِرَنَا، گٝلّࣹ مَكّٝ بٝنْطࣹ طࣹنْ; أٝ سَكَّا إࣹ لَاوٝلْ (پࣹيوَلْ) نگٝلْ. فࣹيرࣹ فرعون ندࣹنْ وٝنَانٝ، سِ وَنَا إࣹ هَيْرࣹرࣹ.

(38) گٝرْكٝ گٝمْطِنْطٝ أٝنْ وِئِ: "كٝ أٝنٝنْ يٝ يِمْࢠࣹ أَمْ، جٝكّࣹي لَمْ، مِي فࣹوْنَ أٝنْ إࣹ لَاوٝلْ پࣹيوَلْ نگٝلْ.

(39) كٝ أٝنٝنْ يٝ يِمْࢠࣹ أَمْ، أَنْندࣹي طَمْ نگُورْندَمْ أَدُنَ، كٝ دَكَمّࣹ-هُنْ تُنْ، كٝ لَاكَرَ أٝنْ وٝنِ، گَلّࣹ ݧِيࢠَلْ طࣹنْ.

(40) مٝ گٝلِّي بٝنࣹ، أٝ يٝࢠِتࣹتَاكࣹ، سِ وَنَا يࣹرُ مَكّٝ; مٝ گٝلِّي كَدِ مٝڃّࣹرࣹ، فٝتِ كٝ گٝرْكٝ مَا دࣹبّٝ، هَرَ كٝ أٝ گٝمْطِنْطٝ، ࢠࣹنْ طٝنْ نَاتَيْ الجنة، ࢠࣹ أَرْسِكٝيࣹي تٝنْ، كٝ أَلْدَا إࣹ لِمٝورࣹ.

(41) كٝ أٝنٝنْ يٝ يِمْࢠࣹ أَمْ، كٝ وَطِ سِ مِطُ نٝدَّ أٝنْ إࣹ دَطَلْ، هِطٝنْ نٝدَّ لَمْ إࣹ يِيتࣹ?

(42) هِطٝنْ نٝدَّ لَمْ إࣹ يࣹدُّگٝلْ الله، سِرْكَنَ مٝ إࣹ كٝ مِي أَلَنَا طُمْ گَنْندَلْ، هَرَ نٝنْ، مِينْ مِطُ نٝدَّ أٝنْ إࣹ فٝولُطٝ، سُورٝووٝ بَكَّاتُ أٝنْ.

(43) سِكّࣹ أَلَا، وٝنْندࣹ كٝ نٝدّٝتٝنْ لَمْ إࣹ مُنْ كٝنْ، دࣹوَلْ أَلَنَا طُمْ أَدُنَ، وَنَا لَاكَرَ. إࣹ وٝنْندࣹ رُتّٝرْدࣹ مࣹنْ ندࣹنْ، كٝ كَ الله، إࣹ وٝنْندࣹ كٝ فَنْتِمْࢠࣹ ࢠࣹنْ، وٝنِ يِيتࣹ نگࣹنْ.

(44) أَرَيْ ندࣹ أَنْندٝتٝنْ، كٝ مِي يࣹوْتَتَ أٝنْ كٝنْ; مِي فَوَ فِيَاكُ أَمْ أٝنْ إࣹ الله. پࣹلّࣹتْ، الله كٝ رࣹينُطٝ جࣹيَاࢠࣹ ࢠࣹنْ".

(45) الله دَطْندِ مٝ، بٝنࣹيجِ طِ ࢠࣹ فࣹوْجُنٝو، أٝ فِيلْتِي يِمْࢠࣹ فرعون ࢠࣹنْ، كٝ ࢠُرِ بٝنْندࣹ إࣹ لࣹپْتࣹ.

(46) يِيتࣹ نگࣹنْ، ࢠࣹ وࣹيࢠِتٝيْتࣹ إࣹ مَگّࣹ، بِمْنبِ إࣹ كِيكِيطࣹ. ݧَلْندࣹ دَرْنگَلْ دَرٝيْتٝو (وِئٝيْتࣹ): "نَاتࣹي يِمْࢠࣹ فرعون ࢠࣹنْ، كَ ࢠُرِ سَطْتُدࣹ إࣹ لࣹپْتࣹ".

(47) جَنْتٝ تُمَ ࢠࣹ يࣹدّٝندِرٝيْتَ كَ يِيتࣹ، لٝؤُنٝوࢠࣹ ࢠࣹنْ وِئَنَ تٝوْنِتِنٝوࢠࣹ ࢠࣹنْ: "مࣹنࣹنْ، كٝ أٝنٝنْ مࣹنْ جٝكُّونٝو، تَوْ سِ أٝنْ دُݧَنَيْ مࣹنْ گࣹࢠَلْ إࣹ يِيتࣹ نگࣹنْ?"

(48) تٝوْنِتِنٝوࢠࣹ ࢠࣹنْ وِئَ: "إࣹنْ فٝوْ، هِطࣹنْ إࣹ مَگّࣹ. گٝمْطِي الله ݧَاوِي هَكُّندࣹ جࣹيَاࢠࣹ ࢠࣹنْ".

(49) وٝنْࢠࣹ كَ يِيتࣹ ࢠࣹنْ كَدِ وِئَنَ أَيْنِنَاࢠࣹ جهنم ࢠࣹنْ: "تٝرَنࣹي مࣹنْ جٝومِ مٝنْ، أٝ هٝيْنَنَ مࣹنْ، ݧَلَاندࣹ (وٝوتࣹرࣹ) إࣹ لࣹپْتࣹ طࣹنْ".

(50) ࢠࣹ وِئَ: "نُلَاࢠࣹ مٝنْ ࢠࣹنْ أَرْدَنَانٝ أٝنْ إࣹ ࢠَنْنگَنْندُيࣹيجِ?" ࢠࣹ وِئَ: "أٝوءٝوٝيࣹ !" أَيْنِنَاࢠࣹ ࢠࣹنْ وِئَ: "أَوَ تٝرࣹي، أٝنٝنْ تِگِرِ !" كٝنٝ تٝرَارࣹ هࣹيفࣹرْࢠࣹ ࢠࣹنْ وٝنَالِ، سِ وَنَا إࣹ مَجّࣹرࣹ.

(51) پࣹلّࣹتْ، مࣹنْ وَلَّيْ نُلَاࢠࣹ أَمࣹنْ ࢠࣹنْ، إࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ، كَ نگُورْندَمْ أَدُنَ، إࣹ ݧَلْندࣹ سࣹيدࣹيࢠࣹ ࢠࣹنْ دَرٝيْتٝو،

(52) ݧَلْندࣹ دࣹ نگَنْتِنَگٝلْ نَفٝيْتَا تٝوݧُنٝوࢠࣹ ࢠࣹنْ، هَرَ كُدِّ هِنٝ وٝودَنِ ࢠࣹ، إࣹ گَلّࣹ بٝنْطࣹ.

(53) گٝمْطِي مࣹنْ أٝكِّي موسى گَنْندَلْ تَوْرࣹيتَ نگَلْ، مࣹنْ رٝنِنِ ࢠِطْࢠࣹ إسرائيل ࢠࣹنْ، دࣹفْتࣹرࣹ ندࣹنْ.

(54) كٝ ندࣹ پࣹيوَلْ إࣹ وَاجُ، وٝنَنْندࣹ جٝومِرَاࢠࣹ هَقِّلّࣹ ࢠࣹنْ.

(55) مُݧّٝ تُنْ، پࣹلّࣹتْ، فٝدٝورࣹ الله ندࣹنْ كٝ گٝونگَ، إِنْسِنَنٝ جُنُوبِ مَا طِنْ، سُبُّنْهِنَنٝطَا جٝومِ مَا، كِيكِيطࣹ إࣹ بِمْنبِ.

(56) پࣹلّࣹتْ، وࣹنْنجِرٝوࢠࣹ إࣹ آيࣹيجࣹ الله طࣹنْ، كٝ أَلْدَا إࣹ دَلِلْ أَرْطٝ إࣹ مَࢠّࣹ، أَلَا إࣹ ࢠࣹرْطࣹ طࣹنْ، سِ وَنَا مَوْنِنْتِنَگٝلْ. ࢠࣹ يٝتّٝتَاكٝ نٝنْ، إࣹ طُمْ. مٝولٝرْ الله، كٝ كَنْكٝ وٝنِ نَنٝووٝ، يِئٝووٝ أٝنْ.

(57) پࣹلّࣹتْ، كٝ تَگْدࣹ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، ࢠُرِ مَوْنُدࣹ إࣹ دِي تَگࣹيندِ يِمْࢠࣹ ࢠࣹنْ. كٝنٝ كٝ ࢠُرِ طُوطُدࣹ إࣹ يِمْࢠࣹ ࢠࣹنْ، ࢠࣹ أَنْندَا.

(58) بُنْطٝ إࣹ يِئٝووٝ فٝتَتَا، گٝمْطِمْࢠࣹ ࢠࣹنْ كَدِ گٝلِّ مٝڃُّطِ، فٝتَتَا إࣹ بٝنُّطٝ. فَنْطِي كٝ وَاجِتٝرْتٝنْ !

(59) پࣹلّࣹتْ، دَرْنگَلْ، كٝ كٝ أَرَتَ، سِكّࣹ أَلَا إࣹ مُؤُنْ. كٝنٝ كٝ ࢠُرِ طُوطُدࣹ إࣹ يِمْࢠࣹ ࢠࣹنْ، ࢠࣹ گٝمْطِنَا.

(60) جٝومِ مٝنْ دَالِ: "تٝرࣹي لَمْ، مِي جَابِنَنَ أٝنْ. مَوْتِنَنْتٝوࢠࣹ رࣹوُگٝلْ لَمْ ࢠࣹنْ، كٝ پࣹلّࣹتْ، ࢠࣹ نَاتٝيَيْ جهنم، كٝ ࢠࣹ هٝيْࢠࣹ".

(61) كٝ الله وَطِرَنِ أٝنْ جࣹمَّ أٝنْ، كَ فٝوْتٝطٝنْ إࣹ مُنْ، إࣹ ݧَلٝرْمَ أٝنْ، يِئِنٝووٝ. پࣹلّࣹتْ، الله كٝ جٝمْ ࢠُرَلْ إࣹ يِمْࢠࣹ ࢠࣹنْ، كٝنٝ كٝ ࢠُرِ طُوطُدࣹ إࣹ يِمْࢠࣹ ࢠࣹنْ، ࢠࣹ يࣹتَّتَا (نࣹيمَ).

(62) كٝ أٝنْ مٝوطٝنْ الله جٝومِ مٝنْ، وٝنِ تَگُطٝ كَلَ هُوندࣹ. رࣹوࣹتࣹيطٝ أَلَا، سِ وَنَا كَنْكٝ. كٝ هٝنّٝ فِرْلِتٝرْتَطٝنْ (إࣹ طُمْ?)

(63) كٝ نٝنْ ࢠࣹيَ يࣹدَّيْنٝوࢠࣹ آيࣹيجࣹ الله طࣹنْ، فࣹرْلِسٝرْتࣹنٝو.

(64) كٝ الله وٝنِ وَطِرَنْطٝ أٝنْ لࣹيْدِ ندِنْ، ݧِيࢠِرْدࣹ، كَمُّ نگُنْ، كِپّٝودِ، أٝ نبَادِنِ أٝنْ - أٝ مٝڃِّنِ نبَادِيجِ مٝنْ -، أٝ أَرْسِكِ أٝنْ إࣹ لَاࢠُطِ. كٝ أٝنْ مٝوطٝنْ وٝنِ الله جٝومِ مٝنْ; سࣹنَارࣹ وٝودَنِي الله، جࣹيْطٝ وِنْندࣹرࣹ ندࣹنْ !

(65) كٝ كَنْكٝ وٝنِ وُورْطٝ أٝنْ. رࣹوࣹتࣹيطٝ أَلَا، سِ وَنَا كَنْكٝ. رࣹوࣹي مٝ كِسَنْ; لَࢠِّنَنٝ مٝ دِينَ كَنْ. يࣹتّٝورࣹ وٝودَنِي الله، جࣹيْطٝ وِنْندࣹرࣹ ندࣹنْ !

(66) مَاكُ: "مِينْ مِي هَطَامَ رࣹوُگٝلْ طِنْ طِ نٝدّٝتٝنْ گَانِنْ الله، تُمَ ندࣹ ࢠَنْنگَندُيࣹيجِ طِنْ أَرِ إࣹ أَمْ، إِمّٝرْدࣹ إࣹ جٝومِ أَمْ، مِي يَمِرَامَ كَدِ، ندࣹ مِي جࣹبِّلَنْتٝو، جࣹيْطٝ وِنْندࣹرࣹ ندࣹنْ".

(67) كٝ كَنْكٝ وٝنِ تَگُطٝ أٝنْ إࣹ نبُنبُلْدِ، رࣹفْتِ إࣹ تٝࢠّࣹرࣹ (مَنِييُ)، رࣹفْتِ إࣹ هࣹطّࣹرࣹ، رࣹفْتِ أٝ يَلْتِنِرِ أٝنْ پࣹنُوࢠࣹ فِي نٝ يٝتّٝرٝنْ ݖࣹمْنبࣹ مٝنْ، إࣹ نٝ لَاتٝرٝنْ نَيࣹيࢠࣹ، - نٝ إࣹ مٝؤٝنْ، ڃࣹتِّتࣹتࣹيࢠࣹ أَدٝ طُمْ، - هَا نٝ يٝتّٝرٝنْ لَجَلْ هَپَّانگَلْ، بࣹلَجٝؤٝ، أٝنْ هَقِّلَيْ.

(68) كٝ كَنْكٝ وٝنِ وُورْنٝووٝ إࣹ وَرٝووٝ. سِ أٝ هَاجِي إࣹ فِيَاكُ، كٝ أٝ وِئَنَيْ مٝ تُنْ: "لَاتٝ"، أٝ لَاتٝو.

(69) إࣹ أَ تِنَالِ، ࢠࣹنْ يࣹدّٝندِرْتࣹيࢠࣹ إࣹ آيࣹيجࣹ الله طࣹنْ، كٝ هٝنّٝ ࢠࣹ فِرْلِتٝرَا?

(70) ࢠࣹنْ فࣹنُّࢠࣹ دࣹفْتࣹرࣹ ندࣹنْ، إࣹ كٝنْ كٝ مࣹنْ نُلِرِ نُلَاࢠࣹ أَمࣹنْ ࢠࣹنْ; ࢠࣹ أَرَيْ أَنْندُدࣹ،

(71) تُمَ كٝلٝنبٝوجࣹ طࣹنْ وَطٝيَا إࣹ دَاطࣹ مَࢠّࣹ، إࣹ جٝلٝكٝوجࣹ(ݖَلَّلّࣹ) طࣹنْ، هَرَ هِࢠࣹ دَاسِرࣹي

(72) كَ ندࣹرْ وُلْطَمْ; رࣹفْتِ ࢠࣹ هُࢠّٝيࣹي كَ ندࣹرْ يِيتࣹ.

(73) رࣹفْتِ ࢠࣹ وِئَنࣹي: "كٝ هٝنْتٝ وٝنِ طِنْ طِ كَفِدَيْنٝطٝنْ?

(74) كٝ أَلْدَهْ إࣹ الله?" ࢠࣹ وِئَ: "طِ مَجِّتِكࣹ مࣹنْ. وُرِنْ، مࣹنْ وٝنَانٝ كٝ أَدِي، إࣹ رࣹوُگٝلْ هَيْ هُوندࣹ". كٝ هٝنَ نِهْ الله مَجِّنِرْتَ هࣹيفࣹرْࢠࣹ ࢠࣹنْ.

(75) كٝ وَطِ أٝنْ طُمْ، كٝ كٝ وࣹلْتٝرَيْنٝطٝنْ كٝنْ كَ هٝورࣹ لࣹيْدِ، كٝ أَلْدَا إࣹ گٝونگَ، إࣹ كٝ وࣹوْلِرَيْنٝطٝنْ كٝنْ.

(76) نَاتِرࣹي دَمࣹ جهنم طࣹنْ، لُتُّگٝلْ تٝنْ. نگࣹ بٝنِي جَاڃٝرْدࣹ وٝنَندࣹ مَوْنِنْتِنِيࢠࣹ ࢠࣹنْ!.

(77) مُݧّٝ. پࣹلّࣹتْ، فٝدٝورࣹ الله ندࣹنْ، كٝ گٝونگَ. كٝ سِ مࣹنْ هٝلّࣹتࣹ يٝگَ إࣹ كٝ مࣹنْ فٝدِ ࢠࣹ كٝنْ، مَا هَرَ مࣹنْ ڃࣹتِّتࣹ (أَدٝ مُنْ) كٝ كَ أَمࣹنْ ࢠࣹ رُتِّرْتࣹي.

(78) گٝمْطِي مࣹنْ إِمِّنِي نُلَاࢠࣹ أَدٝ مَا. هِنٝ إࣹ مَࢠّࣹ، ࢠࣹ مࣹنْ سِفَنِ مَا فِي مُنْ; هِنٝ مَࢠّࣹ، ࢠࣹ مࣹنْ سِفَنَاكِ مَ فِي مُنْ. هَانَنَا نُلَاطٝ، أَدُّگٝلْ آيَ، سِ وَنَا إࣹ دُݝَيࣹي الله. سِ يَمِرٝورࣹ الله ندࣹنْ أَرِي، گٝونگَ ݧَاوِرْتࣹ مࣹيرࣹنْتࣹئࣹنْ سٝويَ ندࣹنْ ݧَلْندࣹ.

(79) كٝ الله وٝنِ وَطِرَنْطٝ أٝنْ نࣹيمٝرَاطِ طِنْ كٝ وَطّٝرٝنْ، ݧَامٝنْ كَدِ إࣹ مَجِّ.

(80) نَفٝوجࣹ كَدِ هِنٝ وٝودَنِ أٝنْ إࣹ مَجِّ، إࣹ نٝ يٝتّٝرٝنْ إࣹ مَجِّ، هَاجُ هِتُّطٝ إࣹ بࣹرْدࣹ مٝنْ. كٝ إࣹ هٝورࣹ مَجِّ إࣹ هٝورࣹ لَاطࣹ طࣹنْ، وٝنِ كٝ رٝنْندࣹتࣹطٝنْ.

(81) هِمٝ هٝلَّ أٝنْ مَاندࣹيجِ مَكّٝ طِنْ. كٝ هٝنْطُمْ إࣹ مَاندࣹيجِ الله طِنْ، وٝنِ كٝ يࣹدّٝتٝنْ?

(82) إࣹ ࢠࣹي يِيلَاكِ كَ لࣹيْدِ، ࢠࣹ ندَارَ كٝ هٝنّٝ بَتَّنࣹ ࢠࣹيَ أَدٝ مَࢠّࣹ وَتِّنِرْنٝو? ࢠࣹنْ نٝ ࢠُرُنٝو ࢠࣹ طُودࣹ إࣹ سَطْتُدࣹ دٝولࣹ إࣹ بَتّࣹيجِ وَطَاطِ كَ لࣹيْدِ. كٝنٝ كٝنْ كٝ ࢠࣹ فَگّٝتٝنٝو، دُنْݖَنَانٝ ࢠࣹ.

(83) ندࣹ نُلَاࢠࣹ مَࢠّࣹ ࢠࣹنْ أَدَّنّٝوࢠࣹ ࢠَنْنگَنْندُيࣹيجِ طِنْ، ࢠࣹ وࣹوْلِرِ گَنْندَلْ كٝ وٝنِ كَ مَࢠّࣹ كٝنْ. ࢠࣹ هُوࢠِتِرَا كٝنْ كٝ ࢠࣹ جَلْكِتَيْنٝو.

(84) ندࣹ ࢠࣹ يِئُنٝو لࣹپْتࣹ أَمࣹنْ طࣹنْ، ࢠࣹ وِئِ: "مࣹنْ گٝمْطِنِي الله تُنْ، مࣹنْ يࣹدِّي كٝنْ كٝ مࣹنْ وٝنُنٝو إࣹ سِرْكَنْندࣹ".

(85) كٝنٝ گٝمْطِنَلْ مَࢠّࣹ وٝنَانٝ كٝ نَفَتَ ࢠࣹ، فࣹوْندٝ ࢠࣹ يِئُنٝو لࣹپْتࣹ أَمࣹنْ طࣹنْ; طُمْ كٝ لَاوٝلْ الله نٝنْ، گٝمْطِي طُنْ يَوْتِينٝ إࣹ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ. يࣹدُّࢠࣹ ࢠࣹنْ سٝويِ طٝنْ ندࣹنْ ݧَلْندࣹ.

41 Fussilat — 54 آيَة

(1) هَا، مِيمْ.

(2) كٝ جِپّٝرِيندࣹ إِمّٝرْدࣹ كَ يُرْمࣹتࣹيطٝ، هِنّٝتٝوطٝ أٝنْ.

(3) دࣹفْتࣹرࣹ فࣹنْسِتَاندࣹ آيࣹيجࣹ مُنْ، جَنْنگࣹتࣹيندࣹ إࣹ طࣹمْنگَلْ آرَبُ، وٝنَنْندࣹ يِمْࢠࣹ أَنْندٝوࢠࣹ.

(4) وࣹوْلِنَيْندࣹ جࣹرْتِنَ. كٝنٝ ࢠُرْࢠࣹ طُوطُدࣹ ࢠࣹنْ إࣹ مَࢠّࣹ، طُورْنِي، ࢠࣹنْ ࢠࣹ نَنَتَا.

(5) ࢠࣹ وِئِ: "ࢠࣹرْطࣹ أَمࣹنْ نٝ أٝمْنبَنِي كٝ نٝدَّتَا مࣹنْ إࣹ مُنْ كٝنْ، نٝتٝ هِنٝ إࣹ نٝپِّ أَمࣹنْ، هࣹيدٝ كَدِ هِنٝ هَكُّندࣹ أَمࣹنْ إࣹ مَا. هَرَيْ وٝنُ إࣹ گٝلّࣹ مَا ; مࣹنࣹنْ كَدِ مࣹطࣹنْ إࣹ گٝلّࣹ أَمࣹنْ".

(6) مَاكُ: "أَنْندࣹي كٝ نࣹطَّنْكࣹ يࣹرُوجٝ مٝنْ. وَهَيِنَامَ إࣹ أَمْ، وٝنْندࣹ رࣹوࣹتࣹيطٝ مٝنْ أٝنْ، كٝ رࣹوࣹتࣹيطٝ گٝوتٝ. ݧِيࢠࣹي إࣹ طُمْ، إِنْسِنَنٝطٝنْ مٝ". بٝنࣹ وٝودَنِي سِرْكٝوࢠࣹ ࢠࣹنْ.

(7) ࢠࣹ جٝنَّتَا ࢠࣹنْ زَكَّ، هَرَ كَدِ هِࢠࣹ يࣹدِّ فِي لَاكَرَ.

(8) پࣹلّࣹتْ، ࢠࣹنْ گٝمْطِمْࢠࣹ ࢠࣹ گٝلِّ مٝڃِّ طِنْ، هِنٝ وٝودَنِ ࢠࣹ نجٝࢠْدِ ندِ تَڃٝندِرْتَا.

(9) مَاكُ: "إࣹ أٝنٝنْ أٝنْ يࣹدَّيْ أٝنْ تَگُطٝ لࣹيْدِ ندِنْ ندࣹرْ بَلْطࣹ طِطِ، وَطَنٝنْ مٝ گٝرࣹيࢠࣹ? أَوَ كٝ أٝنْ وٝنِ جࣹيْطٝ وِنْندࣹرࣹ ندࣹنْ.

(10) أٝ وَطِ پࣹلّࣹ إࣹ دٝوْ مَيْرِ، أٝ هࣹبِّنِ بَرْكِ إࣹ مَيْرِ، أٝ إࣹࢠِّ نگُورࣹيجِ مَيْرِ طِنْ إࣹ ندࣹرْ بَلْطࣹ نَيِ تِمُّطࣹ، وٝنَنْندࣹ لَمْندٝتٝوࢠࣹ ࢠࣹنْ.

(11) رࣹفْتِ أٝ فٝتٝندِرِ كَ كَمُّ، فࣹوْندٝ كَ نگُ ݖُرْكِنّٝو، أٝ دَالِدَنِ نگُنْ كَنْنگُ إࣹ لࣹيْدِ ندِنْ: "أَرْدِرࣹي طٝفْتَارࣹ، مَا كَرْهࣹيدࣹ". طِ وِئِدِ: "مࣹنْ أَرِي كٝ مࣹنْ طٝفْتِي ࢠࣹ".

(12) أٝ تِمِّنِرِ طِ كَمُّولِ جࣹيطِطِ ندࣹرْ بَلْطࣹ طِطِ، أٝ يَمِرِ كَلَ كَمُّ، فِيَاكُ مَگُّ. مࣹنْ ݖُطِنِرِ كَمُّ أَدُنَ نگُنْ، كٝودࣹ إࣹ رࣹينِرْطُمْ. طُمْ كٝ إࣹࢠّٝورࣹ فٝولُطٝ، أَنْندُطٝ أٝنْ.

(13) سِ ࢠࣹ طُورْنِكࣹ، هَرَيْ وِئُ ࢠࣹ: "مࣹنْ جࣹرْتِنِرِي أٝنْ هَاݖَانگُ، سُگُ هَاݖَانگٝ آدِئࣹنْ إࣹ سَمُودَئࣹنْ.

(14) ندࣹ نُلَاࢠࣹ ࢠࣹنْ أَرْدِرَنّٝوࢠࣹ، يࣹيسٝ مَࢠّࣹ إࣹ ࢠَاوٝ مَࢠّࣹ، وٝنْندࣹ: "وَتَ أٝنْ رࣹوُ دࣹيْ، سِ وَنَا الله"، ࢠࣹ وِئِ: "سِ جٝومِ أَمࣹنْ هَاجُنٝ، كٝ الملائكة أٝ جِپِّنَيْنٝو. أَوَ مࣹنࣹنْ، كٝ مࣹنْ يࣹدُّࢠࣹ كٝ نُلَطٝنْ مُنْ كٝنْ".

(15) آدِيئࣹنْ كَنْ، ࢠࣹ مَوْنِنْتِنِ كَ هٝورࣹ لࣹيْدِ، كٝ أَلْدَا إࣹ گٝونگَ، ࢠࣹ وِئِ: "هَرَ كٝ هٝمْنبٝ ࢠُرِ مࣹنْ دٝولࣹ?" إࣹ ࢠࣹي يِئَالِ، وٝنْندࣹ الله تَگُطٝ ࢠࣹ أٝنْ نٝ ࢠُرِ ࢠࣹ تِيطُدࣹ دٝولࣹ? ࢠࣹ وٝنِ إࣹ يࣹدُّگٝلْ آيࣹيجࣹ أَمࣹنْ طࣹنْ.

(16) مࣹنْ وُرْتِ هࣹنْندُ ࢠُوࢠُندُ إࣹ مَࢠّࣹ إࣹ ݧَلَاطࣹ هِيتَاطࣹ، فِي نٝ مࣹنْ مࣹيطِنِرَ ࢠࣹ لࣹپْتࣹ كٝيࣹيرَ طࣹنْ كَ نگُورْندَمْ أَدُنَ. پࣹلّࣹتْ، كٝ لࣹپْتࣹ لَاكَرَ طࣹنْ ࢠُرِ وٝنْندࣹ كٝيࣹيرَ، كَدِ ࢠࣹ فَابٝيْتَاكࣹ.

(17) سَمُودَئࣹنْ كَنْ، مࣹنْ ࢠَنْنگِنَنّٝ ࢠࣹ; كٝنٝ ࢠࣹ سُࢠِي بُمْطَمْ طَمْ إࣹ دٝوْ پࣹيوَلْ نگَلْ. دِلّࣹرࣹ لࣹپْتࣹ هٝيْنٝوجࣹ طࣹنْ نَنْنگِرِ ࢠࣹ، سَبُ كٝ ࢠࣹ فَگِّتٝتٝنٝو كٝنْ.

(18) مࣹنْ دَطْندِ گٝمْطِمْࢠࣹ ࢠࣹ وٝنِينَ هُلْنٝوࢠࣹ ࢠࣹنْ.

(19) ݧَلْندࣹ نٝنْ أَيْࢠࣹ الله ࢠࣹنْ مٝوࢠِرٝيْتࣹي فَادࣹ كَ يِيتࣹ... هَرَ هِࢠࣹ ݧٝڃِّندِرࣹي.

(20) هَا سِ ࢠࣹ هࣹوْتِي نگࣹ، نَنّطࣹ مَࢠّࣹ طࣹنْ، إࣹ گِيطࣹ مَࢠّࣹ طࣹنْ، إࣹ گُرِ ࢠَلِّ مَࢠّࣹ طِنْ سࣹيدِتٝو إࣹ دٝوْ مَࢠّࣹ، كٝنْ كٝ ࢠࣹ گٝلَّيْنٝو.

(21) ࢠࣹ وِئَنَ گُرِ مَࢠّࣹ طࣹنْ: "كٝ فِي هٝنّطُمْ سࣹيدِتَنِطٝنْ إࣹ أَمࣹنْ?" طࣹ وِئَ: "كٝ الله وٝوْلِنْطٝ كَلَ هُوندࣹ أٝنْ، وٝوْلِنِ مࣹنْ. كٝ كَنْكٝ تَگُنٝو أٝنْ گِلَ كَ لَاوٝلْ أَرَنٝلْ، كٝ كَ مَكّٝ كَدِ رُتّࣹتࣹطٝنْ".

(22) أٝنْ وٝنَالِ هِطٝنْ سُوطٝرَ، فِي وَتَ نَنْطࣹ مٝنْ، إࣹ گِيطࣹ مٝنْ، إࣹ گُرِ مٝنْ سࣹيدِتٝ إࣹ مٝنْ. كٝنٝ أٝنْ سِكُّنٝ، وٝنْندࣹ الله أَنْندَا كٝ طُوطِ إࣹ كٝ هُووٝتٝنْ.

(23) كٝ طُمْ مٝوطٝنْ، وٝنُنٝو كٝ سِكُّنٝطٝنْ جٝومِ مٝنْ، طُمْ هَلْكِ أٝنْ. وَطْتِنْطٝنْ جࣹيَاࢠࣹ إࣹ پࣹرْتُࢠࣹ ࢠࣹنْ.

(24) هَيْ سِ ࢠࣹ مُݧِّكࣹ، كٝ يِيتࣹ نگࣹنْ وٝنَتَ جَاڃٝرْدࣹ مَࢠّࣹ; سِ ࢠࣹ طَࢠِّي فࣹلࣹيدࣹ أَݖِّتࣹي، ࢠࣹ وٝنَالِ فࣹلࣹتࣹيࢠࣹ أَݖِّتࣹي.

(25) مࣹنْ وُرْتَنِ ࢠࣹ وٝندِيࢠࣹ، ࢠࣹنْ ݖُطِنَنِ ࢠࣹ كٝ وٝنِ يࣹيسٝ مَࢠّࣹ، إࣹ كٝ وٝنِ ࢠَاوٝ مَࢠّࣹ. كٝنْنگٝلْ لࣹپْتࣹ نگٝلْ جٝجِّ إࣹ مَࢠّࣹ، وٝندُدࣹ إࣹ مٝفْتࣹ يَوْتُنٝوطࣹ أَدٝ مَࢠّࣹ، إِمّٝرْدࣹ إࣹ جِنَّ إࣹ يِمْࢠࣹ. پࣹلّࣹتْ، ࢠࣹ لَاتِنٝ هَيْرُࢠࣹ.

(26) يࣹدُّࢠࣹ ࢠࣹنْ وِئِ: "وَتَ أٝنْ جࣹنْتِتٝ ندࣹي القرآن، سٝنْكࣹي إࣹ مَيْرࣹ (تُمَ ندࣹ جَنْنگࣹتࣹي) بࣹلَجٝؤٝ، أٝنْ فٝولَيْ".

(27) مَ مࣹنْ مࣹيطِنِرٝيْ يࣹدْطُࢠࣹ ࢠࣹنْ، لࣹپْتࣹ سَطْتُطࣹ، مࣹنْ يٝࢠِرَ كٝ ࢠُرِ بٝندࣹ إࣹ كٝ ࢠࣹ هُووَيْنٝو كٝنْ.

(28) كٝ طُمْ وٝنِ نجٝࢠْدِ أَيْࢠࣹ الله ࢠࣹنْ: كٝ يِيتࣹ نگࣹنْ. كٝ گَلّࣹ لُتُّگٝلْ وٝودَنِ ࢠࣹ تٝنْ، نجٝࢠْدِ كٝنْ كٝ ࢠࣹ يࣹدِّرَيْنٝو آيࣹيجࣹ أَمࣹنْ طࣹنْ.

(29) يࣹدُّࢠࣹ ࢠࣹنْ وِئَ: "جٝومِ أَمࣹنْ، هٝلِّي مࣹنْ مَجِّنّٝوࢠࣹ مࣹنْ ࢠࣹنْ كَ جِنَّ إࣹ كَ يِمْࢠࣹ، مࣹنْ يَاࢠَ ࢠࣹ مࣹنْ فُگَ، هَا ࢠࣹ وٝنَ إࣹ ࢠُرْࢠࣹ جٝلْتُدࣹ ࢠࣹنْ".

(30) پࣹلّࣹتْ، وِئُࢠࣹ ࢠࣹنْ : "جٝومِ أَمࣹنْ كٝ الله"، رࣹفْتِ ࢠࣹ تَبِتِ; الملائكةءࣹنْ هِنٝ جِپّٝو إࣹ مَࢠّࣹ (وِيدࣹ): "وَتَ أٝنْ هُلُ، وَتَ أٝنْ سُنٝ; وࣹوْلِرࣹي الجنة، أٝنْ مٝ وٝنَنٝطٝنْ فٝدࣹيدࣹ.

(31) كٝ مࣹنࣹنْ وٝنِ وَلّٝوࢠࣹ أٝنْ ࢠࣹنْ كَ نگُورْندَمْ أَدُنَ إࣹ كَ لَاكَرَ. هِنٝ وٝودَنِ أٝنْ تٝنْ، كَلَ كٝ وٝنْكِيجِ مٝنْ وٝوطࣹتࣹي، هِنٝ وٝودَنِ تٝنْ، كَلَ كٝ نٝدّٝتٝنْ،

(32) كٝ مَرَارِ، إِمّٝرْدࣹ إࣹ هَفٝرٝورࣹ، هِنّٝتٝوطٝ".

(33) هَرَ كٝ هٝمْنبٝ ࢠُرِ مٝڃُّدࣹ كٝنْنگٝلْ، وَنٝ أٝنْ نٝدْطُطٝ إࣹ فِي الله، أٝ گٝلِّ مٝڃّࣹرࣹ، أٝ وِئِ: "مِينْ، كٝ إࣹ جُولْࢠࣹ ࢠࣹنْ مِي جࣹيَا?".

(34) مٝڃّࣹرࣹ ندࣹنْ فٝتَتَا إࣹ بٝنࣹ أٝنْ. دُݧِرْ (گٝلّࣹ بٝنْطࣹ طࣹنْ) كٝنْ كٝ ࢠُرِ مٝڃُّدࣹ; إࣹ جَكَ يٝو، أٝنْ مٝ نگَيْنگُ وٝنِ هَكُّندࣹ مَا إࣹ مُنْ، هِمٝ وَئِتَ وَ گِطٝ سَطْتُطٝ گِگّٝلْ.

(35) كٝنٝ هَوْرَتَا إࣹ طُمْ، سِ وَنَا مُݧِّيࢠࣹ ࢠࣹنْ; هَوْرَتَا إࣹ طُمْ، سِ وَنَا جٝمْ گࣹࢠَلْ مَوْنُنگَلْ.

(36) سِ سٝوِنْسِنٝورࣹ أَرِي هࣹࢠُدࣹ مَ إِمّٝرْدࣹ إࣹ سࣹيْتَانࣹ، مٝولٝرْ الله. كَنْكٝ كٝ أٝ نَنٝووٝ، أَنْندُطٝ.

(37) هِنٝ جࣹيَا إࣹ مَاندࣹيجِ مَكّٝ طِنْ، جࣹمَّ أٝنْ إࣹ ݧَلٝرَمَ أٝنْ، إࣹ نَانگࣹ نگࣹنْ إࣹ لࣹوْرُ ندُنْ. وَتَ أٝنْ سُجَّنْ نَانگࣹ نگࣹنْ وَنَا كَدِ لࣹوْرُ ندُنْ. سُجَّنࣹي الله، تَگُطٝ طِ أٝنْ، سِ وٝنِي كٝ كَنْكٝ رࣹوٝتٝنْ.

(38) سِ ࢠࣹ تٝوْنِتِكࣹ، ࢠࣹنْ وٝنْࢠࣹ كَ جٝومِ مَا كَنْ، هِࢠࣹ سُبُّنْهِنٝرَنَ مٝ جࣹمَّ إࣹ ݧَلٝرْمَ، هَرَ ࢠࣹ هَاࢠَتَا.

(39) هِنٝ جࣹيَا إࣹ مَاندࣹيجِ مَكّٝ طِنْ، تَوْدࣹ أَ يِئَيْ لࣹيْدِ ندِنْ نٝ يٝورِ. سِ وٝنِي مࣹنْ جِپِّنِي ندِيَمْ تٝࢠٝ إࣹ مَيْرِ، ندِ دِلَّ ندِ فٝنْتٝو. أَوَ وُورْنِتُطٝ ندِ أٝنْ، پࣹلّࣹتْ، كٝ وُورْنِتٝووٝ مَايْࢠࣹ ࢠࣹنْ، تَوْدࣹ پࣹلّࣹتْ، كٝ أٝ هَتَّنْطٝ هَكَ هُوندࣹ.

(40) پࣹلّࣹتْ، أٝوݧٝتٝوࢠࣹ ࢠࣹنْ إࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ، ࢠࣹ سُوطٝتَاكٝ مࣹنْ. إࣹ كٝ فࣹرٝيْتࣹيطٝ أٝنْ كَ يِيتࣹ ࢠُرِ مٝڃُّدࣹ كَا كٝ أَرٝيٝووٝ أٝنْ ݧَلْندࣹ دَرْنگَلْ هِنٝ هٝولِي? گٝلّࣹي كٝ فَالَطٝنْ، كٝ پࣹلّࣹتْ، هِمٝ رࣹينِ كٝ گٝلّٝتٝنْ كٝنْ.

(41) يࣹدُّࢠࣹ ࢠࣹنْ القرآن تُمَ ندࣹ أَرُنٝو إࣹ مَࢠّࣹ، پࣹلّࣹتْ نٝنْ، كٝ ندࣹ دࣹفْتࣹرࣹ تࣹدُّندࣹ فٝولُندࣹ ;

(42) مࣹيرࣹ إِوْرَنْتَا ندࣹ يࣹيسٝ مَيْرࣹ، وَنَا ࢠَاوٝ مَيْرࣹ. كٝ جِپّٝرِيندࣹ إِمّٝرْدࣹ كَ ݧࣹينْيٝ، يࣹتِّنِيطٝ أٝنْ.

(43) أَ وِئَاكَ سِ وَنَا كٝ نُلَاࢠࣹ أَدِنٝوࢠࣹ مَ ࢠࣹنْ وِئَنٝو. پࣹلّࣹتْ، جٝومِ مَا، كٝ أٝ جٝمْ يَافُيࣹي، إࣹ جٝمْ لࣹپْتࣹ مُوسُطࣹ.

(44) سِ مࣹنْ وَطُنٝ ندࣹ جَنْنگࣹتࣹيندࣹ إࣹ هَالَ أَجَمِ، ࢠࣹ وِئَيْنٝ: "كٝ هَطْنٝ آيࣹيجࣹ مَيْرࣹ طࣹنْ فࣹنْسِتࣹي? إࣹ ندࣹ وٝنَيْ إࣹ هَالَ أَجَمِ، نُلَاطٝ أٝنْ وٝنَ آرَبُوجٝ?" مَاكُ: "كَيْرࣹ وٝنَنْندࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ، كٝ ندࣹ پࣹيوَلْ إࣹ ݖࣹلَّلْ". ࢠࣹ گٝمْطِنَا ࢠࣹنْ كَنْ، نٝتٝ هِنٝ إࣹ نٝپِّ مَࢠّࣹ طِنْ، كَيْرࣹ كٝ ندࣹ نِࢠࣹ إࣹ دٝوْ مَࢠّࣹ; ࢠࣹنْ هِنٝ نٝدِّرࣹي إࣹ نٝكُّ وٝطُّطُ.

(45) گٝمْطِي مࣹنْ أٝكِّي موسى دࣹفْتࣹرࣹ ندࣹنْ، ندࣹ لُوتٝندِرَ إࣹ مَيْرࣹ. سِ وَنَانٝ كٝنْنگٝلْ أَرْدِنٝونگٝلْ إِمّٝرْدࣹ كَ جٝومِ مَا، هَكُّندࣹ مَࢠّࣹ ݧَاوࣹتࣹنٝ. كَمْࢠࣹ كَنْ، پࣹلّࣹتْ، هِࢠࣹ إࣹ سِكّࣹ سِكِّتِيطٝ ندࣹ.

(46) كَلَ گٝلُّطٝ كٝ مٝڃِّ، كٝ فِي هٝورࣹ-مَكّٝ; كَلَ كَدِ بٝنُّطٝ، يَنْتَيْ إࣹ مَكّٝ. جٝومِ مَا سِفٝرَالِ وٝنْندࣹ، تٝوݧٝووٝ جࣹيَاࢠࣹ ࢠࣹنْ.

(47) كٝ كَ مَكّٝ وٝنِ كٝ گَنْندَلْ دَرْنگَلْ رُتّࣹتࣹي. دِمْطࣹ يَلْتَتَا إࣹ ݖٝومْطِ مُنْ، دࣹيࣹلْ (وٝو) سَاوَتَا،نگࣹلْ جِبِنْتَا سِ وَنَا إࣹ گَنْندَلْ مَكّٝ.ݧَلْندࣹ أٝ نٝدِّرٝيْتَ ࢠࣹ:"كٝ هٝنْتٝ وٝنِ كَفِدَاطِ أَمْ طِنْ?"، ࢠࣹ وِئَ:"مࣹنْ أَنْندِنِي مَ، أَلَهْ إࣹ أَمࣹنْ سࣹيدٝتٝوطٝ طُمْ "

(48) ࢠࣹ مَجِّرَ إࣹ كٝنْ كٝ ࢠࣹ رࣹوَيْنٝو كٝ أَدِي; ࢠࣹ يَنَنࣹي وٝنْندࣹ دَطِرْگَلْ أَلَنَا ࢠࣹ.

(49) نࣹطَّنْكࣹ أٝنْ هَاࢠَتَا تٝرَگٝلْ مٝڃّࣹرࣹ. سِ بٝنࣹ مࣹمِي مٝ، كٝ أٝ تَڃٝتٝوطٝ، يٝنْندِنِيطٝ.

(50) سِ مࣹنْ مࣹيطِنِرْ مٝ مٝڃّࣹرࣹ إِمّٝرْدࣹ كَ أَمࣹنْ، ࢠَاوٝ ندࣹ لٝرَّ مࣹمِ مٝ، كٝ پࣹلّࣹتْ أٝ وِئَيْ: "طُمْ طٝو كٝ مِينْ ! مِي فࣹلِّتَانِ سِ دَرْنگَلْ دَرٝتٝ. سِ مِي رُتَّامَ كَ جٝومِ أَمْ، كٝ پࣹلّࣹتْ، مٝڃّࣹرࣹ هِنٝ وٝودَنِ لَمْ كَ مَكّٝ". مَ مࣹنْ هُمْپِتُ يࣹدْطُࢠࣹ ࢠࣹنْ، فِي كٝ ࢠࣹ هُووِ كٝنْ، كٝ مَ مࣹنْ مࣹيطِنٝيْ ࢠࣹ لࣹپْتࣹ سَطْتُطࣹ.

(51) سِ مࣹنْ نࣹيمِنِي إࣹ نࣹطَّنْكࣹ أٝنْ، أٝ طُورْنٝو أٝ تٝوْنَ بَلَاجࣹ مَكّٝ; سِ بٝنࣹ نٝنْ هࣹࢠِي مٝ، أٝ وٝنَ جٝمْ تٝرَارࣹ يَاجُندࣹ.

(52) مَاكُ: "يࣹيتࣹي لَمْ، سِ وٝنِي لࣹ كٝ كَ الله ندࣹ إِوْرِ، رࣹفْتِ يࣹدُّطٝنْ ندࣹ; هَرَيْ كٝ هٝمْنبٝ ࢠُرِ مَجُّدࣹ أٝنْ وٝنْطٝ إࣹ لُرَّلْ وٝطُّنگَلْ?"

(53) أَرَيْ مࣹنْ هٝلَّ ࢠࣹ مَاندࣹيجِ أَمࣹنْ طِنْ كَ تَتّࣹيجِ إࣹ مَࢠّࣹ تِگِ، هَا ࢠَنْنگَنَ ࢠࣹ وٝنْندࣹ كٝ ندࣹ گٝونگَ. إࣹ يٝنَالِ وٝنْندࣹ جٝومِ مَا، كٝ أٝ سࣹيدِيطٝ كَلَ هُوندࣹ?

(54) أَنْندࣹي هِࢠࣹ إࣹ سِكّࣹ فِي هَوْرٝيْگٝلْ إࣹ جٝومِ مَࢠّࣹ. أَنْندࣹي نٝنْ، كٝ أٝ هُࢠِندِيطٝ كَلَ هُوندࣹ.

42 Ash-Shura — 53 آيَة

(1) هَا، مِيمْ.

(2) أَــِٔنْ، سِينْ، قَافْ.

(3) كٝ وَنٝ نِي وٝنِ كٝ الله تࣹطُّطٝ، ݧࣹيݧٝ أٝنْ وَهَيِنِرِ إࣹ مَا، إࣹ ࢠࣹنْ أَدِنٝوࢠࣹ مَ.

(4) كٝ كَنْكٝ وٝودَنِ كَلَ كٝ وٝنِ كَ كَمُّولِ إࣹ كَ لࣹيْدِ. كٝ كَنْكٝ كَدِ وٝنِ تٝووُطٝ، مَوْطٝ أٝنْ.

(5) كَمُّولِ طِنْ نٝ إࣹࢠَّ سࣹيكٝرْگٝلْ دٝوْ مَجِّ، هَرَ الملائكةءࣹنْ نٝ سُبُّنْهِنَنٝرَ يࣹتُّگٝلْ جٝومِ مَࢠّࣹ، ࢠࣹ إِنْسِنَنٝو وٝنْࢠࣹ كَ لࣹيْدِ ࢠࣹنْ. أَنْندࣹي پࣹلّࣹتْ، كٝ الله وٝنِ هَفٝرٝووٝ، هِنّٝتٝوطٝ أٝنْ.

(6) ࢠࣹنْ نٝنْ جٝگِتِيࢠࣹ يِطْࢠࣹ گٝو گَانِنْ مَكّٝ، الله كٝ رࣹينُطٝ ࢠࣹ. أَ وٝنَالِ رࣹينٝووٝ ࢠࣹ.

(7) كٝ وَنٝ نٝنْ مࣹنْ وَهَيْنِرِ إࣹ مَاطَ، جَنْنگࣹتࣹيندࣹ إࣹ آرَبُ، فِي يٝ أَ جࣹرْتِنْ يِمْࢠࣹ نࣹينࣹ ݖَئࣹ طࣹنْ إࣹ وٝنْࢠࣹ تَكّٝ مَيْرِ ࢠࣹنْ، جࣹرْتِنَا فِي ݧَلْندࣹ دࣹنْتَلْ ندࣹنْ، - ندࣹ سِكّࣹ أَلَا إࣹ مُؤُنْ - فࣹدّࣹ گٝو كَ أَلْجْنَّ، فࣹدّࣹ گٝو كَدِ كَ يِيتࣹ.

(8) سِ الله هَاجُنٝ، أٝ وَطَيْنٝ ࢠࣹ مٝفْتࣹ گٝوتࣹ. كٝنٝ هِمٝ أٝنْ مٝ أٝ مُويِ كَ يُرْمࣹيندࣹ مَكّٝ. تٝوݧٝوࢠࣹ ࢠࣹنْ نٝنْ، گِطٝ أَلَنَا ࢠࣹ، وَنَا بَلّٝ.

(9) كَا ࢠࣹ جٝگِتِ يِطْࢠࣹ گٝو گَانِنْ مَكّٝ? أَوَ كٝ الله وٝنِ گِطٝ أٝنْ، كٝ كَنْكٝ وُورْنِتَتَ مَايْࢠࣹ ࢠࣹنْ; كٝ كَنْكٝ كَدِ وٝنِ كَطْتَنْطٝ كَلَ هُوندࣹ.

(10) كَلَ كٝ لُوتٝندِرْطٝنْ إࣹ فِي مُنْ، هَرَيْ ݧَاوٝورࣹ مُنْ ندࣹنْ كٝ كَ الله وٝنِ. كٝ أٝنْ مٝوطٝنْ الله، وٝنِ جٝومِ أَمْ; كٝ إࣹ مَكّٝ مِي فَوِي، كٝ كَ مَكّٝ كَدِ مِي رُتّٝرْتَ تُوبُگٝلْ .

(11) فُطُّطٝ كَمُولِ طِنْ إࣹ لࣹيْدِ ندِنْ. أٝ وَطِرَنِي أٝنْ إࣹ پِتَّالِ مٝنْ طِنْ گࣹنْندٝوࢠࣹ، كَ نࣹيمٝرَاطِ كَدِ نٝونࣹيجِ; هِمٝ طُطِّنِرَ أٝنْ طُمْ. هَيْ هُوندࣹ يࣹرٝندِرَا إࣹ مَكّٝ; كٝ كَنْكٝ وٝنِ نَنٝووٝ، يِئٝووٝ أٝنْ.

(12) كٝ كَنْكٝ هࣹيرٝرِ سٝكْتِرْطࣹ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ. هِمٝ وࣹرْتَنَ أَرْسِكࣹ أٝنْ مٝ أٝ مُويَنِ، أٝ ࢠِطْتِنَنَ (مٝ أٝ مُويَنِ). پࣹلّࣹتْ، كٝ أٝ أَنْندُطٝ كَلَ هُوندࣹ.

(13) أٝ سَرئِنَنِي أٝنْ إࣹ دِينَ كَنْ، كٝنْكٝ كٝ أٝ يَمِرْنٝو نوح، إࣹ كٝنْ كٝ مࣹنْ وَهَيِنِ إࣹ مَاطَ، إࣹ كٝنْ كٝ مࣹنْ يَمِرْنٝو إبراهيم، إࣹ موسى إࣹ إِيسَا، وٝنْندࣹ: "ݧِࢠْنࣹي دِينَ كَنْ; هَرَ أٝنْ سࣹيرْتَالِ إࣹ مَكَّ". كٝنٝ كٝ نٝدّٝتَا سِرْكٝوࢠࣹ ࢠࣹنْ إࣹ مُنْ كٝنْ، مَوْنِي إࣹ دٝوْ مَࢠّࣹ. الله نٝنْ هِنٝ سُࢠَنٝو طُمْ، مٝ أٝ مُويَنِ، أٝ فࣹوْنَ إࣹ مَكّٝ، كَلَ تُوبٝووٝ.

(14) ࢠࣹ سࣹيرْتَانٝ، سِ وَنَا ࢠَاوٝ ندࣹ گَنْندَلْ نگَلْ أَرْدࣹ إࣹ مَࢠّࣹ، سَبُ بࣹوْرࣹ هَكُّندࣹ مَࢠّࣹ. سِ وَنَانٝ فِي كٝنْنگٝلْ أَرْدِنٝونگٝلْ، إِمّٝرْدࣹ كَ جٝومِ مَا، هَا إࣹ لَجَلْ تَڃَانگَلْ، ݧَاوࣹتࣹنٝ هَكُّندࣹ مَࢠّࣹ. كٝ پࣹلّࣹتْ، رٝنِنَاَࢠࣹ دࣹفْتࣹرࣹ ندࣹنْ ࢠَاوٝ مَࢠّࣹ، هِنٝ إࣹ سِكّࣹ فِي مَيْرࣹ سِكِّتِيࢠࣹ.

(15) نٝدُّ (يِمْࢠࣹ ࢠࣹنْ) كِسَنْ; ݧِيࢠِرَا نٝ يَمِرَطَا نٝنْ، وَتَ أَ جٝكُّ نبࣹلࣹيطࣹ مَࢠّࣹ طࣹنْ، مَاكَا: "مِي گٝمْطِنِي كٝ الله جِپِّنِ كٝنْ (فٝوْ) إࣹ دࣹفْتࣹ، مِي يَمِرَامَ كَدِ، نُنْطُگٝلْ هَكُّندࣹ مٝنْ. الله وٝنِ جٝومِ أَمࣹنْ إࣹ جٝومِ مٝنْ. كٝ مࣹنࣹنْ گٝلّࣹ أَمࣹنْ وٝودَنِ، كٝ أٝنٝنْ كَدِ گٝلّࣹ مٝنْ وٝودَنِ. يࣹدّٝندِرَلْ وٝو أَلَا هَكُّندࣹ أَمࣹنْ إࣹ مٝنْ. الله مٝوࢠٝيَيْ إࣹ دࣹنْندَانَلْ. كٝ كَ مَكّٝ وٝنِ رُتّٝرْدࣹ ندࣹنْ".

(16) ࢠࣹنْ وࣹنْنجِرٝوࢠࣹ إࣹ دِينَ الله كَنْ، ࢠَاوٝ كَ جَࢠَنࣹيدࣹ، هُجَّ مَࢠّࣹ أٝنْ كٝ مࣹيرࣹ، تِكّࣹرࣹ نٝ فَوِي إࣹ مَࢠّࣹ، هِنٝ وٝودَنِ ࢠࣹ كَدِ لࣹپْتࣹ سَطْتُطࣹ.

(17) كٝ الله وٝنِ جِپِّنِرِ دࣹفْتࣹرࣹ ندࣹنْ گٝونگَ، إࣹ مَنْندَكّࣹ طࣹنْ. إࣹ كٝ أَنْندِنْ مَا? هِنٝ گَسَ هَرَ دَرْنگَلْ نگَلْ كٝ كٝ ࢠَدِي?

(18) ࢠࣹيَ ࢠࣹ گٝمْطِنَالِ نگَلْ ࢠࣹنْ، هِࢠࣹ هࣹݧٝرِنگَلْ; گٝمْطِمْࢠࣹ ࢠࣹنْ كَنْ، كٝ ࢠࣹ هُلُ فِي مَگَّلْ، إࣹ هٝورࣹ هِࢠࣹ فࣹلِّتِ وٝنْندࣹ كٝ نگَلْ گٝونگَ. أَوَ أَنْندࣹي وٝنْࢠࣹ يࣹدّٝندِرࣹيدࣹ ࢠࣹنْ فِي دَرْنگَلْ، كٝ پࣹلّࣹتْ، ࢠࣹ وٝنِي إࣹ ندࣹرْ مَجّࣹرࣹ وٝطُّندࣹ.

(19) الله كٝ نࣹوَنِيطٝ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ. هِمٝ أَرْسِكࣹ أٝنْ مٝ أٝ مُويَنِ. كٝ كَنْكٝ وٝنِ دٝولْنُطٝ، فٝولُطٝ أٝنْ.

(20) كَلَ تَوْطٝ كٝ بَرَاجࣹ ندُنْ لَاكَرَ أٝ فَالَا، مࣹنْ ࢠࣹيْدَنَيْ مٝ نبَرْجَارِ مَكّٝ ندِنْ; كَلَ كَدِ تَوْطٝ كٝ جَيْرُ ندُنْ أَدُنَ أٝ يِطِ، مࣹنْ أٝكَّيْ مٝ طُمْ، هَرَ نٝنْ أٝ أَلَا كَ كَالَرَ (تٝنْ) گࣹࢠَلْ.

(21) كَا هَرَ دࣹنْندَاَاطِ هِنٝ وٝودَنِ ࢠࣹ، سَرئِنَنْطِ ࢠࣹ إࣹ دِينَ كَنْ، كٝنْ كٝ الله دُݝَاكِ فِي مُنْ? سِ وَنَانٝ فِي كٝنْنگٝلْ هَپُّنٝونگٝلْ نگٝلْ فِي سࣹنْندُگٝلْ هَكُّندࣹ مَࢠّࣹ ، ݧَاوࣹتࣹنٝ هَكُّندࣹ مَࢠّࣹ. كٝ پࣹلّࣹتْ، هِنٝ وٝودَنِ تٝوݧُࢠࣹ ࢠࣹنْ، لࣹپْتࣹ مُوسُطࣹ.

(22) أَ يِئَيْ تٝوݧُࢠࣹ ࢠࣹنْ هِنٝ هُلِ إࣹ كٝ ࢠࣹ فْگِّتِي كٝنْ، طُمْ لࣹ كٝ يَنْتَيْطُمْ إࣹ مَࢠّࣹ. گٝمْطِمْࢠࣹ ࢠࣹنْ، ࢠࣹ گٝلِّ گٝلّࣹ مٝڃّࣹ، هِࢠࣹ إࣹ دِنْنگِرࣹ نَاكٝوجِ، هَرَ هِنٝ وٝودَنِ ࢠࣹ كَ جٝومِ مَࢠّࣹ، كَلَ كٝ ࢠࣹ فَالࣹتࣹي. كٝ طُمْ تِگِرِ، وٝنِ ࢠُرَلْ مَوْنُنگَلْ نگَلْ.

(23) كٝ طُمْ وٝنِ كٝ الله وࣹوْلِنِرْتَ جࣹيَاࢠࣹ مَكّٝ گٝمْطِمْࢠࣹ ࢠࣹنْ، ࢠࣹ گٝلِّ گٝلّࣹ مٝڃّࣹ. مَاكُ: "مِي لَمْندَاكِ أٝنْ نجٝࢠْدِ إࣹ طُمْ، سِ وَنَا تُنْ گِگّٝلْ إࣹنْطَمْ طَمْ". كَلَ فَگِّتِيطٝ مٝڃّࣹرࣹ، مࣹنْ ࢠࣹيْدَنَيْ مٝ مٝڃّࣹرࣹ إࣹ هٝورࣹ مَيْرࣹ. پࣹلّࣹتْ، الله كٝ هَفٝرٝووٝ، مَنْتٝووٝ.

(24) كَا ࢠࣹ وِئَيْ أٝ فࣹفِندٝرْ ندࣹ إࣹ هٝورࣹ الله فࣹنَاندࣹ. سِ الله هَاجُنٝ، أٝ نٝتَيْنٝ إࣹ ࢠࣹرْندَ مَا ندࣹنْ. الله نٝ مࣹهَ مࣹيرࣹ ندࣹنْ، أٝ تَبِنْتِنِرَ گٝونگَ كَنْ دَالٝلْ مَكّٝ نگٝلْ. پࣹلّࣹتْ، كٝ أٝ أَنْندُطٝ گُنْندٝوجِ ࢠࣹرْطࣹ طࣹنْ.

(25) كٝ كَنْكٝ وٝنِ جَࢠَنٝووٝ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ تُوبُوبُيࣹي، أٝ يَافٝو بٝنْطِ طِنْ، إࣹ هٝورࣹ هِمٝ أَنْندِ كٝ گٝلّٝتٝنْ كٝنْ.

(26) هِمٝ جَابِنَنَ گٝمْطِمْࢠࣹ ࢠࣹنْ، ࢠࣹ گٝلِّ مٝڃُّطِ، أٝ ࢠࣹيْدَنَ ࢠࣹ إࣹ ࢠُرَلْ مَكّٝ نگَلْ. هࣹيفࣹرْࢠࣹ ࢠࣹنْ نٝنْ، هِنٝ وٝودَنِ ࢠࣹ لࣹپْتࣹ سَطْتُطࣹ.

(27) سِ تَوْنٝ الله يَݧَّنّٝ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ فٝوْ أَرْسِكࣹ، پࣹلّࣹتْ، ࢠࣹ بࣹوَيْنٝ كَ لࣹيْدِ; كٝنٝ هِمٝ جِپِّنِرَ إࣹمْنبࣹرࣹ كٝ أٝ مُويِ. پࣹلّࣹتْ، كٝ أٝ هُمّپِتِيطٝ، يِ*أٝووٝ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ.

(28) كٝ كَنٝ وٝنِ جِپِّنٝووٝ تٝࢠٝ نگٝنْ ࢠَاوٝ ࢠࣹ تَڃَادࣹ، أٝ سَاكَ يُرْمࣹيندࣹ مَكّٝ ندࣹنْ. كٝ كَنْكٝ كَدِ وٝنِ گِطٝ، يࣹتِّنِيطٝ أٝنْ.

(29) هِنٝ جࣹيَا إࣹ مَاندࣹيجِ مَكّٝ طِنْ: تَگُگٝلْ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، إࣹ رُوگَيْطِ طِ أٝ سَاكِ طِنْ هَكُّندࣹ مَجِّ. كَنْكٝ إࣹ فِي مٝوࢠِندِرٝيْگٝلْ ࢠࣹ نگٝلْ، سِ أٝ هَاجِي، كٝ أٝ هَتَّنْطٝ.

(30) كَلَ كٝ هࣹࢠِ أٝنْ إࣹ مَسِيبٝ، كٝ سَبُ كٝ جُوطࣹ مٝنْ طࣹنْ فَگِّتِي كٝنْ. هِمٝ يَافٝو نٝنْ، كٝ طُوطِ.

(31) أٝنْ وٝنَالِ لَاوٝتٝوࢠࣹ كَ لࣹيْدِ ; أَلَنَا أٝنْ گَانِنْ الله، گِطٝ، وَنَا بَلّٝ.

(32) هِنٝ جࣹيَا إࣹ مَاندࣹيجِ مَكّٝ طِنْ: دٝگَيْطࣹ طࣹنْ كَ مَايٝ، وَأُطࣹ وَ پࣹلّࣹ.

(33) سِ أٝ هَاجِي، أٝ دࣹيڃِنَيْ هࣹنْندُ ندُنْ، ندُ وَطْتِنِرَ تَبِتْندُ إࣹ هٝورࣹ مَگّٝ. وٝنِي إࣹ طُمْ، مَاندࣹيجِ، وٝنَنْندࣹ كَلَ مُݧّٝتٝوطٝ، يࣹتّٝووٝ.

(34) مَا يٝولِرَطࣹ سَبُ كٝ ࢠࣹ فَگِّتِي كٝنْ. هِمٝ يَافٝو نٝنْ كٝ طُوطِ.

(35) وࣹنْنجٝوࢠࣹ ࢠࣹنْ إࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ، هِنٝ أَنْندِ ࢠࣹ أَلَا لَاوٝرْدࣹ.

(36) كَلَ كٝ أٝكَّطٝنْ إࣹ هُوندࣹ، كٝ دَكَمّࣹ نگُورْندَمْ أَدُنَ; كٝ كٝ وٝنِ كَ الله كٝنْ، ࢠُرِ مٝڃُّدࣹ وٝنِ كَدِ كٝ هࣹدّٝتٝو وٝنَنْندࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ فَوٝتٝوࢠࣹ إࣹ جٝومِ مَࢠّࣹ.

(37) وٝطِّتٝتٝوࢠࣹ بَكَّاتُوجِ مَوْطِ طِنْ إࣹ پَنْكَرࣹ طࣹنْ، سِ وٝنِي ࢠࣹ تِكِّي، هَرَ ࢠࣹ يَافٝتٝ،

(38) ࢠࣹنْ نٝوتِيࢠࣹ جٝومِ مَࢠّࣹ، ࢠࣹ ݧِࢠْنِ جُلْدࣹ ندࣹنْ، ࢠࣹ تَوَتَ فِيَاكُ مَࢠّࣹ أٝنْ كٝ دِيسٝندِرَلْ هَكُّندࣹ مَࢠّࣹ، نَفْقٝوࢠࣹ إࣹ كٝنْ كٝ مࣹنْ أَرْسِكِ ࢠࣹ،

(39) إࣹ ࢠࣹنْ تَوْࢠࣹ سِ بࣹوْرࣹ هࣹࢠِي ࢠࣹ، ࢠࣹ يٝتّٝتٝ.

(40) نجٝࢠْدِ بٝنّࣹرࣹ، كٝ بٝنࣹ يࣹرُ مُنْ. كَلَ نٝنْ يَافِيطٝ أٝ مٝڃِّنْتِنِ، هَرَيْ نجٝࢠْدِ مَكّٝ ندِنْ كٝ كَ الله وٝنِ. پࣹلّࣹتْ نٝنْ، أٝ يِطَا تٝوݧٝوࢠࣹ ࢠࣹنْ.

(41) كَلَ نٝنْ يٝتِّيࢠࣹ ࢠَاوٝ ࢠࣹ تٝوݧࣹيدࣹ، كٝ ࢠࣹنْ وٝنِ ࢠࣹ لَاوٝلْ نَنْنگِتِرࣹيدࣹ فَوَاكِ إࣹ مُنْ.

(42) كٝ لَاوٝلْ نَنْنگِتِرࣹيدࣹ فَوِي إࣹ مُنْ، كٝ تٝوݧٝوࢠࣹ يِمْࢠࣹ ࢠࣹنْ، هَرَ هِࢠࣹ بࣹوِرَ إࣹ لࣹيْدِ ندِنْ كٝ أَلْدَا إࣹ گٝونگَ: ࢠࣹ طٝنْ، هِنٝ وٝودَنِ لࣹپْتࣹ مُوسُطࣹ.

(43) مُݧِّيطٝ أٝنْ كَدِ أٝ يَافِي، كٝ پࣹلّࣹتْ، طُمْ نٝ جࣹيَا إࣹ پࣹلِّتَلْ فِيَكُوجِ.

(44) كَلَ مٝ الله مَجِّنِ، هَرَيْ گِطٝ (گٝو) أَلَنَا مٝ ࢠَاوٝ طُمْ. أَ تِنَيْ تٝوݧُࢠࣹ ࢠࣹنْ، تُمَ ࢠࣹ يِئِ لࣹپْتࣹ طࣹنْ، هِࢠࣹ وِئَ: "هَرَ لَاوٝلْ رُتّٝرْدࣹ نٝ وٝودِ?".

(45) أَ تِنَيْ ࢠࣹ كَدِ، هِࢠࣹ وࣹيࢠِتࣹي إࣹ مَگّࣹ، هَرَ هِࢠࣹ يَنْكِنِي كٝيࣹيرَ أٝنْ، هِࢠࣹ ندَارِرَ ندَارْطࣹ لࣹپَّگٝلْ. گٝمْطِمْࢠࣹ ࢠࣹنْ وِئَيَ: "كٝ وٝنِ هَيْرُ ࢠࣹ ࢠࣹنْ، كٝ ࢠࣹنْ هَيْرِمْࢠࣹ كٝئࣹ مُنْ إࣹ ࢠࣹيْنگُورࣹيجِ مَࢠّࣹ طِنْ ݧَلْندࣹ دَرْنگَلْ". أَنْندࣹي پࣹلّࣹتْ، تٝوݧُࢠࣹ ࢠࣹنْ نٝ إࣹ لࣹپْتࣹ دُومِيطࣹ.

(46) يِطْࢠࣹ أَلَنَا ࢠࣹ، كٝ وَلِّتٝو ࢠࣹ، هَرَ وَنَا الله. كَلَ نٝنْ مٝ الله مَجِّنِ، هَرَيْ لَاوٝلْ أَلَنَا مٝ.

(47) نٝوتࣹي جٝومِ مٝنْ، أَدٝ ݧَلَاندࣹ أَرْدࣹ، ندࣹ رُتّٝووٝ أَلَا طُمْ كَ الله. ندࣹنْ ݧَلْندࣹ، فَابٝرْدࣹ أَلَنَا أٝنْ، يࣹدُّگٝلْ كَدِ أَلَنَا أٝنْ.

(48) سِ ࢠࣹ طُورْنِكࣹ نٝنْ، هَرَيْ مࣹنْ نُلِرَالِ مَ، أَيْنٝووٝ ࢠࣹ: فَوَاكِ مَ، سِ وَنَا يٝتِّنگٝلْ. سِ وٝنِي مࣹنْ مࣹيطِنِرِ نࣹطَّنْكࣹ أٝنْ نࣹيمَ إِمّٝرْدࣹ إࣹ أَمࣹنْ، أٝ وࣹوْلِرَ مٝ; سِ كٝ بٝنࣹ نٝنْ هࣹࢠِ ࢠࣹ، سَبُ كٝنْ كٝ جُوطࣹ مَࢠّࣹ طࣹنْ أَرْدِنِرَ، أَ تَوَيْ پࣹلّࣹتْ، نࣹطّٝ أٝنْ كٝ جࣹدُّطٝ سَنّࣹ !

(49) كٝ الله وٝودَنِ لَامَگٝلْ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ. هِمٝ تَگَ كٝ أٝ مُويِ. هِمٝ يࣹطَ مٝ أٝ مُويَنِ ࢠِطْࢠࣹ رࣹوْࢠࣹ، أٝ يࣹطَ كَدِ مٝ أٝ مُويَنِ ࢠِطْࢠࣹ وٝرْࢠࣹ.

(50) مَا أٝ جِلِّرَ ࢠࣹ وٝرْࢠࣹ إࣹ رࣹوْࢠࣹ; هِمٝ وَطَ دِمَرٝ، أٝنْ مٝ أٝ مُويَنِ. پࣹلّࣹتْ، كٝ أٝ أَنْندُطٝ، هَطْتَنْطٝ.

(51) سࣹلَّنْتَا نࣹطَّنْكࣹ، ندࣹ الله يࣹوْتَتَ مٝ، سِ وَنَا وَهَيُ، مَا إࣹ ࢠَاوٝ هࣹيدٝ، مَا أٝ إِمِّنَ نُلَاطٝ (الملائكةأَجٝ)، أٝنْ وَيَهِنَ إࣹ دُݝَيࣹي(سَكِّتٝورࣹ) مَكّٝ، كٝنْ كٝ أٝ هَاجِ. پࣹلّࣹتْ، كٝ أٝ تٝووُطٝ، ݧࣹيݧُطٝ.

(52) كٝ وَنٝ نٝنْ مࣹنْ وَيَيِنِرْ إࣹ مَاطَ، رُوهُ أٝنْ القرآن إࣹ يَمِرٝورࣹ أَمࣹنْ. أَ أَنْندَانٝ كٝ وٝنِ دࣹفْتࣹرࣹ، مَا گٝمْطِنَلْ; سِ كٝ وٝنِ، مࣹنْ وَطُندࣹ ندَيْگُ، نگُ مࣹنْ فࣹوْنِرْتَ أٝنْ مٝ مࣹنْ مُويَنِ إࣹ جࣹيَاࢠࣹ أَمࣹنْ ࢠࣹنْ. پࣹلّࣹتْ، كٝ أَ فࣹوْنٝووٝ إࣹ لَاوٝلْ فٝݖِّينگٝلْ،

(53) لَاوٝلْ الله، أٝنْ جࣹيْطٝ كٝ وٝنِ كَمُّولِ إࣹ كَ لࣹيْدِ. أَنْندࣹي، كٝ كَ الله، فِيَكُوجِ طِنْ رُتّٝتٝو.

(1) هَا، مِيمْ.

(2) مِي وٝوندِرِي دࣹفࣹرࣹ ࢠَنْنگُندࣹ ندࣹنْ !

(3) مࣹنْ وَطِي ندࣹ جَنْنگࣹتࣹيندࣹ إࣹ طࣹمْنگَلْ آرَبُ، بࣹلَجٝؤٝ، أٝنْ هَقِّلَيْ.

(4) پࣹلّࣹتْ، هِندࣹ وٝنِ كَ أَمࣹنْ گَا، كَ يُمَّ دࣹفْتࣹ، تٝووُندࣹ، ݧࣹݧَاندࣹ.

(5) إࣹ مࣹنْ أَݖِّرَيْ جِپِّنَنگٝلْ أٝنْ جَنْتٝورࣹ ندࣹنْ، طُورْنگَلْ أٝنْ، فِي أٝنْ وٝنِي يِمْࢠࣹ فَنْتِمْࢠࣹ?

(6) طُوطِي إࣹ أَنَّبَاࢠࣹ ࢠࣹ مࣹنْ نُلِ إࣹ أَدِنٝوࢠࣹ ࢠࣹنْ.

(7) أَنَّبَاجٝ أَرَالِ إࣹ مَࢠّࣹ، سِ وَنَا ࢠࣹ جَلْكِتَيْ مٝ.

(8) مࣹنْ هَلْكِ كٝ ࢠُرِ ࢠࣹ سَطْتُدࣹ دٝولࣹ ! تٝودࣹ مِسࣹ أَدِنٝوࢠࣹ ࢠࣹنْ يَوْتِينٝ.

(9) كٝ پࣹلّࣹتْ، سِ أَ لَمْندِكࣹ ࢠࣹ: "كٝ هٝمْنبٝ تَگِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ?" ࢠࣹ وِئَيْ: "كٝ فٝولُطٝ، أَنْندُطٝ أٝنْ تَگِ طِ".

(10) أٝنْ وَطِرَنْطٝ أٝنْ لࣹيْدِ ندِنْ وࣹيرْتَاندِ، أٝ وَطَنِ أٝنْ طَتࣹ إࣹ مَيْرِ، فِي نٝ طٝوٝرٝنْ.

(11) أٝنْ جِپِّنْرْطٝ تٝࢠٝ كَ كَمُّو، يࣹرُ هَاجُ، مࣹنْ وُورْنِتِرِ نگٝ لࣹيْدِ مَايْندِ. كٝ نٝنْ يَلْتِنِرْتࣹطٝنْ.

(12) أٝنْ تَگُطٝ نٝونࣹيجِ طِنْ فٝوْ، أٝ وَطِرَنِ أٝنْ إࣹ لَاطࣹ طࣹنْ إࣹ نࣹيمٝرَاطِ طِنْ، كٝ وَطّٝرٝنْ،

(13) فِي يٝ أٝنْ فٝتٝندِرْ إࣹ هٝورࣹ مَجِّ، رࣹفْتِ أَنْندِتٝنْ نࣹيمَ جٝومِ مٝنْ أٝنْ، تُمَ فٝتٝندِرْطٝنْ إࣹ مَجِّ، وِئٝنْ أٝنْتُمَ: "سࣹنَيࣹي وٝودَنِي إࣹلْتَنْطٝ مࣹنْ طُمْ أٝنْ، مࣹنْ وٝنَانٝ تَاقَنَاࢠࣹ طُمْ.

(14) مࣹنࣹنْ نٝنْ، كٝ كَ جٝومِ أَمࣹنْ وٝنِ كٝ مࣹنْ رُتّٝتٝو".

(15) ࢠࣹ وَطَنِ مٝ گࣹࢠَلْ إࣹ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ ! پࣹلّࣹتْ، أٝنْ نࣹطَّنْكࣹ كٝ يࣹدُّطٝ ࢠَنْنگُطٝ.

(16) كَا وٝو أٝ جٝگِتِ ࢠِطْࢠࣹ رࣹوْࢠࣹ إࣹ كٝ أٝ تَگِ كٝنْ، أٝ هࣹيرْتِنَنِ أٝنْ ࢠِطْࢠࣹ وٝرْࢠࣹ ࢠࣹنْ?

(17) سِ گٝطّٝ إࣹ مَࢠّࣹ وࣹوْلِنِرَامَ كٝ أٝ وَطِرَنِ كٝنْ هِنَرَنْكࣹ أٝنْ مِسَلْ، يࣹيسٝ مَكّٝ نگٝنْ نِࢠِّتَ مࣹتّࣹيندِ هٝنبٝ سُوطِ.

(18) إࣹ كٝ أٝنْ نࣹئࣹتࣹيطٝ إࣹ ݖُطَارِ، هَرَ كَنْكٝ أٝنْ كَ يࣹدّٝندِرَلْ، وَنَا أٝ ࢠَنْنگِنٝووٝ هَالَ?

(19) ࢠࣹ وَطِ الملائكة تَوْࢠࣹ ࢠࣹنْ كٝ جࣹيَاࢠࣹ جٝمْ هِنَرَنْكࣹ أٝنْ، رࣹوْࢠࣹ ! كࣹرࣹ ࢠࣹ سࣹيدِنٝ تَگࣹيدࣹ ࢠࣹنْ? أَوَ فِي سࣹيدَگٝلْ مَࢠّࣹ نگٝلْ وِنْندࣹتࣹ، ࢠࣹ لَمْندِتٝيࣹي.

(20) ࢠࣹ وِئِتِ: "سِ جٝمْ هِنَرَنْكࣹ أٝنْ هَاجُنٝ، مࣹنْ رࣹوَتَانٝ ࢠࣹ". ࢠࣹ أَلَنَا طُمْ هَيْ إࣹ گَنْندَلْ; ࢠࣹ وٝنَالِ، سِ وَنَا هِࢠࣹ فࣹنَ.

(21) كَا كٝ دࣹفْتࣹرࣹ مࣹنْ أَدَّنِ ࢠࣹ أَدٝ ندࣹي طٝو، هَرَ كٝ ندࣹنْ وٝنِ كٝ ࢠࣹ جٝگِتِي?

(22) سِ كٝ وٝنِ، ࢠࣹ وِئِــُٔ: "مࣹنࣹنْ مࣹنْ تَوُ بَابِرَاࢠࣹ أَمࣹنْ ࢠࣹنْ إࣹ دِينَ، هَرَيْ كٝ إࣹ بَتّࣹيجِ مَࢠّࣹ مࣹنْ فࣹيوِرْتَ".

(23) كٝ وَنٝ نٝنْ، مࣹنْ نُلَالِ جࣹرْتِنٝووٝ أَدِتَادࣹ مَ إࣹ سَارࣹ وٝو، سِ وَنَا هٝورࣹيࢠࣹ مَࢠّࣹ نࣹيمِيࢠࣹ وِئَيْ: "مࣹنࣹنْ مࣹنْ تَوُ بَابِرَاࢠࣹ أَمࣹنْ ࢠࣹنْ إࣹ دِينَ، طُمْ نٝنْ كٝ بَتّࣹيجِ مَࢠّࣹ طِنْ مࣹنْ ݧࣹنْتِنْتَ".

(24) أٝ مَاكِ: "هَيْ سِ مِي أَدَّنْ أٝنْ كٝ ࢠُرِ فࣹيوُدࣹ كٝنْ كٝ تَوْرُطٝنْ بَابِرَاࢠࣹ مٝنْ ࢠࣹنْ?" ࢠࣹ وِئِ: "مࣹنࣹنْ دࣹيْ، كٝ مࣹنْ يࣹدُّࢠࣹ كٝ نُلِدَطٝنْ إࣹ مُنْ كٝنْ".

(25) مࣹنْ يٝتِّي إࣹ مَࢠّࣹ. ندَارࣹ نٝ بَتَّندࣹ فࣹنُّࢠࣹ ࢠࣹنْ وَأُنٝو.

(26) جَنْتٝ تُمَ إبراهيم مَاكَنّٝو بࣹنْ مَكّٝ إࣹ يِمْࢠࣹ مَكّٝ ࢠࣹنْ: "مِينْ أٝنْ كٝ مِي دَطْندِيطٝ إࣹ كٝنْ كٝ رࣹوٝتٝنْ،

(27) سِ وَنَا أٝنْ تَگُطٝ لَمْ، كَنْكٝ أٝنْ، أٝ فࣹوْنَيْ لَمْ".

(28) أٝ وَطِ نگٝلْ كٝنْنگٝلْ هࣹدّٝتٝونگٝلْ ࢠَاوٝ مَكّٝ، بࣹلَجٝؤٝ، ࢠࣹ رُتّٝتٝ.

(29) سِ كٝ وٝنِ، مِي دَكْمِتِي ࢠࣹي طٝو إࣹ بَابِرَاࢠࣹ مَࢠّࣹ ࢠࣹنْ، هَا ندࣹ گٝونگَ كَنْ أَرُنٝو إࣹ مَࢠّࣹ، إࣹ نُلَاطٝ ࢠَنْنگُطٝ.

(30) ندࣹ گٝونگَ كَنْ أَرُنٝو إࣹ مَࢠّࣹ، ࢠࣹ وِئِ: "طُمْ كٝ نبِلࣹوُ، مࣹنࣹنْ كٝ مࣹنْ يࣹدُّࢠࣹ طُمْ".

(31) ࢠࣹ وِئِتِ: "كٝ هَنّٝ ندࣹي القرآن جِپِّنࣹي إࣹ گٝرْكٝ مَوْطٝ ݖَئࣹ طࣹنْ طِطِ?"

(32) إࣹ كٝ كَمْࢠࣹ سࣹنْندَتَ يُرْمࣹيندࣹ جٝومِ مَا ندࣹنْ? أَوَ كٝ مࣹنࣹنْ وٝنِ كٝ سࣹنْندِ نگُورࣹيجِ مَࢠّࣹ طِنْ هَكُّندࣹ مَࢠّࣹ، كَ نگُورْندَمْ أَدُنَ. مࣹنْ ࢠَمْتِ يٝگَ مَࢠّࣹ إࣹ هٝورࣹ ࢠࣹيَ، دَرْجَاجِ، فِي نٝ يٝگَ مَࢠّࣹ جٝگِتٝرَ يٝگَ إࣹيلْتَنࣹ. هَرَيْ كٝ يُرْمࣹيندࣹ جٝومِ مَا ندࣹنْ، ࢠُرِ مٝڃُّدࣹ إࣹ كٝنْ كٝ ࢠࣹ مٝوࢠَتَ.

(33) سِ وَنَانٝ فِي وَتَ يِمْࢠࣹ ࢠࣹنْ وٝنُ مٝفْتࣹ گٝوتࣹ، مࣹنْ وَطَيْنٝ إࣹ ݖُوطِ كَلَ يࣹدُّطٝ هِنَّرَنْكࣹ أٝنْ: كِࢠّࣹ كَالِسِ إࣹ ݝَبِّرْطࣹ كَ ࢠࣹ ڃَوْرَ;

(34) (مࣹنْ وَطَيْنٝ كَدِ) إࣹ ݖُوطِ مَࢠّࣹ طِنْ، دَمࣹ إࣹ ندَطُّدࣹ كَ ࢠࣹ سٝࢠِّندٝو،

(35) إࣹ ݖُطَارِ. كٝنٝ طُمْ فٝوْ وٝنَالِ، سِ وَنَا دَكَمّࣹ نگُورْندَمْ أَدُنَ. كٝ لَاكَرَ أٝنْ كَ جٝومِ مَا، وٝنِ فِي گٝمْطُمْࢠࣹ ࢠࣹنْ.

(36) كَلَ طُورْنِيطٝ إࣹ جَنْتٝورࣹ هِنَرَنْكٝ أٝنْ، مࣹنْ وُورْتَ مٝ إࣹ سࣹيْتَنُورُ، هَرَ كٝ ندُنْ وٝنِ كٝ أٝ ݖٝووٝدَتَ.

(37) طِنْ (أَوَسࣹيجِ) نٝنْ، كٝ سَكّٝرَيْطِ ࢠࣹ إࣹ پࣹيوَلْ نگَلْ، هَا كَ ࢠࣹ سِكَّتَ هِࢠࣹ فࣹيوِ.

(38) هَا سِ أٝ أَرٝيِي إࣹ أَمࣹنْ، أٝ وِئَ: "إࣹي بٝنࣹ ! هَرَ هَكُّندࣹ أَمْ إࣹ مَا وٝطِّرَيْنٝ گٝطَّلْ فُطِّرْطࣹ طࣹنْ طِطِ وٝطِّرِ ! أَ بٝنِي ݖِنْندَدِيطٝ !".

(39) طُمْ نَفَتَا أٝنْ هَنْندࣹ - ندࣹ تَوْنٝو أٝنْ تٝوݧِي - وٝنْندࣹ كٝ كَ لࣹپْتࣹ هَوْتُطٝنْ.

(40) إࣹ كٝ أَنْ نَنِنْتَ فَاطُطٝ مَا فࣹوْنَا بُمْطٝ إࣹ أٝنْ وٝنْطٝ إࣹ ندࣹرْ مَجّࣹرࣹ ࢠَنْنگُندࣹ?

(41) سِ مࣹنْ أَرُ ڃࣹتِّتُدࣹ مَ، هَرَيْ كٝ مࣹنْ يٝتّٝتٝوࢠࣹ إࣹ مَࢠّࣹ.

(42) مَا هَرَ مࣹنْ هٝلّࣹ كٝ مࣹنْ كَمْنبِرَنِ ࢠࣹنْ كٝنْ، هَرَيْ نٝونْ كٝ مࣹنْ گَيْنٝوࢠࣹ فِي مَࢠّࣹ.

(43) جٝگِتٝ كٝنْ كٝ وَهَيِنَا إࣹ مَاطَ. پࣹلّࣹتْ، هِطَ جٝكِّ لَاوٝلْ فٝݖِّينگٝلْ.

(44) ندࣹي نٝنْ القرآن ، كٝ ندࣹ تࣹدُّنگَلْ مَا أَنْ إࣹ يِمْࢠࣹ مَا ࢠࣹنْ. أَرَيْ ندࣹ لَمْندِتࣹطٝنْ.

(45) لَمْندِي إࣹ ࢠࣹنْ نُلَاࢠࣹ أَمࣹنْ ࢠࣹ مࣹنْ نُلُنٝو أَدٝ مَا، سِ مࣹنْ وَطُنٝ كٝ وَنَا هِنَّرَنْكࣹ أٝنْ، رࣹوࣹتࣹيطِ گٝو?

(46) گٝمْطِي مࣹنْ إِمِّندِنِي موسى إࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ، هَا إࣹ فرعون إࣹ نبَتُ مُنْ نگُنْ، أٝ مَاكِ: "كٝ مِي نُلَاطٝ جࣹيْطٝ تَگُ نگُنْ".

(47) ندࣹ أٝ أَدَّنّٝوࢠࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ، إࣹ جَكَ يٝو هِࢠࣹ جَلَطِ.

(48) مࣹنْ هٝلَّالِ ࢠࣹ آيَ، سِ وَنَا هَرَيْ أٝنْ ࢠُرِ أَدِيطٝ مٝ أٝنْ مَوْنُدࣹ. مࣹنْ نَنْنگِرِ ࢠࣹ لࣹپْتࣹ طࣹنْ، بࣹلَجٝؤٝ، ࢠࣹ رُتّٝتٝ.

(49) ࢠࣹ وِئِ: "كٝ أَنْ يٝ نبِلࣹيجٝ! تٝرَنٝ مࣹنْ جٝومِ مَا، كٝ أٝ أَهٝدِ كٝنْ إࣹ مَاطَ. پࣹلّࣹتْ، كٝ مࣹنْ فࣹيوٝوࢠࣹ".

(50) ندࣹ مࣹنْ هُنْݖَنّٝوࢠࣹ لࣹپْتࣹ طࣹنْ، إࣹ جَكَ يٝو، هِࢠࣹ فِرْتَ أَهَدِ .

(51) فرعون نٝدِّ إࣹ يِمْࢠࣹ ࢠࣹنْ، أٝ وِئِ: "كٝ أٝنٝنْ يٝ يِمْࢠࣹ أَمْ، وَنَا مِينْ جࣹيْ لَامُ مِسْرَ? إࣹ هِنٝ طِي ݖَنْطِ نٝ إِلَ إࣹ لࣹيْ كُࢠࣹيجِ أَمْ? إࣹ أٝنْ يِيَتَا?

(52) كَا مِي ࢠُرَا تࣹدُّدࣹ أٝو هٝيْطٝ مٝ إࣹࢠَّتَا لَࢠِّنْندࣹ هَالَ?

(53) أَوَ كٝ هَنّٝ جَوْلࣹ كَݝّࣹ وࣹرْلࣹي إࣹ مَكّٝ? مَاطُمْ الملائكةءࣹنْ أَرْدَ إࣹ مَكّٝ، نَ جٝكّٝندِرِ?"

(54) أٝ طَيْتِنِ يِمْࢠࣹ مَكّٝ ࢠࣹنْ، ࢠࣹ طٝوْتِي مٝ. پࣹلّࣹتْ، ࢠࣹنْ طٝنْ لَاتِنٝ يِمْࢠࣹ فَاسِقِيࢠࣹ.

(55) ندࣹ ࢠࣹ تِكِّنّٝو مࣹنْ، مࣹنْ يٝتِّي إࣹ مَࢠّࣹ; مࣹنْ يٝولِ ࢠࣹ فٝوْ.

(56) مࣹنْ وَطِ ࢠࣹ أَرْدِيࢠࣹ إࣹ مِسَلْ، وٝنَنْندࣹ سَكِّيتِيࢠࣹ ࢠࣹنْ.

(57) تُمَ ندࣹ گࣹطَلْ نࣹينࣹ مَرْيَمَ أٝنْ پِيِرَنٝو مِسَلْ، تَوَا يٝو، يِمْࢠࣹ مَا ࢠࣹنْ نٝ سٝوڃِرَ طُمْ.

(58) ࢠࣹ وِئِ: "كٝ رࣹوࣹتࣹيطِ أَمࣹنْ طِنْ ࢠُرِ مٝڃُّدࣹ، كَا كٝ كَنْكٝ?" ࢠࣹ پِيِرَنَالِ مَ نگَلْ (مِسَلْ)، سِ وَنَا فِي وࣹنْنجُگٝلْ. وُرِنْ، كٝ ࢠࣹ يِمْࢠࣹ سَطْتُࢠࣹ وࣹنْنجࣹ.

(59) أَوَ (إِيسَا) وٝنَالِ، سِ وَنَا جࣹيَاطٝ مٝ مࣹنْ نࣹيمِنِ إࣹ مُنْ، مࣹنْ وَطِ مٝ مِسَلْ وٝــَٔنْندࣹ ࢠِطْࢠࣹ إسرائيل ࢠࣹنْ.

(60) سِ مࣹنْ هَاجُنٝ، مࣹنْ وَطْتِتِرَيْنٝ أٝنْ الملائكة، كٝ لٝمْتٝو كَ لࣹيْدِ.

(61) پࣹلّࣹتْ نٝنْ، كٝ أٝ أَنْندِنٝرْدࣹ دَرْنگَلْ. وَتَ أٝنْ سِكِّتٝ فِي مَگَّلْ. جٝكّࣹي لَمْ، كٝ طُمْ وٝنِ لَاوٝلْ فٝݖِّينگٝلْ نگٝلْ.

(62) وَتَ سࣹيْتَانࣹ فَلٝ أٝنْ ! تَوْدࣹ پࣹلّࣹتْ، كٝ أٝ گَݧٝ مٝنْ ࢠَنْنگُطٝ.

(63) ندࣹ إِيسَا أَرْدُنٝو إࣹ ࢠَنْنگُطِ طِنْ، أٝ مَاكِ: "گٝمْطِي مِي أَرْدَنِي أٝنْ إࣹ ݧࣹيݧَلْ، إࣹ نٝ مِي ࢠَنْنگِنِرَنَ أٝنْ يٝگَ إࣹ كٝ لُوتٝندِرْطٝنْ إࣹ مُوطُمْ. هَرَيْ هُلࣹي الله، طٝفْتٝطٝنْ مِي.

(64) پࣹلّࣹتْ، كٝ الله وٝنِ جٝومِ أَمْ إࣹ جٝومِ مٝنْ. رࣹوࣹي مٝ. كٝ طُمْ وٝنِ لَاوٝلْ فٝݖِّينگٝلْ نگٝلْ".

(65) فࣹدّࣹيجِ طِنْ لُوتٝندِرِ هَكُّندࣹ مَجِّ. بٝنࣹ وٝودَنِي ࢠࣹنْ يࣹدُّࢠࣹ، إِمّٝرْدࣹ إࣹ لࣹپْتࣹ ݧَلَاندࣹ مُوسُطࣹ !

(66) هَرَ ࢠࣹ هَبِّ سِ وَنَا ندࣹ دَرْنگَلْ نگَلْ أَرْدَتَ ࢠࣹ إࣹ جُهَلْ، هَرَ ࢠࣹ سٝأَا أَنْندُدࣹ?

(67) وࣹلْدُࢠࣹ ࢠࣹنْ، ندࣹنْ ݧَلْندࣹ، يٝگَ مَࢠّࣹ كٝ أَيْرٝوࢠࣹ يٝگَ; سِ وَنَا گٝمْطِمْࢠࣹ ࢠࣹنْ.

(68) كٝ أٝنٝنْ يٝ جࣹيَاࢠࣹ أَمْ ! كُلٝلْ أَلَنَا أٝنْ هَنْندࣹ; وٝنَا أٝنْ سُنٝتٝوࢠࣹ كَدِ.

(69) ࢠࣹنْ گٝمْطِمْࢠࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ، ࢠࣹ وٝنِ جࣹبِّلِيࢠࣹ،

(70) نَاتࣹي كَ الجنة، أٝنٝنْ إࣹ نَنْندُࢠࣹ إࣹ مٝنْ، هِطٝنْ سَطْنࣹي".

(71) گٝلّٝوࢠࣹ هِنٝ وَنْنگِندِنࣹي إࣹ مَࢠّࣹ، مِرَنْنجِ كَݝّࣹ إࣹ ݧࣹدُّطࣹ; هِنٝ تٝنْ كَدِ: كَلَ كٝ وٝنْكِي مُويࣹتࣹي، گِتࣹ دَكْمِتٝو. هَرَ لࣹ، كٝ أٝنْ دُومٝتٝوࢠࣹ تٝنْ.

(72) كٝ أٝنْ وٝنِ الجنة مٝ رٝنِنِرَطٝنْ، سَبُ كٝنْ كٝ گٝلَّيْنٝطٝنْ.

(73) هِنٝ وٝودَنِ أٝنْ تٝنْ، مُويَنْتࣹيطِ طُوطُطِ، طِ ݧَامٝتٝنْ إࣹ مُنْ.

(74) پࣹلّࣹتْ، بٝمْࢠࣹ ࢠࣹنْ، كٝ كَ لࣹپْتࣹ جهنم ࢠࣹ لُتّٝيْتَ،

(75) ࢠࣹ هٝيْنَنْتَاكࣹ، تࣹي هَرَ كٝ ࢠࣹ تَڃِيࢠࣹ تٝنْ.

(76) مࣹنْ تٝوݧَانِ ࢠࣹ فࣹوْ، سِ كٝ وٝنِ، كٝ كَمْࢠࣹ وٝنِ تٝوݧُࢠࣹ ࢠࣹنْ.

(77) ࢠࣹ نٝدَّ: "كٝ أَنْ يٝ مَالِكُ ! يٝ جٝومِ مَا گَيْنُ فِي أَمࣹنْ !" أٝ وِئِ: "پࣹلّࣹتْ، كٝ أٝنْ لُتّٝوࢠࣹ تٝنْ !"

(78) "گٝمْطِي مࣹنْ أَدَّنِي أٝنْ گٝونگَ كَنْ ; كٝنٝ طُوطُࢠࣹ إࣹ مٝنْ، كٝ أَݧُࢠࣹ گٝونگَ".

(79) كَا ࢠࣹ إࣹࢠُّ فࣹيرࣹ ? دٝو كٝ مࣹنْ فࣹوْجٝوࢠࣹ !

(80) مَا هَرَ ࢠࣹ سِكُّ مࣹنْ نَنَتَا گُنْندٝوجِ مَࢠّࣹ إࣹ سِرُّ مَࢠّࣹ? أٝوءٝوٝيࣹ! نُلَاࢠࣹ أَمࣹنْ ࢠࣹنْ هِنٝ مَاندِنَ تَكّٝ مَࢠّࣹ.

(81) مَاكُ: "سِ هِنَرَنْكٝ أٝنْ نٝ جٝگِتِي ࢠِطّٝ، كٝ مِينْ أَدٝتٝنٝو رࣹوُدࣹ أٝنْ".

(82) سࣹنَيࣹي وٝودَنِي جࣹيْطٝ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، جࣹيْطٝ أَرْسِ كِنْ; لَاࢠُطٝ إࣹ كٝنْ كٝ ࢠࣹ سِفْرٝتَّ مٝ.

(83) أَݖُّࢠࣹ ࢠࣹ جُمْپَ، ࢠࣹ فِجَ هَا ࢠࣹ هَوْرٝيَ إࣹ ݧَلَاندࣹ مَࢠّࣹ ندࣹࢠࣹ فٝدَا.

(84) كٝ كَنْكٝ وٝنِ أٝنْ رࣹوࣹتࣹيطٝ كَ كَمُّ إࣹ رࣹوࣹتࣹيطٝ كَ لࣹيْدِ، كٝ كَنْكٝ كَدِ وٝنِ ݧࣹيݧٝ، أَنْندُطٝ أٝنْ.

(85) بَرْكࣹ هࣹيوَنِي أٝنْ مٝ وٝودَنِ طُمْ، لَامُ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ إࣹ كٝ وٝنِ هَكُّندࣹ مَجِّ. كٝ كَ مَكّٝ أَنْندَلْ فِي دَرْنگَلْ وٝنِ. كٝ كَ مَكّٝ كَدِ رُتّࣹتࣹطٝنْ.

(86) طِ ࢠࣹ رࣹوَتَ طِنْ گَانِنْ مَكّٝ جࣹيَهْ تࣹفٝورࣹ، سِ وَنَا سࣹيدِيࢠࣹ گٝونگَ ࢠࣹنْ، هَرَ هِࢠࣹ أَنْندِ.

(87) كٝ پࣹلّࣹتْ، سِ أَ لَمْندِكࣹ ࢠࣹ، كٝ هٝمْنبٝ وٝنِ كٝ تَگِ ࢠࣹ? ࢠࣹ وِئَيْ: "كٝ الله". كٝ هٝنّٝ نٝنْ ࢠࣹ يِيلِرَا?

(88) هِمٝ أَنْندِ كَدِ فِي كٝ (نُلَاطٝ) أٝنْ مَاكُنٝو كٝنْ: "كٝ أَنْ يٝ جٝومِ، پࣹلّࣹتْ، ࢠࣹي كٝ يِمْࢠࣹ ࢠࣹ گٝمْطِنْتَا".

(89) أَوَ يَافٝ ࢠࣹ، مَاكَا: "كِسَلْ !" أَرَيْ ندࣹبࣹ أَنْندَ.

44 Ad-Dukhan — 59 آيَة

(1) هَا، مِيمْ.

(2) مِي وٝوندِرِي دࣹفࣹرࣹ ࢠَنْنگُندࣹ ندࣹنْ !

(3) مࣹنْ جِپِّنِرِي ندࣹ إࣹ جࣹمَّ بَرْكِنَاطٝ، مࣹنْ سِفٝرِي وٝنْندࣹ جࣹرْتِنٝوࢠࣹ.

(4) كٝ إࣹ أٝنْ (جࣹمَّ) وٝنِ كٝ سࣹنْندࣹتࣹي كَلَ فِيَاكُ ݧࣹݧَاطٝ.

(5) طُمْ كٝ يَمِرٝورࣹ إِمّٝرْدࣹ كَ أَمࣹنْ. مࣹنْ سِفٝرِي وٝنْندࣹ نُلٝوࢠࣹ.

(6) يُرْمࣹيندࣹ إِمّٝرْندࣹ كَ جٝومِ مَا. پࣹلّࣹتْ، كٝ كَنْكٝ وٝنِ نَنٝووٝ، أَنْندُطٝ أٝنْ.

(7) جࣹيْطٝ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ إࣹ كٝ وٝنِ كٝنْ هَكُّندࣹ مَجِّ، سِ وٝنِي كٝ أٝنْ يَنَنَاࢠࣹ.

(8) رࣹوࣹتࣹيطٝ گٝو أَلَا، سِ وَنَا كَنْكٝ. هِمٝ وُورْنَ، هِمٝ وَرَ. جࣹيْطٝ أٝنْ جࣹيِ بَابِرَاࢠࣹ مٝنْ أَرَمْࢠࣹ ࢠࣹنْ.

(9) كٝ وٝنِ تُنْ، ࢠࣹ وٝنِي إࣹ سِكّࣹ، هِࢠࣹ فِجَ.

(10) هَبِّتٝ ݧَلْندࣹ كَمُّ نگُنْ أَرْدَتَ إࣹ ݖُرْكِ فࣹيݧُوكِ،

(11) سُدَّيْكِ يِمْࢠࣹ ࢠࣹنْ. ࢠࣹ وِئَنࣹي : "طُمْ كٝ لࣹپْتࣹ مُوسُطࣹ" !

(12) "جٝومِ أَمࣹنْ، هُنْݖَنْ مࣹنْ لࣹپْتࣹ طࣹنْ. مࣹنࣹنْ كٝ مࣹنْ گٝمْطِمْࢠࣹ".

(13) كٝ هٝنّٝ ࢠࣹ وَاجِتٝرْتَ، كٝ گٝونگٝ نُلَاطٝ ࢠَنْنگِنْطٝ أَرِينٝ ࢠࣹ?

(14) رࣹفْتِ ࢠࣹ هُݖِّتِ، ࢠࣹ وِئِ: "كٝ تِنْندِنَاطٝ، هَانگَاطٝ".

(15) أَوَ كٝ مࣹنْ إِتّٝوࢠࣹ لࣹپْتࣹ طࣹنْ سࣹيطَ; كٝنٝ پࣹلّࣹتْ، كٝ أٝنْ رُتِّتٝتٝوࢠࣹ.

(16) ݧَلْندࣹ مࣹنْ نَنْنگِرٝيْتَ وِپَّنْندࣹ مَوْندࣹ ندࣹنْ، پࣹلّࣹتْ، كٝ مࣹنْ يٝتّٝيْتٝوࢠࣹ.

(17) گٝمْطِي مࣹنْ جَرِّبِنٝكࣹ أَدٝ مَࢠّࣹ، يِمْࢠࣹ فرعون ࢠࣹنْ، نُلَاطٝ تࣹدُّطٝ أَرِ إࣹ مَࢠّࣹ،

(18) وٝنْندࣹ: "وَطࣹي إࣹ جُوطࣹ أَمْ، جࣹيَاࢠࣹ الله ࢠࣹنْ. مِينْ كٝ مِي نُلَاطٝ إࣹ مٝنْ، هٝولْنِيطٝ.

(19) إࣹ وٝندࣹ وَتَ أٝنْ تٝوْنِتَنٝ الله، تَوْدࣹ مِي أَدَّنَيْ أٝنْ هُجَّ ࢠَنْنگُطٝ.

(20) مِينْ كَدِ نٝنْ، مِي مٝولٝرِي جࣹيْطٝ لَمْ أٝنْ جࣹيِ أٝنْ، ندࣹ فِطٝتٝنْمِ كَاڃࣹ.

(21) سِ أٝنْ گٝمْطِنَا لَمْ، هَرَيْ سࣹلࣹي لَمْ".

(22) أٝ نٝدِّ جٝومِ مَكّٝ، وٝنْندࣹ: "ࢠࣹي طٝو، كٝ يِمْࢠࣹ بٝمْࢠࣹ".

(23) "يَادِرْ إࣹ جࣹيَاࢠࣹ أَمْ ࢠࣹنْ جࣹمَّ; أَنْندٝنْ كٝ أٝنْ ݖُكّࣹتࣹيࢠࣹ.

(24) أَݖَّا مَايٝ نگٝنْ دࣹيڃَ سِ أٝنْ لُمْنبِي ࢠࣹيَ كَنْ، كٝ كٝنُ يٝولࣹتࣹينگُ".

(25) طُوطِي كٝ ࢠࣹ أَݖِّ إࣹ گࣹسࣹ إࣹ ࢠُلِّ،

(26) إࣹ رࣹمُرُوجِ إࣹ كٝطَندࣹ تࣹدُّندࣹ،

(27) إࣹ نࣹيمَاجِ طِ ࢠࣹ وٝنُنٝو إࣹ نࣹيمِنِرࣹيدࣹ.

(28) كٝ وَنٝ نٝنْ. هٝوتِ مࣹنْ رٝنِنِ طِ يِمْࢠࣹ گٝو.

(29) كَمُّ نگُنْ تٝࢠَالِ ࢠࣹ، وَنَا لࣹيْدِ ندِنْ، ࢠࣹ وٝنَالِ كَدِ مُݧَنْتࣹيࢠࣹ.

(30) گٝمْطِي مࣹنْ دَطْندِي ࢠِطْࢠࣹ إسرائيل ࢠࣹنْ إࣹ لࣹپْتࣹ هٝيْنࣹيجࣹ;

(31) إِمّٝرْدࣹ إࣹ فرعون، تَوْدࣹ أٝ لَاتِنٝ تٝوْنِتِيطٝ، جࣹيَاطٝ إࣹ فَنْتِمْࢠࣹ ࢠࣹنْ.

(32) گٝمْطِي مࣹنْ سُࢠٝرِكࣹ ࢠࣹ إࣹ دٝوْ گَنْندَلْ إࣹ هٝورࣹ وِنْندࣹرࣹ ندࣹنْ،

(33) مࣹنْ أٝكِّ ࢠࣹ مَاندࣹيجِ، طِ جَرَّبُيࣹي ࢠَنْنگُطٝ وٝنِ إࣹ مُنْ.

(34) ࢠࣹي كَنْ نٝ وِئَ:

(35) "ندࣹي وٝنَالِ، سِ وَنَا مَايْدࣹ أَمࣹنْ أَرَنࣹرࣹ ندࣹنْ تُنْ. مࣹنْ وٝنَتَهْ إِمِّنْتِنٝيْتࣹيࢠࣹ.

(36) أَوَ أَرْتِرࣹي بَابِرَاࢠࣹ أَمࣹنْ ࢠࣹنْ، سِ وٝنِي كٝ أٝنْ گٝونگُࢠࣹ".

(37) كࣹرࣹ كٝ كَمْࢠࣹ ࢠُرِ مٝڃُّدࣹ كَا كٝ يِمْࢠࣹ "تُبَّئِنْ" ࢠࣹنْ إࣹ ࢠࣹيَ أَدِنٝوࢠࣹ ࢠࣹ? مࣹنْ هَلْكُنٝ ࢠࣹنْ، تَوْدࣹ هَرِ كٝ ࢠࣹ بٝمْࢠࣹ.

(38) مࣹنْ تَگِرَالِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ إࣹ كٝ وٝنِ هَكُّندࣹ مَجِّ كٝنْ، فِجٝوࢠࣹ.

(39) مࣹنْ تَگِرَالِ طِنْ طِطِ، سِ وَنَا إࣹ گٝونگَ. كٝنٝ كٝ ࢠُرِ طُوطُدࣹ إࣹ مَࢠّࣹ، ࢠࣹ أَنْندَا.

(40) پࣹلّࣹتْ، كٝ ݧَلَاندࣹ سࣹنْندِندِرٝ نگٝنْ، وٝنِ كٝ ࢠࣹ سَاتِنَنَا، كَمْࢠࣹ فٝفْ.

(41) ݧَلْندࣹ ندࣹ گِطٝ دُنْݖَنٝيْتَا گِطٝ مُنْ هَيْ هُوندࣹ; ࢠࣹ وٝنَالِ كَدِ وَلّٝيْتࣹيࢠࣹ،

(42) سِ وَنَا مٝ الله يُرْمَا وٝو. پࣹلّࣹتْ، كٝ كَنْكٝ وٝنِ فٝولُطٝ، هِنّٝتٝوطٝ أٝنْ.

(43) پࣹلّࣹتْ، لࣹگَّلْ زَقُّومِ نگَلْ،

(44) كٝ ݧَامࣹتࣹي بَكّٝندِمْࢠࣹ ࢠࣹنْ.

(45) أَ سِكَّيْ كٝ تَايِنَا ; هِنٝ فَتَ كَ ندࣹرْ دࣹيدِ.

(46) وَنٝ وُلْطَمْ فَتِرْتَ.

(47) نَنْنگࣹي مٝ پٝوطِرٝنْ كَ هَكُّندࣹ دُلْࢠَيْنگࣹ.

(48) رࣹفْتِ يُپّٝنْ كَ هٝورࣹ مَكّٝ، لࣹپْتࣹ فَتِنَاطࣹ طࣹنْ.

(49) مࣹيطُ نٝنْ ! أَنْ سِكُّنٝوطٝ كٝ أَ فٝولُطٝ، تࣹدُّطٝ.

(50) أَوَ كٝ طُمْ وٝنِ كٝنْ كٝ سِكِّتِطٝنْ إࣹ مُمْ.

(51) پࣹلّࣹتْ، گٝمْطُࢠࣹ ࢠࣹنْ وٝنِي إࣹ وࣹيرْدࣹ هٝولْنِيندࣹ،

(52) إࣹ ندࣹرْ گࣹسࣹ إࣹ ࢠُلِّ،

(53) هِࢠࣹ ࢠٝورْنٝو كٝ طَاتِ إࣹ كٝ تࣹكِّ، هَرَ هِࢠࣹ فࣹوْتٝندِرِ.

(54) كٝ وَنٝ نٝنْ. مࣹنْ رࣹسِنِ ࢠࣹ كَدِ يَاجُࢠࣹ گِتࣹ.

(55) هِࢠࣹ نٝدَّ تٝونْ، كَلَ مُويَنْتࣹيطِ، هَرَ كٝ ࢠࣹ هٝولِيࢠࣹ.

(56) ࢠࣹ مࣹيطَتَا تٝنْ مَايْدࣹ، سِ وَنَا مَايْدࣹ أَرَنࣹرࣹ ندࣹنْ. أٝ دَطْندَ ࢠࣹ لࣹپْتࣹ جَهِيمِ طࣹنْ.

(57) كٝ دٝكَّلْ نٝنْ، إِمّٝرْدࣹ كَ جٝومِ مَا. أَوَ كٝ طُمْ وٝنِ مَلٝورࣹ مَوْندࣹ ندࣹنْ.

(58) كٝ مࣹنْ نࣹوِنِرِ ندࣹ إࣹ طَمْنگَلْ مَاطَ، كٝ نٝ ࢠࣹ وَاجِتٝرَ.

(59) هَبِيتٝ. كَمْࢠࣹ كٝ ࢠࣹ هَبِّتِيࢠࣹ.

45 Al-Jathiya — 37 آيَة

(1) هَا، مِيمْ.

(2) جِپَّگٝلْ دࣹفْتࣹرࣹ ندࣹنْ، كٝ إِمّٝرْدࣹ كَ الله، فٝولُطٝ، ݧࣹيݧٝ أٝنْ.

(3) پࣹلّࣹتْ، هِنٝ إࣹ تَگُگٝلْ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، مَاندࣹيجِ وٝنَنْندࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ.

(4) هِنٝ إࣹ تَگُگٝلْ أٝنْ نگٝلْ كَدِ، إࣹ كٝنْ كٝ أٝ سَاكِ إࣹ رُوگَيْطِ، كٝ مَاندࣹيجِ وٝنَنْندࣹ يِمْࢠࣹ يَنَنَاࢠࣹ ࢠࣹنْ.

(5) إࣹ لُوتٝندِرْگٝلْ جࣹمَّ إࣹ ݧَلٝرْمَ نگٝلْ، إࣹ كٝنْ كٝ الله جِپِّنِرِ كَ كَمُّ، إِمّٝرْدࣹ إࣹ تٝࢠٝ، أٝ وُورْنِتِرِ نگٝ لࣹيْدِ ندِنْ ࢠَاوٝ ندِ يٝورُدࣹ، إࣹ فِرْلِتَگٝلْ كࣹنࣹيلِ طِنْ، كٝ مَاندࣹيجِ وٝنَنْندࣹ يِمْࢠࣹ هَقِّلٝوࢠࣹ ࢠࣹنْ.

(6) طِنْ كٝ مَاندࣹيجِ الله طِ مࣹنْ جَنْنگِرَنْتࣹ هٝورࣹ گٝونگَ. إࣹ كٝ هٝنْطُمْ إࣹ يࣹوْتࣹرࣹ ࢠࣹ گٝمْطِنْتَ، ࢠَاوٝ يࣹوْتࣹرࣹ الله ندࣹنْ إࣹ آيࣹيجࣹ مَكّٝ طࣹنْ?

(7) بٝنࣹ وٝودَنِي كَلَ فࣹنٝووٝ بَكّٝندِنْطٝ !

(8) مٝ نٝ نَنَ آيࣹيجࣹ الله طࣹنْ هِنٝ جَنْنگࣹي إࣹ دٝوْ مَكّٝ، رࣹفْتِ أٝ هࣹدِّتٝرَ إࣹ تٝوْنِتَارࣹ، وَ سِ تَوِي أٝ نَنَالِ طࣹ. أَوَ وࣹوْلِنِرْ مٝ لࣹپْتࣹ مُوسُطࣹ.

(9) سِ أٝ أَنْندِي گٝطُّمْ إࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ، أٝ جٝگِتٝرَ طُمْ جَلْنٝرِ. ࢠࣹنْ طٝنْ، هِنٝ وٝودَنِ لࣹپْتࣹ هٝيْنٝوجࣹ:

(10) جهنم هِنٝ ݖَگّٝلْ مَࢠّࣹ. كٝ ࢠࣹ فَگِّتِنٝو دُنْݖَنٝيْتَاࢠࣹ هَيْ هُوندࣹ، وَنَا كٝنْ كٝ ࢠࣹ جٝگِتِنٝو يِطْࢠࣹ كٝ وٝنَا الله. هِنٝ وٝودَنِ ࢠࣹ لࣹپْتࣹ مَوْطࣹ.

(11) ندࣹي دࣹفْتࣹرࣹ كٝ پࣹيوَلْ. ࢠࣹنْ يࣹدُّࢠࣹ نٝنْ آيࣹيجࣹ جٝومِ مَࢠّࣹ طࣹنْ، هِنٝ وٝودَنِ لࣹپْتࣹ سٝࢠُطࣹ مُوسُطࣹ.

(12) كٝ الله أٝونْ وٝنِ كٝ إࣹيلْتَنْتَ مٝنْ مَايَ نگَمْ لَاطࣹ ندٝگَ إࣹ هࣹنگٝ إࣹ دٝوْ يَمِرٝورࣹ مَكّٝ، نگَمْ طَࢠّٝنْ إࣹ ࢠُرَلْ مَكّٝ إࣹ يَامَ جٝو نجࣹتّٝنْ.

(13) أٝ إࣹيلْتَنِ أٝنْ كٝ وٝنِ إࣹ كَمُّوجِ إࣹ كَ لࣹيْدِ فٝفْ، وٝنِي إࣹ طُومْ كَاوِسَاجِ وٝنَندࣹ يِمْࢠࣹ مِيجَتٝوࢠࣹ.

(14) مَاكَنْ ࢠࣹينْ گٝونْطِࢠࣹ يٝࢠࣹ نجَافَ ࢠࣹينْ ࢠࣹ تِجَّاكِ نْيَلْطِ الله نگَمْ أٝ يٝࢠَ ࢠࣹ يِمْࢠࣹ كٝࢠࣹ پَگِّتّٝنٝ كٝ.

(15) كَلَ گٝلُّطٝ كٝ مٝڃِّ،كٝ فِي هٝورࣹ-مَكّٝ; كَلَ كَدِ بٝنُّطٝ،يَنْتَيْ إࣹ مَكّٝ رࣹفْتِ كٝ كَ جٝومِ مَا وٝنِ رُتّٝرْدࣹ مٝنْ.

(16) مِينْ تٝتِّي بَنِي إِسْرَا*إِيلْ دࣹفْتࣹرࣹ إࣹ نْيࣹينْيَلْ إࣹ أَنَّبَاگَلْ مِينْ نجࣹطِࢠࣹ إࣹ دَگِيطِ مِينْ ࢠُرْنِࢠࣹ كٝ هࣹيوِ إࣹ بِنْندࣹ هࣹي.

(17) مِينْ تٝتِ دَلِّلَّاجِ ࢠَنْنگِنٝوجِ ࢠࣹ لُورَانِ سِ وٝنَهْ ݖَگَّلْ ندࣹ گَنْندَلْ أَرِ إࣹ مُمْ إࣹنْ سَبُ بࣹوْرࣹ هَكُّندࣹ مَࢠّࣹ پࣹلّࣹتْ جٝومَ نَ نْيَاوَ هَكُّندࣹ مَࢠّࣹ نْيَندࣹ دَرْنگَ كٝࢠࣹ نجࣹدَّندِرَتْنٝو كٝ.

(18) رࣹفْتِ مِينْ نبَطْمَ إࣹ لَاوٝلْ إࣹ يَمِرٝورࣹ أَمࣹنْ، وٝتٝ رࣹوْ بࣹلَاطࣹ ࢠࣹينْ ࢠࣹ نگَنْندَهْ.

(19) پࣹلّࣹتْ ࢠࣹينْ ࢠࣹ نَفَتَامَ گَا إࣹ الله هَيْ هُوندࣹ پࣹلّࣹتْ تٝونْيٝوࢠࣹ يٝگٝ إࣹ مُمْ إࣹنْ كٝ وࣹلْدِيࢠࣹ، الله وٝنِ كࣹيدَ هُلْࢠࣹ ࢠࣹ.

(20) أٝ قُرَانَ كٝنبٝ گِتࣹ إࣹ پࣹيوَلْ إࣹ يُرْمࣹيندࣹ وٝنَندࣹ يَنَنَاࢠࣹ.

(21) نبࣹلَ ࢠࣹينْ وَطْࢠࣹ بٝنࣹيجِ نَ نَنْندِ إࣹ گٝونْطِنْࢠࣹ ࢠࣹ گٝلِّ گٝلّࣹ لٝبّࣹ هٝࢠࣹ فٝتَ إࣹ نگُورْندَمْ مُمْ إࣹنْ إࣹ مَايْدࣹ مَࢠّࣹ بٝنِي كٝࢠࣹ نْيَاوَتَ.

(22) الله تَگْرِي كَمُّوجِ إࣹ لࣹيْدِ گٝونگَ نگَمْ كَلَ فِتَّاندُ يٝࢠࣹي كٝندُ فَگِّنٝو كٝ كَمْࢠࣹ ࢠࣹ تٝونْيࣹتَاكࣹ.

(23) نبࣹلَ أَ يِيَانِ أٝونْ جَگِّرْطٝ بࣹلَاطࣹ مَكّٝ دࣹوࣹتࣹيطٝ الله مَجِّنِ نبٝ إࣹ دٝوْ گَنْندَلْ أٝ أُدِّ نَنْطࣹ مَكّٝ إࣹ ࢠࣹرْندࣹ مَكّٝ، أٝ وَطِ إࣹ گِتࣹ مَكّٝ هࣹي ݖُدِّ كٝ هٝنبٝ فࣹيوْنَتَ أٝونْ ݖَگَّلْ الله مَتࣹ أٝنْ مِيݖَّتَاكَ.

(24) هࣹيفࣹرْࢠࣹ وِي طُمْ وٝنَهْ سِ وٝنَهْ نگُورْندَمْ أَدُنَ مِينْ مَايَ مِينْ نگُورَ، هَلْكَتَا مِينْ سِ وٝنَهْ جَمَانُ، ࢠࣹ نگَلَنَهْ طُومْ گَنْندَلْ ࢠࣹ ݖِكِّتَ تَنْ.

(25) سِ آيࣹيجࣹ أَمࣹنْ ࢠَنْنگُطࣹ نجَنْنگَامَ إࣹ مُمْ إࣹنْ هُجَّ مَࢠّࣹ وٝنَتَهْ سِ وٝنَهْ نگَدّࣹي بَابِرَاࢠࣹ أَمࣹنْ سِ أٝنْ نگٝنِي گٝونگُࢠࣹ.

(26) مَاكُ كٝ الله وُورْنَتَ أٝنْ رࣹفْتِ أٝ وَرَ أٝنْ رࣹفْتِ رࣹنْندِنَ أٝنْ فَادࣹ ݧَلَوْمَ دَرْنگَ نبٝ سِكّࣹ أَلَهْ إࣹ مُمْ، كٝنٝ كٝ ࢠُرِ هࣹيوِ إࣹ يِمْࢠࣹ نگَنْندَهْ.

(27) وٝودَنِي الله لَامُ كَمُّوجِ إࣹ لࣹيْدِ، ݧَندࣹ دَرْنگَ دَرِي ݧَندࣹ هࣹينْ مࣹيرࣹنْتࣹيࢠࣹ پࣹرْتَ.

(28) هَطَ يِيَ -آنْ نُلَاطَ- كَلَ لࣹنْيٝلْ نَ گُرْفَنِي إࣹ كٝپِّ مُمْ،كَلَ لࣹنْيٝلْ نَ نٝدِّرࣹي دࣹفْتࣹرࣹ مُمْ هَنْندࣹ وٝنِ كٝ نجٝࢠࣹتࣹطٝنْ كٝ نگٝلَّتْنَطٝنْ إࣹ ندࣹرْ أَدُنَ.

(29) ندࣹي كٝ دࣹفْتࣹرࣹ أَمࣹنْ هٝندࣹ وٝوْلَ گٝونگَ، مِينْ نگٝنِينَ مِينْ ݖࣹيرَانِ وٝندࣹ إࣹمِنْ نجَمِرَ ندࣹ وِنْندَتࣹ كٝ نگٝلَّتْنَ طٝنْ.

(30) تَوِ كٝ ࢠࣹينْ گٝونْطِنْࢠࣹ ࢠࣹ نگٝلِّ گٝلّࣹ لٝبّࣹ أٝ نَاتْنَتْࢠࣹ إࣹ يُرْمࣹيندࣹ مَكّٝ، كٝ طُومْ وٝنِ مَلَ ࢠَنْنگُنگَلْ.

(31) تَوِ كٝ ࢠࣹينْ هࣹيفَرࣹيࢠࣹ نبࣹلَ آيࣹيجࣹ أَمْ طࣹ نجَنْنگࣹتَنٝوكࣹ إࣹ دٝوْ مٝنْ مَوْنِكِنِطٝ لَاتِطٝنْ يِمْࢠࣹ بٝنْࢠࣹ.

(32) سِبْࢠࣹ وِيَامَ پࣹلّࣹتْ فٝدٝورࣹ الله كٝ گٝونگَ سِكّࣹ أَلَهْ هࣹينْ نبِ*أٝنْ مِينْ نگَنْندَهْ كٝ وٝنِ دَرْنگَ، مِينْ ݖِكِّتِ ݖِكِّتَگٝلْ مِينْ نگٝنَانِ يَنَنَاࢠࣹ.

(33) بٝنࣹيجِ طِ ࢠࣹ فَگِّتِنٝو طِنْ فࣹݧَنَ ࢠࣹ، كٝ ࢠࣹ جَلْكِتَيْنٝو كٝنْ فِيلْتٝو ࢠࣹ.

(34) ࢠࣹ وِ*أَنَ هَنْندࣹ مِينْ نجࣹجِّتَ مٝنْ هٝنٝ نٝ نجࣹجِّتِرْطٝنْ پٝتَّلْ ݧَلْنگُ مٝنْ هَنْندࣹ نگُ، كٝ يِيتࣹ وٝنِ هٝطٝرْدࣹ مْنٝ أٝنْ ندَنْيَتَهْ وَلَّدِيࢠࣹ.

(35) طُومْ نٝونْ كٝ سَبُ أٝنْ نجَگَّرِي آيࣹيجࣹ الله طࣹ جَلْكِتٝ نگُورْندَمْ أَدُنَ هٝومْتِ أٝنْ،هَنْندࣹ ࢠࣹ نجَلْتَتَهْ إࣹ يِيتࣹ ࢠࣹ ندُتَّتَاكَ أَدُنَ ࢠࣹ گٝلَّ گٝلّࣹ لٝبّࣹ هَا الله وࣹلࣹيࢠࣹ.

(36) وٝودَنِي الله يࣹتّٝورࣹ ندࣹرْ كَمُّوجِ إࣹ لࣹيْدِ جٝمْ بِنْندࣹ.

(37) نَ وٝودَنِ نبٝ مَنْنگُ ندࣹرْ كَمُّوجِ إࣹ لࣹيْدِ كٝ كَنْكٝ وٝنِ نبٝ گٝوتٝ فٝولَهْ ݧࣹيݧُطٝ.

46 Al-Ahqaf — 35 آيَة

(1) هَا، مِيمْ.

(2) جِپَّگٝلْ دࣹفْتࣹرࣹ ندࣹنْ، كٝ إِمّٝرْدࣹ كَ الله، فٝولُطٝ، ݧࣹيݧٝ أٝنْ.

(3) مࣹنْ تَگِرَالِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ إࣹ كٝنْ كٝ وٝنِ هَكُّندࣹ مَجِّ، سِ وَنَا إࣹ گٝونگَ إࣹ لَجَلَ هَپَّانگَلْ فِي مُنْ . يࣹدُّࢠࣹ ࢠࣹنْ كٝ طُورْنِيࢠࣹ إࣹ كٝ ࢠࣹ رࣹنْتِنِرَا كٝنْ.

(4) مَاكُ: "أٝنْ يِئِي كٝ رࣹوٝتٝنْ كٝنْ هَرَ وَنَا الله، هٝلّࣹي لَمْ كٝ طِ تَگِ إࣹ لࣹيْدِ ندِنْ? كَا هَرَ هِطِ مَرِ گࣹࢠَلْ إࣹ تَگُگٝلْ كَمُّولِ طِنْ? أَوَ أَدَّنࣹي لَمْ دࣹفْتࣹرࣹ جِپِّنَاندࣹ أَدٝ ندࣹي طٝو مَاطُمْ بَتّࣹ گَنْندَلْ، سِ وٝنِي كٝ أٝنْ گٝونگُࢠࣹ !".

(5) كٝ هٝمْنبٝ ࢠُرِ مَجُّدࣹ، وَ أٝنْ رࣹوٝووٝ كٝ وَنَا الله، كٝنْ كٝ جَابٝتَاكٝ مٝ هَا ݧَلْندࣹ دَرْنگَلْ? هَرَ طِنْ رࣹوࣹتࣹيطِ كٝ وࣹلْسِندِيطِ إࣹ دࣹوَلْ مَجِّ.

(6) سِ يِمْࢠࣹ ࢠࣹنْ مٝوࢠٝيَامَ كَ دَرْنگَلْ ، طِ وٝنْتَنٝيَيْ ࢠࣹ أَيْࢠࣹ، ࢠࣹ وٝنَ إࣹ يࣹدُّگٝلْ كٝ ࢠࣹ رࣹوَيْنٝوطِ كٝنْ.

(7) سِ آيࣹيجࣹ أَمࣹنْ ࢠَنْنگُطࣹ طࣹنْ جَنْنگَامَ إࣹ مَࢠّࣹ، يࣹدُنٝوࢠࣹ گٝونگَ ࢠࣹنْ تُمَ ندࣹ أَرْنٝو إࣹ مَࢠّࣹ ࢠࣹنْ وِئَ: "طُمْ كٝ نبِلࣹوُ ࢠَنْنگُ نگُ".

(8) كَا ࢠࣹ وِئَيْ وٝنِ :"أٝ فࣹفِندِ ندࣹ?" أَوَ مَاكُ: "سِ تَوْ مِي فࣹفِندِ ندࣹ، أَوَ دٝو أٝنْ دُنْݖَنْتَا لَمْ هُوندࣹ إࣹ لࣹپْتࣹ الله طࣹنْ. كٝ كَنْكٝ ࢠُرِ أَنْندُدࣹ أٝنْ، تُمَ ندࣹ جُمْپٝتٝنْ إࣹ مَيْرࣹ. أٝ يٝنِي سࣹيدࣹي هَكُّندࣹ أَمْ إࣹ مٝنْ. كٝ كَنْكٝ وٝنِ هَفٝرٝوٝ، هِنّٝتٝوطٝ أٝنْ.

(9) مَاكُ: "مِي وٝنَا أَرَنٝ إࣹ نُلَاࢠࣹ ࢠࣹنْ; مِي أَنْندَا كَدِ كٝ وَطࣹتࣹي مِي، مَا كٝ وَطࣹتࣹطٝنْ. مِي جٝكَّا سِ وَنَا كٝ وَهَيِنَا كٝنْ إࣹ أَمْ; مِي لَاتَاكِ سِ وَنَا جࣹرْتِنٝووٝ ࢠَنْنگِنَ".

(10) مَاكُ: "يࣹيتࣹي لَمْ، سِ تَوِي لࣹ كٝندࣹ إِوْرُندࣹ كَ الله، يࣹدُّطٝنْ طُمْ، سࣹيدٝتٝوطٝ إࣹ ࢠِطْࢠࣹ إِسْرَائِلَ ࢠࣹنْ سࣹيدِي إࣹ سࣹلُّگٝلْ طُمْ، أٝ گٝمْطِنِ، أٝنٝنْ كَنْ تٝوْنِتِطٝنْ...پࣹلّࣹتْ، الله فࣹوْنَتَا يِمْࢠࣹ تٝوݧٝوࢠࣹ ࢠࣹنْ.

(11) يࣹدُّࢠࣹ ࢠࣹنْ وِئَنِ گٝمْطِمْࢠࣹ ࢠࣹنْ: "سِ تَوْنٝ طُمْ كٝ مٝڃّࣹرࣹ، ࢠࣹنْ (وَاسُࢠࣹ) أَدٝتٝتَاكٝ مࣹنْ إࣹ مُنْ". ندࣹ وٝنُنٝو ࢠࣹ فࣹوْنٝرَالِ ندࣹ، مَ ࢠࣹ وِئُ: "ندࣹي كٝ فࣹنَاندࣹ نࣹيࢠُندࣹ!".

(12) هِنٝ أَدِتِيندࣹ، دࣹفْتࣹرࣹ موسى ندࣹنْ، طٝوٝرْندࣹ إࣹ يُرْمࣹيندࣹ. ندࣹي طٝو كَدِ، كٝ دࣹفْتࣹرࣹ گٝونگِنِرْ- ندࣹ طࣹيَ دࣹفْتࣹ إࣹ طࣹمْنگَلْ آرَبُ، فِي يٝ جࣹرْتِنِرْ (ندࣹ) تٝوݧٝوࢠࣹ ࢠࣹنْ إࣹ وࣹوْلِنِرْندࣹ مٝڃِّنٝوࢠࣹ ࢠࣹنْ.

(13) پࣹلّࣹتْ، وِئُࢠࣹ ࢠࣹنْ: "كٝ الله وٝنِ جٝومِ أَمࣹنْ"، رࣹفْتِ ࢠࣹ تَبِتِ (إࣹ گٝمْطِنَلْ)، كُلٝلْ أَلَنَا ࢠࣹنْ، وَنَا كَمْࢠࣹ سُنٝيْتٝو.

(14) كٝ ࢠࣹنْ وٝنِ يِمْࢠࣹ الجنة ࢠࣹنْ دُومٝتٝوࢠࣹ تٝنْ، نجٝࢠْدِ كٝ ࢠࣹ گٝلَّيْنٝو كٝنْ.

(15) مࣹنْ يَمِرِي نࣹطَّنْكࣹ أٝنْ مٝڃَّگٝلْ إࣹ جِبِمْࢠࣹ مٝ ࢠࣹنْ: جِيدٝ مَكّٝ سٝوٝرِكࣹ مٝ سَطْتࣹيندࣹ، أٝ جِبِنِرِ مٝ كَدِ سَطْتࣹيندࣹ; سٝوࣹيدࣹ مٝ إࣹ إࣹنْتࣹيدࣹ مٝ كٝ لࣹبِّ ݖَپَّنْطࣹ تَتِ، هَا كَ أٝ سِ أٝ يٝتِّكࣹ سَگَتَلَاكُ مَكُّ نگُنْ أٝ تِمِّنِي دُوࢠِ ݖَپَّنْطࣹ نَيِ، أٝ وِئَ: "جٝومِ، لٝنگِنَمْ يࣹتُّگٝلْ نࣹيمَ مَ مٝ نࣹيمِنِرْطَا مِي مِينْ إࣹ جِبِمْࢠࣹ لَمْ، إࣹ ندࣹ مِي گٝلَّتَ مٝڃُّطِ وࣹلْتِنَيْطࣹ مَ، مٝڃِّنَنَا لَمْ گࣹيْنگٝلْ أَمْ نگٝلْ. مِي تُوبَنِي مَ، كٝ مِي جࣹيَاطٝ إࣹ جࣹبِّلِيࢠࣹ ࢠࣹنْ".

(16) كٝ ࢠࣹنْ وٝنِ ࢠࣹ مࣹنْ جَࢠَنْتَ كٝ ࢠُرِ مٝڃُّدࣹ إࣹ كٝ ࢠࣹ گٝلِّ، مࣹنْ يَوْتَنَ ࢠࣹ بٝنْطِ مَࢠّࣹ، (ࢠࣹ جࣹيَامَ) إࣹ يِمْࢠࣹ الجنة ࢠࣹنْ. طُمْ كٝ فٝدٝورࣹ گٝونگَ ندࣹ ࢠࣹ فٝدَنٝو.

(17) كَلَ نٝنْ وِئَنْطٝ جِبِمْࢠࣹ مٝ ࢠࣹنْ: "أُفْ مٝؤٝنْ ! كࣹرࣹ أٝنْ فٝدَيْ لَمْ إِمِّنْتِنࣹيدࣹ هَرَ لࣹ گٝمْطِي مٝفْتࣹ يَوْتِينٝ أَدٝ لَمْ?" هَرَ ࢠࣹنْ طِطٝ نٝ فَابْتِنٝرَ الله وِيدࣹ مٝ : "إࣹي بٝنࣹ مَا، گٝمْطِنْ ! پࣹلّࣹتْ، فٝدٝورࣹي الله ندࣹ كٝ گٝونگَ". أٝ وِئَ: "طُمْ وٝنَالِ، سِ وَنَا تِنْندِ أَدِنٝوࢠࣹ ࢠࣹنْ".

(18) كٝ ࢠࣹنْ وٝنِ ࢠࣹ كٝنْنگٝلْ نگٝلْ جٝجِّ إࣹ مُنْ، (ࢠࣹ جࣹيَامَ) إࣹ مٝفْتࣹ يَوْتُنٝوطࣹ أَدٝ مَࢠّࣹ، إِمّٝرْدࣹ إࣹ جِنَّ إࣹ يِمْࢠࣹ. پࣹلّࣹتْ، ࢠࣹ لَاتِنٝ سٝويُࢠࣹ.

(19) هِنٝ وٝودَنِ كَلَ فࣹدّࣹ إࣹ مَࢠّࣹ : دَرْجَاجِ إࣹ كٝ ࢠࣹ گٝلِّ كٝنْ، فِي نٝ أٝ هُنِّرَنَ ࢠࣹ نجٝࢠْدِ گٝلّࣹ مَࢠّࣹ طࣹنْ; إࣹ هٝورࣹ ࢠࣹ تٝوݧࣹتَاكࣹ.

(20) ݧَلْندࣹ هࣹيفࣹرْࢠࣹ ࢠࣹنْ وࣹيࢠِتٝيْتࣹي إࣹ يِيتࣹ نگࣹنْ (ࢠࣹ وِئَنࣹي): "أٝنْ نَࢠِي مٝڃِّ مٝنْ طِنْ كَ نگُورْندَمْ مٝنْ أَدُنَ، دَكْمِتٝرْطٝنْ طِ. هَنْندࣹ نٝنْ، أٝنْ يٝࢠِتِرْتࣹ لࣹپْتࣹ هٝيْنٝوجࣹ، سَبُ كٝ تٝوْنِتٝرَيْنٝطٝنْ كٝنْ كَ هٝورࣹ لࣹيْدِ، كٝ أَلْدَا إࣹ گٝونگَ، إࣹ كٝ هُووِرَيْنٝطٝنْ كٝنْ فَاسِقَاكُ".

(21) جَنْتٝ فِي مُسِطْࢠࣹ آدِ (هُودِ) ࢠࣹنْ، ندࣹ أٝ جࣹرْتِنِرْنٝو يِمْࢠࣹ مَكّٝ ࢠࣹنْ أَيْندࣹ ندࣹنْ - گٝمْطِي هَرِ جࣹرْتِنٝوجِ يَوْتِينٝ يࣹيسٝ مَكّٝ إࣹ ࢠَاوٝ مَكّٝ - وٝنْندࣹ: "وَتَ أٝنْ رࣹوُ سِ وَنَا الله. مِينْ مِطُ هُلَنِ أٝنْ لࣹپْتࣹ ݧَلَاندࣹ مَوْندࣹ".

(22) ࢠࣹ وِئِ: "إࣹ أَ أَرُ مࣹنْ فِي يٝ أَ فِرْلِــََٔ مࣹنْ إࣹ رࣹوࣹتࣹي- طِ أَمࣹنْ طِنْ? أَوَ أَدَّنْ مࣹنْ كٝ كَمْنبِرَنْطَا مࣹنْ كٝنْ، سِ وٝنِي كٝ أَ جࣹيَاطٝ إࣹ گٝونگُࢠࣹ ࢠࣹنْ".

(23) أٝ مَاكِ: "گَنْندَلْ نگَلْ كٝ كَ الله وٝنِ. مِينْ مِي يٝتِّنَنَيْ أٝنْ تُنْ وٝنِ، كٝ مِي نُلَا كٝنْ. كٝ وٝنِ تُنْ مِي يِئِي كٝ أٝنْ يِمْࢠࣹ مَجَّاࢠࣹ".

(24) ندࣹ ࢠࣹ يِئُنٝو رُولْدࣹ هٝندࣹ وࣹيࢠِتِي فࣹوْتِرِ أَيْطࣹ مَࢠّࣹ طࣹنْ، ࢠࣹ وِئِ: "ندࣹي كٝ دُولࣹرࣹ تٝࢠَيْندࣹ إࣹ مࣹنْ". كٝ ندࣹنْ كَنْ وٝنِ ! كٝ هَوْجِرْنٝطٝنْ كٝنْ : كٝ هࣹنْندُ، ندُ لࣹپْتࣹ مُوسُطࣹ وٝنِ إࣹ مُنْ.

(25) هِندُ مُنْݖِرَ كَلَ هُوندࣹ، يَمِرٝورࣹ جٝومِ مَيْرُ". ࢠࣹ وَطْتِنِ، تَوِ يِئࣹتَاكࣹ سِ وَنَا كٝطٝولِ مَࢠّࣹ طِنْ. كٝ وَنٝ نِي مࣹنْ يٝࢠِتِرْتَ يِمْࢠࣹ بٝنّٝوࢠࣹ ࢠࣹنْ.

(26) گٝمْطِي مࣹنْ يࣹطِينٝ ࢠࣹ، إࣹ كٝنْ كٝ مࣹنْ يࣹطَالِ أٝنْ إࣹ مُنْ. مࣹنْ وَطَنِ ࢠࣹ نَنْطࣹ، إࣹ گِيطࣹ إࣹ ࢠࣹرْطࣹ. كٝنٝ طࣹنْ نَنْطࣹ مَࢠّࣹ، إࣹ گِيطࣹ مَࢠّࣹ إࣹ ࢠࣹرْطࣹ مَࢠّࣹ (فٝوْ) دُنْݖَنَالِ ࢠࣹ گٝطُّمْ، تَوْدࣹ ࢠࣹ لَاتِنٝ يࣹدّٝوࢠࣹ آيࣹيجࣹ الله طࣹنْ. كٝ ࢠࣹ جَلْكِتَيْنٝو كٝنْ جِپِّي إࣹ مَࢠّࣹ.

(27) گٝمْطِي مࣹنْ هَلْكِينٝ ݖَئࣹ وٝنُنٝوࢠࣹ سࣹرَ مٝنْ; مࣹنْ فِرْلِتِرِ آيࣹيجࣹ طࣹنْ، بࣹلَجٝؤٝ ࢠࣹ رُتّٝتٝ.

(28) كٝ هَطْنٝ طِنْ طِ ࢠࣹ جٝگِتٝرْنٝو رࣹوࣹتࣹيطِ ࢠَطْتِنٝرْطِ گَانِنْ الله، فَابٝو ࢠࣹ? تِگِ أٝنْ، طِنْ مَجِّرِي ࢠࣹ; كٝ طُمْ وٝنُنٝو فࣹنَاندࣹ مَࢠّࣹ إࣹ كٝ ࢠࣹ فࣹفِندٝتٝنٝو.

(29) جَنْتٝ تُمَ ندࣹ مࣹنْ إِمِّنّٝو فࣹدّࣹ جِنَّ إࣹ مَاطَ، هِطِ هࣹطٝو القرآن. ندࣹ طِ تَوَنٝو طُمْ، طِ وِئِ: "هࣹطࣹي"...ندࣹ جَنْنگِنْ كِنْ تِمُّنٝو، طِ هُݖِّتِرِ كَ يِمْࢠࣹ مَجِّ، جࣹيرْتِنٝوجِ.

(30) طِ وِئِ: "كٝ أٝنٝنْ يٝ نبَتُ أَمࣹنْ، مࣹنْ نَنِي دࣹفْتࣹرࣹ جِپِّنَاندࣹ ࢠَاوٝ موسى، گٝونگِنْندࣹ طࣹنْ أَدِيطࣹ ندࣹ. هِندࣹ فࣹوْنَ إࣹ گٝونگَ إࣹ لَاوٝلْ فٝݖِّينگٝلْ.

(31) كٝ أٝنٝ يٝ نبَتُ أَمࣹنْ، نٝوتࣹي نٝدّٝووٝ إࣹ الله أٝنْ، گٝمْطِنٝنْ مٝ. أٝنْتُمَ أٝ هَفٝرَنَيْ أٝنْ جُنُوبِمٝنْ طِنْ، أٝ دَطْندَ أٝنْ لࣹپْتࣹ مُوسُطࣹ.

(32) كَلَ مٝ نٝوتَاكࣹ نٝدّٝووٝ إࣹ الله أٝنْ، أٝ وٝنَا لَاوٝتٝوطٝ كَ لࣹيْدِ، أٝ أَلَا كَدِ گِطٝ گَانِنْ مَكّٝ. كٝ ࢠࣹنْ وٝنِ إࣹ ندࣹرْ مَجّࣹرࣹ ࢠَنْنگُندࣹ".

(33) إࣹ ࢠࣹي يِئَالِ وٝنْندࣹ كٝ الله تَگِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، أٝ رٝنْكِرَالِ تَگُگٝلْ طِ، وٝنْندࣹ كٝ أٝ هَطْتَنْطٝ وُورْنِتُگٝلْ مَايْࢠࣹ ࢠࣹنْ? كٝ نٝنْ وٝنِرِ. پࣹلّࣹتْ، كٝ أٝ هَطْتَنْطٝ كَلَ هُوندࣹ.

(34) ݧَلْندࣹ هࣹيفࣹرْࢠࣹ ࢠࣹنْ وࣹيࢠِتٝيْتࣹي إࣹ يِيتࣹ نگࣹنْ، ࢠࣹ وِئَنࣹي : "وَنَا گٝونگَ نِي?" ࢠࣹ وِئَ : "كٝ نٝنْ وٝنِرِ! مࣹنْ وٝوندِرِي جٝومِ أَمࣹنْ!". أٝ دَالَ : "أَوَ مࣹيطِرࣹي نٝنْ لࣹپْتࣹ طࣹنْ، سَبُ كٝنْ كٝ يࣹدِّرَيْنٝطٝنْ".

(35) مُݧّٝرْ مُݧَلْ جٝومِرَاࢠࣹ پࣹلِّتَلْ ࢠࣹنْ كَ نُلَاࢠࣹ; وَتَ أَ هَوْجَنْ ࢠࣹ. ݧَلْندࣹ ࢠࣹ يِئٝيْتَ كٝ ࢠࣹ كَمْنبِرَنَا كٝنْ، ࢠࣹ وَأٝيَيْ وَ سِ ࢠࣹ ݧِيࢠَانٝ كَ لࣹيْدِ سِ وَنَا دُمُنَّ-هُنْ ݧَلٝرْمَ. طُنْ كٝ يٝتِّنَلْ. كٝ هٝمْࢠࣹ هَلْكࣹتࣹي سِ وَنَا يِمْࢠࣹ فَاسِقِيࢠࣹ ࢠࣹنْ?

47 Muhammad — 38 آيَة

(1) ࢠࣹنْ يࣹدُّࢠࣹ ࢠࣹ فَلِي لَاوٝلْ الله نگٝلْ، أٝ بٝنِّي گٝلّࣹ مَࢠّࣹ طࣹنْ.

(2) گٝمْطِمْࢠࣹ ࢠࣹنْ ࢠࣹ گٝلِّ گٝلّࣹ مٝڃّࣹ، ࢠࣹ گٝمْطِنِ كٝ جِپِّنَا كٝنْ إࣹ مُهَمَّدْ - أٝنْ گٝونگَ إِمّٝرْطٝ إࣹ جٝومِ مَࢠّࣹ - أٝ يَوْتَنِي ࢠࣹ بٝنْطِ مَࢠّࣹ طِنْ، أٝ مٝڃِّنِي فِيَاكُ مَࢠّࣹ أٝنْ.

(3) كٝ وَطِ طُمْ، كٝ تَوْدࣹ يࣹدُّࢠࣹ ࢠࣹنْ جٝكِّي مࣹيرࣹ ندࣹنْ، گٝمْطِمْࢠࣹ ࢠࣹنْ كَنْ جٝكِّ گٝونگَ كَنْ إِمّٝرْدࣹ إࣹ جٝومِ مَࢠّࣹ أٝنْ. كٝ وَنٝ نِي الله پِيِرَنْتَ يِمْࢠࣹ ࢠࣹنْ مِسࣹ مَࢠّࣹ طࣹنْ.

(4) سِ أٝنْ أَرِي هَوْرُدࣹ إࣹ يࣹدُّࢠࣹ ࢠࣹنْ (كَ هَرࣹ)، كٝ سٝپُّگٝلْ دَاطࣹ طࣹنْ. هَا سِ أٝنْ طُطِّنِي وَرُگٝلْ ࢠࣹ، هَࢠّٝنْ كٝيْطࣹ طࣹنْ. كٝ سِ أٝنْ مٝڃّٝرَيْ إࣹ مَࢠّࣹ أَݖِّتُگٝلْ ࢠࣹ ࢠَاوٝ طُمْ، مَا سِ كٝ سٝكْتِيگُ، هَا هَرࣹ ندࣹنْ وَلِّنَ جٝگَاجِ مَيْرࣹ. كٝ نٝنْ ݧَاوِرَا، سِ الله نٝ مُويُنٝو، أٝ وَلِّتٝتٝنٝ أٝنْ إࣹ مَࢠّࣹ، كٝ وٝنِ تُنْ، كٝ نٝ أٝ جَرِّبٝرَ يٝگَ مٝنْ إࣹ دٝوْ يٝگَ. وَرَاࢠࣹ فِي الله ࢠࣹنْ، پࣹلّࣹتْ، أٝ يࣹيبَتَا گٝلّࣹ مَࢠّࣹ طࣹنْ.

(5) أَرْمَ أٝ فࣹوْنَ ࢠࣹ، أٝ مٝڃِّنَ فِيَاكُ مَࢠّࣹ،

(6) أٝ نَادٝيَ ࢠࣹ الجنة، مٝ أٝ أَنْندِنِ ࢠࣹ فِي مُنْ.

(7) كٝ أٝنٝنْ يٝ گٝمْطِمْࢠࣹ، سِ أٝنْ وٝنِي وَلّٝوࢠࣹ فِي الله، أٝ وَلَّيْ أٝنْ، أٝ تَبِنْتِنَ تࣹپّࣹ مٝنْ طࣹنْ.

(8) يࣹدُّࢠࣹ ࢠࣹي كَنْ، هَلْكَارࣹ وٝودَنِي ࢠࣹ، أٝ مَجِّنِي كَدِ گٝلّࣹ مَࢠّࣹ طࣹنْ.

(9) طُمْ لࣹ كٝ تَوْدࣹ ࢠࣹ أَݧِي كٝ الله جِپِّنِ كٝنْ، أٝ بٝنِّ گٝلّࣹ مَࢠّࣹ طࣹنْ.

(10) إࣹ ࢠࣹي يِيلَاكِ إࣹ لࣹيْدِ ندِنْ، ࢠࣹ تࣹسْكٝو كٝ هٝنّٝ بَتَّنࣹ أَدِنٝوࢠࣹ ࢠࣹنْ وَؤُنٝو? الله هَلْكُنٝ فِي مَࢠّࣹ. سُگُ مُمْ كَدِ هِنٝ وٝودَنِ يࣹدُّࢠࣹ ࢠࣹنْ.

(11) كٝ وَطِ طُمْ، كٝ تَوْدࣹ كٝ الله وٝنِ گِطٝ گٝمْطِمْࢠࣹ ࢠࣹنْ; هَرَ نٝنْ يࣹدُّࢠࣹ ࢠࣹنْ، گِطٝ (وٝو) أَلَنَا ࢠࣹ.

(12) پࣹلّࣹتْ، الله نَادٝيَيْ گٝمْطِمْࢠࣹ گٝلِّ مٝڃُّطِ ࢠࣹنْ، الجنة طِ ݖَنْطِ إِلَتَ لࣹيْ (كُࢠࣹيجࣹ) مَجّࣹ. يࣹدُّࢠࣹ ࢠࣹنْ كَنْ، هِࢠࣹ دَكْمِتٝو، ࢠࣹ ݧَامِرَ وَنٝ نࣹيمٝرَاطِ طِنْ ݧَامِرْتَ; هَرَ نٝنْ كٝ يِيتࣹ نگࣹنْ وٝنِ جَاڃٝرْدࣹ مَࢠّࣹ.

(13) طُوطِي إࣹ ݖَئࣹ، ࢠُرْنٝوࢠࣹ تِيطُدࣹ ݖࣹمْنبࣹ، إࣹ دِي ندࣹينْ سَارࣹ مَا رَطِيندࣹ مَ، ࢠࣹ مࣹنْ هَلْكِ، بَلّٝ أَلَنَا ࢠࣹ.

(14) كࣹرࣹ كٝ أٝنْ وٝنْطٝ إࣹ ࢠَنْنگَنْندُيࣹي إِمّٝرْدࣹ كَ جٝومِ مُنْ، وَئِ وَ ࢠࣹنْ ݖُطِنَنَاࢠࣹ كُوطࣹ مُنْ بٝنْطࣹ طࣹنْ، ࢠࣹ جٝكِّ نبࣹلࣹيطࣹ مَࢠّࣹ?

(15) كٝ وٝنِ سِفَ الجنة فٝدَاطٝ أٝنْ گٝمْطُࢠࣹ ࢠࣹنْ: إِنَ تٝونْ ݖَنْطِ ندِيَمْ طِ دَكَمّࣹ مُنْ وَيْلَاكِ، إࣹ ݖَنْطِ ࢠِرَ طِ وَيْلَتَا دَكَمّࣹ مُنْ، إࣹ ݖَنْطِ كٝنْنجَمْ وࣹلَيْطَمْ يَرٝوࢠࣹ، إࣹ ݖَنْطِ نجُورِ لَࢠِّنَاندِ. هِنٝ وٝودَنِ ࢠࣹ كَدِ تٝنْ: كَلَ دِمْࢠࣹ، إࣹ هَفٝرَنࣹيدࣹ إِمّٝرْدࣹ إࣹ جٝومِ مَࢠّࣹ. ࢠࣹنْ وَؤٝيَيْ وَ لُتِّنٝيْتࣹيࣹࢠࣹ ࢠࣹنْ كَ يِيتࣹ، ࢠࣹ يَرْنࣹي ندِيَمْ فَتَيْطَمْ، طَمْ تَڃَ تࣹتࣹكِّ مَࢠّࣹ طِنْ?

(16) هِنٝ إࣹ مَࢠّࣹ هࣹطِتٝتٝوࢠࣹ مَ. كٝنٝ كٝ هَا سِ ࢠࣹ يَلْتِي كَ مَاطَ، ࢠࣹ وِئَنَ أٝكَّاࢠࣹ گَنْندَلْ ࢠࣹنْ: "كٝ هٝنْطُمْ دࣹي، أٝ وِئُنٝو نَانࣹنْ?" كٝ ࢠࣹنْ وٝنِ ࢠࣹ الله نٝتِ إࣹ ࢠࣹرْطࣹ مُنْ طࣹنْ، ࢠࣹ جٝكِّ نبࣹلࣹيطࣹ مَࢠّࣹ طࣹنْ.

(17) فࣹيوُࢠࣹ ࢠࣹنْ كَنْ، أٝ ࢠࣹيْدُ ࢠࣹ پࣹيوَلْ أٝ أٝكِّࢠࣹ گٝمْطَلْ مَࢠّࣹ.

(18) هَرَ ࢠࣹي نٝ هَبِّي سِ وَنَا دَرْنگَلْ نگَلْ ندࣹ أَرْدَتَ ࢠࣹ إࣹ جُهَلْ? گٝمْطِي لࣹ مَاندࣹيجِ مَگَّلْ أَرِي. كٝ هٝنّٝ ࢠࣹ وَاجِتٝرٝيْتَ، سِ نگَلْ أَرْدَنِي ࢠࣹ (جُهَلْ)?

(19) أَوَ أَنْندُ پࣹلّࣹتْ، رࣹوࣹتࣹيطٝ أَلَا سِ وَنَا الله، إِنْسِنَنٝطَا جُنُوبِ مَا طِنْ، إࣹ گٝمْطِمْࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ گٝمْطِمْࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ. الله نٝ أَنْندِ ݧَلِّرْدࣹ مٝنْ إࣹ هِيرِرْدࣹ مٝنْ.

(20) گٝمْطِمْࢠࣹ ࢠࣹنْ هِنٝ وِئَ: "هَرَ نُنْ ݖٝرْتࣹوٝلْ (هَرࣹ) جِپّٝتٝنٝ!" سِ تَوِي لࣹ ݖٝرْتࣹوٝلْ ݧࣹݧَانگٝلْ جِپِّنَامَ، فِي هَرࣹ جَنْتَامَ إࣹ مَگّٝلْ، أَ تِنَيْ ࢠࣹنْ ࢠࣹ ݧَوُ وٝنِ إࣹ ࢠࣹرْطࣹ مُنْ هِنٝ ندَارࣹ، ندَارْطࣹ مٝ مَايْدࣹ فَطِّنِ. كٝ هَانَنّٝو ࢠࣹ،

(21) كٝ طٝفْتَگٝلْ إࣹ كٝنْنگٝلْ مٝڃّٝلْ. سِ فِيَاكُ أٝنْ فࣹلِّتِي، هَرَ نُنْ ࢠࣹ گٝمْطِنّٝ الله، كٝ طُمْ ࢠُرَيْنٝو مٝڃَّنْندࣹ ࢠࣹ.

(22) إࣹ أٝنْ تَمْڃِنَاكِ، سِ وٝنِي أٝنْ طُورْنِكࣹ، ندࣹ بٝنّٝتٝنْ إࣹ لࣹيْدِ ندِنْ، تَڃٝنْ إࣹنْطَمَاجِ مٝنْ?

(23) كٝ ࢠࣹنْ وٝنِ ࢠࣹ الله هُطِ ࢠࣹنْ، أٝ فَاطِنِ ࢠࣹ، أٝ وُمِنِ گِيطࣹ مَࢠّࣹ طࣹنْ.

(24) إࣹ ࢠࣹي تَسْكْتَاكٝ القرآن، كَا كٝ سٝكِرْطࣹ وٝنِ وٝنِ إࣹ ࢠࣹرْطࣹ مُمْ إࣹنْ?

(25) پࣹلّࣹتْ، ࢠࣹنْ رُتّٝرْࢠࣹ ࢠَبّࣹ مُنْ ࢠَاوٝ ندࣹ پࣹيوَلْ نگَلْ ࢠَنْنگَنّٝو ࢠࣹ، كٝ سࣹيْتَانࣹ ݖُطِنَنِ ࢠࣹ، أٝ هٝدِ ࢠࣹ.

(26) كٝ تَوْدࣹ ࢠࣹ وِئَنِي ࢠࣹنْ أَݧُࢠࣹ كٝ الله جِپِّنِ كٝنْ: "مࣹنْ طٝفْتٝتٝ أٝنْ إࣹ يٝگَ إࣹ پِيجِ". الله نٝنْ نٝ أَنْندِ گُنْندٝوجِ مَࢠّࣹ طِنْ.

(27) كٝ هٝنّٝ (ࢠࣹي وَؤٝيْتَ) تُمَ ندࣹ مَلَائِكَئࣹنْ تِمِّنِ فِي مَࢠّࣹ، هَرَ هِࢠࣹ بَنْتٝو گࣹيݖࣹ مَࢠّࣹ إࣹ دٝتࣹ مَࢠّࣹ?

(28) كٝ تَوْدࣹ كٝ ࢠࣹ جٝكِّي كٝ تِكِّنْتَ الله، ࢠࣹ أَݧِي وࣹلَيْطِ مٝ طِنْ، أٝ بٝنِّ فِي گٝلّࣹ مَࢠّࣹ طࣹنْ.

(29) كَا ࢠࣹنْ ࢠࣹ ݧَوُ وٝنِ إࣹ ࢠࣹرْطࣹ مُنْ سِكُو وٝنْندࣹ الله يَلْتِنْتَا نگَيْنگُوجِ مَࢠّࣹ?

(30) سِنّٝ مࣹنْ هَاجُنٝ، مࣹنْ هٝلّࣹتࣹنٝ ࢠࣹ. أَنْندِرَا ࢠࣹ مَاندࣹيجِ مَࢠّࣹ طِنْ; كٝ پࣹلّࣹتْ، أَ أَنْندِرَيْنٝ ࢠࣹ كَ كٝنْنگٝلْ وࣹلْنگٝلْ. الله نٝنْ هِنٝ أَنْندِ گٝلّࣹ مٝنْ طࣹنْ.

(31) مَ مࣹنْ جَرِّبٝ أٝنْ، هَا مࣹنْ أَنْندَ تِيطْنِيࢠࣹ ࢠࣹنْ إࣹ مٝنْ، إࣹ مُݧِّيࢠࣹ ࢠࣹنْ. مࣹنْ جَرِّبٝو كَدِ كَبَرُوجِ مٝنْ.

(32) پࣹلّࣹتْ، يࣹدُّࢠࣹ ࢠࣹنْ ࢠࣹ فَدِّ لَاوٝلْ الله نگٝلْ، ࢠࣹ لُنْندِي نُلَاطٝ أٝنْ ࢠَاوٝ ندࣹ پࣹيوَلْ نگَلْ ࢠَنْنگَنْندࣹ ࢠࣹ، ࢠࣹنْ لٝرَّتَا الله هَيْ هُوندࣹ. مَ أٝ بٝنُّ فِي گٝلّࣹ مَࢠّࣹ طࣹنْ.

(33) كٝ أٝنٝنْ يٝ گٝمْطِمْࢠࣹ! طٝفْتࣹي الله، طٝفْتٝطٝنْ نُلَاطٝ أٝنْ، وَتَ أٝنْ بٝنُّ گٝلّࣹ مٝنْ طࣹنْ.

(34) پࣹلّࣹتْ، ࢠࣹنْ يࣹدُّࢠࣹ ࢠࣹ سَكِّي إࣹ لَاوٝلْ الله نگٝلْ، هٝوتِ ࢠࣹ مَايِ كٝ ࢠࣹ هࣹيفࣹرْࢠࣹ، الله هَفٝرَنٝيْتَا ࢠࣹ.

(35) وَتَ أٝنْ لٝئِنْكِنٝ، نٝدّٝنْ إࣹ بࣹلْدَلْ، كٝ أٝنٝنْ وٝنِ فٝولُࢠࣹ ࢠࣹنْ، الله نٝ وٝنْندِ إࣹ مٝنْ، أٝ أُسْتَنْتَا أٝنْ نجٝࢠْدِ گٝلّࣹ مٝنْ طࣹنْ.

(36) أَنْندࣹي نگُورْندَمْ أَدُنَ كٝ فِجِندَارُ إࣹ نبࣹلࣹيطࣹ تَنْ; هَرَ لࣹ سِ أٝنْ گٝمْطِيِ، أٝنْ هُلِي الله ، أٝ يࣹطَيْ أٝنْ نجٝࢠْدِ مٝنْ ندِنْ، أٝ لَمْندٝتَاكٝ أٝنْ جَوْلࣹ مٝنْ طࣹنْ.

(37) سِ أٝ أَرُ لَمْندَادࣹ أٝنْ فِي مَجّࣹ، هَا أٝ دُݧِي أٝنْ (إࣹ مُنْ)، وُدّٝنْ، أٝ فࣹݧِّنَ نٝنْ كٝنَّاگُ مٝنْ نگُنْ.

(38) هࣹيْ مٝنْ أٝنٝنْ طٝو، هِطٝ نٝدّࣹي إࣹ نَفْقُگٝلْ فِي لَاوٝلْ الله نگٝلْ. هِنٝ إࣹ مٝؤٝنْ وُدَّيْࢠࣹ. كَلَ نٝنْ وُدُّطٝ، هَرَيْ كٝ هٝورࣹ-مَكّٝ أٝ وُدَّنِ. كٝ الله وٝنِ گَلٝ أٝنْ، أٝنٝنْ وٝنْطٝنْ وَاسُࢠࣹ ࢠࣹنْ. سِ أٝنْ هُݖِّي، أٝ وَطْتِتِرَ أٝنْ وٝطْࢠࣹ گٝو، هَرَ نٝنْ ࢠࣹنْ وَءَتَا وَنٝ مٝنْ.
