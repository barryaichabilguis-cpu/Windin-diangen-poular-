48 Al-Fath — 29 آيَة

(1) مࣹنْ أُدِّتَنِي مَ أُدِّتَلْ ࢠَنْنگُنگَلْ،

(2) فِي نٝ الله يَوْتِرَنِ كٝنْ كٝ أَرْدِنْطَا إࣹ جُنُوبِ، إࣹ كٝ سَكِّتِي، إࣹ فِي نٝ أٝ تِمِّنِرَ نࣹيمَ مَكّٝ أٝنْ إࣹ مَاطَ، أٝ فࣹوْنࣹ إࣹ لَاوٝلْ فٝوݖِّينگٝلْ.

(3) إࣹ نٝ الله وَلِّرࣹ بَلَّلْ تِيطُنگَلْ.

(4) كٝ كَنْكٝ وٝنِ جِپِّنْطٝ دࣹيڃࣹرࣹ ندࣹنْ إࣹ ࢠࣹرْطࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ، فِي نٝ ࢠࣹ ࢠࣹيْدٝرَ گٝمْطِنَلْ إࣹ دٝوْ گٝمْطِنَلْ مَࢠّࣹ نگَلْ. كٝ الله وٝودَنِ كٝنࣹيلِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ; الله سِفٝرِي وٝنْندࣹ أَنْندُطٝ، ݧࣹيݧُطٝ،

(5) إࣹ فِي نٝ أٝ نَادِرَ گٝمْطِمْࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ گٝمْطِمْࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، الجنة طِ ݖَنْطِ إِلَتَ لࣹيْ مُنْ، كٝ ࢠْطࣹ دُومٝتٝوࢠࣹ إࣹ مَجِّ، إࣹ نٝ أٝ مٝنْتِرَ-نَࢠࣹ بَنْطِ مَࢠّࣹ طِنْ. طُمْ وٝنِي كَ الله، مَلَلْ مَوْنُنگَلْ.

(6) إࣹ نٝ أٝ لࣹپْتِرَ نَافِقِيࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ نَافِقِيࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ سِرْكٝوࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ سِرْكٝوࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، سِكّٝوࢠࣹ الله سِكّࣹ بٝنْطࣹ. يِيلٝوندࣹ بٝنْطِ نگٝنِي إࣹ دٝوْ مَࢠّࣹ. الله تِكَّنِيࢠࣹ كَدِ، أٝ هُطِي ࢠࣹ، أٝ هࣹبِلَنِي ࢠࣹ جهنم. نگࣹ بٝنِي رُتّٝرْدࣹ !

(7) كٝ الله وٝودَنِ كٝنُوجِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ. الله سِفٝرِي وٝنْندࣹ فٝولُطٝ، ݧࣹيݧُطٝ.

(8) مࣹنْ نُلِرِي مَ سࣹيدٝتٝوطٝ، وࣹوْلِنٝووٝ، إࣹ جࣹرْتِنٝووٝ،

(9) فِي نٝ گٝمْطِنِرٝنْ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ، مَوْنِنٝنْ مٝ، تࣹدِّنٝنْ مٝ، سُبُّنْهِنَنٝطٝنْ مٝ بِمْنبِ إࣹ كِيكِيطࣹ.

(10) پࣹلّࣹتْ، ࢠࣹنْ وٝنْࢠࣹ أَهٝدُدࣹ إࣹ مَاطَ، أَنْندُ كٝ الله ࢠࣹ وٝنِ إࣹ أَهٝدُدࣹ. جُنْنگٝ الله نگٝنْ نٝ سࣹنْنگٝ دٝوْ جُوطࣹ مَࢠّࣹ طࣹنْ. كَلَ نٝنْ فِرْتُطٝ، هَرَيْ كٝ هٝورࣹ مَكّٝ أٝ فِرْتَنِ; كَلَ كَدِ هُنُّطٝ كٝ أٝ أَهٝدِ كٝنْ إࣹ الله، أَرَيْ ندࣹ أٝ أٝكَّتَ مٝ نجٝࢠْدِ مَوْنُندِ.

(11) أَرَيْ ندࣹ تُلْتُࢠࣹ جࣹيَاࢠࣹ إࣹ يِمْࢠࣹ لَدّࣹ ࢠࣹنْ وِئَنࣹ: "جَوْلࣹ أَمࣹنْ طࣹنْ إࣹ ࢠࣹيْنگُورࣹيجِ أَمࣹنْ طِنْ سٝكٝلِنِي مࣹنْ. إِنْسِنَنٝ مࣹنْ". هِࢠࣹ وٝوْلِرَ طࣹمْطࣹ مَࢠّࣹ، كٝ أَلَا إࣹ ࢠࣹرْطࣹ مَࢠّࣹ. مَاكُ: "كٝ هٝمْنبٝ وَاوَنِ أٝنْ كَ الله، سِ أٝ فَاندَنِكࣹ أٝنْ لٝرَّ مَا هَرَ أٝ فَاندَنِ أٝنْ نَفَ? وُرِنْ الله سِفٝرِينٝ وٝنْندࣹ هُمْپِتِيطٝ كٝ گٝلَّيْنٝطٝنْ كٝنْ.

(12) وُرِنْ أٝنْ سُكُّنٝ نُلَاطٝ أٝنْ إࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ أَرْتَتَانٝ فࣹسْ كَ ࢠࣹيْنگُورࣹيجِ مَࢠّࣹ. طُمْ ݖُطِنَا إࣹ ࢠࣹرْطࣹ مٝنْ; سِكُّطٝنْ سِكّࣹ بٝنْطࣹ، لَاتِطٝنْ يِمْࢠࣹ هَلْكَاࢠࣹ.

(13) كَلَ دࣹي مٝ گٝمْطِنَالِ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ، هَرَيْ مࣹطࣹنْ هࣹبْلَنِ يࣹدُّࢠࣹ ࢠࣹنْ (يِيتࣹ) دُلْبَيْنگࣹ.

(14) كٝ الله هࣹيرَنِي لَامَگٝلْ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ. هِمٝ هَفٝرَنَ مٝ أٝ مُويَنِ، أٝ لࣹپْتَ مٝ أٝ مُويَنِ. الله سِفٝرِي وٝنْندࣹ هَفٝرٝووٝ، هِنّٝتٝوطٝ.

(15) أَرَيْ ندࣹ لُتُّࢠࣹ ࢠࣹنْ وِئَ، تُمَ فٝكِّتَنٝيْتٝنْ هࣹنْندَگٝلْ كٝنُ-هٝيْرِ ندِنْ: "أَݖّࣹي مࣹنْ جٝكَّ أٝنْ". هِࢠࣹ فَالَا وَطْتِتُگٝلْ دَالٝلْ الله نگٝلْ. مَاكُ: "أٝنْ جٝكَّتَا مࣹنْ فࣹسْ. كٝ وَنٝ مٝنْ نِي الله دَالِرْنٝو أٝيَ تُمَ". مَ ࢠࣹ وِئُ: "أٝنْ أَݧَنَيْ مࣹنْ تُنْ وٝنِ". تِگِ أٝنْ، ࢠࣹ فَامَتَانٝ سِ وَنَا سࣹيطَ.

(16) مَاكَنْ تُلْتُࢠࣹ ࢠࣹنْ إࣹ يِمْࢠࣹ لَدّࣹ ࢠࣹنْ: "أَرْمَ ندࣹ نٝدّࣹتࣹطٝنْ إࣹ يِمْࢠࣹ جٝومِرَاࢠࣹ دٝولࣹ تِيطُࢠࣹ. هٝࢠٝنْ ࢠࣹ مَاطُمْ ࢠࣹ سِلْمَ. سِ أٝنْ طٝفْتِكࣹ، الله أٝكَّيْ أٝنْ نبَرْجَارِ مٝڃُّندِ; سِ أٝنْ هُݖِّتِرِي نٝ هُݖِّتِرْنٝطٝنْ أٝيَ تُمَ، أٝ لࣹپْتِرَ لࣹپْتࣹ مُوسُطࣹ".

(17) بَكَّاتُ فَوَاكِ إࣹ هٝورࣹ ࢠُنْطٝ أٝنْ، وَنَا إࣹ لَڃٝووٝ أٝنْ وَنَا كَدِ إࣹ نَوْنُطٝنْ بَكَّاتُ فَوِي. كَلَ طٝفْتِيطٝ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ، أٝ نَادٝيَيْ مٝ الجنة طِ ݖَنْطِ إِلَتَ سࣹنْنگٝ لࣹيْ مُنْ. كَلَ كَدِ طُورْنِيطٝ، أٝ لࣹپْتِرَيْ مٝ لࣹپْتࣹ مُوسُطࣹ.

(18) گٝمْطِي الله وࣹلَنٝ گٝمْطِمْࢠࣹ ࢠࣹنْ، تُمَ ࢠࣹ أَهٝندِرَيْنٝو إࣹ مَاطَ كَ لࣹيْ لࣹكِّ كِنْ. أٝ أَنْندِ كٝ وٝنِ كٝنْ كَ ࢠࣹرْطࣹ مَࢠّࣹ، أٝ جِپِّنِ دࣹيڃࣹرࣹ ندࣹنْ إࣹ مَࢠّࣹ، أٝ وَرْجٝرِيࢠࣹ أُدِّتَلْ ࢠَدِينگَلْ،

(19) إࣹ كٝنُ-هࣹيْرِ طُوطُندِ ندِ ࢠࣹ هࣹنْندٝتٝو. الله سِفٝرِي وٝنْندࣹ فٝولُطٝ، ݧࣹيݧُطٝ.

(20) الله فٝدِي أٝنْ كٝنُ-هࣹيْرِ طُوطُندِ ندِ هࣹنْندٝتٝطٝنْ، أٝ يَݖِّنَنِ أٝنْ ندِي طٝو، أٝ سُرِ جُوطࣹ يِمْࢠࣹ ࢠࣹنْ إࣹ مٝؤٝنْ، فِي يٝ وٝنُ مَاندࣹ وٝنَنْندࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ، أٝ فࣹوْنِرَ أٝنْ لَاوٝلْ فٝݖِّينگٝلْ.

(21) إࣹ وٝنْندِ گٝو ندِ أٝنْ هَطْتَنَالِ تَهٝ، الله كَنْ نٝ هُࢠِندِي ندِ. الله سِفٝرِي وٝنْندࣹ هَطْتَنْطٝ كَلَ هُوندࣹ.

(22) هَيْ سِ يࣹدُّࢠࣹ ࢠࣹنْ هَࢠِينٝ أٝنْ، ࢠࣹ هُݖِّنْتِنَيْنٝ ࢠَبّࣹ مَࢠّࣹ طࣹنْ، رࣹفْتِ ࢠࣹ هࣹࢠَتَانٝ گِطٝ وَنَا بَلَّــٔٝ.

(23) طُمْ كٝ سُنَّ الله فࣹڃُّنٝوطٝ كٝ أَدِي. أَ هࣹࢠَنْتَا لَاوٝلْ الله نگٝلْ، نبَتِّيگُ.

(24) كٝ كَنْكٝ وٝنِ سُرَنْطٝ أٝنْ جُوطࣹ مَࢠّࣹ طࣹنْ إࣹ مٝؤٝنْ، إࣹ سُرُطٝ جُوطࣹ مٝنْ طࣹنْ إࣹ مَࢠّࣹ، كَ أَيْندࣹ مَكَّ، ࢠَاوٝ أٝ فٝولْنُدࣹ أٝنْ إࣹ مَࢠّࣹ. الله سِفٝرِي وٝنْندࣹ رࣹينُطٝ كٝ گٝلّٝتٝنْ كٝنْ.

(25) كٝ ࢠࣹنْ وٝنِ كٝ يࣹدِّ، ࢠࣹ فَدِّي أٝنْ إࣹ جُورْلِرْدࣹ هٝرْمٝرْدࣹ ندࣹنْ، ࢠࣹ هَطِ نࣹلْدَارِ ندِنْ ندࣹ ندِ يٝتّٝتٝو نٝكُّ مَيْرِ. سِ وَنَانٝ سَبُ وٝرْࢠࣹ گٝمْطِمْࢠࣹ إࣹ رࣹوْࢠࣹ گٝمْطِمْࢠࣹ ࢠࣹ أٝنْ أَنْندَتَانٝ هَا هَࢠٝنْ ࢠࣹ، بَكَّاتُ هࣹࢠَ أٝنْ إࣹ فࣹي مَࢠّࣹ، كٝ أَلْدَا إࣹ أَنْندُگٝلْ... فِي يٝ الله نَادُ إࣹ يُرْمࣹيندࣹ مَكّٝ ندࣹنْ، أٝنْ مٝ أٝ مُويَنِ. سِ ࢠࣹ سࣹنْندٝندِرْ-نٝ إࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ ، كٝ پࣹلّࣹتْ، مࣹنْ لࣹپْتِرَيْنٝ يࣹدُّࢠࣹ ࢠࣹنْ إࣹ مَࢠّࣹ، لࣹپْتࣹ مُوسُطࣹ.

(26) تُمَ ندࣹ ࢠࣹيَ يࣹدُّࢠࣹ وَطُنٝو كَ ࢠࣹرْطࣹ مَࢠّࣹ جِنْنگٝ، جِنْنگٝ مَجِّيَنْكࣹ...أَلَّهْ جِپِّنِ دࣹيڃࣹرࣹ مَكّٝ ندࣹنْ إࣹ دٝوْ نُلَاطٝ مَكّٝ أٝنْ إࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ، أٝ فَوِ ࢠࣹ كٝنْنگٝلْ گٝمْطَلْ، ࢠࣹ وٝنِ ࢠُرْࢠࣹ جٝجُّدࣹ إࣹ هَنْندُدࣹ إࣹ مَگّٝلْ. الله سِفٝرِي أَنْندُگٝلْ كَلَ هُوندࣹ.

(27) گٝمْطِي الله گٝونگِنِرَنِي نُلَاطٝ مَكّٝ أٝنْ، كٝيْطٝلْ نگٝلْ إࣹ هٝورࣹ گٝونگَ : كٝ مَا نَاتٝنْ إࣹ جُولِرْدࣹ هٝرْمٝرْدࣹ ندࣹنْ سِ الله مُويِي، هَرَ كٝ أٝنْ هٝولِيࢠࣹ، فࣹمْنبُࢠࣹ كٝ*إࣹ مٝنْ مَا طُيْتُࢠࣹ (ݖُكُلِ)، أٝنْ هُلَتَا. هِمٝ أَنْندِ كٝ أٝنْ أَنْندَا. أٝ وَطِ كٝ وَنَا نگٝلْ (نَاتُگٝلْ مَكّٝ) طٝنْ، أُدِّتَلْ ࢠَدِينگَلْ.

(28) كٝ كَنْكٝ وٝنِ نُلِرْطٝ نُلَاطٝ مَكّٝ أٝنْ پࣹيوَلْ إࣹ دِينَ گٝونگَ، فِي يٝ أٝ فَوَكَ كَ إࣹ هٝورࣹ كَلَ دِينَ. الله يٝنِي سࣹيدࣹي.

(29) مُهَمّدُ كٝ نُلَاطٝ الله. وٝنْندُࢠࣹ إࣹ مَكّٝ ࢠࣹنْ كٝ ݧَاطُࢠࣹ إࣹ يࣹدُّࢠࣹ ࢠࣹنْ، يُرْمٝندِرْࢠࣹ هَكُّندࣹ مَࢠّࣹ. أَ يِئَيْ ࢠࣹ هِࢠࣹ رُڃَّ، هِࢠࣹ سُجَّ، هِࢠࣹ طَࢠِّرَ الله ࢠُرَلْ إࣹ وࣹلَيࣹي. مَاندࣹيجِ مَࢠّࣹ طِنْ نٝ كَ گࣹيݖࣹ مَࢠّࣹ بَتّࣹ سُجُّگٝلْ نگٝلْ. كٝ طُمْ وٝنِ كٝ ࢠࣹ سِفٝرَا كَ تَوْرࣹيتَ. نٝ ࢠࣹ سِفٝرَا كَ لِنجِيلَ كَنْ، كٝ وَنٝ رࣹمُرُ يَلْتِنگُ پُطِ مَيْرُ، نگُ ࢠَنْنگِنِ طِ، طِ تࣹكِّ، طِ فٝتٝندِرِ إࣹ بِنْندࣹيجࣹ مَجِّ، هَرَ طُمْ نٝ هَاوْنَ رࣹمٝوࢠࣹ، فِي نٝ الله تِكِّنِرَ ࢠࣹ يࣹدُّࢠࣹ ࢠࣹنْ. الله فٝدِي إࣹ مَࢠّࣹ ࢠࣹنْ گٝمْطِمْࢠࣹ ࢠࣹ گٝلِّ مٝڃُّطِ طِنْ، هَفٝرَنَلْ إࣹ نجٝࢠْدِ مَوْنُندِ.

49 Al-Hujurat — 18 آيَة

(1) كٝ أٝنٝنْ يٝ ࢠࣹنْ گٝمْطِمْࢠࣹ ! وَتَ أٝنْ أَدِتٝ يࣹيسٝ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ. هُلࣹي الله. پࣹلّࣹتْ، الله كٝ نَنٝووٝ، يِئٝووٝ.

(2) كٝ أٝنٝنْ يٝ ࢠࣹنْ گٝمْطِمْࢠࣹ ! وَتَ أٝنْ تٝوُ سٝوْتٝوجِ مٝنْ إࣹ هٝورࣹ سٝوْتٝ أَنَّبَاجٝ أٝنْ، وَتَ أٝنْ إࣹوْنٝرْ مٝ كَدِ كٝنْنگٝلْ، وَنٝ نٝ يٝگَ مٝنْ إࣹوْنٝرْتَ يٝگَ، سَكّٝ گٝلّࣹ مٝنْ طࣹنْ بٝنَتَ، هَرَ أٝنْ سٝؤَا-َنْندُدࣹ.

(3) پࣹلّࣹتْ، ࢠࣹنْ جٝلْتِنٝوࢠࣹ سٝوْتٝوجِ مَࢠّࣹ طِنْ كَ نُلَاطٝ الله أٝنْ، كٝ ࢠࣹنْ وٝنِ ࢠࣹ الله جَرَبِي ࢠࣹرْطࣹ مُنْ طࣹنْ گٝمْطَلْ. هِنٝ وٝودَنِ ࢠࣹ هَفٝرَنَلْ إࣹ نجٝࢠْدِ مَوْندِ.

(4) پࣹلّࣹتْ، ࢠࣹنْ ࢠࣹ نٝ نٝدِّرࣹ كَ ࢠَاوٝ ݖُوطِ، كٝ ࢠُرِ طُوطُدࣹ إࣹ مَࢠّࣹ، وَنَا هَقِّلٝوࢠࣹ.

(5) سِنّٝ ࢠࣹ مُݧِّنٝ هَا يَلْتَا إࣹ مَࢠّࣹ، كٝ طُمْ ࢠُرَيْنٝو مٝڃَّنْندࣹ ࢠࣹ. الله كٝ هَفٝرٝووٝ، هِنّٝتٝوطٝ.

(6) كٝ أٝنٝنْ يٝ ࢠࣹنْ گٝمْطِمْࢠࣹ ! سِ فَاسِقِيجٝ أَرْدَنِي أٝنْ إࣹ كُمْپِتَلْ، تࣹفࣹي فِي وَتَ أٝنْ هࣹࢠٝرْ إࣹ يِمْࢠࣹ سَبُ مَجّࣹيدࣹ، وٝنْتٝنْ نِنْسُࢠࣹ إࣹ كٝنْ كٝ وَطُطٝنْ.

(7) أَنْندࣹي وٝنْندࣹ نُلَاطٝ الله أٝنْ هِنٝ هَكُّندࣹ مٝنْ. سِ أٝ تࣹوْتِنٝ أٝنْ إࣹ كٝ طُوطِ إࣹ فِيَكُوجِ، أٝنْ تَمْپَيْنٝ. كٝنٝ الله يِطِنِ أٝنْ گٝمْطِنَلْ، أٝ ݖُطِنِ نگَلْ إࣹ ࢠࣹرْطࣹ مٝنْ; أٝ أَݧِنِ أٝنْ كࣹيفࣹرَاكُ، إࣹ فَاسِقَاكُ إࣹ گࣹدِّ. كٝ ࢠࣹنْ وٝنِ فࣹيوُࢠࣹ ࢠࣹنْ،

(8) طُمْ كٝ ࢠُرَلْ إِمّٝرْدࣹ كَ الله إࣹ نࣹيمَ. الله كٝ أَنْندُطٝ، ݧࣹيݧُطٝ.

(9) سِ فࣹدّࣹيجِ طِطِ إࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ هَࢠِي، مٝڃِّنْتِنࣹي هَكُّندࣹ مَࢠّࣹ. سِ فࣹدّࣹ گٝو بࣹوِي إࣹ هٝورࣹ ندࣹيَ، كٝ هَࢠُگٝلْ وٝنْندࣹ إࣹ بࣹوُدࣹ ندࣹنْ، هَا ندࣹ رُتّٝو إࣹ يَمِرٝورࣹ الله ندࣹنْ. سِ ندࣹ رُتِّكࣹ، مٝڃِّنْتِنِرࣹي هَكُّندࣹ مَࢠّࣹ نُنْطَلْ. نُنْطࣹي، تَوْدࣹ پࣹلّࣹتْ، الله نٝ يِطِ نُنْطٝوࢠࣹ ࢠࣹنْ.

(10) أَنْندࣹي گٝمْطِمْࢠࣹ ࢠࣹنْ كٝ جࣹيْدُࢠࣹ. هَرَيْ مٝڃِّنࣹي هَكُّندࣹ جࣹيْدَلْ مٝنْ نگَلْ، هُلٝنْ الله، بࣹلَجٝؤٝ أٝنْ يُرْمِنْتࣹ.

(11) كٝ أٝنٝنْ يٝ ࢠࣹنْ گٝمْطِمْࢠࣹ ! وَتَ يِمْࢠࣹ جَلْنٝرْ يِمْࢠࣹ : هِنٝ گَسَ هَرَ ࢠࣹنْ (جَلَاࢠࣹ) هِنٝ ࢠُرِ ࢠࣹ مٝڃُّدࣹ. وَنَا رࣹوْࢠࣹ وٝنِ كٝ جَلْنٝرْتَ رࣹوْࢠࣹ : هِنٝ گَسَ هَرَ ࢠࣹنْ نٝ ࢠُرِ ࢠࣹ مٝڃُّدࣹ. وَتَ أٝنْ أَيْبِدْنِرْ كٝئࣹ-مٝنْ، وَتَ أٝنْ بُگٝندِرْ جَمّٝوجࣹ (بٝنْطࣹ). إِنْندࣹ فَاسِقَاكُ ندࣹنْ بٝنِي ࢠَاوٝ ندࣹ گٝمْطِنَلْ ندࣹنْ. كَلَ مٝ تُوبَالِ، كٝ ࢠࣹنْ وٝنِ تٝوݧُࢠࣹ ࢠࣹنْ.

(12) كٝ أٝنٝنْ يٝ ࢠࣹنْ گٝمْطِمْࢠࣹ ! وٝطِّتࣹي إࣹ كٝ طُوطِ إࣹ سِكّࣹ. پࣹلّࣹتْ، يٝگَ إࣹ سِكّࣹ كٝ بَكَّاتُ. وَتَ أٝنْ وِطٝندِرْ، وَتَ يٝگَ مٝنْ ݧٝؤُ يٝگَ. هَرَ گٝوتٝ إࣹ مٝؤٝنْ يِطَيْ ݧَامُگٝلْ تࣹيوُ مُسِطّٝ مُنْ مَايْطٝ? هِطٝنْ أَݧِ طُمْ. هُلࣹي الله. پࣹلّࣹتْ، الله كٝ جَࢠٝووٝ تُوبُوبُيࣹي، هِنّٝتٝوطٝ.

(13) كٝ أٝنٝنْ يٝ يِمْࢠࣹ ! مࣹنْ تَگِرِيِ أٝنْ إِمّٝرْدࣹ إࣹ گٝرࣹلْ إࣹ دࣹيࣹلْ، مࣹنْ وَطِرِ أٝنْ لࣹڃِّ إࣹ گٝرِ، فِي يٝ أٝنْ أَنْندِندِرْ. پࣹلّࣹتْ، ࢠُرْࢠࣹ تࣹدُّدࣹ إࣹ مٝنْ ࢠࣹنْ كَ الله، كٝ ࢠُرْࢠࣹ گٝمْطُدࣹ ࢠࣹنْ إࣹ مٝنْ. پࣹلّࣹتْ، الله كٝ أَنْندُطٝ، هُمْپِتِيطٝ.

(14) يِمْࢠࣹ لَدّࣹ ࢠࣹنْ وِئِ: "مࣹنْ گٝمْطِنِي". مَاكُ: "أٝنْ گٝمْطِنَالِ تَهٝ. وِئࣹي كَنْ : مࣹنْ سِلْمِي. گٝمْطِنَلْ نگَلْ نَاتَالِ تَهٝ كَ ࢠࣹرْطࣹ مٝنْ. سِ أٝنْ طٝفْتِكࣹ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ، أٝ طُيْتَنْتَا أٝنْ هُوندࣹ إࣹ گٝلّࣹ مٝنْ طࣹنْ". پࣹلّࣹتْ، الله كٝ هَفٝرٝووٝ، هِنّٝتٝوطٝ.

(15) گٝمْطِمْࢠࣹ (تِگِ) ࢠࣹنْ، كٝ ࢠࣹنْ گٝمْطِمْࢠࣹ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ، رࣹفْتِ ࢠࣹ سِكِّتَاكِ، ࢠࣹ تِيطْنٝرِ كَدِ جَوْلࣹ إࣹ پِتَّالِ مَࢠّࣹ طِنْ فِي لَاوٝلْ الله نگٝلْ. كٝ ࢠࣹنْ وٝنِ گٝونگُࢠࣹ ࢠࣹنْ.

(16) مَاكُ: "كࣹرࣹ أٝنْ أَنْندِنَيْ الله دِينَ مٝنْ كَنْ، هَرَ نٝنْ الله نٝ أَنْندِ كَلَ كٝ وٝنِ كَمُّولِ إࣹ كَ لࣹيْدِ?" الله كٝ أَنْندُطٝ كَلَ هُوندࣹ.

(17) هِࢠࣹ وَجْ مَ فِي ࢠࣹ سِلْمِي. مَاكُ: "وَتَ أٝنْ وَجِتَمْ سِلْمُگٝلْ مٝنْ. سِ كٝ وٝنِ، كٝ الله وَجِتَتَ أٝنْ فِي أٝ فࣹوْنِرِي أٝنْ إࣹ گٝمْطِنَلْ، سِ وٝنِي كٝ أٝنْ گٝونگُࢠࣹ".

(18) پࣹلّࣹتْ، الله نٝ أَنْندِ وِرْنِيطِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ. الله كٝ رࣹينُطٝ كٝ گٝلّٝتٝنْ كٝنْ.

50 Qaf — 45 آيَة

(1) قَافْ. مِي وٝوندِرِي القرآن تࣹدُّندࣹ ندࣹنْ !

(2) سِ كٝ وٝنِ، ࢠࣹ ݝَلْطُ كٝ تَوِ كٝنْ جࣹرْتِنٝووٝ أَرِي ࢠࣹ إِوْطٝ إࣹ مَࢠّࣹ ; هࣹيفࣹرْࢠࣹ ࢠࣹنْ وِئِ: "طُمْ طٝو كٝ هُوندࣹ هَاوْنِيندࣹ.

(3) كࣹرࣹ سِ مࣹنْ مَايِي، مࣹنْ وٝنْتِي نبُلْنبُلْدِ...? طُمْ طٝنْ كٝ رُتِّتَگٝلْ وٝطُّنگٝلْ نٝنْ !"

(4) گٝمْطِي مࣹطࣹنْ أَنْندِ كٝ لࣹيْدِ ندِنْ ݧَامَتَ كٝنْ إࣹ مَࢠّࣹ ; هِنٝ كَ أَمࣹنْ گَا دࣹفْتࣹرࣹ رࣹينُندࣹ.

(5) سِ كٝ وٝنِ، ࢠࣹ فࣹنُّ گٝونگَ كَنْ تُمَ ندࣹ كَ أَرُنٝو إࣹ مَࢠّࣹ، هِࢠࣹ إࣹ ندࣹرْ فِيَاكُ جِلْࢠُطٝ.

(6) إࣹ ࢠࣹي ندَارَالِ كَمُّ نگُنْ دٝوْ مَࢠّࣹ، كٝ هٝنّٝ مࣹنْ مَهِرِ نگُنْ مࣹنْ ݖُطِنِ نگُ ; هَرَ پࣹرْطِ أَلَنَا نگُ?

(7) لࣹيْدِ ندِنْ كَدِ، مࣹنْ وࣹيرْتِ ندِ مࣹنْ وࣹرْلِي إࣹ مَيْرِ پࣹلّࣹ، مࣹنْ فُطِنِ إࣹ مَيْرِ كَلَ نٝونࣹيجِ پُطِ وٝوطُطِ.

(8) فِي تَسَكُيࣹي إࣹ وَاجُ وٝنَنْندࣹ كَلَ جࣹيَاطٝ رُتّٝتٝوطٝ.

(9) مࣹنْ جِپِّنِ كَدِ ندِيَمْ بَرْكِنَاطَمْ أُمَّادࣹ كَ كَمُّ، مࣹنْ فُطِنِرِ طَمْ گࣹسࣹ إࣹ گَبّࣹ سٝݧࣹتࣹيطࣹ.

(10) إࣹ لࣹطّࣹ تَمَرٝ جُوتُطࣹ وٝندُطࣹ إࣹ پِيندِ فَوٝندِرْندِ،

(11) كٝ أَرْسِكࣹ جࣹيَاࢠࣹ ࢠࣹنْ. مࣹنْ وُورْنِتِرِ طَمْ كَدِ لࣹيْدِ يٝورُندِ. أَوَ كٝ نٝنْ وٝنِ فِي أُمِّتَلْ نگَلْ.

(12) أَدٝ مَࢠّࣹ، يِمْࢠࣹ نوح ࢠࣹنْ، إࣹ يِمْࢠࣹ ࢠُنْندُ ࢠࣹنْ إࣹ سَمُودَئࣹنْ فٝوْ فࣹنُّنٝ،

(13) إࣹ آدِئࣹنْ، إࣹ فرعون إࣹ مُسِطْࢠࣹ لوط ࢠࣹنْ.

(14) إࣹ يِمْࢠࣹ فِتَارࣹ ندࣹنْ، إࣹ يِمْࢠࣹ تُبَّأْ ࢠࣹنْ. كَلَ مَࢠّࣹ فࣹنُّنٝ نُلَاࢠࣹ ࢠࣹنْ، كَمْنبَا أَمْ أٝنْ جٝجِّ.

(15) كࣹرࣹ مࣹنْ رٝنْكُنٝ فِي تَگُ أَرَنُ نگُنْ? كٝ وٝنِ كَمْࢠࣹ، هِࢠࣹ إࣹ جِࢠَارࣹ فِي تَگِتࣹيدࣹ لَاوٝلْ كࣹسٝلْ.

(16) گٝمْطِي مࣹنْ تَگِي نࣹطَّنْكࣹ أٝنْ، مࣹطࣹنْ أَنْندِ كٝ وٝنْكِي مَكّٝ كِنْ سٝوِنْسِنِرْتَ مٝ. كٝ مࣹنࣹنْ ࢠُرِ ࢠَدَادࣹ مٝ إࣹ دِي طَطٝلْ (مَكّٝ) دَاندࣹ نگٝلْ.

(17) تُمَ ندࣹ جَࢠّٝيْتٝوࢠࣹ ࢠࣹنْ طِطٝ جَࢠّٝرٝيْتٝو گٝلّࣹ مَكّٝ طࣹنْ ݧَامٝ إࣹ نَنٝ، هَرَ هِࢠࣹ جٝوطِي.

(18) أٝ وٝوْلَتَا كٝنْنگٝلْ، سِ وَنَا هِنٝ تَكّٝ مَكّٝ وِنْندٝووٝ تَوَاطٝ.

(19) طࣹطّࣹرࣹ مَايْدࣹ ندࣹنْ أَرْدِ إࣹ گٝونگَ : "كٝ طُمْ وٝنُنٝو كٝ دٝگَيْنٝطَا".

(20) وُتّࣹي كَ أَلَاندُ ; كٝ ندࣹنْ وٝنِ ݧَلَاندࣹ كَمْنبَا ندࣹنْ.

(21) وٝنْكِي كَلَ أَرَ، هَرَ هِكِ وٝنْندِ إࣹ سٝگّٝووٝ إࣹ سࣹيدٝتٝوطٝ.

(22) "گٝمْطِي أَ لَاتِنٝكࣹ وࣹلْسِندِيطٝ إࣹ ندࣹي. مࣹنْ هُنْݖِ وِرْنگَلّٝ مَا نگٝنْ كِسَنْ; گِتࣹ مَا طࣹنْ هَنْندࣹ كٝ سࣹيࢠُطࣹ".

(23) وٝنْندِيطٝ مَكّٝ أٝنْ وِئَ: "كٝ طُمْ طٝو وٝنِ كَ أَمْ طٝو، مَاندِنَا-طُمْ".

(24) بُگࣹي كَ جهنم أٝنٝنْ طِطٝ، كَلَ يࣹدُّطٝ ݖَنْندَلّٝ.

(25) هَطٝووٝ مٝڃّࣹرࣹ، جَگِّتٝووٝ، سِكِّتِيطٝ،

(26) أٝنْ وَطِدُطٝ إࣹ الله رࣹوࣹتࣹيطٝ گٝو. بُگࣹي مٝ كَ ندࣹرْ لࣹپْتࣹ سَطْتُطࣹ".

(27) ݖِنْندَدِيجٝ مَكّٝ (سࣹيْتَانࣹ) أٝنْ وِئِ: "جٝومِ أَمࣹنْ، مِي بࣹوِنَالِ أٝو; سِ كٝ وٝنِ، كٝ إࣹ مَجّࣹرࣹ وٝطُّندࣹ أٝ وٝنُنٝو إࣹ مُنْ".

(28) الله دَالَ : "وَتَ أٝنْ يࣹدّٝندِرْ كَ أَمْ طٝو ! ندࣹ تَوْنٝو مِي أَرْدِنِينٝ إࣹ مٝنْ كَمْنبَا.

(29) كٝنْنگٝلْ وَطْتِتࣹتَاكࣹ كَ أَمْ طٝو ; مِي سِفٝرَا كَدِ تٝوݧٝووٝ جࣹيَاࢠࣹ ࢠࣹنْ".

(30) ݧَلْندࣹ مࣹنْ وِئَنٝيْتَ جهنم أٝنْ : "إࣹ أَ هࣹيوِي?" نگࣹ وِئَ: "كٝ ࢠࣹيْدࣹي هِنٝ وٝودِ?"

(31) الجنة أٝنْ ࢠَطْتِنَنࣹي گٝمْطُࢠࣹ ࢠࣹنْ، هَرَ وَنَا كَ وٝطِّ.

(32) "كٝ طُمْ طٝو وٝنِ كٝ فٝدَطٝنْ، وٝنَنْندࣹ كَلَ رُتِّيطٝ إࣹ جٝومِ مُنْ رࣹينِيطٝ.

(33) أٝنْ هُلُطٝ جٝمْ هِنَرَنْكࣹ أٝنْ كَ وِرْنِي، أٝ أَرْدِ إࣹ ࢠࣹرْندࣹ لَاࢠُندࣹ.

(34) نَاتِرࣹي مٝ إࣹ جَمْ". كٝ ندࣹنْ وٝنِ ݧَلَاندࣹ لُتَّيْندࣹ ندࣹنْ.

(35) هِنٝ وٝودَنِ ࢠࣹ تٝنْ كَلَ كٝ ࢠࣹ فَالࣹتࣹي، هَرَ كَدِ ࢠࣹيْدَارِ هِنٝ كَ أَمࣹنْ.

(36) طُوطِي إࣹ گِرࣹ طࣹنْ مࣹنْ هَلْكِ أَدٝ مَࢠّࣹ، كٝ ࢠࣹنْ ࢠُرْنٝوࢠࣹ دٝولࣹ، ࢠࣹنْ وَنْنگٝرِي إࣹ لࣹيْدِ ندِنْ، ندَارُگٝلْ تَوْ سِ ࢠࣹ هࣹࢠَيْ لَاوٝرْدࣹ.

(37) پࣹلّࣹتْ، وٝنِي إࣹ طُمْ، وَاجُ وٝنَنْندࣹ كَلَ جٝمْ ࢠࣹرْندࣹ، مَا تُطُطٝ نٝپِّ هَرَ كٝ تَوَاطٝ.

(38) گٝمْطِي مࣹنْ تَگِي كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ إࣹ كٝنْ كٝ وٝنِ هَكُّندࣹ مَجِّ، ندࣹرْ بَلْطࣹ جࣹيگٝو، تَمْپࣹرࣹ وٝو هࣹࢠَالِ مࣹنْ.

(39) مُݧّٝ إࣹ كٝ ࢠࣹ وٝوْلَتَ كٝنْ، سُبُّنْهِنَنٝطَا جٝومِ مَا أَدٝ پُطَلْ نَانگࣹ نگࣹنْ إࣹ أَدٝ مُتَلْ نگَلْ.

(40) جࣹمَّ أٝنْ كَدِ، سُبُّنْهِنَنٝطَا مٝ إࣹ كَ بَتَّنࣹ جُلْطࣹ فَرْلَاطࣹ.

(41) هࣹطٝ، ݧَلْندࣹ نٝدّٝووٝ أٝنْ نٝدّٝيْتَ إࣹ نٝكُّورࣹ ࢠَدِيندࣹ،

(42) ݧَلْندࣹ ࢠࣹ نَنِرٝيْتَ هَاݖَانگٝ نگٝنْ إࣹ هٝورࣹ گٝونگَ. كٝ ندࣹنْ وٝنِ ݧَلْنْندࣹ يَلْتُگٝلْ نگٝلْ كَ گࣹنَالࣹ .

(43) پࣹلّࣹتْ، كٝ مࣹنࣹنْ وُورْنَتَ مࣹنْ وَرَ، كٝ كَ أَمࣹنْ كَدِ وٝنِ رُتّٝرْدࣹ ندࣹنْ.

(44) ݧَلْندࣹ لࣹيْدِ ندِنْ سࣹيكٝتٝو ࢠࣹ يَلْتِرَ هࣹݧَگٝلْ; كٝ ندࣹ وٝنِ مٝوࢠِرْدࣹ نࣹوَنِيندࣹ مࣹنْ.

(45) كٝ مࣹنࣹنْ ࢠُرِ أَنْندُدࣹ كٝ ࢠࣹ وٝوْلَتَ كٝنْ. أَ وٝنَا دٝولٝووٝࢠࣹ. هَرَيْ وَاجٝرْ القرآن ندࣹنْ، كَلَ هُلٝووٝ كَمْنبَا أَنْ أٝنْ.

51 Adh-Dhariyat — 60 آيَة

(1) مِي وٝوندِرِي كࣹنࣹيلِ جَرٝرَيْطِ (لࣹيْدِ) جَرَگٝلْ !

(2) مِي وٝوندِرِي دُولࣹ رٝنْندٝتٝوطࣹ تٝࢠٝ تِيطْنگٝ !

(3) مِي وٝوندِرِي لَاطࣹ دٝگِرَيْطࣹ نࣹوَارࣹ !

(4) إࣹ پࣹݖّٝوجِ يَمِرَاندࣹ.

(5) كٝ پٝدࣹتࣹ طٝنْ كٝ، كٝ گٝونگٝ.

(6) هَاسْبࣹيدࣹ كٝ كٝ وٝنَتَ.

(7) مِي وٝوندِرِي كَمُّ جٝمْ طَتࣹ نگُنْ !

(8) پࣹلّࣹتْ، هِطٝنْ إࣹ كٝنْنگٝلْ لُورَّنگٝلْ.

(9) هِنٝ يِلْتࣹي إࣹ مَكّٝ كَلَ فِرْلِتَاطٝ إࣹ گٝمْطِنَلْ.

(10) فࣹنٝوࢠࣹ ࢠࣹنْ هُطَامَ.

(11) ࢠࣹنْ يٝلِّيࢠࣹ إࣹ مَجّࣹرࣹ وࣹلْسِندِيࢠࣹ.

(12) هِࢠࣹ لَمْندٝو : "كٝ هٝنْندࣹ وٝنِ ݧَلَاندࣹ نجٝࢠْدِ كُوطࣹ?"

(13) ݧَلَاندࣹ ندࣹ ࢠࣹ لࣹپْتٝيْتࣹيࢠࣹ كَ يِيتࣹ :

(14) "مࣹيطࣹي لࣹپْتࣹ مٝنْ طࣹنْ; كٝ طُمْ وٝنِ كٝ كࣹپَّتْنٝطٝنْ كٝنْ".

(15) پࣹلّࣹتْ، گٝمْطُࢠࣹ ࢠࣹنْ وٝنِي إࣹ الجنة إࣹ ࢠُلِّ.

(16) جَگَّتْنٝوࢠࣹ كٝ جٝومْ مَࢠّࣹ أٝكِّࢠࣹ ࢠࣹ تࣹ كٝ أَدِي طُومْ كٝࢠࣹ مٝڃِّنَتْنٝوࢠࣹ.

(17) هٝࢠࣹ پَمْطُنٝ كٝࢠࣹ طَانَتٝو جࣹمَّ.

(18) إࣹ سَانگَ ندٝوگُ هٝࢠࣹ تٝرٝ يَافࣹيدࣹ.

(19) إِنَ إࣹ جَوْطࣹ مُمࣹنْ هَكّࣹ وٝنَندࣹ ݧَاگَتٝوࢠࣹ إࣹ ࢠࣹ ݧَاگَتَاكٝ.

(20) إِنَ إࣹ ندࣹرْ لࣹيْدِ هࣹ مَاندࣹيجِ وٝنَندࣹ يَنَنَاࢠࣹ.

(21) إࣹ ندࣹرْ پِتَّالِ مٝنْ مَتࣹ أٝنْ تࣹسْكَتَاكَ.

(22) إِنَ إࣹ كَمُّ أَرْسُكَ مٝنْ إࣹ كٝ پٝدَنْتࣹ طٝنْ.

(23) إࣹ جٝومْ كَمُّ إࣹ لࣹيْدِ كٝطُمْ گٝونگٝ هٝنٝ نٝ كَالِرْتٝنْ نِهْ.

(24) نبࣹلَ أَرِي إࣹ مَا يࣹيوْتࣹرࣹ هٝطْࢠࣹ إبراهيم تࣹدِّنَاࢠࣹ.

(25) تُمَ ࢠࣹ نَاتُنٝو كَ مَكّٝ ࢠࣹ وِئِ: "سَلَامْ (كِسِيࣹي مٝنْ)". أٝ مَاكِ: "مࣹنࣹنْ كٝ مࣹنْ طࣹنْڃِي ࢠࣹ أٝنْ".

(26) أٝ نَايِي فَادࣹ إࣹ كٝرࣹيجِ مَكّٝ أٝ أَدّٝرِ نگَارِ پَيْندِ.

(27) أٝ ࢠَدَنِࢠࣹ ندِ أٝ وِي أٝنْ ݧَامَتَ?

(28) أٝ سٝ*إِ إࣹ مُمْ إࣹنْ كُلٝلْ، ࢠࣹ وِي وٝتٝ هُلْ، ࢠࣹ وࣹلْتِنِرِࢠࣹ ݖُكْلࣹلْ.

(29) جٝمْ سُودُ مَكّٝ أَرِ هٝنبٝ وُلَّ أٝ هࣹلِّي يࣹيسَ مَكّٝ، أٝ وِي نَيࣹيجٝ دِمَرَ.

(30) مَلَيْكَاجِ طِ نبِي كٝ هٝنٝ نِهْ جٝومْ مَ دَالِ كٝ كَنْكٝ وٝنِ ݧࣹيݧٝ گَنْندٝ.

(31) أَنَّبَــِٔ إبراهيم مَاكِ: "كٝ وٝنِ فِي مٝنْ، أٝنٝنْ ࢠࣹي نࣹلَاࢠࣹ?"

(32) ࢠࣹ وِئِ: "مࣹنࣹنْ مࣹنْ نࣹلَ إࣹ يِمْࢠࣹ بٝمْࢠࣹ ࢠࣹنْ،

(33) فِي يٝ مࣹنْ أَݖِّتُ إࣹ مَࢠّࣹ، كَاڃࣹ إِوْطࣹ إࣹ لٝوپࣹ (جُطَاطࣹ)،

(34) مَاندِنِرَاطࣹ كَ جٝومِ مَا، فِي فَنْتِنٝوࢠࣹ ࢠࣹنْ".

(35) مࣹنْ يَلْتِنِ تٝونْ كَلَ گٝمْطِمْࢠࣹ وٝنُنٝوࢠࣹ تٝنْ.

(36) مࣹنْ تَوَانٝ تٝنْ، سِ وَنَا سُودُ جُولْࢠࣹ وٝوتُرُ.

(37) مࣹنْ أَݖِّ تٝنْ مَاندࣹ، وٝنَنْندࣹ ࢠࣹنْ هُلٝوࢠࣹ لࣹپْتࣹ مُوسُطࣹ طࣹنْ.

(38) مࣹنْ أَݖِّ كَدِ مَاندࣹ إࣹ موسى، تُمَ ندࣹ مࣹنْ نُلُنٝو مٝ إࣹ فرعون، وٝنْندُدࣹ إࣹ هُجَّ ࢠَنْنگُطٝ.

(39) كٝنٝ أٝ طُورْنٝدِي إࣹ ݖࣹمْنبࣹ مَكّٝ، أٝ وِئِ موسى : "كٝ نبِلࣹيجٝ مَا فࣹيتُطٝ".

(40) مࣹنْ نَنْنگِ مٝ كَنْكٝ إࣹ كٝنࣹيلِ مَكّٝ طِنْ، مࣹنْ بُگٝدِي ࢠࣹ كَ مَايٝ، تَوْدࣹ كٝ أٝ فࣹلْنِيطٝ.

(41) إࣹ أَدِئࣹنْ، تُمَ مࣹنْ وُرْتُنٝو هࣹنْندُ بٝنْندُ ندُنْ إࣹ مَࢠّࣹ.

(42) ندُ أَݖَّتَا هَيْ هُوندࣹ ندࣹ ندُ أَرَنِ، سِ وَنَا ندُ وَطْتَيْ ندࣹ وَ ندٝوندِ.

(43) إࣹ سَمُودَئࣹنْ كَدِ، تُمَ ندࣹ ࢠࣹ وِئَنَنٝو : "دَكْمِتࣹي هَا سࣹيطَ !"

(44) ࢠࣹ تٝوْنِتَنِي يَمِرٝورࣹ جٝومِ مَࢠّࣹ ندࣹنْ، هَاݖَانگٝ نگٝنْ نَنْنگِ ࢠࣹ، تَوِ هِࢠࣹ ڃࣹيوَ.

(45) ࢠࣹ وَاوَانٝ دَرَگٝلْ (لَاوٝو)، ࢠࣹ وٝنَانٝ كَدِ فَابَاࢠࣹ.

(46) كٝ نٝنْ كَدِ يِمْࢠࣹ نوح ࢠࣹنْ كٝ أَدِي ࢠࣹ . ࢠࣹنْ كٝ يِمْࢠࣹ فَاسِقِيࢠࣹ وٝنُنٝو.

(47) كَمُّ نگُنْ، مࣹنْ مَهِرْ نگُ دٝولࣹ أَمࣹنْ. پࣹلّࣹتْ، كٝ مࣹنْ يَݧُّࢠࣹ.

(48) لࣹيْدِ ندِنْ كَنْ، مࣹنْ وࣹيرْتُندِ. مࣹنْ مٝڃِّي وࣹيرْتُࢠࣹ.

(49) إࣹ كَلَ هُوندࣹ، مࣹنْ تَگِي نٝونࣹ طِطِ; بࣹلَجٝؤٝ أٝنْ وَاجِتٝرَيْ.

(50) "أَوَ دٝگࣹي كِسَنْ فَادࣹ إࣹ الله. مِينْ، وٝنَنْندࣹ أٝنْ إِمّٝرْدࣹ كَ مَكّٝ، كٝ مِي جࣹرْتِنٝووٝ ࢠَنْنگُطٝ.

(51) وَتَ أٝنْ وَطِدَنࣹ الله رࣹوࣹتࣹيطٝ گٝو. مِينْ، وٝنَنْندࣹ أٝنْ إِمّٝرْدࣹ كَ مَكّٝ، كٝ مِي جࣹرْتِنٝووٝ ࢠَنْنگُطٝ".

(52) كٝ وَنٝ نٝنْ. نُلَاطٝ أَرَالِ إࣹ ࢠࣹنْ أَدِنٝوࢠࣹ ࢠࣹ، سِ وَنَا ࢠࣹ وِئَيْ : "كٝ نبِلࣹيجٝ مَا فࣹيتُطٝ!".

(53) هَرَ ࢠࣹ وَسٝندِرْ إࣹ طُمْ? كٝ وٝنِ كٝنْ، ࢠࣹي كٝ يِمْࢠࣹ بࣹوُࢠࣹ.

(54) رُنْتٝ ࢠࣹ، أَ فࣹلْنَاكِ إࣹ فِيَࢠّࣹ.

(55) وَاجٝ تُنْ; سَبُ وَاجَگٝلْ نگٝلْ، هِنٝ نَفَ گٝمْطِمْࢠࣹ ࢠࣹنْ.

(56) مِي تَگِرَالِ جِنَّ إࣹ يِمْࢠࣹ، سِ وَنَا فِي يٝ ࢠࣹ رࣹوَنْ.

(57) مِي طَࢠِّرَا ࢠࣹ أَرْسِكࣹ; مِي طَࢠَّا يٝ ࢠࣹ ݧَمِّنَ.

(58) پࣹلّࣹتْ، كٝ الله وٝنِ أَرْسِكٝووٝ أٝنْ، جٝمْ ݖࣹمْنبࣹ تِيطُطࣹ.

(59) هَرَيْ هِنٝ وٝودَنِ ࢠࣹنْ تٝوݧُࢠࣹ، بَكَّاتْ يࣹرُ بَكَّاتْ نَنْندٝ مَࢠّࣹئࣹنْ، وَتَ ࢠࣹ هَوْجُ.

(60) بٝنࣹ وٝودَنِي ࢠࣹنْ يࣹدُّࢠࣹ، إࣹ ندࣹنْ ݧَلَاندࣹ مَࢠّࣹ ندࣹ ࢠࣹ كَمْنبَنَا!

(21) ࢠࣹنْ گٝمْطِمْࢠࣹ، گࣹيْنگٝلْ مَࢠّࣹ نگٝلْ كَدِ جٝكِّ ࢠࣹ إࣹ گٝمْطِنَلْ، مࣹنْ تَوْتِندِنَيْ ࢠࣹ إࣹ گࣹيْنگٝلْ مَࢠّࣹ نگٝلْ. مࣹنْ طُيْتَنْتَا ࢠࣹ هُوندࣹ إࣹ گٝلّࣹ مَࢠّࣹ طࣹنْ; نࣹطّٝ كَلَ كٝ فَگِّتِي كٝنْ نٝ تِگِنَا.

(22) مࣹنْ ࢠࣹيْدَنِ ࢠࣹ مُويَنْتࣹيطِ إࣹ تࣹبّࣹيلِ كَلَ كٝ تُويࣹتࣹي.

(23) هِࢠࣹ جَاࢠٝندِرَ تٝنْ، ݧࣹدُّطࣹ طࣹ جِلَّا مࣹيرࣹ وَنَا كٝ بَكّٝندِنْتَ.

(24) سُكَاࢠࣹ نٝ إࣹلْتَنَاࢠࣹ هِنٝ وَنْنگٝو هَكُّندࣹ مَࢠّࣹ، وَؤُࢠࣹ وَ كَݝّࣹ مَرَاطࣹ.

(25) يٝگَ مَࢠّࣹ فࣹوْتِتَ يٝگَ، ࢠࣹ لَمْندٝندِرَ;

(26) ࢠࣹ وِئَ: "پࣹلّࣹتْ، مࣹنْ لَاتِنٝ أَدٝ هَنْندࣹ ، هُلُنٝوࢠࣹ لࣹپْتࣹ طࣹنْ هَكُّندࣹ يِمْࢠࣹ أَمࣹنْ";

(27) الله مٝڃِّي إࣹ أَمࣹنْ، أٝ دَطْنِ مࣹنْ لࣹپْتࣹ نٝرْنگَلْ.

(28) مࣹنْ لَاتِنٝ أَدٝ هَنْندࣹ ، تٝرٝتٝنٝوࢠࣹ مٝ. پࣹلّࣹتْ، كٝ كَنْكٝ وٝنِ مٝڃّٝ، هِنّٝتٝوطٝ أٝنْ".

(29) أَنْندِنْتِنْ تُنْ، سَبُ نࣹيمَ جٝومِ مَا أٝنْ، أَ وٝنَا ندُرࣹيجٝ وَنَا أَ فࣹيتُطٝ.

(30) كَا ࢠࣹ وِئَيْ وٝنْندࣹ : "كٝ يِمٝووٝ ! مࣹنْ هَبِّتَنِ مٝ تُنْ جِكّࣹ مَايْدࣹ".

(31) مَاكُ : "أَوَ هَبِّتࣹي ! مِينْ كَدِ مِيطُ هَبِّتٝدِ إࣹ مٝنْ".

(32) كَا كٝ هَقِّلّࣹ مَࢠّࣹ طࣹنْ يَمِرْࢠࣹ طُمْ? كَا هَرَ كٝ ࢠࣹ يِمْࢠࣹ بࣹوُࢠࣹ تُنْ?

(33) كَا ࢠࣹ وِئَيْ وٝنْندࣹ : "أٝ سِنْݖُ ندࣹ?" سِ كٝ وٝنِ، وَنَا ࢠࣹ گٝمْطِنٝوࢠࣹ.

(34) أَوَ يٝ ࢠࣹ أَدِّي يࣹوْتࣹرࣹ سُگُ مَيْرࣹ، سِ وٝنِي كٝ ࢠࣹ گٝونگُࢠࣹ.

(35) كَا ࢠࣹ تَگِرَ ࢠَاوٝ تَگُطٝ? كَا كٝ كَمْࢠࣹ وٝنِ تَگُࢠࣹ ࢠࣹنْ?

(36) كَا (كٝ) كَمْࢠࣹ تَگِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ? كٝ وٝنِ كٝنْ، ࢠࣹ يَنَنَاكَ.

(37) كَا ࢠࣹ جٝگِتِ نگَلُوجِ جٝومِ مَا طِنْ? كَا كٝ كَمْࢠࣹ وٝنِ هُࢠِندِيࢠࣹ ࢠࣹنْ?

(38) كَا ࢠࣹ جٝگِتِ ࢠَرْگَلْ نگَلْ ࢠࣹ جࣹنْتٝرْتَ? أَوَ يٝ هࣹطِتِيطٝ مَࢠّࣹ أٝنْ أَدُّ هُجَّ ࢠَنْنگُطٝ !

(39) كَا هَرَ كٝ ࢠِطْࢠࣹ رࣹوْࢠࣹ وٝودَنْتَ مٝ، أٝنٝنْ كَنْ وٝرْࢠࣹ ࢠࣹنْ?

(40) كَا أَ لَمْندِ ࢠࣹ نجٝࢠْدِ، هَرَ يٝࢠُگٝلْ ندِ نٝ تࣹدَّنِ ࢠࣹ?

(41) كَا ࢠࣹ أَنْندُ فِي وِرْنِيطِ طِنْ، هَرَ هِࢠࣹ وِنْندَ?

(42) كَا ࢠࣹ فَالَ فࣹوْجٝورࣹ? دٝو كٝ يࣹدُّࢠࣹ ࢠࣹنْ، وٝنِ فٝولࣹتࣹيࢠࣹ پࣹوْجࣹ ࢠࣹنْ.

(43) كَا هِࢠࣹ مَرِ رࣹوࣹتࣹيطٝ كٝ وٝورِ الله? سࣹنَادࣹ وٝودَنِي الله، إࣹ كٝنْ كٝ ࢠࣹ سِرْكَتَ !

(44) سِ ࢠࣹ يِئِي تَڃَتِنࣹ إِمّٝرْندࣹ كَ كَمُّ نٝ يَنِ، ࢠࣹ وِئَ: "كٝ دُولࣹ فَوٝندِرْطࣹ".

(45) تࣹرْتٝ ࢠࣹ، هَا ࢠࣹ هَوْرٝيَ إࣹ ݧَلَاندࣹ مَࢠّࣹ ندࣹ ࢠࣹ لࣹپْتٝيْتࣹي إࣹ مُنْ;

(46) ݧَلْندࣹ ندࣹ پࣹوْجࣹ مَࢠّࣹ دُنْݖَنٝيْتَا ࢠࣹ هَيْ هُوندࣹ، ࢠࣹ وٝنَا كَدِ وَلّٝيْتࣹيࢠࣹ.

(47) كٝ پࣹلّࣹتْ، هِنٝ وٝودَنِ تٝوݧُࢠࣹ ࢠࣹنْ، لࣹپْتࣹ گٝو كٝ وَنَا طࣹنْ طٝنْ. كٝنٝ كٝ ࢠُرِ طُوطُدࣹ كٝنْ إࣹ مَࢠّࣹ، ࢠࣹ أَنْندَا.

(48) مُݧّٝ ݧَاوٝورࣹ جٝومِ مَا ندࣹنْ. تَوْدࣹ پࣹلّࣹتْ، هِطَ إࣹ ندࣹنْكَ أَمࣹنْ. سࣹنِنِرَا يࣹتُّگٝلْ جٝومِ مَا، تُمَ ࢠَݝٝتٝطَا.

(49) جࣹمَّ أٝنْ كَدِ، سُبُّنْهِنَنٝطَا مٝ إࣹ كَ مُتُگٝلْ كٝودࣹ.

54 Al-Qamar — 55 آيَة

(1) دَرْنگَلْ نگَلْ ࢠَدِكࣹ، لࣹوْرُ ندُنْ كَدِ سࣹيكِكࣹ.

(2) سِ ࢠࣹ يِئِي مَاندࣹ، ࢠࣹ طُورْنٝو ࢠࣹ وِئَ: "كٝ نبِلࣹوُ جٝكّٝندِرْنگُ".

(3) ࢠࣹ فࣹنِّ، ࢠࣹ جٝكِّ نبࣹلࣹيطࣹ مَࢠّࣹ. هَرَ لࣹ هُوندࣹ كَلَ نٝ مَرِ ݧِيࢠِرْدࣹ.

(4) گٝمْطِي كُمْپِتࣹ أَرِي إࣹ مَࢠّࣹ، طࣹ هَطِتٝرْطُمْ وٝنِ إࣹ مُنْ;

(5) طُمْ كٝ ݧࣹݧْڃَلْ يٝتِّينگَلْ. كٝنٝ جࣹرْتِنگٝلْ نگٝلْ نَفَالِ.

(6) سࣹلُ ࢠࣹ. ݧَلْندࣹ نٝدّٝووٝ أٝنْ نٝدّٝيْتَ إࣹ هُوندࣹ أَݧِنِيندࣹ.

(7) جَاسُࢠࣹ ندَارْطࣹ، هِࢠࣹ يَلْتِرَ إࣹ گࣹنَالࣹ طࣹنْ وَنٝ كَنُوجِ ݖَرَتٝوطِ.

(8) هࣹݧَنِيࢠࣹ نٝدّٝووٝ أٝنْ. هࣹيفࣹرْࢠࣹ ࢠࣹنْ وِئَ: "ندࣹي كٝ ݧَلَاندࣹ سَتُّندࣹ".

(9) أَدٝ مَࢠّࣹ، يِمْࢠࣹ نوح ࢠࣹنْ يࣹدُّنٝ. ࢠࣹ فࣹنِّ جࣹيَاطٝ أَمࣹنْ أٝنْ، ࢠࣹ وِئِ : "كٝ جِنَّاطٝ!"، أٝ كَمْنبَنَا.

(10) أٝ تٝرِي جٝومِ مَكّٝ، وٝنْندࣹ : "مِينْ أٝنْ كٝ مِي فٝولَاطٝ، فَابٝ لَمْ ".

(11) مࣹنْ أُدِّتِرِ دَمْنبُطࣹ كَمُّ نگُنْ، ندِيَمْ وَنّݖٝوجَمْ،

(12) مࣹنْ فُݖِّتِرِ لࣹيْدِ ندِنْ ࢠُلِّ. ندِيࣹ طࣹنْ فٝتِّرِ إࣹ يَمِرٝورࣹ هٝدِّرَنٝوندࣹ.

(13) مࣹنْ رٝوندٝرِ مٝ إࣹ لَانَ لࣹسَاكَ مَرْكَ أَلُّوࣹ إࣹ پࣹݝࣹ،

(14) هِكَ دٝگِرَ إࣹ ندࣹنْكَ أَمࣹنْ. طُمْ كٝ نجٝࢠْدِ نوح فࣹنَّنٝوطٝ أٝنْ.

(15) گٝمْطِي مࣹنْ أَݖِّرِي كٝنْ كٝ مࣹنْ وَطِ ࢠࣹ تَسَكُيࣹي. تَوْ سِ وٝودِ وَاجِتٝرَيْطٝ?

(16) تَسْكِي كٝ هٝنّٝ لࣹپْتࣹ أَمْ إࣹ جࣹرْتِنٝلْ (أَمْ) وَؤُنٝو?

(17) گٝمْطِي مࣹنْ نࣹوِنِرِي القرآن ندࣹنْ وَاجِتٝرْگٝلْ. هَرَ وٝودَيْ وَاجِتٝرْطٝ?

(18) آدِئࣹنْ فࣹنُّنٝ. تَسْكࣹي كٝ هٝنّٝ لࣹپْتࣹ أَمْ إࣹ جࣹرْتِنٝلْ (أَمْ) وَؤُنٝو?

(19) مࣹنْ وُرْتُ إࣹ مَࢠّࣹ هࣹنْندُ سَتُّندُ، إࣹ ݧَلَاندࣹ هِيتَاندࣹ، دُومِيندࣹ;

(20) هِندُ وِپِّتِرَ يِمْࢠࣹ ࢠࣹنْ، وَ سِ تَوِي كٝ ࢠࣹ دَڃّࣹ تَمَرٝ إِرْتَاطࣹ.

(21) تَسْكࣹي كٝ هٝنّٝ لࣹپْتࣹ أَمْ طࣹنْ إࣹ جࣹرْتِنٝلْ (أَمْ) نگٝلْ وَؤُنٝو?

(22) گٝمْطِي مࣹنْ نࣹوِنِرِي القرآن ندࣹنْ سَبُ وَاجِتٝرْگٝلْ. هَرَ وٝودَيْ وَاجِتٝرْطٝ?

(23) سَمُودَئࣹنْ فࣹنُّنٝ وَاجُ نگُنْ،

(24) ࢠࣹ وِئِ: "كࣹرࣹ كٝ نࣹطّٝ گٝوتٝ إࣹ أَمࣹنْ وٝنِ كٝ مࣹنْ جٝكَّتَ? هَرَيْ مࣹنْ وٝنِي أٝنْتُمَ إࣹ ندࣹرْ مَجّࣹرࣹ إࣹ تَمْپࣹرࣹرࣹ.

(25) هَرَ كٝ إࣹ مَكّٝ جَنْتٝورࣹ ندࣹنْ وࣹرْلَا هَكُّندࣹ أَمࣹنْ? سِ كٝ وٝنِ، كٝ فࣹنٝووٝ نِي، تٝوْنِتِيطٝ".

(26) أَرَيْ ࢠࣹ أَنْندَ (ݧَلْندࣹ) جَنْنگٝ، كٝ هٝمْنبٝ وٝنِ فࣹنٝووٝ تٝوْنِتِيطٝ أٝنْ.

(27) كٝ مࣹنْ وُرْتِرٝوࢠࣹ نگࣹلٝوبَ ندࣹوَ نبَمْ، جَرِّبَگٝلْ ࢠࣹ. أَيْنِتُ ࢠࣹ، مُݧّٝطَا.

(28) هُمْپِتُ ࢠࣹ وٝنْندࣹ ندِيَمْ طَمْ كٝ سࣹنّندَاطَمْ هَكُّندࣹ مَࢠّࣹ إࣹ نگࣹلٝوبَ نبَمْ ; كَلَ يَرَيْ گࣹࢠَلْ مُنْ.

(29) ࢠࣹ نٝدِّ وٝنْندِيطٝ مَࢠّࣹ، أٝ جٝنِّ كَافَ كَنْ ، أٝ سٝدِ نبَ .

(30) تࣹسْكِي كٝ هٝنّٝ لࣹپْتࣹ أَمْ طࣹنْ إࣹ جࣹرْتِنٝلْ (أَمْ) نگٝلْ وَؤُنٝو?

(31) كٝ هَاݖَانگٝ وٝوتُرُ تُنْ مࣹنْ وُرْتِ إࣹ مَࢠّࣹ، ࢠࣹ وَئِ وَ لࣹدّࣹ يٝورُطࣹ مُنْݖَاطࣹ.

(32) گٝمْطِي مࣹنْ نࣹوِنِرِي القرآن ندࣹنْ وَاجِتٝرْگٝلْ. هَرَ وٝودَيْ وَاجِتٝرْطٝ?

(33) يِمْࢠࣹ لوط ࢠࣹنْ فࣹنُّنٝ جࣹرْتِنࣹ طࣹنْ.

(34) مࣹنْ وُرْتُنٝ هࣹنْندُ كَاڃࣹ إࣹ مَࢠّࣹ، سِ وَنَا إࣹ ࢠࣹيْنگُورࣹ لوط ندࣹنْ، مࣹنْ دَطْندِرْنٝ ࢠࣹنْ كَ وࣹنْندٝ،

(35) مٝڃّࣹرࣹ إِمّٝرْدࣹ إࣹ أَمࣹنْ. كٝ نٝنْ مࣹنْ يٝࢠِتِرْتَ كَلَ يࣹتُّطٝ نࣹيمَ الله أٝنْ .

(36) گٝمْطِي لࣹ أٝ جࣹرْتِنِرْنٝ ࢠࣹ نَنْنگَلْ أَمࣹنْ ݖَتُّنگَلْ نگَلْ. كٝنٝ ࢠࣹ سِكِّتِي جࣹرْتِنٝلْ نگٝلْ.

(37) گٝمْطِي ࢠࣹ طَࢠِّرْنٝ مٝ بٝنِدُگٝلْ إࣹ هٝطْࢠࣹ مَكّٝ ࢠࣹنْ . مࣹنْ مࣹهِ گِتࣹ مَࢠّࣹ طࣹنْ. "مࣹيطࣹي لࣹپْتࣹ طࣹنْ إࣹ جࣹرْتِنٝلْ أَمْ نگٝلْ".

(38) گٝمْطِي كٝ بِمْنبِ لَوْ، وٝنِ كٝ لࣹپْتࣹ دُومِيطࣹ دَوِرَنِ ࢠࣹ.

(39) "مࣹيطࣹي لࣹپْتࣹ إࣹ جࣹرْتِنٝلْ أَمْ".

(40) گٝمْطِي مࣹنْ نࣹوِنِرِي القرآن ندࣹنْ وَاجِتٝرْگٝلْ. هَرَ وٝودَيْ وَاجِتٝرْطٝ?

(41) گٝمْطِي جࣹرْتِنَانگٝ نگٝنْ أَرِي إࣹ يِمْࢠࣹ فرعون ࢠࣹنْ.

(42) ࢠࣹ فࣹنِّ آيࣹيجࣹ أَمࣹنْ طࣹنْ فٝوْ. مࣹنْ نَنْنگِرِ ࢠࣹ، نَنْنگَلْ فٝولُطٝ هَتَّنْطٝ.

(43) كࣹرࣹ كٝ هࣹيفࣹرْࢠࣹ مٝنْ ࢠࣹنْ ࢠُرِ ࢠࣹيَ مٝڃُّدࣹ? كَا كٝ دَطُگٝلْ وٝودَنِ أٝنْ كَ طࣹرِ?

(44) كَا هَرَ ࢠࣹ وِئَيْ : "مࣹنࣹنْ كٝ مࣹنْ دࣹنْتَلْ وَلِّندِرٝوࢠࣹ"?

(45) أَرَيْ ندࣹ دࣹنْتَلْ مَࢠّࣹ نگَلْ فٝولࣹي، ࢠࣹ رُنّتِنَ ࢠَبّࣹ مَࢠّࣹ.

(46) وُرِنْ كٝ دَرْنگَلْ نگَلْ وٝنِ كَمْنبَا مَࢠّࣹ أٝنْ، كٝ دَرْنگَلْ نٝنْ ࢠُرِ مَوْنُدࣹ إࣹ هَاطُدࣹ.

(47) پࣹلّࣹتْ، بٝمْࢠࣹ ࢠࣹنْ وٝنِي إࣹ مَجّࣹرࣹ إࣹ أٝويُنگࣹ.

(48) ݧَلْندࣹ ࢠࣹ دَاسِرٝيْتࣹي گࣹيݖࣹ مَࢠّࣹ طࣹنْ كَ ندࣹرْ يِيتࣹ، (ࢠࣹ وِئَنࣹي): "مࣹيطࣹي مࣹيمُگٝلْ سَقَرَ".

(49) مࣹنࣹنْ مࣹنْ تَگِرْ هُوندࣹ كَلَ إࣹ إࣹࢠّٝورࣹ،

(50) يَمِرٝورࣹ أَمࣹنْ وٝنَالِ، سِ وَنَا كٝنْنگٝلْ گٝوتٝلْ پࣹتْ، يَاوِرَ وَ هِنْڃَانگٝ يِيتࣹرࣹ.

(51) گٝمْطِي مࣹنْ هَلْكِينٝ يࣹرَدِيࢠࣹ مٝنْ. هَرَ وٝودَيْ وَاجِتٝرَيْطٝ?

(52) كَلَ هُوندࣹ ندࣹ ࢠࣹ گٝلِّ نٝ كَ ندࣹرْ طࣹرِ،

(53) كَلَ كَدِ تٝكْسُمْ إࣹ مَوْنُكُمْ، كٝ وِنْندَاطُمْ.

(54) پࣹلّࣹتْ، گٝمْطُࢠࣹ ࢠࣹنْ وٝنِي إࣹ الجنة إࣹ ݖَنْطِ،

(55) إࣹ تٝولٝرْدࣹ گٝونگَ، كَ لَامْطٝ هَتَّنْطٝ.

55 Ar-Rahman — 78 آيَة

(1) جٝمْ هِنَرَݝْكࣹ أٝنْ.

(2) أٝ أَنْندِنِي القرآن.

(3) أٝ تَگِ نࣹطَّݝْكࣹ.

(4) أٝ أَنْندِنِ مٝ ࢠَنْنگِنگٝلْ.

(5) نَانگࣹ نگࣹنْ إࣹ لࣹوْرُ ندُنْ هِنٝ دٝگِرَ لِمٝورࣹ،

(6) كٝودࣹ طࣹنْ كَدِ إࣹ لࣹطّࣹ طࣹنْ، هِنٝ سُجُّدࣹ.

(7) كَمُّ نگُنْ، أٝ تٝووْنُ نگُ. أٝ وَطِ نُنْطَلْ نگَلْ،

(8) فِي وَتَ أٝنْ بࣹوُ كَ پࣹيسِرْطࣹ.

(9) ݧِنِّرٝنْ إࣹتُگٝلْ نگٝلْ نٝ نُنْطِرِ. وَتَ أٝنْ طُيْتُ پࣹيسِرْطࣹ طࣹنْ.

(10) لࣹيْدِ ندِنْ، أٝ وَطَنْندِ تَگٝورࣹ ندࣹنْ:

(11) مُويَنْتࣹيطِ هِنٝ إࣹ مَيْرِ، إࣹ لࣹطّࣹ تَمَرٝ مَرْطࣹ پٝڃِّ.

(12) إࣹ گَبّࣹ مَرْطࣹ ݧَاندࣹ، إࣹ أُورْنگٝلْ.

(13) إࣹ كٝ هٝنْطِ إࣹ نࣹيمَاجِ جٝومِ مٝنْ طِنْ وٝنِ كٝ فࣹنّٝتٝنْ?

(14) أٝ تَگِرِ نࣹطّٝ أٝنْ إِمّٝرْدࣹ إࣹ لٝوپَلْ، وَؤُنگَلْ وَ كٝ جُطَا.

(15) أٝ تَگِرِ جِنَّاجِ طِنْ، إِمّٝرْدࣹ إࣹ لࣹوْ لࣹوْندُ يِيتࣹ لَاࢠُكِ.

(16) إࣹ كٝ هٝنْطِ إࣹ نࣹيمَاجِ جٝومِ مٝنْ طِنْ يࣹدّٝتٝنْ?

(17) جࣹيْطٝ فُطِرْطࣹ طࣹنْ طِطِ جࣹيِ كَدِ مُتِرْطࣹ طࣹنْ طِطِ.

(18) إࣹ كٝ هٝنْطِ إࣹ نࣹيمَاجِ جٝومِ مٝنْ طِنْ يࣹدّٝتٝنْ?

(19) أٝ تُپِّندِرِ مَاجࣹ طࣹنْ طِطِ، هِطࣹ فٝتَّ.

(20) هࣹيدٝ هِنٝ هَكُّندࣹ مَجّࣹ، طࣹ جِلّٝندِرْتَا.

(21) إࣹ كٝ هٝنْطِ إࣹ نࣹيمَاجِ جٝومِ مٝنْ طِنْ يࣹدّٝتٝنْ?

(22) هِنٝ يَلْتَ إࣹ طࣹنْ مَاجࣹ طِطِ: كَݝّࣹ "لُؤُلُؤُ" إࣹ "مَرْجَانْ".

(23) إࣹ كٝ هٝنْطِ إࣹ نࣹيمَاجِ جٝومِ مٝنْ طِنْ يࣹدّٝتٝنْ?

(24) كٝ كَنْكٝ وٝودَنِ دٝگٝوجࣹ لࣹسَاطࣹ كَ ندࣹرْ مَايٝ، وَؤُطࣹ وَ پࣹلّࣹ.

(25) إࣹ كٝ هٝنْطِ إࣹ نࣹيمَاجِ جٝومِ مٝنْ طِنْ يࣹدّٝتٝنْ?

(26) كَلَ كٝ وٝنِ إࣹ هٝورࣹ مَيْرِ كٝ كٝ رࣹؤَتَ،

(27) هࣹدّٝو جَاتِ جٝومِ مَا كِنْ، جٝمْ مَوْنگُ إࣹ تࣹدُّنگَلْ.

(28) إࣹ كٝ هٝنْطِ إࣹ نࣹيمَاجِ جٝومِ مٝنْ طِنْ يࣹدّٝتٝنْ?

(29) ࢠࣹنْ وٝنْࢠࣹ كَ كَمُّولِ إࣹ كَ لࣹيْدِ هِنٝ نَامْنٝو مٝ. ݧَلْندࣹ وٝو، هِمٝ إࣹ فِيَاكُ.

(30) إࣹ كٝ هٝنْطِ إࣹ نࣹيمَاجِ جٝومِ مٝنْ طِنْ يࣹدّٝتٝنْ?

(31) أَرَيْ ندࣹ مࣹنْ فَاندِي هَسْبُدࣹ أٝنْ، أٝنٝنْ يٝ تࣹدَّ كٝئࣹ إࣹنْ طِطٝ.

(32) إࣹ كٝ هٝنْطِ إࣹ نࣹيمَاجِ جٝومِ مٝنْ طِنْ يࣹدّٝتٝنْ?

(33) كٝ أٝنٝنْ يٝ دࣹنْتَلْ جِنَّ إࣹ يِمْࢠࣹ ! سِ أٝنْ هَتَّنِي وُطِّرْگٝلْ كَ سࣹرَاجِ كَمُّولِ إࣹ كَ لࣹيْدِ، هَرَيْ وُطّࣹي. أٝنْ وُطِّرْتَا سِ وَنَا إࣹ دٝولࣹ.

(34) إࣹ كٝ هٝنْطِ إࣹ نࣹيمَاجِ جٝومِ مٝنْ طِنْ يࣹدّٝتٝنْ?

(35) هِنٝ وُرْتࣹي إࣹ مٝنْ - أٝنٝنْ جِنَّ إࣹ يِمْࢠࣹ - طࣹمْنگَلْ يِيتࣹ إࣹ ݖُرْكِ مَا سِلَ تَايُطٝ ، أٝنْ فَابٝندِرْتَا.

(36) إࣹ كٝ هٝنْطِ إࣹ نࣹيمَاجِ جٝومِ مٝنْ طِنْ يࣹدّٝتٝنْ?

(74) سُبُّنْهِنٝرْ إِنْندࣹ جٝومِ مَا، مَوْطٝ أٝنْ!

(75) مِي وٝوندِرِي مُتِرْطࣹ كٝودࣹ طࣹنْ.

(76) پࣹلّࣹتْ، طُمْ كٝ وٝوندٝورࣹ مَوْندࣹ، سِ هِطٝنْ أَنْندُنٝو.

(77) پࣹلّࣹتْ ندࣹي كٝ القرآن كٝندࣹ تࣹدُّندࣹ،

(78) وٝنْندࣹ إࣹ دࣹفْتࣹرࣹ رࣹينَاندࣹ

(79) مࣹيمَتَا ندࣹنْ سِ وَنَا لَࢠِّنَاࢠࣹ ࢠࣹنْ;

(80) كٝ جِپِّنِرَاندࣹ كَ جࣹيْطٝ وِنْندࣹرࣹ ندࣹنْ.

(81) إࣹ كٝ ندࣹي يࣹوْتْرࣹ وٝنْكِ كٝ پࣹنّٝتٝنْ أٝنٝنْ?

(82) هِطٝنْ وَطَ أَرْسِكࣹ مٝنْ (تٝࢠٝ نگٝنْ) كٝ أٝنْ فࣹنّٝوࢠࣹ أَرْسِكُطٝ أٝنْ ?

(83) كٝ هَطَتَ ندࣹ وٝنْكِي يٝتِّي كَ لٝوكٝڃࣹرࣹ،

(84) هَرَ أٝنٝنْ هِطٝنْ ندَارَ أٝنْتُمَ،

(85) هَرَ نٝنْ كٝ مࣹنࣹنْ ࢠُرِ أٝنْ ࢠَدَادࣹ مٝ، كٝنٝ أٝنْ يِئَتَا.

(86) كٝ هَطَتَ أٝنْ، سِ تَوِينٝ هَرَ أٝنْ وَرْجࣹتَاكࣹ.

(87) رُتّٝنْ كِنْ وٝنْكِي ، سِ تَوِي كٝ أٝنْ گٝونگُࢠࣹ?

(88) سِ تَوِي أٝنْ كٝ جࣹيَاطٝ إࣹ ࢠَتِّنَاࢠࣹ ࢠࣹنْ،

(89) هَرَيْ كٝ فٝوْتࣹرࣹ، إࣹ مٝڃّࣹرࣹ، إࣹ الجنة نࣹيمَ (وٝودَنِ مٝ).

(90) سِ تَوِي كَدِ كٝ جࣹيَاطٝ إࣹ يِمْࢠࣹ ݧَامٝ ࢠࣹنْ،

(91) هَرَيْ أٝ جَࢠّٝرٝيْتࣹ : "كِسَلْ مَا" إࣹ ࢠَݝگࣹ يِمْࢠࣹ ݧَامٝ ࢠࣹنْ.

(92) سِ تَوِي نٝنْ كٝ جࣹيَنٝوطٝ إࣹ فࣹنُّࢠࣹ مَجُّࢠࣹ ࢠࣹنْ،

(93) هَرَيْ هٝطْتٝرْدࣹ مَكّٝ أٝنْ كٝ فَتَيْطَمْ،

(94) إࣹ سُمࣹيدࣹ جَهِيمِ.

(95) پࣹلّࣹتْ، كٝ طُمْ وٝنِ گٝونگَ يَنَنَاطٝ أٝنْ.

(96) سُبُّنْهِنٝرْ إِنْندࣹ جٝومِ مَا، مَوْطٝ أٝنْ!

57 Al-Hadid — 29 آيَة

(1) كَلَ كٝ وٝنِ كَ كَمُّولِ إࣹ كَ لࣹيْدِ سࣹنِنِي الله. كٝ كَنْكٝ وٝنِ فٝولُطٝ، ݧࣹيݧٝ أٝنْ.

(2) كٝ كَنْكٝ هࣹيرَنِي لَامُ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ. هِمٝ وُورْنَ هِمٝ وَرَ، كَنْكٝ كٝ أٝ هَتَّنْطٝ كَلَ هُوندࣹ.

(3) كٝ كَنْكٝ وٝنِ أَرَنٝ إࣹ سَكِّتِيطٝ، فࣹيݧُطٝ إࣹ وِرْنِيطٝ، كَنْكٝ كٝ أٝ أَنْندُطٝ كَلَ هُوندࣹ.

(4) كٝ كَنْكٝ تَگِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ ندࣹرْ بَلْطࣹ جࣹيگٝو، رࣹفْتِ أٝ فٝتٝندِرِ كَ دٝوْ "أَرْسِ"; هِمٝ أَنْندِ كٝ مُتَتَ كٝنْ كَ لࣹيْدِ، إࣹ كٝ ڃَوَتَ إࣹ مَيْرِ، إࣹ كٝ جِپّٝرْتَ كَ كَمُّ إࣹ كٝ ڃَوَتَ إࣹ مَگُّ. هِمٝ وٝنْندِ إࣹ مٝنْ كَلَ كَ وٝنْطٝنْ. الله نٝ رࣹينِ كٝ گٝلّٝتٝنْ كٝنْ.

(5) كٝ كَنْكٝ هࣹيرَنِي لَامُ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ. كٝ كَ الله فِيَكُوجِ طِنْ رُتِّرْتࣹي.

(6) هِمٝ نَادِــَٔ جࣹمَّ أٝنْ إࣹ ندࣹرْ ݧَلٝرْمَ أٝنْ، أٝ نَادِرَ ݧَلٝرْمَ أٝنْ كَدِ إࣹ ندࣹرْ جࣹمَّ أٝنْ. كَنْكٝ كٝ أٝ أَنْندُطٝ كٝ وٝنِ إࣹ ندࣹرْ ࢠࣹرْطࣹ.

(7) گٝمْطِنࣹي الله إࣹ نُلَاطٝ مَكّٝ أٝنْ، نَفْقٝنْ كٝنْ كٝ أٝ وَطِرِ أٝنْ لٝمْتِنَاࢠࣹ إࣹ مُنْ. گٝمْطِمْࢠࣹ ࢠࣹنْ إࣹ مٝنْ، ࢠࣹ نَفْقِ جَوْلࣹ مَࢠّࣹ طࣹنْ ، هِنٝ وٝودَنِ ࢠࣹنْ نجٝࢠْدِ مَوْنُندِ.

(8) إࣹ كٝ هࣹࢠِ أٝنْ، أٝنْ گٝمْطِنْتَا الله? إࣹ هِنٝ نُلَاطٝ أٝنْ هِنٝ نٝدَّ أٝنْ إࣹ گٝمْطِنگٝلْ جٝومِ مٝنْ. گٝمْطِي الله ڃࣹتِّي إࣹ مٝنْ أَهَدِ، سِ أٝنْ لَاتِكࣹ گٝمْطِمْࢠࣹ.

(9) كٝ كَنْكٝ جِپِّنْتَ إࣹ جࣹيَاطٝ مَكّٝ أٝنْ آيࣹيجࣹ ࢠَنْنگُطࣹ، فِي نٝ أٝ يَلْتِنِرَ أٝنْ إࣹ نِࢠࣹ فَادࣹ إࣹ ندَيْگُ. پࣹلّࣹتْ، الله كٝ يُمِنٝووٝ أٝنْ، هِنّٝتٝوطٝ.

(10) إࣹ كٝ هࣹࢠِ أٝنْ، أٝنْ سَكّٝتَاكٝ فِي لَاوٝلْ الله نگٝلْ، إࣹ هِنٝ لࣹ كٝ الله وٝودَنِ رٝنُگٝلْ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ? فٝتَتَا إࣹ مٝنْ، ࢠࣹنْ سَكِّنٝوࢠࣹ أَدٝ أُدِّتَلْ نگَلْ ࢠࣹ هَࢠِدَا; كٝ ࢠࣹنْ ࢠُرِ مَوْنُدࣹ دَرْجَ إࣹ دِي ࢠࣹنْ سَكِّيࢠࣹ ࢠَاوٝ طُمْ ࢠࣹ هَࢠِدَا. ࢠࣹ فٝوْ مَࢠّࣹ نٝنْ، الله فٝدِي ࢠࣹ مٝڃّࣹرࣹ. الله نٝ هُمْپِتِي كٝ گٝلّٝتٝنْ كٝنْ.

(11) كٝ هٝمْنبٝ ݧَوْلِرْتَ الله ݧَمَاندࣹ لَبَاندࣹ، أٝ سٝوَنَ مٝ ندࣹ، هَرَ نٝ وٝودَنِ مٝ نجٝࢠْدِ تࣹدُّندِ?

(12) ݧَلْندࣹ يِئٝيْتَا گٝمْطِمْࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ گٝمْطِمْࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، هَرَ أَنّٝورَ مَࢠّࣹ يَارَ يࣹيسٝ مَࢠّࣹ إࣹ ݧَامٝ مَࢠّࣹ (ࢠࣹ وِئَنࣹي): "وࣹلْتَارࣹ مٝنْ هَنْندࣹ كٝ الجنة، طِ ݖَنْطِ إِلَتَ سࣹنْنگٝ لࣹيْ مُنْ، كٝ أٝنْ دُومٝتٝوࢠࣹ ندࣹرْ تٝنْ". كٝ طُمْ وٝنِ مَلَلْ مَوْنگَلْ نگَلْ.

(13) ݧَلْندࣹ نَافِقِيࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ نَافِقِيࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ وِئَنٝيْتَ گٝمْطِمْࢠࣹ ࢠࣹنْ: "هَبّࣹي مࣹنْ، مࣹنْ جَلْبِنٝرَ أَنّٝورَ مٝنْ أٝنْ". ࢠࣹ وِئَنࣹي : "يِلْتٝرࣹي ࢠَاوٝ مٝنْ، طَࢠّٝيٝنْ أَنّٝورَ". هࣹيدٝ وَطࣹي هَكُّندࣹ مَࢠّࣹ، نگٝ دَمَلْ نگَلْ يُرْمࣹيندࣹ وٝنِ إࣹ ندࣹرْ مُنْ، ࢠَاوٝ مَگَّلْ نٝونْ، كٝ لࣹپْتࣹ وٝنِ إࣹ بَنْنگࣹ مُنْ.

(14) هِࢠࣹ نٝدَّ ࢠࣹ : "إࣹ مࣹنْ وٝندَانٝ إࣹ مٝنْ?" ࢠࣹ وِئَ : "كٝ نٝنْ وٝنِرِ. كٝ وٝنِ تُنْ، أٝنْ فِتِنّٝ كٝئࣹ مٝنْ، هَبِّتَنِطٝنْ گٝمْطِمْࢠࣹ ࢠࣹنْ بٝنࣹ ، سِكِّتِطٝنْ، يࣹلَاجِ هٝدِ أٝنْ هَا يَمِرٝورࣹ الله ندࣹنْ أَرِ. هٝدٝووٝ أٝنْ هٝدِ أٝنْ إࣹ مٝڃّࣹرࣹ الله ندࣹنْ.

(15) هَنْندࣹ أٝنْ، أٝنْ جَࢠَنْتَاكٝ سٝكْتٝرْطُمْ وَنَا إࣹ ࢠࣹنْ يࣹدُّࢠࣹ. دࣹپّٝرْدࣹ مٝنْ ندࣹنْ كٝ كَ يِيتࣹ : كٝ نگࣹنْ وٝنِ كٝ هَنْندُطٝنْ. نگࣹ بٝنِي رُتّٝرْدࣹ.

(16) إࣹ يٝنْتَانِ گٝمْطِمْࢠࣹ ࢠࣹنْ ندࣹ ࢠࣹرْطࣹ مَࢠّࣹ طࣹنْ دࣹيڃِرْتَ جَنْتٝورࣹ الله ندࣹنْ إࣹ كٝنْ كٝ جِپِّي إِمّٝرْدࣹ إࣹ گٝونگَ? هَرَ ࢠࣹ وَءَالِ وَ ࢠࣹنْ أٝكَّنٝوࢠࣹ دࣹفْتࣹ ࢠࣹنْ كٝ أَدِي، دُمُنَّ أٝنْ جُوتَنِ ࢠࣹنْ، ࢠࣹرْطࣹ مَࢠّࣹ طࣹنْ يٝورِ، كٝ طُوطُنٝو إࣹ مَࢠّࣹ كٝنْ كٝ فَاسِقِيࢠࣹ.

(17) أَنْندࣹي پࣹلّࣹتْ، الله وُورْنِتَيْ لࣹيْدِ ندِنْ ࢠَاوٝ ندِ يٝورِي. گٝمْطِي مࣹنْ ࢠَنْنگِنِرَنِي أٝنْ آيࣹيجࣹ طࣹنْ، فِي نٝ هَقِّلِرٝنْ.

(18) پࣹلّࣹتْ، سَكّٝتٝوࢠࣹ (جَوْلࣹ مُنْ ࢠࣹنْ) كَ وٝرْࢠࣹ إࣹ كَ رࣹوْࢠࣹ ࢠࣹ ݧَوْلِ الله ݧَمَاندࣹ لَبَاندࣹ، ࢠࣹ سٝوَنٝيْتࣹ (نبَرْجَارِ) هَرَ نٝ وٝودَنِ ࢠࣹ نجٝࢠْدِ مَوْندِ.

(19) ࢠࣹنْ گٝمْطِمْࢠࣹ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ، كٝ ࢠࣹنْ وٝنِ گٝونگِيَݝْكࣹ إࣹنْ. سَهٝدِنْࢠࣹ ࢠࣹنْ كَ جٝومِ مُنْ، هِنٝ وٝودَنِ ࢠࣹ نجٝࢠْدِ مَࢠّࣹ إࣹ أَنّٝورَ مَࢠّࣹ. ࢠࣹنْ نٝنْ يࣹدُّࢠࣹ ࢠࣹ فࣹنِّ آيࣹيجࣹ أَمࣹنْ طࣹنْ، كٝ ࢠࣹنْ وٝنِ يِمْࢠࣹ جَهِيمِ نگࣹنْ.

(20) أَنْندࣹي وٝنْندࣹ نگُورْندَمْ أَدُنَ طَمْ كٝ فِجِندَارُ، إࣹ نبࣹلࣹيطࣹ، إࣹ ݖُطَارِ، إࣹ ࢠُرٝندِرٝ هَكُّندࣹ مٝنْ إࣹ جَوْلࣹ إࣹ ࢠِطْࢠࣹ. وَنٝ تٝࢠٝ هَاوْنُنگٝ رࣹمٝوࢠࣹ ࢠࣹنْ پُطِ مُنْ، رࣹفْتِ نگٝ يٝورَ، يِئَا هِنگٝ سٝيْطِ; رࣹفْتِ نگٝ وٝنْتَ مُنْݖِيطُمْ. هِنٝ كَ لَاكَرَ، لࣹتّࣹ سَطْتُطࣹ، إࣹ يَافُيࣹي إِمّٝرْدࣹ إࣹ الله إࣹ يَرْلُيࣹي. نگُورْندَمْ أَدُنَ طَمْ وٝنَالِ، سِ وَنَا دَكَمّࣹهٝيْ هٝدٝ.

(21) أَدٝندِرࣹي إࣹ (گٝلّࣹ) يَافَاوُيَ كَ جٝومِ مٝنْ، إࣹ الجنة مٝ يَاجࣹيندِ مُنْ وَئِ يَاجࣹيندِ كَمُّ نگُنْ إࣹ لࣹيْدِ ندِنْ، أٝنْ مَرَنَامَ ࢠࣹنْ گٝمْطِمْࢠࣹ الله إࣹ نُلَاࢠࣹ مَكّٝ ࢠࣹنْ. طُمْ كٝ ࢠُرَلْ الله نگَلْ أٝ أٝكَّتَ أٝنْ مٝ أٝ مُويَنِ. الله كٝ جٝمْ ࢠُرَلْ مَوْنُنگَلْ.

(22) بٝنࣹ وٝو هࣹࢠَتَا أٝنْ إࣹ لࣹيْدِ ندِنْ مَا إࣹ پِتَّالِ مٝنْ طِنْ، سِ وَنَا طُمْ وٝنِينٝ إࣹ دࣹفْتࣹرࣹ أَدٝ مࣹنْ تَگُدࣹ ندࣹي تَگٝورࣹ . پࣹلّࣹتْ، طُمْ كٝ كٝ نࣹوَنِي الله.

(23) كٝ فِي وَتَ أٝنْ سُنٝ إࣹ كٝ لَاوِي أٝنْ كٝنْ، وَنَا كَدِ وࣹلْتٝرْگٝلْ كٝ أٝ أٝكِّ أٝنْ كٝنْ. الله يِطَا كَلَ ݧَاڃٝووٝ، مَنْتِيطٝ.

(24) ࢠࣹنْ وُدّٝوࢠࣹ ࢠࣹ يَمِرَ يِمْࢠࣹ ࢠࣹنْ نگُدَّمْ... كَلَ طُورْنِيطٝ، پࣹلّࣹتْ، الله كَنْ كٝ گَلٝ، يࣹتِّنِيطٝ.

(25) گٝمْطِي مࣹنْ نُلِرِي نُلَاࢠࣹ أَمࣹنْ ࢠࣹنْ ࢠَنْنگَنْندُيࣹيجِ، مࣹنْ جِپِّندِنَنِ ࢠࣹ دࣹفْتࣹ إࣹ سَرِيَارࣹ، فِي نٝ يِمْࢠࣹ ࢠࣹنْ ݧِيࢠِرَ إࣹ نُنْطَلْ. مࣹنْ جِپِّنِ كَدِ نجَمْندِ ندِنْ، ندِ دٝولࣹ تِيطُطࣹ وٝنِ إࣹ مُنْ إࣹ نَفٝوجࣹ وٝنَنْندࣹ يِمْࢠࣹ ࢠࣹنْ، إࣹ نٝ الله فࣹݧِّنِرَ ࢠࣹنْ وَلّٝوࢠࣹ مٝ إࣹ نُلَاطٝ مَكّٝ أٝنْ كَ وِرْنِي. پࣹلّࣹتْ، الله كٝ جٝمْ ݖࣹمْنبࣹ، فٝولُطٝ.

(26) گٝمْطِي مࣹنْ نُلِي نوح إࣹ إبراهيم، مࣹنْ وَطِ إࣹ گࣹيْنگٝلْ مَࢠّࣹ نگٝلْ أَنَّبَاكُ إࣹ دࣹفْتࣹ. هِنٝ إࣹ مَࢠّࣹ فࣹيوُࢠࣹ، كٝ طُوطِ كٝنْ إࣹ مَࢠّࣹ كٝ فَاسِقِيࢠࣹ.

(27) رࣹفْتِ مࣹنْ جٝكِّندِنِ إࣹ بَتّࣹيجِ مَࢠّࣹ طِنْ، نُلَاࢠࣹ أَمࣹنْ گٝو ، مࣹنْ جٝكِّنْتِنِرِ ࢠࣹ إِيسَا گࣹطَلْ مَرْيَمَ أٝنْ، مࣹنْ أٝكِّ مٝ لِنجِيلَ، مࣹنْ وَطِ إࣹ ࢠࣹرْطࣹ جٝكُّࢠࣹ مٝ ࢠࣹنْ : يُرْمࣹيندࣹ إࣹ گِگّٝلْ. لَبّࣹيَاگَلْ نگَلْ ࢠࣹ سِنْݖِ نگَلْ، مࣹنࣹنْ مࣹنْ وَطِّنَا نگَلْ إࣹ مَࢠّࣹ. مࣹنْ لَمْندَاكِ سِ وَنَا طَࢠُّگٝلْ وࣹلَيࣹي الله. كٝنٝ ࢠࣹ أَيْنِرَالِ طُمْ نٝ هَانِ أَيْنِرࣹيدࣹ. مࣹنْ أٝكِّ نجٝࢠْدِ گٝمْطِمْࢠࣹ ࢠࣹنْ إࣹ مَࢠّࣹ. كٝ طُوطُنٝو كٝنْ إࣹ مَࢠّࣹ كٝ فَاسِقِيࢠࣹ.

(28) كٝ أٝنٝنْ يٝ گٝمْطِمْࢠࣹ! هُلࣹي الله، گٝمْطِنٝنْ نُلَاطٝ مَكّٝ أٝنْ; أٝ يࣹطَيْ أٝنْ گࣹطࣹ طِطِ إࣹ نبَرْجَارِ مَكّٝ ندِنْ، أٝ وَطَنَ أٝنْ أَنّٝورَ مٝ ندَيْگِنٝرْتٝنْ، أٝ هَفٝرَنَ أٝنْ. پࣹلّࣹتْ، الله كٝ هَفٝرٝووٝ، هِنّٝتٝوطٝ.

(29) فِي نٝ يِمْࢠࣹ دࣹفْتࣹ ࢠࣹنْ أَنْندِرَ وٝنْندࣹ ࢠࣹ هَتَّنَا هَيْ هُوندࣹ إࣹ ࢠُرَلْ الله نگَلْ، إࣹ وٝنْندࣹ ࢠُرَلْ نگَلْ كٝ إࣹ جُوطࣹ الله وٝنِ; هِمٝ يࣹطَ نگَلْ مٝ أٝ مُويَنِ. پࣹلّࣹتْ، الله كٝ جٝمْ ࢠُرَلْ مَوْنگَلْ.
