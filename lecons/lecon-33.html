31 Luqman — 34 آيَة

(1) أَلِفْ، لَامْ، مِيمْ

(2) طࣹي، كٝ آيࣹيجࣹ دࣹفْتࣹرࣹ ݧࣹݧَاندࣹ ندࣹنْ.

(3) كٝ ندࣹ پࣹيوَلْ إࣹ يُرْمࣹيندࣹ وٝنَنْندࣹ مٝڃِّنٝوࢠࣹ ࢠࣹنْ،

(4) ࢠࣹنْ ݧِنّٝوࢠࣹ جُلْدࣹ ندࣹنْ، ࢠࣹ تٝتَّ آكَّلْ أٝنْ، هَرَ كَمْࢠࣹ كٝ ࢠࣹ يَنَنَاࢠࣹ (فِي لَاكَرَ).

(5) كٝ ࢠࣹنْ وٝنِ إࣹ پࣹيوَلْ إِمّٝرْدࣹ إࣹ جٝومِ مَࢠّࣹ ࢠࣹ، كٝ كَمْࢠࣹ وٝنِ مَلَاࢠࣹ ࢠࣹنْ.

(6) هِنٝ وٝودِ إࣹ يِمْࢠࣹ ࢠࣹنْ، سٝودِرٝووٝ نبࣹلࣹيطࣹ كٝنْنگُطِ هَا أٝ مَجِّرَ إࣹ لَاوٝلْ الله نگٝلْ كٝ أَلْدَا إࣹ گَنْندَلْ، أٝ جٝگِتٝرَ (َايࣹيجࣹ) طࣹنْ جَلْنِتْگٝلْ. ࢠࣹنْ طٝنْ، هِنٝ وٝودَنِ ࢠࣹ لࣹپْتࣹ هٝيْنٝوجࣹ.

(7) سِ آيࣹيجࣹ أَمࣹنْ طࣹنْ جَنْنگَامَ إࣹ مَكّٝ، أٝ هُݖِّتِرَ(رُنْتٝرَ) تٝوْنِتَارࣹ، وَ سِ تَوِي أٝ نَنَالِ طࣹ، وَ سِ تَوِي كٝ پَهُ وٝنِ إࣹ نٝپِّ مَكّٝ. وࣹلْتَنِرْ-مٝ لࣹپْتࣹ مُوسُطࣹ.

(8) پࣹلّࣹتْ، ࢠࣹنْ گٝمْطِمْࢠࣹ ࢠࣹ گٝلِّ مٝڃِّ طِنْ، هِنٝ وٝودَنِ ࢠࣹ أَلْجَنَّاجِ نࣹيمَ.

(9) كٝ ࢠࣹ دُومٝتٝوࢠࣹ تٝنْ، طُمْ كٝ فٝدٝورࣹ الله گٝونگُندࣹ. كٝ كَنْكٝ وٝنِ أٝن فٝولُطٝ، ݧࣹيݧُطٝ.

(10) أٝ تَگِرِي كَمُّولِ طِنْ كٝ أَلْدَا إࣹ تُگَالࣹ، طࣹ يِئٝتٝنْ; أٝ وࣹرْلِي كَ لࣹيْدِ پࣹلّࣹ فِي وَتَ ندِ دِلِّدُ إࣹ مٝنْ; أٝ سَاكِ إࣹ مَيْرِ كَلَ رُوگَيْنگࣹلْ. مࣹنْ جِپِّنِرِ ندِيَمْ كَ كَمُّ، مࣹنْ فُطِنِ إࣹ مَيْرِ كَلَ نٝونࣹيجِ تࣹدُّطِ.

(11) "طُمْ طٝو كٝ تَگُ الله. هٝلّࣹي لَمْ كٝ طِ (رࣹوٝتٝنْ طِنْ) گَانِنْ مَكّٝ تَگِ?" سِ كٝ وٝنِ، تٝوݧُࢠࣹ ࢠࣹنْ نٝ إࣹ مَجّࣹرࣹ ࢠَنْنگُندࣹ.

(12) گٝمْطِي مࣹنْ أٝكِّي لُقْمَانْ ݧࣹنْڃَلْ: "وٝنْندࣹ يࣹتُّ الله. كَلَ يࣹتُّطٝ، هَرَيْ كٝ هٝورࣹ-مَكّٝ أٝ يࣹتَّنِ; كَلَ نٝنْ يࣹدُّطٝ، أَنْندُ پࣹلّࣹتْ، الله كٝ گَلٝ، يࣹتِّنِيطٝ".

(13) جَنْتٝ تُمَ لُقْمَانَ وِئَنّٝو گࣹطَلْ مَكّٝ أٝنْ، هَرِ هِمٝ وَاجٝو مٝ: "كٝ أَن ࢠِنْنگࣹلْ أَمْ، وَتَ أَ سِرْكَنْ الله. پࣹلّࣹتْ، سِرْكُ أٝن كٝ تٝوݧࣹ مَوْطࣹ".

(14) مࣹنْ يَمِرِي نࣹطَّنْكࣹ أٝن طٝفْتَگٝلْ جِبِمْࢠࣹ مٝ ࢠࣹنْ; يُمَّ مَكّٝ سَاوِرِي مٝ لٝئࣹرࣹ إࣹ هٝورࣹ لٝئࣹرࣹ: إࣹنْتُگٝلْ مٝ نگٝلْ، كٝ إࣹ ندࣹرْ دُوࢠِ طِطِ. "وٝنْندࣹ يࣹتَّمْ إࣹ جِنَّاࢠࣹ مَا ࢠࣹنْ. كٝ كَ أَم وٝنِ وَطْتِنِرْدࣹ ندࣹنْ".

(15) سِ ࢠࣹ تِيطْنٝرِي مَ فِي سِرْكَنگٝلْ لَمْ إࣹ كٝ أَ أَلَنَا طُمْ گَنْندَلْ، وَتَ أَ طٝفْتٝ ࢠࣹ; وٝنْندِرْ إࣹ مَࢠّࣹ كَ أَدُنَ طٝو مٝڃّࣹرࣹ. جٝكُّ لَاوٝلْ أٝن رُتِّيطٝ إࣹ أَمْ. رࣹفْتِ كٝ كَ أَم وٝنِ رُتّٝرْدࣹ مٝنْ، مِي يࣹيتِتٝو أٝنْ، كٝ گٝلَّيْنٝطٝنْ.

(16) "كٝ أَن ࢠِنْنگࣹلْ أَمْ، أَنْندُ طࣹنْ گٝلّࣹ بٝنْطࣹ مَا مٝڃّࣹ ، سِ فٝتِي وَ أَبّࣹرࣹ، إِمّٝرْدࣹ إࣹ نࣹنْندِ فٝݧّࣹ، طُمْ وٝنَ إࣹ فࣹتࣹرࣹ، مَا كَ كَمُّولِ مَا كَ لࣹيْدِ، الله أَرْتِرَيْ ندࣹ. پࣹلّࣹتْ، الله كٝ نࣹوِيطٝ، هُمْپِتِيطٝ.

(17) كٝ أَن يٝ ࢠِنْنگࣹلْ أَمْ، ݧِيࢠْنُ جُلْدࣹ ندࣹنْ، يَمِرَا مٝڃّࣹرࣹ، هَطَا ݝِݧࣹرࣹ(پَنْكَرࣹ)، مُݧّٝ-طَا إࣹ كٝ هࣹࢠُ-مَا. پࣹلّࣹتْ، طُمْ كٝ جࣹيَاطُمْ إࣹ فِيَكُّجِ مَوْطِ.

(18) وَتَ فَلْ يِمْࢠࣹ ࢠࣹنْ يࣹيسٝ مَا نگٝنْ، وَتَ أَ يَهْرُ كَ لࣹيْدِ ݧَاڃَگٝلْ. پࣹلّࣹتْ، الله يِطَا كَلَ ݧَاڃٝتٝوطٝ مَنْتٝو.

(19) هَكِندٝ إࣹ يَهْدُ-مَا ندُنْ، جٝلْتِنَا(ُسْتُ) هِتٝ مَا نگٝنْ; كٝ ࢠُرِ أَݧࣹيدࣹ إࣹ هِتٝوجِ طِنْ، پࣹلّࣹتْ، كٝ لُوكَادُ نبَبَّ.

(20) إࣹ أٝن يِئَالِ وٝنْندࣹ الله إࣹلْتَنِي أٝن كٝ وٝنِ كٝنْ كَ كَمُّولِ إࣹ كَ لࣹيْدِ? أٝ هُوࢠِتِرِ أٝن نࣹيمَاجِ مَكّٝ فࣹيݧُطِ إࣹ سُوطِيطِ. هِنٝ وٝودِ إࣹ يِمْࢠࣹ ࢠࣹنْ، أٝن وࣹنْنجِرٝووٝ إࣹ فِي الله، كٝ أَلْدَا إࣹ گَنّندَلْ، وَنَا پࣹيوَلْ، وَنَا دࣹفْتࣹرࣹ جَبُندࣹ.

(21) سِ ࢠࣹ وِئَنَا:"جٝكّࣹي كٝنْ كٝ الله جِپِّنِ"، ࢠࣹ وِئَ: "سِ كٝ وٝنِ، مࣹنْ جٝكَّيْ كٝ مࣹنْ تَوْرِ كٝنْ بَابِرَاࢠࣹ أَمࣹنْ ࢠࣹنْ"; إࣹ هَيْ سِ تَوِي سࣹيْتَانࣹ وٝنِي هِنٝ نٝدَّ ࢠࣹ إࣹ لࣹپْتࣹ دُلْبٝووࣹ نگࣹنْ!

(22) كَلَ جࣹبِّلُطٝ يࣹيسٝ مُنْ نگٝنْ فَادࣹ إࣹ الله، هَرَ كٝ أٝ مٝڃِّنْطٝ، گٝمْطِي أٝ جَگِّيمَ إࣹ ࢠٝگّٝلْ تِيطُنگٝلْ. كٝ كَ الله وٝنِ بَتَّنࣹ فِيَكُوجِ طِنْ.

(23) كَلَ يࣹدُّطٝ، وَتَ كࣹيفࣹرَاكُ مَكّٝ سُنِنࣹ. كٝ كَ أَمࣹنْ وٝنِ رُتّٝرْدࣹ مَࢠّࣹ، مࣹنْ هُمْپِتَ ࢠࣹ كٝ ࢠࣹ هُووُنٝو. پࣹلّࣹتْ، الله كٝ أَنْندُطٝ جَاتِ ࢠࣹرْطࣹ.

(24) مࣹنْ دَكْمِتَيْ ࢠࣹ سࣹيطَ، رࣹفْتِ مࣹنْ فٝرْسَࢠࣹ إࣹ لࣹپْتࣹ تࣹكُّطࣹ.

(25) پࣹلّࣹتْ، سِ أَ لَمْندِكࣹ ࢠࣹ : "كٝ هٝمْنبٝ تَگِ كَمِّيلِ طِنْ إࣹ لࣹيْدِ ندِنْ?" مَ ࢠࣹ وِئُ: "كٝ الله!" مَاكُ: "يࣹتّٝورࣹ وٝودَنِي الله!". كٝنٝ كٝ ࢠُرِ طُوطُدࣹ إࣹ مَࢠّࣹ أَنْندَا.

(26) كٝ الله وٝودَنِ كَلَ كٝ وٝنِ كَمُّولِ إࣹ كَ لࣹيْدِ. پࣹلّࣹتْ، الله، كٝ كَنْكٝ وٝنِ گَلٝ، يࣹتِّنِيطٝ.

(27) هَيْ سِ تَوِي كٝ وٝنِ كَ هٝورࣹ لࣹيْدِ كٝنْ إࣹ لࣹطّࣹ، سࣹتِّرَامَ كَرَنبِ، مَايٝ نگٝنْ ࢠࣹيْدٝرَ (وٝنْتُگٝلْ ندَهَ) ࢠَاوٝ طُمْ، يࣹرُ مَاجࣹ جࣹيطِطِ، كٝنْنگُطِ الله طِنْ لَنَّتَا. پࣹلّࣹتْ، الله كٝ فٝولُطٝ، ݧࣹيݧُطٝ.

(28) تَگُگٝلْ أٝن إࣹ إِمِّنْتِنگٝلْ وٝنَالِ، سِ وَنَا وَ (تَگُگٝلْ) وٝنْكِي گٝوتِ. پࣹلّࣹتْ، الله كٝ نَنٝووٝ، يِئٝووٝ.

(29) إࣹ أَ يِئَالِ وٝنْندࣹ الله نٝ نَادَ جࣹمَّ أٝن إࣹ ندࣹرْ ݧَلٝرَ أٝنْ، أٝ نَادَ ݧَلٝرْمَ كَدِ إࣹ ندࣹرْ جࣹمَّ أٝنْ، أٝ إࣹلْتِي كَدِ نَانگࣹ نگࣹنْ إࣹ لࣹوْرُ ندُنْ، كَلَ مَجِّ نٝ دٝگَ هَا إࣹ لَجَلْ إِنَّانگَلْ? إࣹ وٝنْندࣹ الله كٝ هُمْپِتِيطٝ كٝ گٝلّٝتٝنْ كٝنْ.

(30) طُمْ كٝ جَݝْتَا طٝو ، كٝ تَوْدࣹ پࣹلّࣹتْ، الله كٝ كَنْكٝ وٝنِ گٝونگَ، إࣹ وٝنْندࣹ كٝنْ كٝ ࢠࣹ نٝدَّتَ گَانِنْ مَكّٝ، كٝ طُمْ وٝنِ مࣹيرࣹ ندࣹنْ، إࣹ وٝنْندࣹ كٝ الله وٝنِ تٝووُطٝ، مَوْنُطٝ أٝنْ.

(31) إࣹ أَ يِئَالِ وٝنْندࣹ لَاطࣹ طࣹنْ هِنٝ دٝگِرَ كَ مَايٝ إࣹ نࣹيمَ الله، فِي يٝ أٝ هٝلُّ أٝن مَاندࣹيجِ مَكّٝ طِنْ? وٝنِي إࣹ طُمْ، مَاندࣹيجِ، وٝنَنْندࣹ كَلَ هࣹيوُطٝ-مُݧَلْ، يࣹتّٝووٝ (نࣹيمَ).

(32) سِ تَوِي مٝرْلٝوطࣹ(بࣹمْپࣹجّࣹ) هِپِّرِيࢠࣹ وَ دُولࣹ مَا پࣹلّࣹ ، ࢠࣹ نٝدَّيْ الله، هَرَ هِࢠࣹ لَࢠِّنَنِ مٝ دࣹوَلْ; تُمَ ندࣹ أٝ دَطْندُنٝوࢠࣹ كَ نجٝرْندِ، تَوِ نٝ إࣹ مَࢠّࣹ هَكِندِيطٝ. يࣹدَّتَا آيࣹيجࣹ أَمࣹنْ طࣹنْ، سِ وَنَا كَلَ هࣹيوُطٝ جَنْفَ، كࣹيفࣹيرٝ.

(33) كٝ أٝنٝنْ يٝ يِمْࢠࣹ! هُلࣹي جٝومِ مٝنْ، هُلٝنْ ݧَلَاندࣹ، ندࣹ بَابَ دُنْݖَنٝيْتَا ࢠِطّٝ مُنْ، وَنَا ࢠِطّٝ أٝن كَدِ دُنْݖَنٝيْتَ بࣹنْ مُنْ هَيْ هُوندࣹ. پࣹلّࣹتْ، فٝدٝورࣹ الله ندࣹنْ كٝ گٝونگَ. وَتَ نگُورْندَمْ أَدُنَ هٝدُ أٝنْ، وَتَ هٝدُ أٝن إࣹ الله، (سࣹيْتَانࣹ) هٝدٝووٝ!

(34) پࣹلّࣹتْ، كٝ كَ الله وٝنِ أَنْندَلْ دَرْنگَلْ، هِمٝ جِپِّنَ تٝࢠٝ; هِمٝ أَنْندِ كٝ وٝنِ كَ ندࣹرْ رࣹنْنگَاجِ. وٝنْكِي (وٝو) أَنْندَا كٝ كِ فَگِّتٝتٝ جَنْنگٝ، وٝنْكِي (وٝو) كَدِ أَنْندَا كٝ إࣹ لࣹيْدِ هٝنْندِ مَايَتَ. پࣹلّࣹتْ، الله كٝ أَنْندُطٝ، هُمْپِتِيطٝ.

32 As-Sajdah — 30 آيَة

(1) أَلِفْ، لَامْ، مِيمْ

(2) جِپِّنࣹيدࣹ دࣹفْتࣹرࣹ ندࣹنْ، سِكّࣹ وٝو أَلَا إࣹ مُؤُنْ، كٝ إِمّٝرْدࣹ كَ جࣹيْطٝ وِنْندࣹرࣹ ندࣹنْ.

(3) وَلَّ ࢠࣹ وِئَيْ كَنْكٝ (مُهَمَّدُ) أٝ فࣹفِندِ ندࣹ? سِ كٝ وٝنِ، كٝ ندࣹ گٝونگَارࣹ، إِمّٝرْدࣹ كَ جٝومِ مَا، فِي يٝ أَ جࣹرْتِنْ يِمْࢠࣹ، ࢠࣹ جࣹرْتِنٝووٝ أَرَانٝ إࣹ مُنْ أَدٝ مَاطَ، بࣹلَجٝؤٝ، ࢠࣹ فࣹيوَيْ.

(4) كٝ الله وٝنِ تَگُطٝ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، إࣹ كٝنْ كٝ وٝنِ هَكُّندࣹ مَجِّ، إࣹ ندࣹرْ بَلْطࣹ جࣹيگٝو. رࣹفْتِ أٝ فٝتٝندِرِ كَ دٝوْ "َرْسْكِ". أَلَنَا أٝن گَانِنْ مَكّٝ، گِطٝ وَنَا تࣹفٝووٝ. إࣹ أٝن وَاجِتٝتَاكٝ?

(5) هِمٝ فࣹوْجَ فِيَاكُوجِ طِنْ كَ كَمُّ هَا كَ لࣹيْدِ، رࣹفْتِ طِ ڃࣹينگَ كَ مَكّٝ، إࣹ ݧَلَاندࣹ ندࣹ إࣹمْنبࣹرࣹ مُنْ وٝنِ، دُوࢠِ أُجُنࣹرࣹ إࣹ كٝ لِمٝتٝنْ.

(6) أٝن فࣹوْجٝووٝ فِي طُمْ أٝن ، كٝ أَنْندُطٝ وِرْنِيطِ إࣹ فࣹيݧُطِ، فٝولُطٝ، هِنّٝتٝوطٝ.

(7) أٝن لَبِنْطٝ كَلَ هُوندࣹ ندࣹ أٝ تَگِ. أٝ فُطِّ تَگُدࣹ نࣹطَّنْكࣹ ندِنْ إࣹ لٝوپَلْ،

(8) رࣹفْتِ أٝ وَطِ گࣹيْنگٝلْ مَكّٝ نگٝلْ، إࣹ سٝرْتٝندِرَلْ، إِمّٝرْدࣹ ندِهَلْ مَنِييُ لٝؤُطَمْ;

(9) رࣹفِ أٝ فٝتْندِندِرِ مٝ، أٝ وُتِّ إࣹ مَكّٝ رُوهُ مَكّٝ أٝنْ. أٝ وَطَنِ أٝن نَنْطࣹ، إࣹ گِيطࣹ إࣹ ࢠࣹرْطࣹ. كٝنٝ فَنْطِي كٝ يࣹتّٝنْتٝنْ!

(10) ࢠࣹ وِأْ: "ࣹ نࣹي، سِ مِينْ مَجِّي كَ لࣹيْدِ، إࣹيٝو مࣹنْ وٝنْتَيْ إࣹ تَگُ هࣹسُ?" كٝ وٝنِ كَمْࢠࣹ إࣹ هَوْرُگٝلْ إࣹ جٝومِ مَࢠّࣹ، كٝ ࢠࣹ يࣹدُّࢠࣹ طُمْ .

(11) مَاكُ: "كٝ هُنَّتَ أٝنْ، كٝ الملائكة مَايْدࣹ أٝن وَكِّلَاطٝ أٝنْ. رࣹفْتِ كٝ كَ جٝومِ مٝنْ رُتّࣹتࣹطٝنْ".

(12) سِنّٝ أَ يِئَيْنٝ، تُمَ ندࣹ بٝمْࢠࣹ ࢠࣹنْ أُگِّنْتَ كٝئࣹ-مَࢠّࣹ كَ جٝومِ مَࢠّࣹ! لْࢠࣹ وِئَ : جٝومِ" أَمࣹنْ، مࣹنْ يِئِي مࣹنْ نَنِي، رُتّٝنْ مࣹنْ، مࣹنْ گٝلَّ كٝ مٝڃِّ; مࣹنࣹنْ مࣹنْ يَنَنَامَ".

(13) سِنّٝ مࣹنْ مُويُنٝ، مࣹنْ أٝكَّيْنٝ كَلَ وٝنْكِي پࣹيوَلْ مَكِّ. كٝ وٝنِ تُنْ، كٝنْنگٝلْ جٝجِّي إِمّٝرْدࣹ إࣹ أَن : "مَ مِي هࣹبِّنِرَيْ جهنم أٝن جِنَّ إࣹ يِمْࢠࣹ، ࢠࣹ دࣹنْندَانگَلْ".

(14) "مࣹيطِرࣹي (لࣹپْتࣹ طࣹنْ) سَبُ وࣹلْسِندَگٝلْ مٝنْ هَوْرُگٝلْ إࣹ ندࣹي ݧَلَاندࣹ مٝنْ طٝو. مࣹنࣹنْ (كَدِ) مࣹنْ يࣹجِّتِي أٝنْ. مࣹيطِرࣹي لࣹپْتࣹ لُتَّلْ، سَبُ كٝ هُووَيْنٝطٝنْ كٝنْ".

(15) أَنْندࣹي كٝ گٝمْطِنْتَ آيࣹيجࣹ أَمࣹنْ طࣹنْ، كٝ ࢠࣹنْ تَوْࢠࣹ سِ ࢠࣹ وَاجٝرَامَ طࣹ، ࢠࣹ سُويِرَيْ سُجُّگٝلْ، ࢠࣹ سُبُّهِنٝرَ يࣹتُّگٝلْ جٝومِ مَࢠّࣹ، كَمْࢠࣹ ࢠࣹ تٝوْنِتٝتَاكٝ.

(16) بࣹݖّࣹ مَࢠّࣹ طࣹنْ هِنٝ وٝطِّتٝو إࣹ لࣹلٝرْطࣹ طࣹنْ، هِࢠࣹ نٝدِّرَ جٝومِ مَࢠّࣹ كُلٝلْ إࣹ تَمَؤُ; إࣹ كٝنْ كٝ مࣹنْ أَرْسِكِ ࢠࣹ، هِࢠࣹ وِنْتٝو.

(17) وٝنْكِي (وٝو) أَنْندَا كٝ مَرَنَا ࢠࣹ إࣹ كٝ ࢠُࢠِّنْتَ ࢠࣹرْندࣹ، طُمْ كٝ نجٝࢠْدِ كٝ ࢠࣹ گٝلَّيْنٝو كٝنْ !

(18) إࣹ أٝن لَاتِيطٝ گٝمْطِنْطٝ وَأَيْ وَ أٝن فَاسِقࣹ? (َاءَا)، ࢠࣹ فٝتَتَا.

(19) سِ كٝ ࢠࣹنْ أٝمْطِمْࢠࣹ ࢠࣹ گٝلِّ گٝلّࣹ مٝڃّࣹ، هِنٝ وٝودَنِ ࢠࣹ أَلْجَنَّاجِ وࣹيرْدࣹ جِپّٝرْدࣹ، سَبُ كٝࢠࣹ گٝلَّيْنٝو كٝنْ.

(20) سِ تَوِي نٝنْ كٝ ࢠࣹ فَاسِقِيࢠࣹ، هَرَيْ فࣹمْپٝرْدࣹ ࢠࣹنْ طٝنْ ندࣹنْ، كٝ كَ يِيتࣹ: تُمَ كَلَ ࢠࣹ فَالَا يَلْتُگٝلْ إࣹ مَگّࣹ، ࢠࣹ رُتِّتࣹي إࣹ مَگّࣹ، ࢠࣹ وِئَنࣹي: "مࣹيطࣹي لࣹپْتࣹ يِيتࣹ نگࣹ فࣹنَّيٝطٝنْ نگࣹنْ".

(21) مَ مࣹنْ مࣹيطِنِرْ ࢠࣹ لࣹپْتࣹ أَدُنَ، أَدٝ لࣹپْتࣹ مَوْطࣹ طࣹنْ (لَاكَرَ)، بࣹلَجٝؤٝ، ࢠࣹ رُتِّتٝتٝ.

(22) أَلَا ࢠُرْطٝ تٝوݧُدࣹ وَنٝ وَاجٝرَاطٝ آيࣹيجࣹ جٝومِ مُنْ طࣹنْ، رࣹفْتِ أٝ طُورْنِي طࣹ ! پࣹلّࣹتْ، مࣹنࣹنْ، كٝ مࣹنْ يٝفْتٝتٝوࢠࣹ إࣹ بٝمْࢠࣹ ࢠࣹنْ.

(23) گٝمْطِي مࣹنْ أٝكِّي موسى دࣹفْتࣹرࣹ ندࣹنْ، - وَتَ أَ وٝنُ إࣹ سِكّࣹ إࣹ فِي هَوْرٝيْگٝلْ إࣹ مَكّٝ -، مࣹنْ وَطِ ندࣹ طٝوٝرْدࣹ ࢠِطْࢠࣹ إسرائيل ࢠࣹنْ.

(24) مࣹنْ وَطِ إࣹ مَࢠّࣹ أَرْدِيࢠࣹ، طٝوِرٝوࢠࣹ (يِمْࢠࣹ ࢠࣹنْ) يَمِرٝورࣹ أَمࣹنْ، ندࣹ ࢠࣹ مُݧِّنٝو، ࢠࣹ وٝنِ يَنَنَاࢠࣹ إࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ.

(25) پࣹلّࣹتْ، جٝومِ مَا، كٝ كَنْكٝ سࣹنْندٝيْتَ هَكُّندࣹ مَࢠّࣹ، ݧَلْندࣹ دَرْنگَلْ، إࣹ كٝنْ كٝ ࢠࣹ لُرَّيْنٝو إࣹ فِي مُنْ.

(26) إࣹ فࣹيݧَنَالِ ࢠࣹ، نٝ فٝتِ طُوطُدࣹ إࣹ كٝ مࣹنْ هَلْكِ أَدٝ مَࢠّࣹ إࣹ گِرࣹ، هِࢠࣹ يَهَ إࣹ كٝطٝولِ مَࢠّࣹ? پࣹلّࣹتْ، نٝ إࣹ طُمْ، تَسَكُيࣹيجِ ! إࣹ ࢠࣹي هَقِّلْتَا?

(27) إࣹ ࢠࣹ يِئَالِ، وٝنْندࣹ مࣹطࣹنْ سٝگِّرَ ندِيَمْ طَمْ كَ لࣹيْدِ هٝكُّندِ، مࣹنْ يَلْتِنِرَ طَمْ رࣹمُرُ، ندُ نࣹيمٝرَاطِ مَࢠّࣹ إࣹ كَمْࢠࣹ ݧَامَتَ إࣹ مُنْ? إࣹ ࢠࣹ يِئَتَا?

(28) هِࢠࣹ وِئَ: "كٝ هٝنْندࣹ تُمَ وٝنِ نگَلْ أُدِّتَلْ، سِ أٝن لَاتِكࣹ گٝونگُࢠࣹ?"

(29) مَاكُ: "ݧَلْندࣹ أُدِّــَٔلْ نگَلْ، كٝ هࣹيفࣹرْࢠࣹ ࢠࣹنْ گٝمْطِنٝيْتَ نَفٝيْتَاࢠࣹ، وَنَا كَمْࢠࣹ وٝنِ نࣹنَّتࣹيࢠࣹ !".

(30) طُورْنٝ ࢠࣹ، هَࢠِّتٝ-طَا. كَمْࢠࣹ كَنْ، كٝ ࢠࣹ هَࢠِّتِيࢠࣹ (لࣹپْتࣹ).

33 Al-Ahzab — 73 آيَة

(1) كٝ أَن يٝ أَنَّبِيجٝ! هُلُ الله، وَتَ أَ طٝفْتٝ هࣹيفࣹرْࢠࣹ ࢠࣹنْ إࣹ نَافِقِيࢠࣹ ࢠࣹنْ. پࣹلّࣹتْ، الله لَاتِكࣹ أَنْندُطٝ، ݧࣹيݧُطٝ.

(2) جٝكُّ كٝ وَهَيِنَا كٝنْ إࣹ مَاطَ، إِمّٝرْدࣹ كَ جٝومِ مَا. پࣹلّࣹتْ، الله لَاتِكࣹ هُمْپِتِيطٝ كٝ گٝلّٝتٝنْ كٝنْ.

(3) فَوٝ إࣹ الله. الله يٝنِي فَوࣹتࣹيطٝ إࣹ مُنْ.

(4) الله وَطَنَالِ نࣹطَّنْكࣹ ࢠࣹرْطࣹ طِطِ إࣹ ندࣹرْ فُلْكُرُ مَكّٝ. أٝ وَطَالِ كَدِ رࣹسٝندِرَاࢠࣹ مٝنْ ࢠࣹنْ ࢠࣹ نَنْندِنِرْتٝنْ، نࣹينِرَاࢠࣹ مٝنْ. أٝ وَطَالِ كَدِ نٝدِّتِرَيْࢠࣹ أٝن (بَابَ ࢠࣹنْ)، ࢠِطْࢠࣹ مٝوطٝنْ. طُمْ مٝوطٝنْ، كٝ كٝنگٝلْ مٝنْ يَلْتُنگٝلْ إࣹ كُنْندُلࣹ مٝنْ. الله نٝنْ، نٝ دَالَ گٝونگَ، كٝ كَنْكٝ وٝنِ فࣹوْنٝووٝ إࣹ فٝݖِّينگٝلْ نگٝلْ.

(5) نٝدِّرࣹي ࢠࣹ بَابِرَاࢠࣹ مَࢠّࣹ ࢠࣹنْ: كٝ طُمْ ࢠُرِ نُنْطُدࣹ كَ الله. سِ تَوِي نٝنْ أٝن أَنْندَا بَابِرَاࢠࣹ مَࢠّࣹ ࢠࣹنْ، هَرَيْ كٝ مُسِطْࢠࣹ مٝنْ كَ دِينَ إࣹ هࣹفْتٝدِيࢠࣹ مٝنْ. بَكَّاتُ فَوَاكِ أٝن إࣹ كٝ وٝوپٝتٝنْ (فَايٝنْ) إࣹ مُنْ، كٝنٝ (ٝن بَكّٝندِنَيْ إࣹ) كٝ فَاندٝرِطٝنْ ࢠࣹرْطࣹ مٝنْ طࣹنْ. الله لَاتِكࣹ هَفٝرٝووٝ، هِنّٝتٝوطٝ.

(6) كٝ أَنَّبِيجٝ أٝن ࢠُرِ هَنْندُدࣹ إࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ دِينِ كٝئࣹ مَࢠّࣹ; سُدِّيࢠࣹ مَكّٝ ࢠࣹنْ، كٝ نࣹينِرَاࢠࣹ مَࢠّࣹ. جٝومِرَاࢠࣹ إࣹنْطَمَاجِ ࢠࣹنْ، كٝ يٝگَ مَࢠّࣹ ࢠُرِ هَنْندُدࣹ يٝگَ إࣹ ندٝنْندِ ندِنْ كَ دࣹفْتࣹرࣹ الله، دِينِ گٝمْطِمْࢠࣹ ࢠࣹنْ إࣹ فࣹرُࢠࣹ ࢠࣹنْ، سِ وَنَا هَرَ أٝن وَطَنْ وࣹلْدِيࢠࣹ مٝنْ ࢠࣹ رٝنَتَا ࢠࣹنْ مٝڃّࣹرࣹ. طُمْ وٝنِي كٝ وِنْندَا كَ دࣹفْتࣹرࣹ.

(7) جَنْتٝ تُمَ ندࣹ مࣹنْ ڃࣹتِّدُنٝو إࣹ أَنَّبَاࢠࣹ ࢠࣹنْ أَهَدِ مَࢠّࣹ، إࣹ مَأَ أَن كَدِ، إࣹ نوح، إࣹ إبراهيم، إࣹ موسى إࣹ إِيسَا ࢠِطّٝ مَرْيَمَ أٝنْ: مࣹنْ ڃࣹتِّدِ إࣹ مَࢠّࣹ أَهَدِ تِيطُندِ،

(8) فِي نٝ الله لَمْدٝرَ گٝونگُࢠࣹ ࢠࣹنْ إࣹ گٝونگَ مَࢠّࣹ كَنْ. أٝ هࣹبِلَنِي يࣹدُّࢠࣹ ࢠࣹنْ، لࣹپْتࣹ مُوسُوطࣹ.

(9) كٝ أٝنٝنْ يٝ گٝمْطِمْࢠࣹ ! أَنْندِتࣹي نࣹيمَ الله أٝن إࣹ دٝوْ مٝوطٝنْ، تُمَ ندࣹ كٝنࣹيلِ أَرُنٝو أٝنْ، مࣹنْ وُرْتِ إࣹ مَࢠّࣹ هࣹنْندُ إࣹ كٝنُولِ، نگُ أٝن يِئَالِ. الله سِفٝرِي رࣹنْتُطٝ كٝ گٝلّٝتٝنْ.

(10) تُمَ ندࣹ ࢠࣹ أَرْدِرَنّٝو أٝنْ، كَ سࣹنْنگٝ دٝوْ مٝنْ إࣹ كَ لࣹيْ سَارࣹ مٝنْ، إࣹ تُمَ ندࣹ گِتࣹ أٝوݧِنٝو، ࢠࣹرْطࣹ طࣹنْ هࣹوْتِرِ كَ لٝوكٝڃࣹ، هِطٝنْ جِكَّ الله (ٝنْتُمَ) جِكّࣹ لُوتٝندِرْطࣹ.

(11) كٝ طٝنْ گٝمْطِمْࢠࣹ ࢠࣹنْ نجَرِبَ، ࢠࣹ دِلِّنِرَا دِلّࣹرࣹ سَطْتُندࣹ.

(12) جَنْتٝ كَدِ، تُمَ ندࣹ نَافِقِيࢠࣹ ࢠࣹنْ، إࣹ ࢠࣹنْ ࢠࣹ ݧَوُ وٝنِ إࣹ ࢠࣹرْطࣹ مُنْ وِئَيْنٝو: الله" إࣹ نُلَاطٝ مَكّٝ أٝن فٝدَالِ مࣹنْ، سِ وَنَا هٝدٝ".

(13) إࣹ تُمَ ندࣹ فࣹدّࣹ گٝو إࣹ مَࢠّࣹ وِئُنٝو: "كٝ أٝنٝنْ يٝ يِمْࢠࣹ يَتْهْرِبْ ! ݧِيࢠِرْدࣹ أَلَنَا أٝن (طٝو)، يِلْتِتࣹي كَ مٝنْ ". فࣹدّࣹ گٝو إࣹ مَࢠّࣹ نٝ وَيْنِتٝو أَنَّبِيجٝ أٝنْ، هِࢠࣹ وِئَ : "ݖُوطِ أَمࣹنْ طِنْ كٝ نَ هُنْݖِي"، هَرَ نٝنْ وَنَا طِ هُنْݖِيطِ: ࢠࣹ فَاندَاكِ، سِ وَنَا دٝگُگٝلْ.

(14) هَيْ سِ تَوْنٝ (َيْࢠࣹ) ࢠࣹنْ نَاتِرَنّٝ ࢠࣹ ࢠَنْنگࣹ مَيْرࣹ، رࣹفْتِ ࢠࣹ لَنْندَا فِتِنَ (يࣹدُّگٝلْ)، پࣹلّࣹتْ، ࢠࣹ وٝنَيْنٝ إࣹ مُنْ، ࢠࣹ لࣹيلْنَتَانٝ إࣹ طُمْ، سِ وَنَا سࣹيطَ،

(15) هَرَ لࣹ گٝمْطِي ࢠࣹ أَهٝدِينٝ إࣹ الله كٝ أَدِي، وٝنْندࣹ ࢠࣹ رُنْتٝتَاكٝ ࢠَبّࣹ. أَهَدِ الله ندِنْ نٝنْ، ندِ لَاتِكࣹ لَنْندِتٝرْتࣹيندِ.

(16) مَاكُ: "دٝگُگٝلْ نَفَتَا أٝنْ، سِ أٝن دٝگُ مَايْدࣹ مَا هَرࣹ; هَرَيْ طُمْ، أٝن دَكْمِتࣹتَاكࣹ، سِ وَنَا سࣹيطَ".

(17) مَاكُ: "كٝ هٝمْنبٝ وٝنِ دَطْندٝووٝ أٝن إࣹ الله، سِ أٝ فَاندَنِكࣹ أٝن بٝنࣹ، مَا سِ أٝ فَاندَنِكࣹ أٝن يُرْمࣹيندࣹ?" ࢠࣹ هࣹࢠَنْتَا كٝئࣹ مَࢠّࣹ گَانِنْ الله، گِطٝ وَنَا بَلّٝ.

(18) گٝمْطِي الله نٝ أَنْندِ فَلَــٔٝتٝوࢠࣹ ࢠࣹنْ إࣹ مٝنْ، إࣹ ࢠࣹنْ وِئَنٝوࢠࣹ مُسِطْࢠࣹ مَࢠّࣹ ࢠࣹنْ :"َرࣹي إࣹ أَمࣹنْ"، هَرَ نٝنْ ࢠࣹ أَرَتَا إࣹ هَرࣹ ندࣹنْ، سِ وَنَا سࣹيطَ.

(19) كٝ وُدَّنْ ࢠࣹ أٝنْ. سِ كُلٝلْ نگٝلْ أَرِي، أَ يِئَيْ هِࢠࣹ ندَارࣹ، هَرَ گِتࣹ مَࢠّࣹ نٝ يِيلٝرَ وَ مٝ مَايْدࣹ سُدَّا إࣹ مُنْ. سِ كُلٝ نگٝلْ نٝنْ يࣹهِي، ࢠࣹ لَگّٝرَ طࣹمْطࣹ مُوسُطࣹ، وُدَّنگٝلْ مٝڃّࣹرࣹ ندࣹنْ. ࢠࣹنْ طٝنْ، ࢠࣹ گٝمْطِنَالِ، الله بٝنِّي گٝلّࣹ مَࢠّࣹ طࣹنْ. طُمْ وٝنِي كٝ نࣹوَنِي الله.

(20) هِࢠࣹ سِكَّ كٝنࣹيلِ طِنْ يَهَتَا. هَيْ سِ كٝنࣹيلِ طِنْ أَرْتِي، ࢠࣹي نَافِقِيࢠࣹ يࣹلٝتٝنٝ سِنّٝ هَرَ نُنْ هِࢠࣹ كَ لَدّࣹ هَكُّندࣹ يِمْࢠࣹ لَدّࣹ ࢠࣹنْ، هَرَ هِࢠࣹ لَنْندٝ فِي مٝنْ. هَيْ سِ ࢠࣹ تَوْتٝدَنٝ إࣹ مٝنْ، ࢠࣹ هَࢠَتَانٝ سِ وَنَا سࣹيطَ نِي.

(21) گٝمْطِي لَاتَنِكࣹ أٝن إࣹ نُلَاطٝ الله أٝنْ، ݧࣹمْتِنَلْ مٝڃَّلْ، وٝنَنْندࣹ أٝن جٝرْتِيطٝ الله، إࣹ ݧَلَاندࣹ سَكِّتِيندࣹ ندࣹنْ، أٝ جَنْتِي الله كٝ طُوطِ.

(22) ندࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ يِئُنٝو كٝنࣹيلِ طِنْ، ࢠࣹ وِئِ: "كٝ طُمْ وٝنِ كٝ الله إࣹ نُلَاطٝ مَكّٝ أٝن فٝدِ مࣹنْ; هَرَيْ الله إࣹ نُلَاطٝ مَكّٝ أٝن گٝونگِي". طُمْ ࢠࣹيْدَالِ ࢠࣹ، سِ وَنَا گٝمْطِنَلْ إࣹ جࣹبِّلَارࣹ.

(23) هِنٝ إࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ، وٝرْࢠࣹ هُنُّࢠࣹ كٝ ࢠࣹ أَهٝدِ إࣹ الله كٝنْ : هِنٝ إࣹ مَࢠّࣹ تِمِّنْطٝ تَكّࣹ مُنْ طࣹنْ (هَا أٝ سَهٝدِنَا)، هِنٝ إࣹ مَࢠّࣹ كَدِ هَࢠِّتِيطٝ; ࢠࣹ وَطْتِتَالِ (َهَدِ ندِنْ) نبَتِّيگُ.

(24) فِي يٝ الله يٝࢠِرْ هُنُّࢠࣹ ࢠࣹنْ سَبُ گٝونگُدࣹ مَࢠّࣹ، أٝ لࣹپْتَ نَافِقِيࢠࣹ ࢠࣹنْ، سِ مٝ يِطِ، مَا أٝ جَࢠَنَ ࢠࣹ تُوبُوبُيࣹي. تَوْدࣹ پࣹلّࣹتْ، الله لَاتِكࣹ هَفٝرٝووٝ، هِنّٝتٝوطٝ.

(25) الله رُتِّدِي يࣹدُّࢠࣹ ࢠࣹنْ إࣹ تِكّࣹرࣹ مَࢠّࣹ ندࣹنْ، ࢠࣹ هࣹࢠَالِ مٝڃّࣹرࣹ. الله يٝنِي گٝمْطِمْࢠࣹ ࢠࣹنْ إࣹ هَرࣹ. الله سِفٝرِي ݖࣹمْنبࣹ إࣹ پٝولَلْ.

(26) أٝ جِپِينِ ࢠࣹنْ وَلِّندِرْنٝوࢠࣹ جࣹيَاࢠࣹ إࣹ يِمْࢠࣹ دࣹفْتࣹ ࢠࣹنْ، كَ تَتَاجِ مَࢠّࣹ طِنْ، أٝ وࣹدِّي طࣹنْڃࣹرࣹ إࣹ ࢠࣹرْطࣹ مَࢠّࣹ; فࣹدّࣹ گٝو (ࣹ مَࢠّࣹ) هِطٝنْ وَرَ، هِطٝنْ دَهَ كَدِ فࣹدࣹ گٝو.

(27) أٝ رٝنِنِ أٝن لࣹيْدِ مَࢠّࣹ، إࣹ گَلّࣹيجِ مَࢠّࣹ، إࣹ جَوْلࣹ مَࢠّࣹ، إࣹ لࣹيْدِ ندِ أٝن تِپَّانٝ إࣹ مُنْ. الله سِفٝرِي هَتَّنْطٝ كَلَ هُوندࣹ.

(28) كٝ أَن يٝ أَنَّبِيجٝ ! مَاكَنْ سُدِّيࢠࣹ مَا ࢠࣹنْ: "سِ وٝنِي كٝ نگُورْندَمْ أَدُنَ فَالَطٝنْ إࣹ ݖُطَارِ مَجَّمْ، هَرَيْ أَرࣹي مِي دَكْمِتَ أٝنْ، مِي سࣹيرَ أٝن أٝنْتُمَ ݖࣹرْگَلْ مٝڃَّلْ.

(29) سِ تَوِي نٝنْ كٝ الله فَالَطٝنْ، إࣹ نُلَاطٝ مَكّٝ أٝنْ، إࣹ گَلّࣹ لَاكَرَ طࣹنْ، هَرَيْ پࣹلّࣹتْ، الله مَرَنِي مٝڃِّمْࢠࣹ ࢠࣹنْ إࣹ مٝوطٝنْ، نجٝࢠْدِ مَوْندِ".

(30) كٝ أٝنٝنْ يٝ سُدِّيࢠࣹ أَنَّبِيجٝ أٝن ! كَلَ أَدُّطٝ إࣹ مٝوطٝنْ بٝنࣹ ࢠَنْنگُطٝ، لࣹپْتࣹ طࣹنْ سٝوِرَنْتࣹ مٝ ݖٝوَلّࣹ طِطِ! طُمْ وٝنِي كٝ نࣹوَنِي الله.

(31) كَلَ إࣹ مٝوطٝنْ طٝفْتِيطٝ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ، أٝ گٝلِّ كٝ مٝڃِّ، مࣹنْ يٝࢠِرَيْ مٝ نجٝࢠْدِ مَكّٝ ندِنْ لَاࢠِ طِطِ، مࣹطࣹنْ مَرَنِ مٝ كَدِ، أَرْسِكࣹ تࣹدُّطٝ.

(32) كٝ أٝنٝنْ يٝ سُدِّيࢠࣹ أَنَّبِيجٝ! أٝن وَأَا وَنٝ گٝوتٝ إࣹ سُدِّيࢠࣹ ࢠࣹنْ. سِ أٝن گٝمْطِي، هَرَيْ وَتَ أٝن يࣹرْگِنْ كٝنْنگٝلْ، سَكّٝ أٝن مٝ ݧَوُ تُويٝ وٝنِ إࣹ ࢠࣹرْندࣹ مُنْ تَمْڃِنٝو. وٝوْلٝنْ كٝنْنگٝلْ مٝڃّٝلْ.

(33) ݧِيࢠࣹي كَ ݖُوطِ مٝنْ; وَتَ أٝن فࣹݧِّنِرْ ݖُطَارِ مٝنْ، وَنٝ وِرْتَارࣹ مَجّࣹرࣹ أَدِنٝوࢠࣹ. ݧِيࢠْنࣹي جُلْدࣹ ندࣹنْ، تٝتّٝنْ أَسْكَلْ أٝنْ، طٝفْتٝطٝنْ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ. أَنْندࣹي كٝ الله فَاندِي، كٝ إِتَّنْتَ أٝن إࣹ تُوطࣹ، أٝنٝنْ يِمْࢠࣹ سُودُ أَنَّبِيجٝ ، أٝ لَࢠِّنِرَ أٝن لَࢠِّنگٝلْ.

(34) جَنْتࣹي كٝ وٝنَا إࣹ جَنْنگࣹيدࣹ كَ ݖُوطِ مٝنْ، إِمّٝرْدࣹ إࣹ آيࣹيجࣹ الله، إࣹ سُنَّ. پࣹلّࣹتْ، الله لَاتِكࣹ نࣹوِيطٝ، هُمْپِتِيطٝ.

(35) پࣹلّࣹتْ، جُولْࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ جُولْࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ گٝمْطِمْࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ گٝمْطِمْࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ طٝفْتِيࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ طٝفْتِيࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ گٝونگُࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ گٝونگُࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ مُݧِّيࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ مُݧِّيࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ يَنْكِنِيࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ يَنْكِنِيࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ سَكّٝتٝوࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ سَكّٝتٝوࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ هٝورٝوࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ هٝورٝوࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ وٝرْࢠࣹ رࣹينٝوࢠࣹ فَرْجُوجِ مُنْ إࣹ رࣹوْࢠࣹ رࣹينٝوࢠࣹ فَرْجُوجِ مُنْ، إࣹ وٝرْࢠࣹ جَنْتٝتٝوࢠࣹ الله كٝ طُوطِ ࢠࣹنْ إࣹ رࣹوْࢠࣹ جَنْتٝتٝوࢠࣹ ࢠࣹنْ: الله مَرَنِي ࢠࣹنْ، هَفٝرَنَلْ إࣹ نجٝࢠْدِ مَوْندِ.

(36) أَلَنَا گٝمْطِطٝ گٝرْكٝ، وَنَا گٝمْطِنْطٝ دࣹبّٝ، سِ الله إࣹ نُلَاطٝ مَكّٝ أٝن ݧَاوِي فِيَاكُ، ندࣹ سُࢠَگٝلْ إࣹ فِيَاكُ مَࢠّࣹ أٝنْ. كَلَ نٝنْ يࣹدُّطٝ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ، هَرَيْ أٝ مَجِّي مَجّࣹرࣹ ࢠَنْنگُندࣹ.

(37) جَنْتٝ تُمَ مَاكَنَيْنٝ-طَا أٝن مٝ الله نࣹيمِنِ إࣹ مُنْ، مٝڃُّ-طَا إࣹ مَكّٝ: "جٝگِتَنٝ سُدِّيطٝ مَا أٝنْ، هٝورࣹ-مَا، هُلَا الله"، هِطَ سُوطَ إࣹ فٝنْندٝ مَا، كٝ الله فࣹݧِّنْتَ. هِطَ هُلَ يِمْࢠࣹ ࢠࣹنْ، كٝنٝ كٝ الله فٝتُ-طَا هُلُدࣹ. تُمَ زَيْدُ هُنْتُنٝو إࣹ (سُدِّيطٝ) أٝن هَاجُ، مࣹنْ رࣹسِنْ-مَا مٝ، فِي وَتَ ࢠِتّࣹرࣹ وٝنُ إࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ، إࣹ (رࣹسُگٝلْ) سُدِّيࢠࣹ نٝدِّتِرَاࢠࣹ مَࢠّࣹ ࢠࣹنْ، سِ ࢠࣹنْ هُنْتِي إࣹ مَࢠّࣹ هَاجُ. يَمِرٝورࣹ الله ندࣹنْ لَاتِكࣹ گٝلّࣹتࣹيندࣹ.

(38) فࣹلٝورࣹ أَلَنَا أَنَّبِيجٝ أٝنْ، إࣹ كٝنْ كٝ الله دَگِنَــِٔ مٝ، طُمْ كٝ ݧَاوٝورࣹ الله إࣹ ࢠࣹنْ فࣹڃُّࢠࣹ أَدٝ مَكّٝ. ݧَاوٝورࣹ الله ندࣹ وٝنِي إࣹࢠَّاندࣹ هٝدِّرَاندࣹ.

(39) ࢠࣹنْ يٝتِّينٝوࢠࣹ نُلࣹ الله طࣹنْ، ࢠࣹ نٝ هُلَ مٝ، ࢠࣹ هُلَ گٝطّٝ گٝو سِ وَنَا الله. الله يٝنِي هَسْبٝووٝ.

(40) مُهَمَّدُ وٝنَالِ بَابَ هَيْ گٝوتْ إࣹ وٝرْࢠࣹ مٝنْ، كٝ أٝ وٝنِ، كٝ نُلَاطٝ الله إࣹ تِمّٝودࣹ أَنَّبَاࢠࣹ. الله لَاتِكࣹ أَنْندُطٝ كَلَ هُوندࣹ.

(41) كٝ أٝنٝنْ يٝ ࢠࣹنْ گٝمْطِمْࢠࣹ، جَنْتٝرࣹي الله جَنْتٝورࣹ طُوطُندࣹ،

(42) سُبْهِنࣹي مٝ بِمْنبِ إࣹ كِيكِيطࣹ.

(43) كٝ كَنْكٝ وٝنِ يُرْمࣹتࣹيطٝ أٝنْ، -ࣹ الملائكة مَكّٝ ࢠࣹنْ (دُؤٝرَنَ أٝنْ)، - فِي نٝ أٝ يَلْتِنِرَ أٝن إࣹ نِࢠࣹ فَادࣹ إࣹ أَنّٝورَ. أٝ سِفٝرِي هِنّٝتٝوطٝ گٝمْطِمْࢠࣹ ࢠࣹنْ.

(44) هِوْرٝندِرَلْ مَࢠّࣹ نگَلْ ݧَلْندࣹ ࢠࣹ هَوْرَتَ إࣹ مَكّٝ، كٝ "سَلَامْ" كِسِيࣹي ، أٝ مَرَنِي ࢠࣹ نبَرْجَارِ تࣹدُّندِ.

(45) كٝ أَن يٝ أَنَّبِيجٝ ! مࣹنْ نُلِرِي مَ فِي يٝ وٝنُ سࣹيدِيطٝ، إࣹ وࣹوْلِنٝووٝ إࣹ جࣹرْتِنٝووٝ،

(46) إࣹ نٝدّٝووٝ فَادࣹ إࣹ الله، دُݝَيࣹي مَكّٝ; إࣹ پِتِلٝلْ جَلْࢠُنگٝلْ.

(47) وࣹوْلِنِرْ گٝمْطِمْࢠࣹ ࢠࣹنْ، وٝنْندࣹ نٝ وٝودَنِ ࢠࣹ كَ الله، ࢠُرَلْ مَوْنُنگَلْ.

(48) وَتَ أَ طٝفْتٝ هࣹيفࣹرْࢠࣹ ࢠࣹنْ إࣹ نَافِقِيࢠࣹ ࢠࣹنْ، طُورْنٝ (ࣹ) لٝرَّ مَࢠّࣹ كَنْ، فَوٝ-طَا إࣹ الله. سَبُ الله يٝنِي كَ فَوࣹي.

(49) كٝ أٝنٝنْ يٝ ࢠࣹنْ گٝمْطِمْࢠࣹ ! سِ رࣹسِي رࣹوْࢠࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ، رࣹفْتِ أٝن سࣹيرِي ࢠࣹ أَدٝ أٝن مࣹيمٝندِرْدࣹ إࣹ مَࢠّࣹ، هَرَيْ أَلَنَا أٝن إࣹ مَࢠّࣹ، إࣹدَّ مٝ ࢠࣹ إࣹدِّنٝتٝو. دَكْمِتِرࣹي ࢠࣹ نجٝوࢠَ ، سࣹيرِرٝنْ ࢠࣹ ݖࣹرْگَلْ لَࢠَانگَلْ.

(50) كٝ أَن يٝ أَنَّبِيجٝ ! مࣹنْ دَگِنَنِي مَ سُدِّيࢠࣹ مَا ࢠࣹنْ ࢠࣹ جٝنُّ-طَا تࣹݝࣹ مُنْ ࢠࣹنْ، إࣹ ࢠࣹنْ هٝرْࢠࣹ ࢠࣹ جࣹيْرُ-طَا ݧَامٝ مَا، إࣹ ࢠࣹنْ دَهَاࢠࣹ ࢠࣹ الله نࣹلْدُ-مَا، إࣹ جِوْࢠࣹ بَپَّ مَا، إࣹ جِوْࢠࣹ يَايِرَاࢠࣹ مَا، إࣹ جِوْࢠࣹ كَاوُ-مَا، إࣹ جِوْࢠࣹ نࣹينِرَاࢠࣹ مَا، - ࢠࣹنْ فࣹرِدُنٝوࢠࣹ إࣹ مَا، - إࣹ سُدِّيطٝ گٝمْطِنْطٝ، سِ أٝ يࣹطِي هٝورࣹ-مَكّٝ أَنَّبِيجٝ أٝنْ، سِ أَنَّبِيجٝ أٝن كَدِ فَاندِكࣹ رࣹسُگٝلْ مٝ : طُمْ كٝ كٝ هࣹيرَنِ مَا، گَانِنْ گٝمْطِمْࢠࣹ ࢠࣹنْ. گٝمْطِي مࣹطࣹنْ أَنْندِ كٝ مࣹنْ فَرْلِ إࣹ مَࢠّࣹ كٝنْ، إࣹ فِي سُدِّيࢠࣹ مَࢠّࣹ ࢠࣹنْ، إࣹ كٝ ࢠࣹ جࣹيْتِرِ ݧَامࣹ مَࢠّࣹ طࣹنْ، فِي وَتَ ࢠِطْتࣹيندࣹ وٝنُ إࣹ مَاطَ. الله لَاتِكࣹ هَفٝرٝووٝ، هِنّٝتٝوطٝ.

(51) هِطَ نࣹنِّنْتِنَ أٝن مٝ يِطُ-طَا (نࣹنِّنْتِنْندࣹ) إࣹ مَࢠّࣹ، ࢠَطْتِنَا إࣹ مَاطَ أٝن مٝ يِطُ-طَا. إࣹ أٝن مٝ طَࢠُّ-طَا (بࣹلَگٝلْ كَ مُنْ) إࣹ ࢠࣹنْ ࢠࣹ نࣹنِّتُ-طَا، بَكَّاكُ وٝو فَوَاكِ مَ. كٝ طُمْ ࢠُرِ ࢠَدَادࣹ إࣹ كٝ ࢠُࢠِّنْتَ ࢠࣹرْطࣹ مَࢠّࣹ، هَرَ ࢠࣹ سُنٝتَاكٝ، ࢠࣹ وࣹلْتٝرَ كَدِ كٝ أٝكُّ-طَا ࢠࣹ كٝنْ، ࢠࣹ فٝوْ مَࢠّࣹ. الله نٝ أَنْندِ كٝ وٝنِ إࣹ ࢠࣹرْطࣹ مٝنْ. الله لَاتِكࣹ أَنْندُطٝ، مُݧِّيطٝ.

(52) رࣹوْࢠࣹ ࢠࣹنْ دَگِتَنْتَاكٝ مَ ࢠَاوٝ ࢠࣹي طٝو ، وَنَا ندࣹ وَطْتِتِرْتَا ࢠࣹ گࣹنْندِرَاࢠࣹ گٝو، هَيْ سِ ݝَرِ ࢠࣹنْ طٝنْ هَاوْنِي مَ; سِ وَنَا ࢠࣹ جࣹيْرُ-طَا ݧَامٝ مَا ࢠࣹنْ. الله لَاتِكࣹ تٝمْنبِيطٝ كَلَ هُوندࣹ.

(53) كٝ أٝنٝنْ يٝ ࢠࣹنْ گٝمْطِمْࢠࣹ ! وَتَ أٝن نَاتُ كَ ݖُوطِ أَنَّبِيجٝ أٝنْ، سِ وَنَا هَرَ أٝن نٝدَّ إࣹ ݧَامࣹتࣹي، هَرَ وَنَا يٝ أٝن هَبِّتٝ كٝ ࢠࣹنْندِ كَ مَكّٝ. كٝنٝ سِ أٝن نٝدَّا مَ، نَاتࣹي. سِ أٝن ݧَامِي، سࣹنْتࣹي، هَرَ أٝن طَࢠَّالِ هَالَ. أَنْندٝنْ طُمْ مٝوطٝنْ، هِنٝ لٝرَّيْنٝو أَنَّبِيجٝ أٝنْ، كٝنٝ أٝ هࣹرْسَ (هَالَنْندࣹ) أٝنْ، الله نٝنْ طَلَتَا يَمِرْدࣹ گٝونگَ. سِ أٝن لَنْندٝتٝ (سُدِّيࢠࣹ مَكّٝ) ࢠࣹنْ هَاجُ، لَنْندٝرࣹي ࢠࣹ ࢠَاوٝ وِرْنگَلّٝ: كٝ طُمْ مٝوطٝنْ، ࢠُرِ لَاࢠَنْندࣹ ࢠࣹرْطࣹ مٝنْ إࣹ ࢠࣹرْطࣹ مَࢠّࣹ. هَانَنَا أٝن ندࣹ لٝرّٝتٝنْ نُلَاطٝ الله أٝنْ، وَنَا ندࣹ رࣹسٝتٝنْ سُدِّيࢠࣹ مَࢠّࣹ ࢠࣹنْ ࢠَاوٝ مَكّٝ، پٝومَا; پࣹلّࣹتْ، وَطُگٝلْ مٝنْ طُمْ، وٝنِي كٝ مَوْنِ كَ الله.

(54) سِ أٝن فࣹݧِّنِي هُوندࣹ، مَا أٝن سُوطِي ندࣹ، پࣹلّࣹتْ، الله لَاتِكࣹ أَنْندُطٝ كَلَ هُوندࣹ.

(55) بَكَّاتُ فَوَاكِ ࢠࣹ إࣹ بَابِرَاࢠࣹ مَࢠّࣹ ࢠࣹنْ، وَنَا كَدِ ࢠِطْࢠࣹ مَࢠّࣹ ࢠࣹنْ، وَنَا بَنْندِرَاࢠࣹ مَࢠّࣹ ࢠࣹنْ، وَنَا إࣹ ࢠِطْࢠࣹ بَنْندِرَاࢠࣹ مَࢠّࣹ ࢠࣹنْ، وَنَا إࣹ ࢠِطْࢠࣹ مِݧِّرَاࢠࣹ مَࢠّࣹ رࣹوْࢠࣹ ࢠࣹنْ، وَنَا إࣹ رࣹوْࢠࣹ مَࢠّࣹ ࢠࣹنْ، وَنَا إࣹ ࢠࣹنْ ࢠࣹ ࢠࣹ جࣹيِرِ ݧَامࣹ مَࢠّࣹ. هُلࣹي الله (يٝ سُدِّيࢠࣹ). پࣹلّࣹتْ، الله سِفٝرِي سࣹيدِيطٝ كَلَ هُوندࣹ.

(56) پࣹلّࣹتْ، الله إࣹ الملائكة مَكّٝ ࢠࣹنْ، هِࢠࣹ دُؤَنَادࣹ أَنَّبِيجٝ أٝنْ; كٝ أٝنٝنْ يٝ ࢠࣹنْ گٝمْطِمْࢠࣹ، جُولࣹي إࣹ مَكّٝ، تٝرَنٝطٝنْ مٝ كِسِيࣹي.

(57) پࣹلّࣹتْ، ࢠࣹنْ لٝرّٝوࢠࣹ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ، الله هُطِي ࢠࣹنْ أَدُنَ إࣹ لَاكَرَ، أٝ هࣹبِلَنِي ࢠࣹ كَدِ لࣹپْتࣹ هٝيْنٝوجࣹ.

(58) ࢠࣹنْ كَدِ لٝرّٝوࢠࣹ گٝمْطِمْࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ گٝمْطِمْࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ هٝورࣹ كٝ ࢠࣹ فَگِّتَاكِ، گٝمْطِي ࢠࣹ رٝنْندِكࣹ فࣹنَاندࣹ إࣹ بَكَّاتُ ࢠَنْنگُطٝ.

(59) كٝ أَن يٝ أَنَّبِيجٝ! مَاكَنْ سُدِّيࢠࣹ مَا ࢠࣹنْ، إࣹ جِوْࢠࣹ مَا ࢠࣹنْ، إࣹ گࣹنْندِرَاࢠࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ، ندࣹ ࢠࣹ جٝلْتِنْتَ إࣹ مَࢠّࣹ، وُتّࣹيجِ مَࢠّࣹ طِنْ: كٝ طُمْ ࢠُرَتَ ࢠَطْتَادࣹ، ندࣹ ࢠࣹ أَنْندِتࣹتࣹي، هَرَ ࢠࣹ تٝرَّاكَ. الله لَاتِكࣹ هَفٝرٝووٝ، هِنّٝتٝوطٝ.

(60) پࣹلّࣹتْ، سِ نَافِقِيࢠࣹ ࢠࣹنْ هَطِتَاكِ، إࣹ ࢠࣹنْ ࢠࣹ ݧَوُ (سِكِّتَارࣹ) وٝنِ إࣹ ࢠࣹرْطࣹ مُنْ، إࣹ فࣹنَ-فࣹنْتَئࣹنْ كَ سَارࣹ مَدِينَ، مَ مࣹنْ وُرْتِتࣹ إࣹ مَࢠّࣹ، رࣹفِ ࢠࣹ كَوْتِرْتَا إࣹ مَاطَ تٝنْ، سِ وَنَا سࣹيطَ.

(61) كٝ هُطَا ࢠࣹ. كَلَ كَ ࢠࣹ تَوَا، كٝ ࢠࣹ نَنْنگࣹتࣹي ࢠࣹ، ࢠࣹ وَرࣹي وَرْگٝلْ:

(62) طُمْ كٝ سُنَّ الله إࣹ ࢠࣹنْ وُلِنٝوࢠࣹ كٝ أَدِي. أَ هࣹࢠَنْتَا لَاوٝلْ (سُنَّ) الله نگٝلْ، بَتِّيگُ.

(63) يِمْࢠࣹ ࢠࣹنْ أٝ لَنْندٝ مَا فِي دَرْنگَلْ. مَاكُ : "َنْندࣹي گَنْندَلْ دَرَگٝلْ مَگَّلْ، كٝ كَ الله وٝنِ". إࣹ كٝ أَنْندِنْ مَا، وٝنْندࣹ نٝ هَسِي هَرَ دَرْنگَلْ نگَلْ ࢠَدِكࣹ?

(64) پࣹلّࣹتْ، الله هُطِي هࣹيفࣹرْࢠࣹ ࢠࣹنْ، أٝ كَمْنبِرَنِي ࢠࣹ كَدِ دُلْبٝووࣹ،

(65) كٝ ࢠࣹ لُتّٝوࢠࣹ تٝنْ پٝومَا. ࢠࣹ هࣹࢠَتَا گِطٝ، وَنَا بَلّٝ.

(66) ݧَلْندࣹ گࣹيݖࣹ مَࢠّࣹ طࣹنْ وَيْلِتࣹتࣹي كَ يِيتࣹ، ࢠࣹ وِئَيْ: "ࣹي جَلَّ أَمࣹنْ! هَرَ مࣹنْ طٝفْتِنٝ الله، مࣹنْ طٝفْتِي نُلَاطٝ أٝنْ!".

(67) ࢠࣹ وِئَ: جٝومِ" أَمࣹنْ، مࣹنࣹنْ مࣹنْ طٝفْتِ هٝورࣹيࢠࣹ أَمࣹنْ إࣹ مَوْࢠࣹ أَمࣹنْ، ࢠࣹ مَجِّنِ مࣹنْ إࣹ لَاوٝلْ نگٝلْ.

(68) جٝومِ أَمࣹنْ، تٝتُّ ࢠࣹ ݖٝوَنْلّࣹ طِطِ إࣹ لࣹپْتࣹ طࣹنْ، هُطَا ࢠࣹ كُدِّ مَوْندِ".

(69) كٝ أٝنٝنْ يٝ ࢠࣹنْ گٝمْطِمْࢠࣹ! وَتَ أٝن وَؤُ وَ ࢠࣹنْ لٝرُّنٝوࢠࣹ موسى، الله وٝطِّنِ مٝ إࣹ كٝ ࢠࣹ وِئِ كٝنْ، تَوْدࣹ أٝ وٝنِي تࣹدُّطٝ كَ الله.

(70) كٝ أٝن يٝ ࢠࣹنْ گٝمْطِمْࢠࣹ! هُلࣹي الله، وٝوْلٝنْ كٝنْنگٝلْ فࣹيوُنگٝلْ.

(71) سِ أٝن وَطِي طُمْ ، أٝ مٝڃِّنَنَيْ أٝن گٝلّࣹ مٝنْ، أٝ هَفٝرَنَ أٝن جُنُوبِ مٝنْ. كَلَ طٝفْتِيطٝ الله إࣹ نُلَاطٝ مَكّٝ أٝنْ، هَرَيْ أٝ مَلِرَامَ مَلٝورࣹ مَوْندࣹ.

(72) مࣹنْ وࣹيࢠِتِي هٝولَارࣹ ندࣹنْ إࣹ دٝوْ كَمُّولِ طِنْ، إࣹ لࣹيْدِ ندِنْ إࣹ پࣹلّࣹ طࣹنْ، طِ هَطِتِي رٝنْندَگٝلْ ندࣹ، طِ هُلِ (بَتَّنࣹ) مَجِّ، نࣹطّٝ أٝن كَنْ رٝنْندِي ندࣹ; تَوْدࣹ پࣹلّࣹتْ، أٝ وٝنِي طُوطُطٝ-تٝوݧُدࣹ هٝورࣹ-مُنْ ، مَجَّاطٝ بَتَّنࣹ طُمْ .

(73) نࣹطّٝ أٝن رٝنْندِرَامَ طُمْ ، نٝ الله لࣹپْتِرَ نَافِقِيࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ نَافِقِيࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ سِرْكٝوࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ سِرْكٝوࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ، إࣹ (نٝ) الله جَࢠِرَنَ گٝمْطِمْࢠࣹ وٝرْࢠࣹ ࢠࣹنْ إࣹ گٝمْطِمْࢠࣹ رࣹوْࢠࣹ ࢠࣹنْ تُوبُبُيࣹي. الله لَاتِكࣹ هَفٝرٝووٝ، هِنّٝتٝوطٝ.

34 Saba' — 54 آيَة

(1) يࣹتّٝورࣹ وٝودَنِي الله، أٝنْ جࣹيْطٝ كَلَ كٝ وٝنِ كَ كَمُّولِ إࣹ كَ لࣹيْدِ. هِنٝ وٝودَنِ مٝ كَدِ يࣹتّٝورࣹ كَ لَاكَرَ. كٝ كَنْكٝ وٝنِ ݧࣹيݧٝ، هُمْپِتِيطٝ.

(2) هِمٝ أَنْندِ كٝ نَاتَتَ كَ لࣹيْدِ، إࣹ كٝ يَلْتَتَ إࣹ مَيْرِ، كٝ جِپّٝتٝو إِوْرَ كَ كَمُّ إࣹ كٝ ڃَوَتَ إࣹ مَگُّ. كٝ كَنْكٝ وٝنِ هِنّٝتٝو، هَفٝرٝووٝ.

(3) يࣹدُّࢠࣹ ࢠࣹنْ وِئِ: "دَرْنگَلْ نگَلْ أَرَتَا مࣹنْ". مَاكُ: "آءَ! مِي وٝوندِرِي جٝومِ أَمْ! پࣹلّࣹتْ، كٝ أَرَيْ إࣹ مٝنْ. جٝومِ مَ كٝ أَنْندُطٝ وِرْنِيطِ. هُوندࣹ وِرْنَنَاكِ مٝ، كٝ فٝتَتَ إࣹ أَبّࣹرࣹ جَرَّ، وٝنِ كَ كَمُّولِ مَا كَ لࣹيْدِ. وَنَا كٝ ࢠُرِ طُمْ فَنْطُدࣹ، وَنَا كٝ ࢠُرِ طُمْ مَوْنُدࣹ، سِ وَنَا نٝ إࣹ دࣹفْتࣹرࣹ فࣹݧِنْندࣹ.

(4) فِي يٝ أٝ يٝࢠٝيْ ࢠࣹنْ گٝمْطِمْࢠࣹ، ࢠࣹ گٝلِّ مٝڃُّطِ. ࢠࣹنْ طٝنْ، هِنٝ وٝودَنِ ࢠࣹ هَفٝرَنࣹيدࣹ، إࣹ أَرْسِكࣹ تࣹدُّطٝ.

(5) ࢠࣹنْ نٝنْ يَݖِّي ࢠࣹ إࣹ بٝنِّتُگٝلْ آيࣹيجࣹ أَمࣹنْ طࣹنْ، ࢠࣹنْ طٝنْ، هِنٝ وٝودَنِ ࢠࣹ لࣹپْتࣹ، إِمّٝرْدࣹ إࣹ ࢠُرْطࣹ مُوسُدࣹ.

(6) ࢠࣹنْ أٝكَّاࢠࣹ گَنْندَلْ هِنٝ يِئِرَ جِپِّنِرَاندࣹ ندࣹنْ إࣹ مَاطَ، إِمّٝرْدࣹ كَ جٝومِ مَا، كٝ كَيْرࣹ وٝنِ گٝونگَ، هِندࣹ فࣹوْنِرَ إࣹ لَاوٝلْ فٝولُطٝ، يࣹتِّنِيطٝ أٝنْ.

(7) يࣹدُّࢠࣹ ࢠࣹنْ كَدِ وِئِ: "أٝنْ أَݖَّيْ مࣹنْ تِنْندِنَ أٝنْ فِي گٝرْكٝ، مٝ نٝ هُمْپِتَ أٝنْ، وٝنْندࣹ سِ أٝنْ مُنْݖِتَامَ كَلَ مُنْݖِتَنْندࣹ، أٝنْ وٝنْتٝيَيْ إࣹ تَگُ هࣹسُ?

(8) أٝ فࣹفِندِ فࣹنَاندࣹ إࣹ هٝورࣹ الله? كَا هَرَ كٝ جِنَّ نَنْنگِ مٝ? سِ كٝ وٝنِ، ࢠࣹنْ ࢠࣹ گٝمْطِنَا لَاكَرَ أٝنْ، ࢠࣹ وٝنِي إࣹ لࣹپْتࣹ إࣹ مَجّࣹرࣹ وٝطِّيندࣹ (گٝونگَ).

(9) إࣹ ࢠࣹي يِئَالِ كٝ وٝنِ يࣹيسٝ مَاࢠࣹ إࣹ كٝ وٝنِ ࢠَاوٝ مَࢠّࣹ إِمّٝرْدࣹ كَ كَمُّ إࣹ كَ لࣹيْدِ? سِ مࣹنْ مُويِينٝ، مࣹنْ مُنِّرَيْنٝ ࢠࣹ لࣹيْدِ ندِنْ، مَا مࣹنْ فُكِّنَ إࣹ مَࢠّࣹ تَڃٝنْتِنࣹ (لࣹپْتࣹ)، إِمّٝرْدࣹ كَ كَمُّ. وٝنِي إࣹ طُمْ، مَاندࣹ، وٝنَنْندࣹ كَلَ جࣹيَاطٝ رُتّٝتٝوطٝ (إࣹ جٝومِ مُنْ).

(10) گٝمْطِي مࣹنْ أٝكِّي دَاوُودَ ࢠُرَلْ، إِمّٝرْدࣹ إࣹ أَمࣹنْ. مࣹنْ وِئِ : "كٝ أٝنٝنْ پࣹلّࣹ إࣹ ݖٝلِّ، سُبُّنْهِنٝدࣹي إࣹ مَكّٝ. مࣹنْ نࣹوِنَنِ مٝ (تَفُگٝلْ) نجَمْندِ ندِنْ.

(11) (مࣹنْ وِئَنِ مٝ وٝنْندࣹ): "گٝلُّ وُتّࣹيجِ نجَمْندِ، إࣹࢠَّا كَ تَفٝودِ. گٝلّٝنْ گٝلّࣹ مٝڃّࣹ، پࣹلّࣹتْ، كٝ مِي رࣹينُطٝ كٝ گٝلّٝتٝنْ كٝنْ".

(12) (مࣹنْ إࣹلْتَنِ كَدِ) سُلَيْمَانَ هࣹنْندُ ندُنْ: دَوٝلْ مَيْرُ كٝ يࣹرُ (يَادُ) لࣹوْرُ، كِرْندٝلْ مَيْرُ كَدِ كٝ يࣹرُ (يَادُ) لࣹوُ. مࣹنْ هٝنْݖِنَنِ مٝ ࢠُنْندُ سِلَ (كِرࣹ). هِنٝ جِنَّاجِ طِنْ، گٝلِّرَيْنٝوطِ يࣹيسٝ مَكّٝ، إࣹ دُݝَيࣹي جٝومِ مَكّٝ. كَلَ أٝوݧِينگِي إࣹ مَجِّ، گَايْ إࣹ يَمِرٝورࣹ أَمࣹنْ ندࣹنْ، مࣹنْ مࣹيطِنَ نگِي لࣹپْتࣹ دُلْࢠٝووࣹ نگࣹنْ.

(13) هِطِ گٝلَّنَ مٝ كٝ أٝ فَالَا: گِلَ إࣹ تَتَاجِ، إࣹ نَنْندٝلَّاجِ، إࣹ لٝوطࣹ وَؤُطࣹ وَ بࣹيلِ، إࣹ پَيَنْطࣹ فࣹݝِيطࣹ. "گٝلّࣹي، أٝنٝنْ ࢠࣹيْنگُورࣹ دَاوُودَ، يࣹتّٝنْ (الله)"، فَنْطِي إࣹ جࣹيَاࢠࣹ أَمْ ࢠࣹنْ، يࣹتّٝووٝ (لَمْ).

(14) ندࣹ مࣹنْ ݧَاوُنٝو إࣹ مَكّٝ مَايْدࣹ ندࣹنْ، أَنْندِنَالِ طِ فِي مَايْدࣹ مَكّٝ ندࣹنْ، سِ وَنَا گَنْنگُوجِ لࣹيْدِ(مٝوڃُ) طِنْ، هِنٝ ݧَامَ تُگّٝرْدُ مَكّٝ ندُنْ. ندࣹ أٝ فُكِّنٝو، فࣹيݧَنِ جِنَّاجِ طِنْ، وٝنْندࣹ سِ طِ أَنْندُنٝو كٝ وِرْنِي، طِ ݧِيࢠَتَانٝ إࣹ لࣹپْتࣹ جࣹيَاگُ هٝيْنٝوجࣹ.

(15) گٝمْطِي وٝنَنِينٝ يِمْࢠࣹ سَبَئِ ࢠࣹنْ إࣹ كٝطٝولِ مَࢠّࣹ طِنْ، مَاندࣹ; گࣹسࣹ طِطِ كَ بَنْنگࣹ ݧَامٝ إࣹ كَ نَنٝ. (مࣹنْ وِئَنِ ࢠࣹ): "ݧَامࣹي إࣹ أَرْسِكࣹ جٝومِ مٝنْ، يࣹتّٝنْ مٝ. لࣹيْدِ ندِنْ كٝ لَاࢠُندِ، إࣹ جٝومِرَاوٝ هَفٝرٝووٝ".

(16) (كٝنٝ) ࢠࣹ طُونِي. مࣹنْ وُرْتِ إࣹ مَࢠّࣹ ندِيَمْ وَامࣹ، مࣹنْ وَطْتِتَنِ ࢠࣹ گࣹسࣹ مَࢠّࣹ طࣹنْ طِطِ، گࣹسࣹ (گٝو) طِطِ، مَرْطࣹ ݧَامࣹتࣹي هَاطُطٝ، إࣹ (لࣹطّࣹ) دُࢠّࣹيجࣹ، إࣹ هُوندࣹ إࣹ جَاࢠࣹ سࣹيطَاجࣹ.

(17) نگٝولْ (يَگِّنگٝلْ)، مࣹنْ يٝࢠِرْࢠࣹ كٝ ࢠࣹ يࣹدِّ كٝنْ. إࣹ هَرَ مࣹنْ يٝࢠَيْ سِ وَنَا يࣹدُّࢠࣹ ࢠࣹنْ?

(18) مࣹنْ وَطِ هَكُّنࣹ مَࢠّࣹ إࣹ كٝطٝولِ طِ مࣹنْ وَطِ بَرْكِ إࣹ مُنْ طِنْ، كٝطٝولِ فࣹيݧٝندِرْطِ، مࣹنْ إࣹࢠِّ إࣹ مَجِّ يَادُ. (مࣹنْ وِئَنِ ࢠࣹ): "يࣹهࣹي تٝنْ جࣹمَّاجِ إࣹ ݧَلٝرْمَاجِ، كٝ أٝنْ هٝولِيࢠࣹ".

(19) ࢠࣹ وِئِ: "جٝومِ أَمࣹنْ، وٝطِّندِرْ هَكُّندࣹ يَادُوجِ أَمࣹنْ"، ࢠࣹ تٝوݧِ كٝؤࣹ مَࢠّࣹ، مࣹنْ وَطْتِ فِلِّتࣹي ࢠࣹ فِي مُنْ، مࣹنْ سࣹنْندِندِرِ ࢠࣹ هَتِّرْدࣹ إࣹ سࣹنْندِرْگٝلْ. پࣹلّࣹتْ، وٝنِي إࣹ طُمْ، مَاندࣹيجِ وٝنَنْندࣹ كَلَ هࣹيوُطٝ مُݧَلْ، يࣹتّٝووٝ (نࣹيمَ).

(20) گٝمْطِي إبليس گٝونگِنِي جِكّࣹ مُنْ طࣹنْ إࣹ مَࢠّࣹ. ࢠࣹ جٝكِّ مٝ، سِ وَنَا فࣹدّࣹ إࣹ گٝمْطِمْࢠࣹ ࢠࣹنْ.

(21) إࣹ دٝولࣹ أَلَنَا مٝ إࣹ دٝوْ مَࢠّࣹ، ݖࣹمْنبࣹ (وُرْتُگٝلْ)، سِ وَنَا تُنْ، كٝ فِي نٝ مࣹنْ أَنْندِرَ أٝنْ گٝمْطِنْطٝ لَاكَرَ إࣹ أٝنْ وٝنْطٝ إࣹ سِكّࣹ فِي مُنْ. جٝومِ مَا نٝنْ، إࣹ دٝوْ كَلَ هُوندࣹ، كٝ أٝ رࣹينُطٝ.

(22) مَاكُ: "نٝدّࣹي طِنْ طِ طَيْنِتٝرْطٝنْ رࣹوࣹتࣹيطِ گَانِنْ الله. طِ جࣹيَا هَيْ يࣹرُ أَبّࣹرࣹ جَرَّ، كَ كَمُّولِ وَنَا كَ لࣹيْدِ. أَلَنَا طِ إࣹ طِنْ (كَمُّولِ إࣹ لࣹيْدِ ندِنْ) كَفِدَاطٝ (گࣹࢠَلْ)، أَلَنَا مٝ إࣹ طِنْ، هَيْ وَلّٝندِرْتࣹيطٝ".

(23) تࣹفٝورࣹ وٝو نَفَتَا كَ مَكّٝ، سِ وَنَا أٝنْ مٝ أٝ دُݝَنِي. هَا سِ طࣹنْڃࣹرࣹ ندࣹنْ يَلْتِنَامَ إࣹ ࢠࣹرْطࣹ مَࢠّࣹ طࣹنْ، ࢠࣹ وِئَ: "كٝ هٝنْطُمْ جٝومِ مٝنْ دَالِ?" ࢠࣹ وِئَ: "كٝ گٝونگَ كَنْ; كٝ كَنْكٝ وٝنِ تٝووُطٝ، مَوْطٝ أٝنْ".

(24) مَاكُ: "كٝنْ هٝنبٝ وٝنِ أَرْسِكِرٝووٝ أٝنْ كَ كَمُّولِ إࣹ كَ لࣹيْدِ?" مَاكُ: "كٝ الله. كٝ سِ كٝ مࣹنࣹنْ مَا سِ كٝ أٝنٝنْ، وٝنِ إࣹ پࣹيوَلْ مَا إࣹ ندࣹرْ مَجّࣹرࣹ ࢠَنْنگُندࣹ".

(25) مَاكُ: "أٝنْ لَمْندِتٝيْتَاكࣹ فِي كٝ مࣹنْ بٝنِّ، مࣹنࣹنْ كَدِ مࣹنْ لَمْندِتٝيْتَاكࣹ فِي كٝ هُووٝتٝنْ".

(26) مَاكُ: "جٝومِ مࣹنْ رࣹنْندِنٝيَيْ إࣹنْ، رࣹفْتِ أٝ ݧَاوِرَ هَكُّندࣹ مࣹنْ گٝونگَ; كٝ كَنْكٝ وٝنِ ݧَاوٝووٝ، أَنْندُطٝ أٝنْ".

(27) مَاكُ: "هٝلّࣹي لَمْ طِنْ طِ وَطْتِدُ-طٝنْ إࣹ مَكّٝ كَفِدِيطِ. وَنَا نٝنْ وٝنِرِ! سِ كٝ وٝنِ، كٝ كَنْكٝ وٝنِ الله فٝولُطٝ، ݧࣹيݧُطٝ أٝنْ".

(28) مࣹنْ نُلِرَالِ مَ، سِ وَنَا إࣹ يِمْࢠࣹ ࢠࣹنْ فٝوْ: وࣹوْلِنٝووٝ إࣹ جࣹرْتِنٝووٝ. كٝنٝ كٝ ࢠُرِ طُوطُدࣹ إࣹ يِمْࢠࣹ ࢠࣹنْ، ࢠࣹ أَنْندَا.

(29) هِࢠࣹ وِئَ: "كٝ هٝنْندࣹ تُمَ وٝنِ ندࣹي فٝدٝورࣹ، سِ أٝنْ لَاتِكࣹ گٝونگُࢠࣹ?"

(30) مَاكُ: "هِنٝ وٝودَنِ أٝنْ فٝدٝورࣹ، ندࣹ أٝ نࣹنِّتَنْتَاكࣹ إࣹ مَيْرࣹ فٝتْدࣹ سَائِ، أٝنْ أَرْدِنْتِنَنْتَاكࣹ ندࣹ كَدِ!".

(31) يࣹدُّࢠࣹ ࢠࣹنْ وِئِ: "مࣹنْ گٝمْطِنْتَا ندࣹي القرآن مُكْ، وَنَا ندࣹنْ أَدِيندࣹ ندࣹ". سِنّٝ أَ تِنَيْنٝ تُمَ ندࣹ تٝوݧُࢠࣹ ࢠࣹنْ دَرْنࣹتࣹي كَ جٝومِ مَࢠّࣹ، هَرَ يٝگَ مَࢠّࣹ نٝ رُتَّ يٝگَ كٝنْنگٝلْ، لٝئِنَنٝوࢠࣹ ࢠࣹنْ وِئَنَ تٝوْنِتِنٝوࢠࣹ ࢠࣹنْ: "سِ وَنَانٝ أٝنٝنْ نُنْ، مࣹنْ وٝنَيْنٝ گٝمْطِمْࢠࣹ".

(32) تٝوْنِتِنٝوࢠࣹ ࢠࣹنْ وِئَنِ لٝئِنَنٝوࢠࣹ ࢠࣹنْ: "إࣹ كٝ مࣹنࣹنْ فَلِنٝو أٝنْ إࣹ پࣹيوَلْ ࢠَاوٝ نگَلْ أَرْدࣹ إࣹ مٝنْ? سِ كٝ وٝنِ، أٝنْ وٝنِينٝ بٝمْࢠࣹ".

(33) لٝئِنَنٝوࢠࣹ ࢠࣹنْ كَدِ وِئَنِ تٝوْنِتِنٝوࢠࣹ ࢠࣹنْ: "سِ كٝ وٝنِ، كٝ پࣹوْجࣹ (مٝنْ) جࣹمَّ إࣹ ݧَلٝرْمَ، تُمَ يَمِرَيْنٝطٝنْ مࣹنْ يٝ مࣹنْ يࣹدُّ الله، مࣹنْ وَطَنَ مٝ فٝتٝندِرْطِ". ࢠࣹ سُوطِ نِنْسࣹ طࣹنْ، ندࣹ ࢠࣹ يِئُنٝو لࣹپْتࣹ طࣹنْ. مࣹنْ وَطِ كٝلٝنبٝوجࣹ إࣹ دَاطࣹ ࢠࣹنْ يࣹدُّࢠࣹ. هَرَ ࢠࣹ يٝࢠِتࣹتࣹ سِ وَنَا كٝ ࢠࣹ هُووَيْنٝو كٝنْ?

(34) مࣹنْ نُلَالِ جࣹرْتِنٝووٝ إࣹ يِمْࢠࣹ سَارࣹ، سِ وَنَا نࣹيمِنَاࢠࣹ مَيْرࣹ ࢠࣹنْ وِئَيْ: "مࣹنࣹنْ كٝ مࣹنْ يࣹدُّࢠࣹ كٝ نُلِدَطٝنْ كٝنْ".

(35) ࢠࣹ وِئِ كَدِ: "كٝ مࣹنࣹنْ ࢠُرِ طُوطُدࣹ جَوْلࣹ إࣹ ࢠِࢠّࣹ، مࣹنْ وٝنَالِ كَدِ لࣹپْتࣹتࣹيࢠࣹ".

(36) مَاكُ: "پࣹلّࣹتْ، جٝومِ أَمْ هِنٝ وࣹيرْتَنَ أَرْسِكࣹ أٝنْ، مٝ أٝ مُويِ، أٝ ࢠِلِّنِ (مٝ أٝ مُويِ). كٝنٝ ࢠُرْࢠࣹ طُوطُدࣹ إࣹ يِمْࢠࣹ ࢠࣹنْ، ࢠࣹ أَنْندَا".

(37) وَنَا جَوْلࣹ مٝنْ، وَنَا جِيدِ مٝنْ وٝنِ كٝ ࢠَطْتِنِرْتَ أٝنْ كَ أَمࣹنْ ࢠَدَلْ. سِ وَنَا ࢠࣹنْ گٝمْطِمْࢠࣹ ࢠࣹ گٝلِّ گٝلّࣹ مٝڃّࣹ; ࢠࣹنْ طٝنْ، هِنٝ وٝودَنِ ࢠࣹ نجٝࢠْدِ سٝوَاندࣹ، سَبُ كٝ ࢠࣹ گٝلِّ كٝنْ، كَمْࢠࣹ كَ سٝورٝوجِ، كٝ ࢠࣹ هٝولِيࢠࣹ.

(38) ࢠࣹنْ يَݖّٝتٝوࢠࣹ إࣹ رٝنْكِنگٝلْ (يِمْࢠࣹ ࢠࣹنْ) إࣹ آيࣹيجࣹ أَمࣹنْ طࣹنْ، ࢠࣹنْ طٝنْ، كَ كَ ندࣹرْ لࣹپْتࣹ وٝنِ كٝ ࢠࣹ تَوْنٝيْتࣹي.

(39) مَاكُ: "پࣹلّࣹتْ، جٝومِ أَمْ هِنٝ وࣹيرْتَنَ أَرْسِكࣹ أٝنْ، مٝ أٝ مُويَنِ إࣹ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ، أٝ ࢠِلِّنِ مٝ (أٝ مُويِ إࣹ مَࢠّࣹ). كَلَ نٝنْ كٝ وِنْتِطٝنْ إࣹ هُوندࣹ، كَنْكٝ أٝ لٝنْتِنَيْ ندࣹ. كٝ كَنْكٝ وٝنِ مٝڃّٝ أَرْسِكٝووٝ".

(40) ݧَلْندࣹ أٝ مٝوࢠْتِندِرٝيْتَ ࢠࣹ دࣹندَانگَلْ، رࣹفْتِ أٝ دَالَنَ الملائكة ࢠࣹنْ: "إࣹ كٝ ࢠࣹي طٝو (يِمْࢠࣹ) رࣹوَيْنٝو أٝنْ?"

(41) ࢠࣹ وِئَ: "سࣹنَيࣹي وٝودَنِي مَ! كٝ أَنْ وٝنِ گِطٝ أَمࣹنْ گَانِنْ مَࢠّࣹ. تٝودࣹ كٝ جِنَّاجِ طِنْ ࢠࣹ رࣹوَيْنٝو، كٝ ࢠُرِ طُوطُدࣹ إࣹ مَࢠّࣹ، كٝ طِنْ (جِنَّاجِ) گٝمْطِنِ".

(42) هَنْندࣹ نٝنْ، يٝگَ مٝنْ وَاوَنَا يٝگَ نَفَ وَنَا لٝرَّ. مࣹنْ وِئَنٝيَ ࢠࣹنْ تٝوݧُࢠࣹ: "مࣹيطࣹي لࣹپْتࣹ يِيتࣹ نگࣹ فࣹنَّيْنٝطٝنْ فِي مُنْ".

(43) سِ آيࣹيجࣹ أَمࣹنْ ࢠَنْنگُطࣹ طࣹنْ جَنْنگَامَّ إࣹ مَࢠّࣹ، ࢠࣹ وِئَ: "أٝو وٝنَالِ، سِ وَنَا گٝرْكٝ فَالَاطٝ فَلَگٝلْ أٝنْ إࣹ كٝ بَابِرَاࢠࣹ مٝنْ رࣹوَيْنٝ رࣹوَيْنٝو". ࢠࣹ وِئِ: "ندࣹي (القرآن) وٝنَالِ، سِ وَنَا فࣹنَاندࣹ فࣹفِندَاندࣹ". يࣹدُّࢠࣹ ࢠࣹنْ وِئَنِ گٝونگَ كَنْ ندࣹ كَ أَرُنٝو إࣹ مَࢠّࣹ: "ندࣹي وٝنَالِ، سِ وَنَا نبِلࣹوُ ࢠَنْنگُ نگُ".

(44) مࣹنْ أَدَّنَالِ ࢠࣹي، دࣹفْتࣹ طࣹ ࢠࣹ جَنْنگَتَ. مࣹنْ مࣹيطَالِ نُلُدࣹ إࣹ مَࢠّࣹ، جࣹرْتِنٝووٝ أَدٝ مَا.

(45) أَدِنٝوࢠࣹ ࢠࣹ ࢠࣹنْ كَدِ فࣹنِّينٝ (نُلَاࢠࣹ مُنْ). يِمْࢠࣹ مَا ࢠࣹنْ يٝتَّاكِ سَپَّࢠَلْ كٝ مࣹنْ أٝكُّنٝو ࢠࣹنْ كٝنْ. ࢠࣹنْ فࣹنِّ نُلَاࢠࣹ أَمࣹنْ ࢠࣹنْ، تَسْكِي نٝ لࣹپْتࣹ أَمْ طࣹنْ وَؤُنٝو!

(46) مَاكُ: "أَنْندࣹي كٝ مِي وَاجٝرْتَ أٝنْ، كٝ (هُوندࣹ) وٝوتࣹرࣹ: "كٝ ندࣹ دَرٝرَنْتٝنْ الله طِطٝ طِطٝ إࣹ گٝوتٝ گٝوتٝ، رࣹفْتِ مِيجِتٝطٝنْ. (أَنْندٝنْ) وٝندِيطٝ مٝنْ أٝنْ أَلْدَا إࣹ جِنَّ: أٝ وٝنَالِ، سِ وَنَا جࣹرْتِنٝووٝ أٝنْ، إࣹ كٝ وٝنِ يࣹيسٝ (مٝنْ)، إࣹ لࣹپْتࣹ سَطْتُطࣹ".

(47) مَاكُ: "مِي لَمْندَاكِ أٝنْ نجٝࢠْدِ، طُمْ كٝ فِي مٝنْ. نجٝࢠْدِ أَمْ ندِنْ وٝنَالِ، سِ وَنَا كَ الله. كَنْكٝ كٝ أٝ سࣹيدِيطٝ إࣹ كَلَ هُوندࣹ".

(48) مَاكُ: "پࣹلّࣹتْ، جٝومِ أَمْ كٝ وࣹرْلٝرْتٝوطٝ مࣹيرࣹ ندࣹنْ، هُجَّاجِ گٝونگَ كَنْ، تِمِّنْطٝ أَنْندُگٝلْ وِرْنِيطِ".

(49) مَاكُ: "گٝونگَ كَنْ أَرِي إِيسْلَامُ أٝنْ ، مࣹيرࣹ كࣹيفࣹرَاكُ نگُنْ أَلَا كَ فࣹيݧَ، كَدِ، ندࣹ رُتّٝتَاكٝ".

(50) مَاكُ: "سِ مِي مَجِّي، هَرَيْ كٝ هٝورࣹ-أَمْ مِي مَجَّنِ; سِ مِي فࣹيوِي كَدِ، كٝ سَبُ كٝ سَبُ كٝ جٝومِ أَمْ وَهَيِنِ إࣹ أَمْ كٝنْ، تَوْدࣹ پࣹلّࣹتْ، كٝ أٝ نَنٝووٝ، ࢠَدِيطٝ".

(51) سِنّٝ أَ تِنَيْنٝ، تُمَ ندࣹ ࢠࣹ طࣹنْيِي، - لَاوٝرْدࣹ أَلَنَا (ࢠࣹ)، - ࢠࣹ نَنْنگِرࣹي إࣹ نٝكُّورࣹ ࢠَدِيندࣹ!

(52) ࢠࣹ وِئَ: "مࣹنْ گٝمْطِنِي دَرْنگَلْ نگَلْ"، إࣹ كٝ هٝنّٝ گٝمْطِنَلْ وٝونِرَنِ ࢠࣹ إࣹ نٝكُّ وٝطُّطٝ?

(53) هَرَ نٝنْ ࢠࣹ يࣹدِّينٝ ندࣹ كٝ أَدِي، هِࢠࣹ وࣹرْلٝرَ كٝ وِرْنِي إࣹ نٝكُّ وٝطُّطٝ!

(54) هࣹيدࣹي هَكُّندࣹ إࣹ كٝ ࢠࣹ تُويࣹتࣹي، وَنٝ گٝلِّرَنٝو إࣹ نَنْندٝ مَࢠّࣹ كٝ أَدِي، تَوْدࣹ ࢠࣹ وٝنِينٝ إࣹ سِكّࣹ ݖِكِّنِيطَ.

35 Fatir — 45 آيَة

(1) يࣹتّٝورࣹ وٝودَنِي الله، تَگُطٝ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، وَطُطٝ الملائكة نُلَاࢠࣹ، مَرْࢠࣹ گَبِتَنجِ(بِبْجࣹ) طِ طِطِ، طِ تَتِ إࣹ نَيِ. هِمٝ ࢠࣹيْدَ كٝ أٝ مُويِ إࣹ تَگُ مَكّٝ نگُنْ، تَوْدࣹ پࣹلّࣹتْ، الله كٝ هَتَّنْطٝ كَلَ هُوندࣹ.

(2) كَلَ يُرْمࣹيندࣹ ندࣹ الله أُدِّتَنِ يِمْࢠࣹ ࢠࣹنْ، جٝگِتٝتٝوطٝ ندࣹ أَلَا; كَلَ كَدِ كٝ أٝ جٝگِتِي، أَݖِّتٝووٝ طُمْ أَلَا ࢠَاوٝ (جٝگِتَگٝلْ) مَكّٝ. كٝ كَنْكٝ وٝنِ فٝولُطٝ، ݧࣹيݧُطٝ.

(3) كٝ أٝنٝنْ يٝ يِمْࢠࣹ! جَنْتࣹي نࣹيمَ الله أٝنْ إࣹ مٝنْ. هَرَ تَگٝووٝ كٝ وٝورِ الله نٝ وٝودِ، مٝ نٝ أَرْسِكِرَ أٝنْ كَ كَمُّ إࣹ كَ لࣹيْدِ? رࣹوࣹتࣹيطٝ أَلَا سِ وَنَا كَنْكٝ! إࣹ كٝ هٝنّٝ يِيلِرْتࣹطٝنْ إࣹ كَا گٝونگَ ?

(4) سِ ࢠࣹ فࣹنِّي مَ، پࣹلّࣹتْ نُلَاࢠࣹ (گٝو) فࣹنَّنٝومَ أَدٝ مَا. كٝ فَادࣹ إࣹ الله، وٝنِ رُتّٝرْدࣹ فِيَكُوجِ طِنْ.

(5) كٝ أٝنٝنْ يٝ يِمْࢠࣹ! أَنْندࣹي پࣹلّࣹتْ فٝدٝورࣹ الله ندࣹنْ كٝ گٝونگَ. وَتَ نگُورْندَمْ أَدُنَ هٝدُ أٝنْ، وَتَ (سࣹيْتَانࣹ) هٝدٝووٝ أٝنْ هٝدُ أٝنْ إࣹ فِي الله.

(6) پࣹلّࣹتْ، سࣹيْتَانࣹ كٝ گَݧٝ وٝنَنِ أٝنْ. هَرَيْ جٝگِتٝرࣹي مٝ گَݧٝ. أَنْندࣹي كٝ أٝ نٝدِّرْتَ فࣹدّࣹ مَكّٝ ندࣹنْ، كٝ نٝ ࢠࣹ جࣹيْدِرࣹي إࣹ يِمْࢠࣹ دُلْࢠٝووࣹ نگࣹنْ.

(7) هِنٝ وٝودَنِ يࣹدُّࢠࣹ ࢠࣹنْ، لࣹپْتࣹ سَطْتُطࣹ. ࢠࣹيَ گٝمْطِمْࢠࣹ ࢠࣹ گٝلِّ گٝلّࣹ مٝڃّࣹ، هِنٝ وٝودَنِ ࢠࣹ هَفٝرَنࣹيدࣹ إࣹ نجٝࢠْدِ مَوْندِ.

(8) إࣹ كٝ أٝنْ ݖُطِنَنَاطٝ گٝلّࣹ مُنْ بٝنْطࣹ، هَرَ هِمٝ يِئَ طࣹ مٝڃُّطࣹ? - تَوْدࣹ پࣹلّࣹتْ، الله نٝ مَجِّنَ أٝنْ مٝ أٝ مُويِ، أٝ فࣹوْنَ مٝ أٝ مُويِ. وَتَ أَ تَمْپِنِرْ هٝورࣹ-مَا، سُنَگٝلْ فِي مَࢠّࣹ. پࣹلّࣹتْ، الله كٝ أَنْندُطٝ كٝ ࢠࣹ وٝنِ هُووُدࣹ.

(9) كٝ الله وٝنِ وُرْتُطٝ كࣹنࣹيلِ طِنْ، طِ سٝگَّ دُولࣹ، مࣹنْ يَرْنِرَ طُمْ لࣹيْدِ يٝورُندِ; مࣹنْ وُورْنِتِرَ طُمْ لࣹيْدِ ندِنْ، ࢠَاوٝ ندِ يٝورُدࣹ. كٝ وَنٝ نٝنْ أُمِّتَلْ نگَلْ.

(10) كَلَ تَوْطٝ كٝ تࣹدُّنگَلْ فَالَا (يٝ أٝ طَࢠِّرْ الله)، تَوْدࣹ كٝ الله جࣹيِ تࣹدُّنگَلْ نگَلْ فٝوْ: كٝ كَ مَكّٝ وٝنِ كٝ كٝنْنگٝلْ لَاࢠُنگٝلْ ڃَوَتَ، گٝلّࣹ مٝڃّࣹ طࣹنْ ࢠَمْتَ نگٝلْ. ࢠࣹنْ فࣹوْجٝوࢠࣹ بٝنࣹيجِ، هِنٝ وٝودَنِ ࢠࣹ لࣹپْتࣹ سَطْتُطࣹ. پࣹوْجࣹ ࢠࣹنْ طࣹنْ، كٝ بٝنَيْ طࣹ.

(11) كٝ الله تَگِرِ أٝنْ إࣹ نبُلْبُلْدِ، رࣹفْتِ إࣹ تٝࢠّࣹرࣹ (مَنِييُ)، رࣹفْتِ أٝ وَطِ أٝنْ نٝونࣹيجِ. دࣹيࣹلْ سَاوَتَا، نگࣹلْ لࣹلْنَتَا كَدِ، سِ وَنَا إࣹ أَنْندَلْ مَكّٝ. وُورْنَاطٝ وُورْنࣹتَاكࣹ، طُيْتࣹتَاكࣹ كَدِ إࣹ نگُورْندَمْ مَكّٝ، سِ وَنَا نٝ إࣹ دࣹفْتࣹرࣹ. پࣹلّࣹتْ، طُمْ كٝ نࣹوَنِي طُمْ الله.

(12) مَاجࣹ طࣹنْ طِطِ فٝتَتَا: ندِيَمْ گٝو طٝو طَمْ، كٝ وࣹلْطَمْ، لَاࢠُنگٝ، نࣹوِينگٝ نجَرَمْ مُنْ، نگٝو طَا، كٝ لَنْطَمْ هَاطُطَمْ. هِطٝنْ ݧَامَ إࣹ كَلَ مَجَّمْ، تࣹيوُ كࣹݖُّ، هِطٝنْ يَلْتِنَ كَدِ هࣹنْ، ݖُطَارِ ندِ ࢠٝورْنٝتٝطٝنْ. هِطَ يِئَ لَاطࣹ طࣹنْ نٝ أَوْجَ إࣹ مَگّٝ، فِي يٝ أٝنْ طَࢠِّرٝنْ إࣹ ࢠُرَلْ مَكّٝ. بࣹلَجٝو أٝنْ يࣹتَّيْ مٝ .

(13) هِمٝ نَادَ جࣹمَّ أٝنْ إࣹ دࣹرْ ݧَلٝرْمَ أٝنْ، أٝ نَادَ ݧَلٝرْمَ أٝنْ إࣹ ندࣹرْ جࣹمَّ أٝنْ. أٝ إࣹلْتِ كَدِ نَانگࣹ نگࣹنْ إࣹ لࣹوْرُ ندُنْ. كَلَ مُنْ نٝ دٝگَ هَا إࣹ لَجَلْ إِنَّانگَلْ. كٝ أٝنْ مٝوطٝنْ وٝنِ الله، جٝومِ مٝنْ: كٝ كَنْكٝ لَامُ وٝودَنِ. طِيَ طِ نٝدّٝتٝنْ گَانِنْ مَكّٝ، طِ جࣹيْتَا يࣹرُ گٝرٝلْ تَمَرَ.

(14) سِ أٝنْ نٝدُّ طِ، طِ نَنَتَا نٝدَّاندُ مٝنْ; هَيْ سِ طِ نَنٝيْ، طِ جَابِتٝتَاكٝ أٝنْ. ݧَلْندࣹ دَرْنگَلْ، طِ يࣹدّٝيَيْ سِرْكُ مٝنْ أٝنْ. أَلَا هَبْرٝووٝ مَ، وَنٝ هُمْپِتِيطٝ أٝنْ.

(15) كٝ أٝنٝنْ يٝ يِمْࢠࣹ، كٝ أٝنٝنْ وٝنِ هَتٝنجِمْࢠࣹ إࣹ الله ࢠࣹنْ، الله كَنْ، كٝ كَنْكٝ وٝنِ گَلٝ يࣹتِّنِيطٝ أٝنْ.

(16) سِ أٝ يِطِي، أٝ نَࢠَ أٝنْ، أٝ أَدَّ تَگُ هࣹسُ.

(17) طُمْ وٝنَالِ سَتُّطُمْ كَ الله.

(18) وٝنْكِ رٝنْندٝتَاكٝ دٝݝْگَلْ وٝنْكِي گٝو. هَيْ سِ تࣹدِّرَاكِ جُنُوبِ كِنْ نٝدِّي إࣹ رٝنْندَنࣹيدࣹ، كِ رٝنْندَنْتَاكࣹ هَيْ هُوندࣹ، هَيْ سِ كٝ ࢠَدٝندِرَاطٝ أَنْندُ كٝ جࣹرْتِنْتَا نَنَ وَاجُ ، كٝ ࢠࣹنْ هُلِرٝوࢠࣹ جٝومِ مُنْ كَ وِرْنِي، ࢠࣹ ݧِيࢠِ إࣹ جُلْدࣹ. كَلَ أٝنْ لَࢠِّنِيطٝ، هَرَيْ كٝ هٝورࣹ مَكّٝ أٝ وٝنِ لَࢠِّنَنَادࣹ. كٝ كَ الله وٝنِ رُتّٝرْدࣹ ندࣹنْ.

(19) بُنْطٝ إࣹ يِئٝووٝ فٝتَتَا،

(20) وَنَا نِࢠࣹ طࣹنْ إࣹ ندَيْگُ نگُنْ،

(21) وَنَا طٝوْدِ ندِنْ إࣹ هࣹنْندُ وُلْندُ ندُنْ.

(22) وَنَا كَدِ وُورْࢠࣹ ࢠࣹنْ فٝتَتَ إࣹ مَايْࢠࣹ ࢠࣹنْ. پࣹلّࣹتْ، الله نٝ نَنِنَ أٝنْ مٝ أٝ مُويَنِ. وَنَا أَنْ نُلَاطٝ ، وٝنِ نَنِنٝووٝ، وٝنْࢠࣹ كَ گࣹنَالࣹ ࢠࣹنْ.

(23) أَنْ أَ وٝنَالِ، سِ وَنَا جࣹرْتِنٝووٝ.

(24) پࣹلّࣹتْ، مࣹنْ نُلِرِي مَ گٝونگَ، كٝ أَ وࣹوْلِنٝووٝ إࣹ جࣹرْتِنٝووٝ. أَلَا إࣹ مٝفْتࣹ، سِ وَنَا جࣹرْتِنٝووٝ فࣹڃِّي إࣹ مَࢠّࣹ.

(25) سِ ࢠࣹ فࣹنِّي مَ، هَرِ أَدِنٝوࢠࣹ ࢠࣹ ࢠࣹنْ فࣹنِّينٝ (نُلَاࢠࣹ مُنْ)، فࣹوْندٝ ࢠࣹنْ نُلَاࢠࣹ مَࢠّࣹ أَدَّنّٝو ࢠَنْنگَنْندُيࣹيجِ، إࣹ طࣹرِ إࣹ دࣹفْتࣹرࣹ ندَيْگِنٝورࣹ.

(26) رࣹفْتِ نَنْنگِتِرْ مِي يࣹدُّࢠࣹ ࢠࣹنْ (هَلَكُيࣹي). تَسْكِي نٝ لࣹپْتࣹ أَنْ طࣹنْ وَطْتِنِرْنٝو.

(27) إࣹ أَ تِنَالِ وٝنْندࣹ الله جِپِّنِرِي ندِيَمْ كَ كَمُّ? مࣹنْ فُطِنِرِ طَمْ دِمْطࣹ سࣹيرْتُطࣹ نٝونࣹ. هِنٝ كَ پࣹلّࣹ، طَتࣹ دَنࣹيجࣹ إࣹ بٝطࣹيجࣹ سࣹيرْتْيُطࣹ نٝونࣹ، إࣹ پࣹتࣹ ࢠَلࣹيجࣹ.

(28) هِنٝ إࣹ يِمْࢠࣹ ࢠࣹنْ، إࣹ دَابࣹيجِ طِنْ، إࣹ نࣹيمٝرَاطِ طِنْ، طِ نٝونࣹيجِ مُنْ كَدِ لُوتٝندِرِ. أَنْندُ كٝ هُلَتَ الله إࣹ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ، كٝ أَنْندُࢠࣹ ࢠࣹنْ. پࣹلّࣹتْ، الله كٝ فٝولُطٝ، هَفٝرٝووٝ.

(29) ࢠࣹنْ جَنْنگٝوࢠࣹ دࣹفْتࣹرࣹ الله ندࣹنْ، ࢠࣹ ݧِيࢠِ إࣹ جُلْدࣹ، ࢠࣹ نَفْقِي كٝ مࣹنْ أَرْسِكِ ࢠࣹ كٝنْ گُنْندٝو إࣹ كࣹنࣹ، هِࢠࣹ جٝرْتِي نجُلَاكُ نگُ سٝنْتَتَا.

(30) فِي نٝ الله هُنِّرَنَ ࢠࣹ نجٝࢠْدِيجِ مَࢠّࣹ طِنْ، أٝ ࢠࣹيْدَ ࢠࣹ إࣹ ࢠُرَلْ مَكّٝ نگَلْ. كَنْكٝ، كٝ أٝ هَفٝرٝووٝ، أَنْندِتَنٝووٝ أٝنْ .

(31) كٝنْ كٝ مࣹنْ وَهَيِنِ إࣹ مَاطَ إࣹ دࣹفْتࣹرࣹ ندࣹنْ، كٝ طُمْ وٝنِ گٝونگَ، گٝمْطِندࣹ إࣹ طࣹنْ وٝنْطࣹ يࣹيسٝ مَيْرࣹ. پࣹلّࣹتْ، الله كٝ هُمْپِتِيطٝ رࣹنْتُطٝ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ.

(32) رࣹفْتِ مࣹنْ رٝنِنِ دࣹفْتࣹرࣹ ندࣹنْ، ࢠࣹنْ ࢠࣹ مࣹنْ سُࢠِي إࣹ جࣹيَاࢠࣹ أَمࣹنْ. هِنٝ إࣹ مَࢠّࣹ تٝوݧُࢠࣹ هٝورࣹ-مُنْ، هِنٝ إࣹ مَࢠّࣹ هَكِندِنْطٝ، هِنٝ إࣹ مَࢠّࣹ فٝولٝندِرٝوࢠࣹ إࣹ مٝڃّࣹرࣹيجِ، دُݝَيࣹي الله; كٝ طُمْ وٝنِ ࢠُرَلْ مَوْنُنگَلْ نگَلْ.

(33) الجنة ݧِيࢠَلْ، طِ نَاتَتَ إࣹ مُنْ، هَرَ هِࢠࣹ ݖُطِرࣹي تٝنْ، جَوࣹ كَݝّࣹ إࣹ كَالِسࣹ إࣹ لُؤُ-لُؤُ; ݖٝمْݖِ مَࢠّࣹ طِنْ تٝنْ، كٝ سٝويِ.

(34) ࢠࣹ وِئَ: "يࣹتّٝورࣹ وٝودَنِي الله، أٝنْ إِتَّنْطٝ إࣹنْ سُنٝ. پࣹلّࣹتْ، جٝومِ مࣹنْ، كٝ هَفٝرٝووٝ، وَرْجٝتٝوطٝ.

(35) أٝنْ وࣹيرْنِرْطٝ إࣹنْ ࢠُرَلْ مَكّٝ نگَلْ، إࣹ گَلّࣹ ݧِيࢠَلْ، تَمْپࣹرࣹ (وٝو) هࣹࢠَتَا إࣹنْ تٝنْ، وَنَا رٝنْكࣹرࣹ هࣹࢠَتَا إࣹنْ تٝنْ.

(36) يࣹدُّࢠࣹ ࢠࣹنْ كَنْ، كٝ يِيتࣹ جهنم وٝودَنِ ࢠࣹنْ; مَايْدࣹ ݧَاوࣹتَاكࣹ إࣹ مَࢠّࣹ، سَكّٝ ࢠࣹ مَايَ، ࢠࣹ هٝيْنَنْتَاكࣹ كَدِ لࣹپْتࣹ مَگّࣹ طࣹنْ. كٝ وَنٝ نِي مࣹنْ يٝࢠِتِرْتَ كَلَ يࣹدُّطٝ.

(37) هَرَ هِࢠࣹ أٝولٝو ندࣹرْ تٝنْ ࢠࣹ وِئَ : "جٝومِ أَمࣹنْ، يَلْتِنْ مࣹنْ; مࣹنْ گٝلَّ مٝڃّࣹرࣹ، هَرَ وٝنَهْ كٝ مࣹنْ گٝلَّيْنٝو كٝنْ". أٝ جَابٝو ࢠࣹ : "إࣹ مࣹنْ وُورْنَانٝ أٝنْ يࣹرُ كٝ وَاجِتٝتٝوطٝ وَاجِتٝرَنٝو إࣹ مُنْ? جࣹرْتِنٝووٝ كَدِ أَرِ إࣹ مٝنْ? أَوَ مࣹيطࣹي، سَبُ فَابٝ أَلَنَا تٝوݧٝوࢠࣹ ࢠࣹنْ".

(38) پࣹلّࣹتْ، الله كٝ أَنْندُطٝ وِرْنِيطِ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ. كٝ أٝ أَنْندُطٝ كٝ سُوطِي إࣹ ࢠࣹرْطࣹ.

(39) كٝ كَنْكٝ وٝنِ وَطُطٝ أٝنْ لٝنْتِيࢠࣹ كَ لࣹيْدِ. كَلَ يࣹدُّطٝ، كٝ إࣹ مَكّٝ لٝرَّ گࣹدِّ مَكّٝ يَنْتَتَ. يࣹدُّگٝلْ هࣹيفࣹرْࢠࣹ ࢠࣹنْ، ࢠࣹيْدَتَا ࢠࣹ كَ جٝومِ مَࢠّࣹ، سِ وَنَا تِكّࣹرࣹ. يࣹدُّگٝلْ ࢠࣹيْدَتَا هࣹيفࣹرْࢠࣹ ࢠࣹنْ، سِ وَنَا هَيْرࣹرࣹ.

(40) مَاكُ: "إࣹ أٝنْ يِئِي رࣹندِيطِ مٝنْ رࣹوٝتٝنْ گَانِنْ الله طِنْ? هٝلّࣹي لَمْ كٝ طِ تَگِ كَ لࣹيْدِ. كَا هِطِ مَرِ گࣹࢠَلْ إࣹ تَگُگٝلْ كَمُّولِ طِنْ? كَا مࣹنْ أَدَّنْ ࢠࣹ دࣹفْتࣹرࣹ، ندࣹ ࢠࣹ وٝنِ إࣹ ࢠَنْندَنْندُيࣹي مُنْ? سِ كٝ وٝنِ، تٝوݧٝوࢠࣹ ࢠࣹنْ أَلَا إࣹ فٝدُدࣹ يٝگَ مَࢠّࣹ، سِ وَنَا هٝدٝ".

(41) پࣹلّࣹتْ، كٝ الله جٝگِتِي كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، فِي وَتَ طِ دِرْتُ. هَيْ سِ طِ دِرْتَيْنٝ، هَيْ گٝوتٝ جٝگِتٝتَاكٝ طِ إࣹ ࢠَاوٝ مَكّٝ. أٝ لَاتِكࣹ مُݧِّيطٝ، هَفٝرٝووٝ.

(42) ࢠࣹ وٝودْندِرِ الله، هَتِّرْدࣹ إࣹ گٝوندٝوجࣹ مَࢠّࣹ، وٝنْندࣹ سِ جࣹرْتِنٝووٝ أَرِي إࣹ مَࢠّࣹ، پࣹلّࣹتْ، ࢠࣹ وٝنَيْ ࢠُرْࢠࣹ فࣹيوُدࣹ وٝوتࣹرࣹ إࣹ مٝفْتࣹ طࣹنْ. ندࣹ جࣹرْتِنٝووٝ نٝنْ أَرُنٝو إࣹ مَࢠّࣹ، طُمْ ࢠࣹيْدَالِ ࢠࣹ، سِ وَنَا وٝطِّتَگٝلْ.

(43) ࢠࣹ وَطِرِ طُمْ مَوْنِنْتِنَارࣹ كَ كࣹيْدِ، إࣹ پࣹوْجࣹ ࢠࣹنْطࣹ. پࣹوْجࣹ بٝنْطࣹ طࣹنْ نٝنْ هُوࢠِتَتَا، سِ وَنَا جٝمْ-مُنئࣹنْ. هَرَ هِࢠࣹ سَبِّتِي، سِ وَنَا سُنَّ (الله) أٝنْ إࣹ أَدِنٝوࢠࣹ ࢠࣹنْ? أَ هࣹࢠَنْتَا سُنَّ الله أٝنْ بَتِّيگُ، أَ هࣹࢠَنْتَا سُنَّ الله وَيْلِرْطُمْ.

(44) إࣹ ࢠࣹي يِيلَاكِ كَ لࣹيْدِ، ࢠࣹ تَسْكٝو كٝ هٝنّٝ بَتَّنࣹ أَدِنٝوࢠࣹ ࢠࣹنْ وٝنِرْنٝو? هَرِ نٝ ࢠُرِ ࢠࣹ تِيطُدࣹ دٝولࣹ. الله سِفٝرَالِ وٝنْندࣹ گٝطُّمْ نٝ رٝنْكِنَ مٝ كَ كَمُّولِ مَا كَ لࣹيْدِ. تَوْدࣹ پࣹلّࣹتْ، كٝ أٝ أَنْندُطٝ، هَتَّنْطٝ.

(45) سِ تَوْنٝ الله نَنْنگِتِرَيْنٝ يِمْࢠࣹ ࢠࣹنْ كٝ ࢠࣹ فَگِّتٝرِي كٝنْ، أٝ أَݖَّتَانٝ رُوگَيْنگࣹلْ إࣹ دٝوْ مَيْرِ. كٝ وٝنِ تُنْ، هِمٝ نࣹنِّتَنَ ࢠࣹ هَا إࣹ لَجَلْ أَنْندَا نگَلْ. سِ لَجَلْ مَࢠّࣹ نگَلْ هࣹوْتِي، أٝ نَنْنگَيْ ࢠࣹ ، تَوْدࣹ پࣹلّࣹتْ، الله كٝ رࣹينُطٝ جࣹيَاࢠࣹ مَكّٝ ࢠࣹنْ.

(9) مࣹنْ وَطِ يࣹيسٝ مَࢠّࣹ هࣹيدٝ، ࢠَاوٝ مَࢠّࣹ كَدِ هࣹيدٝ، مࣹنْ هِپِّ ࢠࣹ (گِتࣹ طࣹنْ); وَطْتِ ࢠࣹ يِئَتَا گٝونگَ .

(10) هِنٝ فٝتَ إࣹ مَࢠّࣹ، فٝتِ أَ جࣹرْتِنِ ࢠࣹ، مَا أَ جࣹرْتِنَالِ ࢠࣹ; ࢠࣹ گٝمْطِنْتَا.

(11) أَنْندُ كٝ جࣹرْتِنْتَا (جَࢠَ)، كٝ أٝنْ جٝكُّطٝ جَنْتٝورࣹ (القرآن) ندࣹنْ، أٝ هُلِ جٝمْ-يُرْمࣹيندࣹ أٝنْ كَ وِرْنِي. وࣹوْلِنِرْ مٝ هَفٝرَنࣹيدࣹ إࣹ نبَرْجَارِ تࣹدُّندِ.

(12) كٝ مࣹنࣹنْ وٝنِ وُورْنِتٝوࢠࣹ مَايْࢠࣹ ࢠࣹنْ، مࣹنْ وِنْندَ كٝ ࢠࣹ أَرْدِنِ إࣹ بَتّࣹيجِ مَࢠّࣹ. كَلَ هُوندࣹ، مࣹطࣹنْ مٝفْتِ ندࣹ إࣹ دࣹفْتࣹرࣹ ࢠَنْنگُندࣹ.

(13) پِيَنْ-ࢠࣹ مِسَلْ، فِي يِمْࢠࣹ سَارࣹ، تُمَ ندࣹ نُلَاࢠࣹ أَرُنٝو ࢠࣹ،

(14) ندࣹ مࣹنْ نُلُنٝو إࣹ مَࢠّࣹ، نُلَاࢠࣹ طِطٝ، ࢠࣹ فࣹنِّ ࢠࣹنْ، مࣹنْ سَمْنبِنِرِ ࢠࣹنْ، تَتَࢠٝ، ࢠࣹ وِئِ: "مࣹنࣹنْ، كٝ مࣹنْ نُلَاࢠࣹ إࣹ مٝؤٝنْ".

(15) يِمْࢠࣹ سَارࣹ ندࣹنْ وِئِ: "أٝنْ وٝنَالِ، سِ وَنَا كٝ أٝنْ ࢠَنْندِنْكࣹ وَنٝ مࣹنࣹنْ. جٝمْ-هِنَرَنْكࣹ أٝنْ جِپِّنَالِ هَيْ هُوندࣹ، أٝنْ وٝنَالِ، سِ وَنَا كٝ أٝنْ فࣹنٝوࢠࣹ".

(16) نُلَاࢠࣹ ࢠࣹنْ وِئِ: "جٝومِ أَمࣹنْ نٝ أَنْندِ پࣹلّࣹتْ، كٝ مࣹنْ نُلَاࢠࣹ إࣹ مٝؤٝنْ.

(17) فَوَاكِ مࣹنْ نٝنْ، سِ وَنَا يٝتِّنگٝلْ ࢠَنْنگِنَ".

(18) يِمْࢠࣹ سَارࣹ ندࣹنْ وِئِتِ: "مࣹنْ هِيتٝرِ أٝنْ كَتَا. سِ تَوِي نٝنْ أٝنْ هَطِتَاكِ، مَ مࣹنْ فِطُ أٝنْ، لࣹپْتࣹ مُوسُطࣹ كَدِ مࣹمِرَ أٝنْ إِمّٝرْدࣹ إࣹ أَمࣹنْ".

(19) نُلَاࢠࣹ ࢠࣹنْ كَدِ وِئِتِ: "كِيتࣹ مٝنْ طِنْ، هِطٝنْ وٝنْندِ إࣹ مُنْ. كٝ سِ أٝنْ وَاجَامَ (هِيتِنٝطٝنْ)? تِگِ أٝنْ، كٝ أٝنْ يِمْࢠࣹ فَنْتِنٝوࢠࣹ!".

(20) گٝرْكٝ إِوْرِ كَ ࢠُرِ وٝطُّدࣹ إࣹ سَارࣹ ندࣹنْ، أٝ أَرِ هِمٝ يَݖّٝو، أٝ وِئِ: "كٝ أٝنٝنْ يٝ يِمْࢠࣹ أَمْ، كٝرِ جٝكّࣹي نُلَاࢠࣹ ࢠࣹنْ،

(21) جٝكّࣹي ࢠࣹنْ ࢠࣹ لَمْندٝتَاكٝ أٝنْ نجٝࢠْدِ، ࢠࣹنْ كٝ فࣹوْنَاࢠࣹ.

(22) إࣹ كٝ هَطَتَ لَمْ رࣹوُگٝلْ أٝنْ تَگُطٝ لَمْ? كٝ كَ مَكّٝ وٝنِ رُتّٝرْدࣹ مٝنْ.

(23) إࣹ مِي جٝگْتٝتٝ رࣹوࣹتࣹيطِ گَانِنْ مَكّٝ? سِ جٝمْ-يُرْمࣹيندࣹ أٝنْ مُويَنِي لَمْ لٝرَّ، تࣹفٝورࣹ مَجِّ دُنْݖَنْتَالِ لَمْ هَيْ هُوندࣹ، طِ دْطَندَتَا لَمْ كَدِ.

(24) هَرَيْ مِينْ أٝنْتُمَ، مِي وٝنِي إࣹ مَجّࣹرࣹ ࢠَنْنگُندࣹ.

(25) مِينْ أٝنْ، مِي گٝمْطِنِي جٝومِ مٝنْ، نَنَنࣹي لَمْ".

(26) أٝ وِئَنَا: "نَاتُ الجنة". أٝ وِئِ: "هَانِينٝ يِمْࢠࣹ أَمْ ࢠࣹنْ أَنْندَ،

(27) كٝ جٝومِ أَمْ هَفٝرَنِ لَمْ، أٝ وَطِ لَمْ إࣹ تࣹدِّنَاࢠࣹ ࢠࣹنْ".

(28) مࣹنْ جِپِّنِرَانٝ كٝنࣹيلِ كَ كَمُّ، إࣹ يِمْࢠࣹ مَكّٝ ࢠࣹنْ، ࢠَاوٝ مَاكٝ. مࣹنْ وٝنَانِي جِپِّنٝوࢠࣹ طِنْ كٝنࣹيلِ .

(29) (هَلْكࣹيدࣹ مَࢠّࣹ) وٝنَالِ، سِ وَنَا هَاݖَانگٝ وٝوتٝ، تَوِ ࢠࣹ لَاتِي مَايْࢠࣹ-ࢠُتِّ.

(30) إࣹي هَيْرࣹيدࣹ جࣹيَاࢠࣹ فࣹنُنٝوࢠࣹ ! نُلَاطٝ أَرَالِ إࣹ ࢠّࣹ، سِ وَنَا ࢠࣹ وَطْتَيْنٝ مٝ جَلْنٝرِ.

(31) إࣹ ࢠࣹي يِئَالِ طُوطِي إࣹ گِرࣹ طࣹ مࣹنْ هَلْكِ أَدٝ مَࢠّࣹ? وٝنْندࣹ ࢠࣹنْ رُتِّتٝتَاكٝ إࣹ مَࢠّࣹ.

(32) ࢠࣹ فٝوْ وٝنَالِ، سِ وَنَا كٝ مٝوࢠٝيْتࣹيࢠࣹ كَ أَمࣹنْ.

(33) هِنٝ وٝنَنِ ࢠࣹ مَاندࣹ: لࣹيْدِ يٝورُندِ ندِنْ، مࣹنْ وُورْنِتِ ندِ، مࣹنْ يَلْتِنِ گَبّࣹ إࣹ مَيْرِ، تَوِ كٝ إࣹ طࣹنْ ࢠࣹ ݧَامَتَ.

(34) مࣹنْ وَطِ إࣹ مَيْرِ گࣹسࣹ تَمَرٝوجࣹ إࣹ إِنَبُوجࣹ، مࣹنْ ࢠُلْنِ إࣹ مَيْرِ ࢠُلِّ(ݖࣹيوْلࣹ)

(35) فِي نٝ ࢠࣹ ݧَامِرَ دِمْطࣹ مُنْ طࣹنْ إࣹ كٝنْ كٝ ࢠࣹ گٝلِّرِ جُوطࣹ مَࢠّࣹ طࣹنْ. إࣹ ࢠࣹ يࣹتَّتَا (الله)?

(36) سࣹنَادࣹ وٝودَنِي تَگُطٝ نٝونࣹيجِ طِنْ أٝنْ فٝوْ: گِلَ إࣹ كٝ لࣹيْدِ ندِنْ فُطِنْتَ كٝنْ، هَا إࣹ وٝنْكِيجِ مَࢠّࣹ طِنْ، إࣹ كٝنْ كٝ ࢠࣹ أَنْندَا!

(37) مَاندࣹ گٝو كَدِ وٝنَنِي ࢠࣹ: إࣹ جࣹمَّ أٝنْ، مࣹطࣹنْ سٝرْتَ إࣹ مَكّٝ ݧَلٝرْمَ أٝنْ، تَوَ ࢠࣹ وٝنْتِي إࣹ نِوْرࣹ.

(38) إࣹ نَانگࣹ نگࣹنْ، هِنگࣹ دٝگَ هَا إࣹ ݧِيࢠِرْدࣹ مَگّࣹ; طُمْ كٝ إࣹࢠّٝورࣹ فٝولُطٝ، أَنْندُطٝ أٝنْ.

(39) إࣹ لࣹوْرُ ندُنْ، مࣹنْ إࣹࢠَّنِ ندُ جِپّٝرْدࣹ، هَا ندُ وَطْتِنِرَ وَ بَاݝَلْ تَمَرٝ كِيطْنگَلْ.

(40) وَنَا نَانگࣹ نگࣹنْ وٝنِ كٝ هَانِنِ هࣹوْتِتَگٝلْ لࣹوْرُ ندُنْ، وَنَا جࣹمَّ أٝنْ وٝنِ كٝ أَدِتٝتٝو ݧَلٝرْمَ أٝنْ; كَلَ مَجِّ نٝنْ، كٝ إࣹ لَاوٝلْ مُنْ وٝنِ دٝگُدࣹ.

(41) مَاندࣹ گٝو وٝنَنِي ࢠࣹ، وٝنْندࣹ مࣹنْ رٝوندِكࣹ گࣹيْنگٝلْ مَࢠّࣹ نگٝلْ، كَ لَانَ هࣹبِّنَاكَ;

(42) مࣹنْ تَگَنِ ࢠࣹ سُگُ مَكّٝ، كٝ ࢠࣹ وَطّٝو.

(43) سِ مࣹطࣹنْ يِطُنٝ، مࣹنْ يٝولَيْنٝ ࢠࣹ; هَرَ فَابٝ أَلَنَا ࢠࣹ، وَنَا كَدِ ࢠࣹ دَطْندࣹتࣹيࢠࣹ،

(44) سِ وَنَا يُرْمࣹيندࣹ إِمّٝرْندࣹ إࣹ أَمࣹنْ، إࣹ دَكْمِتَگٝلْ هَا إࣹ سَانگَ.

(45) سِ ࢠࣹ وِئَنَامَ: "رࣹنْتࣹي إࣹ كٝنْ كٝ وٝنِ يࣹيسٝ مٝنْ إࣹ ࢠَاوٝ مٝنْ، بࣹلَجٝؤٝ، أٝنْ يُرْمِنْتࣹ"...!

(46) آيَ وٝو إࣹ آيࣹيجࣹ جٝومِ مَࢠّࣹ طࣹنْ أَرَالِ إࣹ مَࢠّࣹ، سِ وَنَا ࢠࣹ طُورْنٝوطࣹ.

(47) سِ ࢠࣹ وِئَنَامَ كَدِ: "وِنْتࣹي(نَفْقࣹ) إࣹ كٝنْ الله أَرْسِكِ أٝنْ"، يࣹدُّࢠࣹ ࢠࣹنْ وِئَنَ گٝمْطِمْࢠࣹ ࢠࣹنْ: "إࣹ مࣹنْ ݧَمِّنَيْ أٝنْ تَوْطٝ سِ الله مُويُنٝ، أٝ ݧَمِّنَيْنٝ مٝ? أٝنْ وٝنَالِ، سِ وَنَا إࣹ ندࣹرْ مَجّࣹرࣹ ࢠَنْنگُندࣹ".

(48) هِࢠࣹ وِئَ: "كٝ هٝنْندࣹ تُمَ وٝنِ ندࣹي فٝدٝورࣹ، سِ أٝنْ لَاتِكࣹ گٝونگُࢠࣹ?"

(49) ࢠࣹ هَبَّاكِ، سِ وَنَا هَاݖَانگٝ وٝوتٝ، نَنْنگَيْنگٝ ࢠࣹ، هَرَ هِࢠࣹ يࣹدّٝندِرَ.

(50) ࢠࣹ وَاوٝيْتَا وَسِيَگٝلْ أٝنْتُمَ، وَنَا ࢠࣹ رُتّٝيْتٝوࢠࣹ إࣹ يِمْࢠࣹ مَࢠّࣹ ࢠࣹنْ.

(51) وُتّࣹ كَ أَلَاندُ، ࢠࣹ يَلْتِرَ إࣹ گࣹنَالࣹ طࣹنْ، ࢠࣹ هࣹݧٝرَ كَ جٝومِ مَࢠّࣹ،

(52) هِࢠࣹ وِئَ: "إࣹي بٝنࣹ أَمࣹنْ! كٝ هٝمْنبٝ إِمِّنِ مࣹنْ إࣹ گࣹنَالࣹ أَمࣹنْ?" كٝ طُمْ وٝنِ كٝ هِنَرَنْكࣹ أٝنْ فٝدُنٝو; نُلَاࢠࣹ ࢠࣹنْ گٝونگِنِي.

(53) (أُمِّتَلْ) نگَلْ وٝنَانٝ، سِ وَنَا هَاݖَانگٝ گٝوتٝ، تَوِ ࢠࣹ فٝوْ ࢠࣹ أَدَّامَ كَ أَمࣹنْ.

(54) هَنْندࣹ نٝنْ، وٝنْكِي تٝوݧِرْتَاكࣹ هَيْ هُوندࣹ. أٝنْ يٝࢠِتِرْتَاكࣹ، سِ وَنَا كٝنْ كٝ گٝلَّيْنٝطٝنْ.

(55) يِمْࢠࣹ الجنة ࢠࣹ هَنْندࣹ، پࣹلّࣹتْ، هِࢠࣹ إࣹ ندࣹرْ سٝكْلِــٔٝ، هِࢠࣹ نࣹيمِنٝو;

(56) كَمْࢠࣹ إࣹ گࣹنْندِرَاࢠࣹ مَࢠّࣹ ࢠࣹنْ،أࣹ لࣹيْ طٝوْطࣹ، هِࢠࣹ سٝࢠِّندِي إࣹ هٝورࣹ دَنْطࣹ.

(57) هِنٝ وٝودَنِ ࢠࣹ تٝنْ، ࢠِطْࢠࣹ لࣹطّࣹ، هِنٝ وٝودَنِ كَدِ، كٝنْ كٝ ࢠࣹ نٝدَّتَ.

(58) "سَلَامُنْ"، كٝنْنگٝلْ كِسِيࣹي ، إِمّٝرْدࣹ إࣹ جٝومِرَطࣹ، هِنّٝتٝوطٝ.

(59) "سࣹرْتࣹي نٝنْ هَندࣹ، أٝنٝنْ ࢠࣹي بٝمْࢠࣹ!

(60) مِي أَهٝدَانٝ إࣹ مٝنْ، أٝنٝنْ ࢠِطْࢠࣹ آدَمَ، وٝنْندࣹ وَتَ أٝنْ رࣹوُ سࣹيْتَانࣹ? تَوْدࣹ پࣹلّࣹتْ، كٝ أٝ گَݧٝ مٝنْ ࢠَنْنگُطٝ،

(61) مِي يَمِرِ أٝنْ كَدِ رࣹوُگٝلْ لَمْ. كٝ نگٝلْ وٝنِ لَاوٝلْ فٝݖِّينگٝلْ نگٝلْ.

(62) گٝمْطِي أٝ مَجِّنِي تَگُ طُوطُنگُ إࣹ مٝنْ. إࣹ أٝنْ وٝنَانٝ هَقِّلٝوࢠࣹ?

(63) كٝ نگࣹي وٝنِ جهنم، نگࣹ وٝنَنٝطٝنْ فٝدَنࣹيدࣹ نگࣹنْ.

(64) نَاتِرࣹي نگࣹ هَنْندࣹ، سَبُ كٝنْ كٝ يࣹدِّرَيْنٝطٝنْ.

(65) هَنْندࣹ، مࣹنْ نٝتَيْ إࣹ كُنْندُلࣹ مَࢠّࣹ طࣹنْ، جُوطࣹ مَࢠّࣹ طࣹنْ يࣹوْتَ مࣹنْ، كٝيْطࣹ مَࢠّࣹ طࣹنْ كَدِ سࣹيدِتٝو إࣹ كٝنْ كٝ ࢠࣹ وٝنُنٝو إࣹ فَگِّتَگٝلْ".

(66) سِ مࣹنْ يِطُنٝ، مࣹنْ مࣹهَيْنٝ گِتࣹ مَࢠّࣹ طࣹنْ، ࢠࣹ أَدٝندِرَ إࣹ لَاوٝلْ نگٝلْ. كٝنٝ كٝ هٝنّٝ ࢠࣹ يِيرَتَ?

(67) سِ مࣹنْ يِطُنٝ كَدِ، مࣹنْ مٝنْتَيْنٝ ࢠࣹ إࣹ نٝكُّ مَࢠّࣹ أٝنْ; هَرَ ࢠࣹ هَتَّنْتَا فࣹڃُّگٝلْ ࢠࣹ يِلْتٝتَاكٝ.

(68) كَلَ مٝ مࣹنْ جُوتْنِ نگُورْندَمْ مُنْ، مࣹنْ يِلْتِتَيْ مٝ إࣹ تَگُ نگُنْ. إࣹ ࢠࣹي هَقِّلْتَا?

(69) مࣹنْ أَنْندِنَالِ مٝ (كَنْكٝ مُهَمَّدُ) گِمْطِ; طُمْ هَانَنَا مٝ. دࣹفْتࣹرࣹ ندࣹنْ وٝنَالِ، سِ وَنَا جَنْتٝورࣹ إࣹ جَنْنگࣹتࣹيندࣹ ࢠَنْنگُندࣹ،

(70) فِي يٝ أٝ جࣹرْتِنْ ندࣹ، أٝنْ وٝنْطٝ وُورْطٝ (ࢠࣹرْندࣹ)، كٝنْنگٝلْ لࣹپْتࣹ نگٝلْ جٝجَّ إࣹ هࣹيفࣹرْࢠࣹ ࢠࣹنْ.

(71) إࣹ ࢠࣹ يِئَالِ، وٝنْندࣹ مࣹنْ تَگِرَنِي ࢠࣹ نࣹيمٝرَاطِ، إࣹ كٝ جُوطࣹ أَمࣹنْ گٝلِّ، هَرَ هِࢠࣹ جࣹيِنٝوطِ;

(72) مࣹنْ إࣹلْتَنِ ࢠࣹ طِ: نٝ إࣹ مَجِّ طِ ࢠࣹ وَطّٝتٝو، نٝ إࣹ مَجِّ طِ ࢠࣹ ݧَامَتَ;

(73) هِنٝ وٝودَنِ ࢠࣹ تٝنْ، نَفٝوجࣹ إࣹ نجَرَمْ. إࣹ ࢠࣹي كَنْ يࣹتَّتَا (مٝ)?

(74) ࢠࣹ جٝگِتٝرِي گَانِنْ الله رࣹوࣹتࣹيطِ، جٝرْتَگٝلْ ࢠࣹ فَابࣹيدࣹي.

(75) طِ هَتَّنَا فَابَگٝلْ ࢠࣹ; كٝ ࢠࣹ كٝنُ تَوْنࣹتࣹينگُ إࣹ مَجِّ.

(76) وَتَ كٝنْنگٝلْ مَࢠّࣹ سُنِنࣹ. پࣹلّࣹتْ، مࣹطࣹنْ أَنْندِ كٝ ࢠࣹ سُوطَتَ إࣹ كٝ ࢠࣹ فࣹݧِّنْتَ.

(77) إࣹ نࣹطَّنْكࣹ أٝنْ يِئَالِ، وٝنْندࣹ مࣹنْ تَگِرِي مٝ إࣹ تٝࢠّࣹرࣹ مَنِييُ? كٝنٝ تَوَا مٝ كٝ أٝ وࣹنْنجٝووٝ ࢠَنْنگُطٝ!

(78) أٝ فِيَنِ مࣹنْ مِسَلْ، كٝنٝ أٝ يࣹجِّتِ تَگࣹيدࣹ مَكّٝ; أٝ وِئِ: "كٝ هٝمْنبٝ وُورْنِتَتَ ڃِئࣹ طࣹنْ، سِ طࣹ رَپِّي?"

(79) مَاكُ: "كٝ وُورْنِتٝووٝ طࣹ، كٝ تَگُطٝ طࣹ أٝنْ كَ لَاوٝلْ أَرَنٝلْ. كٝ كَنْكٝ وٝنِ أَنْندُطٝ فِي كَلَ تَگُ;

(80) أٝنْ وَطَنْطٝ أٝنْ يِيتࣹ، إِوْدࣹ إࣹ لࣹطّࣹ كࣹݖّࣹ، تَوَ كٝ إࣹ مَجّࣹ هُࢠِّرْتٝنْ نگࣹ.

(81) كࣹرࣹ تَگُطٝ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ وٝنَالِ هَطْتَنْطٝ تَگُگٝلْ يࣹرُ مَࢠّࣹ? طُمْ كٝ پࣹلّࣹتْ! كٝ كَنْكٝ وٝنِ طُوطُطٝ تَگࣹيفٝ، أَنْندُطٝ.

(82) أَنْندٝنْ فِيَاكُ مَكّٝ أٝنْ، سِ أٝ فَاندَنِكࣹ هُوندࣹ، كٝ أٝ وِئَنَيْ ندࣹ تُنْ: "لَاتٝ"، ندࣹ لَاتٝو.

(83) سࣹنَيࣹي وٝودَنِي أٝنْ مٝ لَامَتࣹيرِ كَلَ هُوندࣹ وٝنِ إࣹ جُوطࣹ مُنْ! كٝ كَ مَكّٝ رُتّࣹتࣹطٝنْ.

37 As-Saffat — 182 آيَة

(1) مِي وٝوندِرِي سَفِّنِرَاطِ طِنْ سَفَّاجِ.

(2) مِي وٝوندِرِي سٝگِّرَيْطِ (دُولࣹ) طِنْ سٝگُّگٝلْ.

(3) مِي وٝوندِرِي جَنْنگِرَيْطِ طِنْ، أَنْندِنْتِنٝرْگٝلْ:

(4) "پࣹلّࣹتْ، رࣹوࣹتࣹيطٝ مٝنْ أٝنْ، كٝ بَجّٝ،

(5) جࣹيْطٝ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، إࣹ كٝنْ كٝ وٝنِ هَكُّندࣹ طِنْ طِطِ، جࣹيْطٝ كَدِ فُطِرْطࣹ طࣹنْ.

(6) مࣹنْ ݖُطِرِي كَمُّ (هِكِّي نگُ) أَدُنَ نگُنْ: ݖُطَارِ كٝودࣹ طࣹنْ،

(7) إࣹ رࣹينِرْگٝلْ (نگُ) كَلَ سࣹيْتَانࣹ رَطَاندُ.

(8) طِ هَطْتَنَاكَ هࣹطِتَگٝلْ نبَتُ الملائكة تٝووُ نگُ نگُنْ; تَوْدࣹ هِطِ فࣹرٝرࣹي إࣹ كَلَ بَݝّࣹ،

(9) كَ رَطَگٝلْ (طِ). هِنٝ وٝودَنِ طِ كَدِ، لࣹپْتࣹ دُومِيطࣹ.

(10) سِ وَنَا ندُنْ بِفْتِ ندُ إࣹ مَجِّ، جِلَنْندࣹ هَالَ ، ندُ رࣹوْنِرࣹي هٝودࣹرࣹ تُفَيْندࣹ.

(11) لَمْندٝ ࢠࣹ: سِ كٝ كَمْࢠࣹ ࢠُرِ سَطْتُدࣹ تَگُدࣹ كَا كٝ ࢠࣹنْ ࢠࣹ مࣹنْ تَگِ? مࣹنْ تَگِرِي ࢠࣹ كَمْࢠࣹ، لٝوپَلْ نَرْنُنگَلْ.

(12) سِ كٝ وٝنِ، أَ هَاوِ، هِࢠࣹ جَلْنٝرَ!

(13) سِ ࢠࣹ وَاجَامَ، ࢠࣹ وَاجِتٝرْتَا;

(14) سِ ࢠࣹ يِئِي مَاندࣹ، ࢠࣹ جَلْكِتَ،

(15) ࢠࣹ وِئَ: "طُمْ وٝنَالِ، سِ وَنَا نبِلࣹوُ ࢠَنْنگُ نگُ.

(16) إࣹيٝو سِ مࣹنْ مَايِي، مࣹنْ وٝنْتِي نبُلَّارِ إࣹ ڃِئࣹ، هَرَيْ كٝ مࣹنْ إِمِّنْتِنٝيْتࣹيࢠࣹ?"

(17) بَابِرَاࢠࣹ أَمࣹنْ أَدِنٝوࢠࣹ ࢠࣹنْ كَدِ إࣹ مِنࣹنْ?"

(18) مَاكُ: "پࣹلّࣹتْ! هَرَ لࣹ كٝ أٝنْ هٝيْنَاࢠࣹ".

(19) أَنْندٝنْ طُمْ، كٝ وُتَّاندُ وٝوتُرُ پࣹتْ، تَوࣹي جَكَ يٝو، هِࢠࣹ ندَارَ،

(20) ࢠࣹ وِئَ: إࣹي بٝنࣹ أَمࣹنْ ! كٝ ندࣹي وٝنِ ݧَلَاندࣹ نجٝࢠْدِ كُوطࣹ".

(21) "كٝ ندࣹي وٝنِ ݧَلَاندࣹ ݧَاوٝورࣹ، كٝ فࣹنَّيْنٝطٝنْ فِي مُنْ".

(22) مٝوࢠࣹي تٝوݧُࢠࣹ ࢠࣹنْ، إࣹ نَندَدِيࢠࣹ مَࢠّࣹ ࢠࣹنْ، إࣹ كٝنْ كٝ ࢠࣹ وٝنُنٝو إࣹ رࣹوْدࣹ،

(23) كٝ وٝنَا الله. سٝگِّرࣹيࢠࣹ كَ لَاوٝلْ جَهِيمِ.

(24) دَرْنٝنْ ࢠࣹ، تَوْدࣹ كٝ ࢠࣹ لَمْندِتࣹتࣹيࢠࣹ".

(25) "كٝ هࣹࢠِ أٝنْ، أٝنْ وَلِّندِرْتَا?"

(26) كٝ وٝنِ كَمْࢠࣹ أٝنْ هَنْندࣹ، كٝ ࢠࣹ جࣹبِّلِي ࢠࣹ،

(27) وٝطْࢠࣹ إࣹ مَࢠّࣹ فࣹوْتِتَ ࢠࣹيَ، ࢠࣹ لَمْندٝندِرَ;

(28) ࢠࣹ وِئَ: "كٝ أٝنٝنْ وٝنِ كٝ أَرْدِرَنَيْنٝو مࣹنْ، كَ ࢠَݝّࣹ ݧَامٝ".

(29) (كٝوهٝوࢠࣹ) ࢠࣹنْ وِئَ: "كٝ وٝنِ كٝنْ، أٝنْ مࣹيطَانٝ وٝنْندࣹ گٝمْطِمْࢠࣹ.

(30) دٝولࣹ وٝو مࣹنْ أَلَنَانٝ أٝنْ. سِ كٝ وٝنِ، أٝنْ وٝنُنٝ يِمْࢠࣹ بࣹوْࢠࣹ.

(31) كٝنْنگٝلْ جٝومِ مࣹنْ نگٝلْ، جٝجِّي إࣹ مࣹنْ; پࣹلّࣹتْ، كٝ إࣹنْ مࣹيطٝوࢠࣹ لࣹپْتࣹ طࣹنْ .

(32) مࣹنْ بࣹوِنّٝ أٝنْ، تَوْدࣹ هَيْ مࣹنࣹنْ، كٝ مࣹنْ بࣹوْنٝوࢠࣹ".

(33) كَمْࢠࣹ أٝنْ ندࣹنْ ݧَلْندࣹ، كٝ ࢠࣹ هَوْتٝيْتࣹيࢠࣹ كَ لࣹپْتࣹ.

(34) كٝ نِي مࣹنࣹنْ مࣹنْ هُووِرْتَ بٝمْࢠࣹ ࢠࣹنْ.

(35) كَمْࢠࣹ سِ ࢠࣹ وِئَنٝومَ: "رࣹوࣹتࣹيطٝ أَلَا سِ وَنَا الله"، ࢠࣹ مَوْنِنْتِنٝو.

(36) ࢠࣹ وِئَ: "كࣹرࣹ مࣹنْ أَݖِّتِرَيْ رࣹوࣹتࣹيطِ أَمࣹنْ طِنْ، فِي يِمٝووٝ هَانگَاطٝ?"

(37) سِ كٝ وٝنِ، (نُلَاطٝ) أٝنْ كٝ گٝونگَ أَرْدِ، أٝ گٝونگِنِ نُلَاࢠࣹ ࢠࣹنْ.

(38) پࣹلّࣹتْ، كٝ أٝنْ مࣹيطَيْ ࢠࣹ لࣹپْتࣹ مُوسُطࣹ طࣹنْ.

(39) أٝنْ يٝࢠِتࣹتَاكࣹ، سِ وَنَا يࣹرُ كٝنْ كٝ هُووَيْنٝطٝنْ،

(40) سِ وَنَا جࣹيَاࢠࣹ الله لَࢠِّنَاࢠࣹ ࢠࣹنْ،

(41) هِنٝ وٝودَنِ ࢠࣹنْ طٝنْ، فَنْندَارِ أَنْندَاندِ:

(42) مُويَنْتࣹيطِ، كَدِ كٝ ࢠࣹ تࣹدِّنَاࢠࣹ،

(43) كَ الجنة نࣹيمِنِيطِ،

(44) إࣹ هٝورࣹ دَنْطࣹ، كٝ ࢠࣹ فࣹوْتٝندِرْࢠࣹ.

(45) هَرَ كٝرࣹ هِنٝ وَنْنگِنِي إࣹ مَࢠّࣹ، إِمّٝرْدࣹ إࣹ ࢠُلِرْدࣹ

(46) دَنࣹيرࣹ، لࣹوْݧُندࣹ، فِي يَرٝوࢠࣹ،

(47) مُوسُ هٝورࣹ أَلَا إࣹ مَجَّمْ، وَنَا طَمْ سِوْرِنِرْتࣹيࢠࣹ طَمْ.

(48) هِنٝ وٝودَنِ ࢠࣹ (تٝنْ): رَࢠِّدُࢠࣹ ندَارْطࣹ، يَاجَ گِتࣹيࢠࣹ،

(49) وَؤُࢠࣹ وَ بٝوفٝ مَرَاطࣹ.

(50) وٝطْࢠࣹ إࣹ مَࢠّࣹ فࣹوْتِتَ ࢠࣹيَ، ࢠࣹ لَمْندٝندِرَ.

(51) گٝطّٝ إࣹ مَࢠّࣹ وِئَ: "مِينْ مِطُ مَرُنٝو وٝنْندِيطٝ،

(52) وِئَيْنٝوطٝ: "كࣹرࣹ كٝ أَ جࣹيَاطٝ إࣹ گٝوگِمْࢠࣹ ࢠࣹنْ?

(53) إࣹيٝو سِ إࣹنْ مَايِي، إࣹنْ وٝنْتِي نبُلَّارِ إࣹ ڃِئࣹ، هَرَيْ كٝ إࣹنْ يٝࢠِتٝيْتࣹيࢠࣹ?"

(54) أٝ وِئَ: "تَوْ سِ طٝنْ يِطِ ڃࣹيوُدࣹ?"

(55) أٝ ڃࣹيوَ، أٝ هَيْنٝو أٝيَ كَ تُمْنبٝ دُلْبُنگࣹ،

(56) أٝ وِئَ: "مِي وٝوندِرِي الله ! أَ إࣹࢠُّنٝ فِي نٝ مِي هَلْكٝرْدࣹ!

(57) سِ وَنَانٝ مٝڃّࣹرࣹ جٝومِ أَمْ ندࣹنْ، مِي تَوْدࣹتࣹنٝ إࣹ لࣹپْتࣹتࣹيࢠࣹ ࢠࣹنْ .

(58) وٝنَا نٝنْ إࣹنࣹنْ مَايَتَا،

(59) سِ وَنَا مَايْدࣹ مࣹنْ أَرَنࣹرࣹ ندࣹنْ، إࣹنْ وٝنَالِ كَدِ لࣹپْتࣹتࣹيࢠࣹ?"

(60) پࣹلّࣹتْ، كٝ طُمْ طٝو وٝنِ مَلٝورࣹ مَوْنُندࣹ ندࣹنْ.

(61) كٝ فِي سُگُ طُمْ طٝو، وٝنِ يٝ گٝلّٝوࢠࣹ گٝلَّنْ.

(62) هَرَ كٝ طُمْ ࢠُرِ مٝڃُّدࣹ جِپّٝرْدࣹ، كَا كٝ لࣹگَّلْ "زَقُّومِ" نگَلْ هَاطُنگَلْ ?

(63) مࣹنْ وَطِي نگَلْ جَرَّبُيࣹي تٝوݧُࢠࣹ ࢠࣹنْ.

(64) نگَلْ، كٝ لࣹگَّلْ يَلْتَيْنگَلْ كَ تُمْنبࣹرࣹ دُلْبُنگࣹ.

(65) دِمْطࣹ مَگَّلْ، هِنٝ وَئِ وَ كٝؤࣹ سࣹيْتَنٝوجِ.

(66) كَمْࢠࣹ أٝنْ، پࣹلّࣹتْ، كٝ إࣹ مَگَّلْ ࢠࣹ نْيَامِرْتَ، ࢠࣹ هࣹبِّنِرَ نگَلْ دࣹيدِ مَࢠّࣹ.

(67) رࣹفْتِ هِنٝ وٝودَنِ ࢠࣹ إࣹ دٝوْ مَگَّلْ، نجَرَمْ فَتِنَاطَمْ.

(68) رࣹفْتِ رُتّٝرْدࣹ مَࢠّࣹ ندࣹنْ، كٝ كَ دُلْࢠٝووࣹ.

(69) ࢠࣹ تَوُ بَابِرَاࢠࣹ مَࢠّࣹ ࢠࣹنْ، كٝ مَجُّنٝوࢠࣹ،

(70) ࢠࣹ وٝنِ إࣹ يَݖَّگٝلْ جٝكَّ بَطْتࣹيجِ مَࢠّࣹ.

(71) گٝمْطِي، كٝ ࢠُرِ طُوطُدࣹ إࣹ أَرَمْࢠࣹ ࢠࣹنْ، مَجِّينٝ، أَدٝ مَࢠّࣹ.

(72) گٝمْطِي لࣹ، مࣹنْ نُلُنٝ إࣹ مَࢠّࣹ، جࣹرْتِنٝوࢠࣹ.

(73) كٝنٝ ندَارُ بَطْتَنࣹ ࢠࣹنْ جࣹرْتِنَنٝوࢠࣹ، نٝ وَؤُنٝو!

(74) سِ وَنَا جࣹيَاࢠࣹ الله، لَࢠِّنَاࢠࣹ ࢠࣹنْ.

(75) گٝمْطِي نوح نٝدُّنٝ مࣹنْ، مࣹنْ مٝڃِّي جَابٝتٝوࢠࣹ.

(76) مࣹنْ دَطْنِ مٝ، كَنْكٝ إࣹ يِمْࢠࣹ مَكّٝ ࢠࣹنْ، إࣹ سُنٝ مَوْنُنگٝ نگٝنْ،

(77) مࣹنْ وَطِ گࣹيْنگٝلْ مَكّٝ نگٝلْ، هࣹدّٝتٝوࢠࣹ ࢠࣹنْ.

(78) مࣹنْ أَݖَّنِ مٝ مَنْتٝورࣹ، إࣹ سَكِّتِيࢠࣹ ࢠࣹنْ.

(79) كِسَلْ وٝنِي إࣹ نوح، إࣹ ندࣹرْ وِنْندࣹرࣹ ندࣹنْ.

(80) كٝ نٝنْ وٝنِ كٝ مࣹنْ وَرْجٝرْتَ مٝڃِّمْࢠࣹ ࢠࣹنْ.

(81) كَنْكٝ، كٝ أٝ جࣹيَنٝوطٝ إࣹ جࣹيَاࢠࣹ أَمࣹنْ گٝمْطِمْࢠࣹ ࢠࣹنْ.

(82) رࣹفْتِ مࣹنْ يٝولِ هࣹدِّي ࢠࣹ ࢠࣹنْ.

(83) هِنٝ جࣹيَنٝو إࣹ جٝكُّࢠࣹ مٝ ࢠࣹنْ، إبراهيم.

(84) تُمَ ندࣹ أَرْدُنٝو كَ جٝومِ مَكّٝ، إࣹ ࢠࣹرْندࣹ هِسُندࣹ.

(85) ندࣹ أٝ مَاكَنّٝو بࣹنْ مَكّٝ إࣹ يِمْࢠࣹ مُنْ ࢠࣹنْ: "كٝ هٝنْطُمْ رࣹوٝتٝنْ?

(86) إࣹ كٝ رࣹوࣹتࣹيطِ فࣹفِندَاطِ، كٝ وَنَا الله، فَالَطٝنْ?

(87) كٝ سِكُّطٝنْ إࣹ جٝومِرَاوٝ تَگࣹيفٝ نگٝنْ?"

(88) أٝ ندَارِرِ ندَارَندࣹ كَ كٝودࣹ،

(89) أٝ مَاكِ: "كٝ مِي نَوْنُطٝ".

(90) ࢠࣹ أَݖِّ مٝ، ࢠࣹ هُݖِّتِ.

(91) أٝ مٝلْتِي إࣹ رࣹوࣹتࣹيطِ مَࢠّࣹ طِنْ، أٝ مَاكِ: "إࣹ أٝنْ ݧَامَتَا?

(92) كٝ هࣹࢠِ أٝنْ، أٝنْ وٝوْلَتَا?"

(93) أٝ مٝلْتٝرَنِ طِ، پِيِرْگٝلْ طِ ݧَامٝ نگٝنْ.

(94) ࢠࣹ فࣹوْرِ كَ مَكّٝ، هِࢠࣹ يَاوْنٝ.

(95) أٝ مَاكِ: "إࣹ أٝنْ رࣹوَيْ كٝنْ كٝ لࣹسُطٝنْ،

(96) هَرَ لࣹ كٝ الله تَگِ أٝنْ، إࣹ كٝنْ كٝ گٝلّٝتٝنْ?"

(97) ࢠࣹ وِئِ: "مَهَنࣹي مٝ مَادِ، بُگٝطٝنْ مٝ كَ دُلْࢠَيَنگࣹ".

(98) ࢠࣹ فَاندَنِي مٝ فࣹيرࣹ; كٝنٝ مࣹنْ وَطِ ࢠࣹ سٝويُبࣹ ࢠࣹنْ.

(99) أٝ مَاكِ: "مِينْ أٝنْ، كٝ مِي فࣹرٝووٝ كَ جٝومِ أَمْ، أَرَيْ ندࣹ أٝ فࣹوْنَتَ لَمْ.

(100) جٝومِ أَمْ، يࣹطَمْ گࣹࢠَلْ مٝڃُّࢠࣹ".

(101) مࣹنْ وࣹوْلِنِرِ مٝ، گࣹطَلْ مُݧِّيطٝ.

(1) سَادْ. مِي وٝوندِرِي القرآن، جٝمْ وَاجُ !

(6) نبَتُ مَࢠّࣹ نگُنْ هُݖِّتِرِ وِيدࣹ: "يࣹهࣹي، تَبِتٝنْ إࣹ رࣹوࣹتࣹيطِ مٝنْ طِنْ. طُمْ طٝو، كٝ هُوندࣹ إࣹࢠَّنَاندࣹ.

(8) كࣹرࣹ كٝ إࣹ مَكّٝ جَنْتٝورࣹ ندࣹنْ جِپّٝتٝو هَكُّندࣹ أَمࣹنْ?" سِ كٝ وٝنِ، ࢠࣹ وٝنِي إࣹ سِكِّتَگٝلْ جَنْتٝورࣹ أَمْ ندࣹنْ. وُرِنْ، كٝ فِي ࢠࣹ مࣹيطَالِ تَهٝ، لࣹپْتࣹ أَمْ طࣹنْ!

(16) ࢠࣹ وِ: "جٝومِ أَمࣹنْ، يَݖِّنَنْ مࣹنْ گࣹࢠَلْ لࣹپْتࣹ أَمࣹنْ، أَدٝ ݧَلَاندࣹ هَسْبٝورࣹ ندࣹنْ".

(17) مُݧّٝ إࣹ كٝنْ كٝ ࢠࣹ وِئَتَ; أَنْندِتَا جࣹيَاطٝ أَمࣹنْ أٝنْ، دَاوُودَ، جٝمْ دٝولࣹ. كٝ أٝ طُوطُنٝوطٝ تُوبُگٝلْ.

(18) مࣹنْ إࣹلْتِي پࣹلّࣹ طࣹنْ، هِطࣹ سُبُّنْهِنٝدَ إࣹ مَكّٝ، كَ كِيكِيطࣹ إࣹ كَ وࣹيتُگٝلْ.

(19) إࣹ ݖٝلِّ طِنْ، هِطِ مٝوࢠِندِرَا، كَلَ إࣹ مَجِّ نٝ تُوبَنَ مٝ.

(20) مࣹنْ سࣹمْنبِنِ لَامُ مَكّٝ نگُنْ، مࣹنْ يࣹطِ مٝ أَنَّبَاكُ إࣹ فَسِنْتࣹيگَلْ.

(21) هَرَ كُمْپِتَلْ وࣹنجُنٝوࢠࣹ ࢠࣹنْ أَرِي مَا، ندࣹ ࢠࣹ ڃَوُنٝو كَ گَنْنگُنَلْ !

(22) ندࣹ ࢠࣹ نَاتُــٔٝو كَ دَاوُودَ، أٝ طࣹنْڃِي ࢠࣹ. ࢠࣹ وِئِ: "وَتَ أَ هُلُ ! كٝ وࣹنْنجُࢠࣹ طِطٝ; گٝطّٝ إࣹ أَمࣹنْ تٝوݧِي أٝيَ. ݧَاوِرْ گٝونگَ هَكُّندࣹ أَمࣹنْ، وَتَ أَ تٝوݧُ، تِنْندِنَا مࣹنْ إࣹ لَاوٝلْ فٝݖِّينگٝلْ نگٝلْ.

(23) أٝو كٝ مُسِطّٝ أَمْ: هِمٝ جٝگِي بَالِ ݖَپَّنْطࣹ جࣹينَيْ إࣹ جࣹينَيْ، مِينْ مِطُ جٝگِي بَالِي گٝوتِي. أٝ وِئِ: "هࣹنْندِنَمْ نگِي"; أٝ فٝولِمِ إࣹ يࣹوْتࣹرࣹ.

(24) أٝ مَاكِ: "پࣹلّࣹتْ، أٝ تٝوݧِرِي مَ لَمْندَگٝلْ بَالِي مَا نگِي، وَطْتُگٝلْ إࣹ بَالِ مَكّٝ طِنْ". طُوطِي كٝ هَوْتُࢠࣹ تٝوݧِندِرْتَ هَكُّندࣹ مَࢠّࣹ، سِ وَنَا ࢠࣹنْ گٝمْطِمْࢠࣹ، ࢠࣹ گٝلِّ مٝڃُّطِ - ࢠࣹنْ نٝنْ نٝ فَنْطِ. - دَاوُودَ فࣹلِّتِ، وٝنْندࣹ مࣹنْ جَرِّبِكࣹ مٝ. أٝ إِنْسِنَنِي جٝومِ مَكّٝ، أٝ سُويِرِ سُجُّگٝلْ، أٝ تُوبِ.

(25) مࣹنْ يَافَنِي مٝ طُمْ. پࣹلّࣹتْ نٝنْ، هِنٝ وٝودَنِ مٝ كٝ أَمࣹنْ، ࢠَدَلْ إࣹ مٝڃُّدࣹ رُتّٝرْدࣹ.

(26) كٝ أَنْ يٝ دَاوُودَ، مࣹنْ وَطِي مَ لٝمْتٝ كَ لࣹيْدِ. ݧَاوِرْ گٝونگَ هَكُّندࣹ يِمْࢠࣹ ࢠࣹنْ، وَتَ أَ جٝكُّ نبࣹلࣹيطࣹ، سَكّٝ طࣹ مَجِّنࣹ إࣹ لَاوٝلْ الله نگٝلْ". پࣹلّࣹتْ، ࢠࣹنْ مَجّٝوࢠࣹ إࣹ لَاوٝلْ الله نگٝلْ، هِنٝ وٝودَنِ ࢠࣹ لࣹپْتࣹ تِيطُطࣹ، سَبُ كٝ ࢠࣹ يࣹجِّتِرِ كٝنْ، ݧَلَاندࣹ هَسْبٝورࣹ ندࣹنْ.

(27) مࣹنْ تَگِرَالِ كَمُّ نگُنْ إࣹ لࣹيْدِ ندِنْ إࣹ كٝ وٝنِ هَكُّندࣹ مَجِّ كٝنْ، مࣹيرࣹ. طُمْ كٝ سِكّࣹ يࣹدُّࢠࣹ ࢠࣹنْ. بٝنࣹ وٝودَنِي ࢠࣹنْ يࣹدُّࢠࣹ، إࣹ يِيتࣹ نگࣹنْ!

(28) كَ مࣹنْ وَطَيْ ࢠࣹنْ گٝمْطِمْࢠࣹ ࢠࣹ گٝلِّ مٝڃِّ، وَنٝ بٝنّٝوࢠࣹ ࢠࣹنْ كَ هٝورࣹ لࣹيْدِ? كَا مࣹنْ وَطَيْ گٝمْطُࢠࣹ ࢠࣹنْ، وَنٝ فَاجِرࣹيࢠࣹ ࢠࣹنْ?

(29) كٝ دࣹفْتࣹرࣹ بَرْكِنَاندࣹ، وٝنِ كٝ مࣹنْ جِپِّنِ إࣹ مَاطَ، فِي نٝ ࢠࣹ تَسْكٝرَ آيࣹيجࣹ مَيْرࣹ، إࣹ كٝ جٝومِرَاࢠࣹ هَقِّلّࣹ ࢠࣹنْ وَاجِتٝرَ.

(30) مࣹنْ يࣹطِ دَاوُودَ سُلَيْمَانَ، - جࣹيَاطٝ مٝڃّٝ - كٝ أٝ طُوطُطٝ تُوبُگٝلْ-رُتّٝو.

(31) جَنْتٝ ندࣹ أٝ هٝلَّنٝو كِيكِيطࣹ، پُݖِّ يٝوطُطِ يَاوُطِ،

(32) أٝ مَاكِ: "مِينْ ࢠُرْنٝرِي يِطُگٝلْ جَوْدِ، دِي كٝ مِي جَنْتٝتٝو جٝومِ أَمْ كٝنْ، هَا نَانگࣹ نگࣹنْ مُتٝيِي.

(33) رُتّࣹي لَمْ طِ". أٝ وٝنِ إࣹ سٝدُگٝلْ طِ كَ تَاپࣹ إࣹ كَ دَاطࣹ.

(34) گٝمْطِي مࣹنْ جَرِّبِكࣹ سُلَيْمَانَ، مࣹنْ وࣹرْلِي إࣹ جُلّࣹرࣹ مَكّٝ ندࣹنْ ࢠَنْندُ. رࣹفْتِ أٝ تُوبِ.

(35) أٝ مَاكِ: "جٝومِ أَمْ، هَفٝرَنَمْ، يࣹطَا لَمْ لَامُ، نگُ هࣹيرْتَنْتَاكٝ گٝطّٝ گٝو، ࢠَاوٝ أَمْ. پࣹلّࣹتْ، كٝ أَنْ وٝنِ يࣹطٝووٝ أٝنْ".

(36) مࣹنْ إࣹلْتَنِ مٝ هࣹنْندُ ندُنْ، تَوِ هِندُ وِفِرَ إࣹ سَگٝ مَكّٝ، نࣹوࣹيندࣹ، نٝكُّ كَلَ كٝ أٝ يِطِ.

(37) إࣹ سࣹيْتَنُوجِ طِنْ، مَهَيْنٝوطِ إࣹ مُتَيْنٝوطِ.

(38) گٝطِّ طِنْ، كٝ دُمْنبَاطِ إࣹ پࣹنْطࣹ.

(39) "طُمْ كٝ دٝكَّلْ أَمࣹنْ. أٝكِّتِرْ، مَا جٝگِتٝرَا كٝ أَلْدَا إࣹ لَمْندِتٝرࣹيدࣹ".

(40) كٝ پࣹلّࣹتْ، هِنٝ وٝودَنِ مٝ كَ أَمࣹنْ، ࢠَدَلْ إࣹ مٝڃُّدࣹ رُتّٝرْدࣹ.

(41) جَنْتٝ كَدِ جࣹيَاطٝ أَمࣹنْ أٝنْ أَيُّوبَ، ندࣹ أٝ نٝدُّنٝو جٝومِ مَكّٝ، وٝنْندࣹ : "سࣹيْتَانࣹ مࣹمِرِي لَمْ تَمْپࣹرࣹ إࣹ مُوسࣹيندِ".

(42) الله وِئَنِ مٝ : "لَتِرْ تُنْندَوَلْ مَا نگَلْ كَ لࣹيْدِ : طَمْ طٝو، كٝ لٝوتٝرْطَمْ، ࢠُوࢠُطَلْ إࣹ نجَرَمْ".

(43) مࣹنْ يࣹطِ مٝ ࢠࣹيْنگُورࣹ مَكّٝ ندࣹنْ، إࣹ يࣹرُ مَࢠّࣹ تَوْتِ ࢠࣹ، يُرْمࣹيندࣹ إِمّٝرْدࣹ إࣹ أَمࣹنْ، إࣹ وَاجُ وٝنَنْندࣹ مَرْࢠࣹ هَقِّلّࣹ ࢠࣹنْ.

(44) "نَنْنگِرْ سٝوكࣹوٝ مَا نگٝنْ، تَمࣹرࣹ هُطٝ، پِيِرَا طُمْ سُدِّيطٝ مَا أٝنْ ، فِي وَتَ أَ فِرْتُ (وٝوندٝورࣹ ندࣹنْ)". مࣹنْ تَوُ مٝ، مُݧِّيطٝ. أٝ مٝڃِّي جࣹيَاطٝ. پࣹلّࣹتْ، كٝ أٝ طُوطُطٝ تُوبُگٝلْ-رُتّٝو.

(45) جَنْتٝ كَدِ، جࣹيَاࢠࣹ أَمࣹنْ ࢠࣹنْ، إبراهيم إࣹ إسحاق إࣹ يعقوب، جٝومِرَاࢠࣹ ݖࣹمْنبࣹ إࣹ يِئَنْطࣹ.

(46) مࣹنْ هࣹيرْتِنَنِي ࢠࣹ هࣹيرْتِنَنْندࣹ، أَنْندِتُگٝلْ گَلّࣹ (لَاكَرَ) طࣹنْ.

(47) إࣹ كَدِ ࢠࣹي كَ أَمࣹنْ، كٝ جࣹيَاࢠࣹ إࣹ سُࢠَاࢠࣹ مٝڃُّࢠࣹ سُࢠَاࢠࣹ.

(48) جَنْتٝ كَدِ إِسْمَائِيلَ، إࣹ يُوسَؤُ، إࣹ زَلْ-كِفْلِ، كَلَ مَࢠّࣹ، كٝ جࣹيَاࢠࣹ إࣹ سُࢠَاࢠࣹ.

(49) طُمْ كٝ وَاجُ. كٝ پࣹلّࣹتْ، هِنٝ وٝودَنِ گٝمْطُࢠࣹ ࢠࣹنْ، رُتّٝرْدࣹ مٝڃُّندࣹ.

(50) طُمْ كٝ الجنة ݧِيࢠَلْ، طِ ࢠࣹ أُدِّتَنَا دَمࣹ مُنْ،

(51) هَرَ كٝ ࢠࣹ سٝࢠِّندِي ࢠࣹ تٝنْ، هِࢠࣹ نٝدَّ تٝنْ مُويَنْتࣹيطِ طُوطُطِ إࣹ نجَرَمْ.

(52) هِنٝ وٝودَنِ ࢠࣹ (تٝنْ): رَࢠِّدُࢠࣹ ندَارْطࣹ، گٝرࣹيࢠࣹ.

(53) كٝ طُمْ وٝنِ كٝ فٝدَطٝنْ، فِي ݧَلَاندࣹ هَسْبٝورࣹ ندࣹنْ.

(54) پࣹلّࣹتْ، طُمْ كٝ أَرْسِكࣹ أَمࣹنْ، گَسُگٝلْ أَلَنَا مٝ.

(55) طُمْ، كٝ نجٝࢠْدِ مَࢠّࣹ . پࣹلّࣹتْ، هِنٝ وٝودَنِ كَدِ بࣹوْࢠࣹ ࢠࣹنْ، كٝ ࢠُرِ بٝنْندࣹ إࣹ رُتّٝرْدࣹ،

(56) وٝنِ جهنم، نگࣹ ࢠࣹ نَاتٝيْتَ. نگࣹ بٝنِي ندَطُّدِ!

(57) طُمْ، كٝ يٝ ࢠࣹ مࣹيطُ: كٝ فَتَتَ، إࣹ ڃࣹڃِّرِ،

(58) گٝطّࣹ طࣹنْ، نَنْندُطࣹ إࣹ مَجّࣹ، كٝ نٝونࣹيجِ.

(59) إࣹ هِنٝ دࣹنْتَلْ نَاتِرٝيَيْ نگَلْ دٝولࣹ إࣹ مٝنْ، تٝولِي أَلَنَا ࢠࣹ; تَوْدࣹ كٝ ࢠࣹ نَاتٝوࢠࣹ يِيتࣹ !

(60) ࢠࣹنْ وِئَ: بِسْمِلَّ أَلَنَا مٝنْ، كٝ أٝنٝنْ دࣹفِ مࣹنْ طُمْ". نگࣹ بٝنِي ݧِيࢠِرْدࣹ.

(61) ࢠࣹ وِئَ: "جٝومِ أَمࣹنْ، دࣹفُطٝ مࣹنْ طُمْ أٝنْ، ࢠࣹيْدَنْ مٝ لࣹپْتّࣹ ݖٝوَلّࣹ كَ يِيتࣹ".

(62) ࢠࣹيَ وِئِ: "كٝ هَطِ مࣹنْ يِئُگٝلْ وٝرْࢠࣹ، ࢠࣹ مࣹنْ لِمْتَيْنٝو إࣹ بٝمْࢠࣹ ࢠࣹنْ?

(63) هَرَ مࣹنْ جٝگِتٝرْنٝ ࢠࣹ تٝنْ جَلْنٝرِ، كَا كٝ گِتࣹ (أَمࣹنْ) يَوْتِ ࢠࣹ?"

(64) پࣹلّࣹتْ، كَ گٝونگَ، كٝ طُمْ وٝنِ يࣹدّٝندِرَلْ يِمْࢠࣹ يِيتࣹ ࢠࣹنْ.

(65) مَاكُ: "أَنْندࣹي كٝ مِي جࣹرْتِنٝووٝ. رࣹوࣹتࣹيطٝ وٝو أَلَا، سِ وَنَا الله گٝوتٝ، فٝولُطٝ أٝنْ،

(66) جࣹيْطٝ كَمُّولِ طِنْ إࣹ لࣹيْدِ ندِنْ، إࣹ كٝنْ كٝ وٝنِ هَكُّندࣹ طِنْ طِطِ، فٝولُطٝ، هَفٝرٝووٝ أٝنْ".

(67) مَاكُ: "(ندࣹي القرآن)، كٝ ندࣹ كَبَارُ مَوْطٝ،

(68) كٝنٝ هِطٝنْ طُورْنِي طُمْ.

(69) گَنْندَلْ أَلَنَانٝ لَمْ، إࣹ فِي نبَتُ نگُنْ دٝوْرٝوُ، تُمَ ندࣹ ࢠࣹ وࣹنْنجَيْنٝو.

(70) وَهَيِنَاكَ إࣹ أَمْ، سِ وَنَا وٝنْندࣹ مِينْ أٝنْ، كٝ مِي جࣹرْتِنٝووٝ ࢠَنْنگِنَ".

(71) جَنْتٝ ندࣹ جٝومِ مَا دَالَنّٝو الملائكة: "مِينْ كٝ مِي تَگِرٝووٝ ࢠَنْندِنْكࣹ، إِمّٝرْدࣹ لٝوپَلْ.

(72) سِ مِي فٝتِندِرِي مٝ، مِي وُتِّي إࣹ مَكّٝ رُوهُ أَمْ أٝنْ، هَرَيْ سُويَنࣹي مٝ سُجّٝنْ".

(73) الملائكة فٝوْ سُجِّ ࢠࣹ دࣹنْندَانگَلْ.

(74) سِ وَنَا إبليس، أٝ مَوْنِنْتِنِي، أٝ وٝنِ جࣹيَاطٝ إࣹ يࣹدُّࢠࣹ ࢠࣹنْ.

(75) (الله) دَالِ: "كٝ أَنْ يٝ إبليس، كٝ هَطُ مَا سُجَّنگٝلْ كٝ مِي تَگِرِ جُوطࣹ أَمْ? أَ مَوْنِنْتِنِ كَا أَ جࣹيَ إࣹ تٝوْنِتِيࢠࣹ ࢠࣹنْ?"

(76) أٝ وِئِ: "كٝ مِينْ ࢠُرِ مٝ مٝڃُّدࣹ. أَ تَگِرِي لَمْ إࣹ يِيتࣹ، أَ تَگِرِي مٝ كَنْكٝ إࣹ لٝوپَلْ".

(77) أٝ دَالِ: "يَلْتُ تٝنْ، أَنْ أٝنْ، كٝ أَ رِيوَاطٝ;

(78) پࣹلّࣹتْ، كُدِّ أَمْ ندِنْ فَوِكࣹ مَ هَا ݧَلْندࣹ نجٝࢠْدِ دِينَ".

(79) (إبليس) وِئِ : "جٝومِ أَمْ، نࣹنِّتَنَمْ، هَا ݧَلْندࣹ ندࣹ ࢠࣹ إِمِّرْتِنْتࣹي".

(80) أٝ دَالِ: "أَ جࣹيَامَ إࣹ نࣹنِّتَنْتࣹيࢠࣹ ࢠࣹنْ،

(81) هَا إࣹ ݧَلَاندࣹ هَپُّ أَنْندَاطٝ".

(82) أٝ وِئِ: "مِي وٝوندِرِي مَوْنگُ مَا نگُنْ ! مَ مِي بࣹوْنُࢠࣹ فٝوْ مَࢠّࣹ،

(83) سِ وَنَا جࣹيَاࢠࣹ مَا، لَࢠِّنَاࢠࣹ ࢠࣹنْ إࣹ مَࢠّࣹ".

(84) (الله) دَالِ: "مِي وٝوندِرِي گٝونگَ، كٝ گٝونگَ مِي وٝوْلَتَ،

(85) مَ مِي هࣹبِّنِرࣹ جهنم، إࣹ ࢠࣹنْ جٝكُّࢠࣹ مَ فٝوْ إࣹ مَࢠّࣹ".

(86) مَاكُ: "مِي لَمْندَاكِ أٝنْ نجٝࢠْدِ إࣹ فِي طُمْ; مِي وٝنَالِ جࣹيَاطٝ إࣹ سِنْݖٝوࢠࣹ(دࣹفْتَتٝوبࣹ).

(87) ندࣹي القرآن وٝنَالِ، سِ وَنَا وَاجُ تَگࣹيفٝ نگٝنْ.

(88) مَ أٝنْ أَنْندُ (فِي) كَبَارُ مَيْرࣹ، كٝ نࣹيࢠَا !".
